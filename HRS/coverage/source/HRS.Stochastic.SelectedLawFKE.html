94      ext j
95      simp only [Finset.mem_filter, Finset.mem_erase, Finset.mem_univ, and_true, true_and]

100      ext j

105      · intro h
106        exact ⟨ne_of_lt h, not_lt.2 h.le⟩

108      = -∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j), B i j := by

169/-- Itô's drift identity: the drift of `configLift Φ` along a lift `Y` equals the generator and
170interaction of Proposition B.3 at the projected configuration (pointwise, for every `s`). -/
171theorem fkeIto_eq (D : DynData d) (hsd : D.s < d) (T : ℝ) (Φ : ℝ → Config d N → ℝ) (s : ℝ)
172    (Y : Fin N → Euc d) :
173    fkeIto D T Φ s Y = fkeGenerator D T Φ s (fun i => Torus.mk (Y i))
174      + fkeInteraction D Φ s (fun i => Torus.mk (Y i)) := by
175  obtain ⟨v, hv, hYr⟩ := exists_lattice_shift (Y := Y)
176    (Y' := repC (fun i => Torus.mk (Y i))) (fun i => (Torus.mk_rep _).symm)
177  have hr : (fun i => Torus.rep (Torus.mk (Y i))) = Y + v := hYr
178  have e1 : derivWithin (fun r => configLift Φ r (fun i => Torus.rep (Torus.mk (Y i))))
179      (Set.Icc 0 T) s = derivWithin (fun r => configLift Φ r Y) (Set.Icc 0 T) s := by
180    congr 1
181    funext r
182    rw [hr]
183    exact configLift_add_lattice Φ r hv Y
184  have e2 : fderiv ℝ (configLift Φ s) (fun i => Torus.rep (Torus.mk (Y i)))
185      = fderiv ℝ (configLift Φ s) Y := by
186    rw [hr]
187    exact fderiv_configLift_add_lattice Φ s hv Y
188  have e3 : fderiv ℝ (fderiv ℝ (configLift Φ s)) (fun i => Torus.rep (Torus.mk (Y i)))
189      = fderiv ℝ (fderiv ℝ (configLift Φ s)) Y := by
190    rw [hr]
191    exact fderiv2_configLift_add_lattice Φ s hv Y
192  have hσ : ∀ k : Fin N × Fin d,
193      fderiv ℝ (fderiv ℝ (configLift Φ s)) Y (liftedNoise D k) (liftedNoise D k)
194        = (2 * D.invβ) * fderiv ℝ (fderiv ℝ (configLift Φ s)) Y (configBasis d N k)
195            (configBasis d N k) := by
196    intro k
197    have hc : liftedNoise D k = Real.sqrt (2 * D.invβ) • configBasis d N k := rfl
198    have h2 : Real.sqrt (2 * D.invβ) ^ 2 = 2 * D.invβ := Real.sq_sqrt (by linarith [D.hβ])
199    rw [hc]
200    simp only [map_smul, smul_apply, smul_eq_mul]
201    linear_combination (fderiv ℝ (fderiv ℝ (configLift Φ s)) Y (configBasis d N k)
202      (configBasis d N k)) * h2
203  unfold fkeIto
204  simp only [fkeGenerator, fkeInteraction]
205  rw [e1, e2, e3, clm_driftVec_filter D hsd, Finset.sum_congr rfl (fun k _ => hσ k),
206    ← Finset.mul_sum]
207  simp only [pairForce]
208  ring

307/-- The Itô drift of `configLift Φ` is bounded, uniformly in `s ∈ [0,T]` and in the lift. -/
308theorem fkeIto_bound {T : ℝ} (hT : 0 ≤ T) (D : DynData d) (hN : 0 < N) (hV : Torus.IsCk 2 D.V)
309    (Φ : ℝ → Config d N → ℝ) (hΦ : IsC12On T Set.univ (configLift Φ))
310    {K : Set (Config d N)} (hK : IsCompact K) (hdisj : Disjoint K (collisionSet d N))
311    (hsupp : ∀ r (X : Config d N), X ∉ K → Φ r X = 0) :
312    ∃ C : ℝ, ∀ s ∈ Icc 0 T, ∀ Y, ‖fkeIto D T Φ s Y‖ ≤ C := by
313  obtain ⟨C0, hC0⟩ := bounded_of_periodic (T := T)
314    (fun s Y => derivWithin (fun r => configLift Φ r Y) (Icc 0 T) s)
315    (by simpa [Function.uncurry] using hΦ.cont_deriv)
316    (fun s Y v hv => by simp only [configLift_add_lattice Φ _ hv])
317  obtain ⟨C1, hC1⟩ := bounded_of_periodic (T := T)
318    (fun s Y => fderiv ℝ (configLift Φ s) Y)
319    (by simpa [Function.uncurry] using hΦ.cont_fderiv)
320    (fun s Y v hv => fderiv_configLift_add_lattice Φ s hv Y)
321  obtain ⟨C2, hC2⟩ := bounded_of_periodic (T := T)
322    (fun s Y => fderiv ℝ (fderiv ℝ (configLift Φ s)) Y)
323    (by simpa [Function.uncurry] using hΦ.cont_fderiv₂)
324    (fun s Y v hv => fderiv2_configLift_add_lattice Φ s hv Y)
325  obtain ⟨B, hB⟩ := driftVec_bound_on_K D hN hV hK hdisj
326  set C1' := max C1 0
327  set C2' := max C2 0
328  set B' := max B 0
329  set C0' := max C0 0
330  have hσ : ∑ k : Fin N × Fin d, ‖liftedNoise D k‖ * ‖liftedNoise D k‖ ≥ 0 :=
331    Finset.sum_nonneg fun k _ => mul_nonneg (norm_nonneg _) (norm_nonneg _)
332  refine ⟨C0' + C1' * B' + (1 / 2) * (C2' * ∑ k : Fin N × Fin d,
333      ‖liftedNoise D k‖ * ‖liftedNoise D k‖), fun s hs Y => ?_⟩
334  have hd : ‖fderiv ℝ (configLift Φ s) Y (driftVec D (fun j => Torus.mk (Y j)))‖ ≤ C1' * B' := by
335    have hL : ‖fderiv ℝ (configLift Φ s) Y‖ ≤ C1' := (hC1 s hs Y).trans (le_max_left _ _)
336    by_cases hmem : (fun j => Torus.mk (Y j)) ∈ K
337    · calc ‖fderiv ℝ (configLift Φ s) Y (driftVec D (fun j => Torus.mk (Y j)))‖
338          ≤ ‖fderiv ℝ (configLift Φ s) Y‖ * ‖driftVec D (fun j => Torus.mk (Y j))‖ :=
339            ContinuousLinearMap.le_opNorm _ _
340        _ ≤ C1' * B' := mul_le_mul hL ((hB _ hmem).trans (le_max_left _ _))
341            (norm_nonneg _) ((norm_nonneg _).trans hL)
342    · rw [fderiv_configLift_eq_zero hK Φ hsupp s Y hmem]
