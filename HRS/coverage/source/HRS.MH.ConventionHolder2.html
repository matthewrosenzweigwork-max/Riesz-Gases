23/-- Extraction of the bound and the Hölder constant of a `Hol0` function. -/
24theorem ch_hol_data {α : ℝ} {ρ : Torus d → ℝ} (hρ : PDE.MFHolder.Hol0 α ρ) :
25    ∃ B Λ : ℝ, 0 ≤ B ∧ 0 ≤ Λ ∧ (∀ x, |ρ x| ≤ B) ∧
26      ∀ x z : Torus d, ‖z‖ ≤ 1 / 4 → |ρ (x - z) - ρ x| ≤ Λ * ‖z‖ ^ α := by
27  obtain ⟨B, hB0, hB⟩ := hρ.bdd
28  obtain ⟨hc, M, hM⟩ := hρ
29  refine ⟨B, |M|, hB0, abs_nonneg M, hB, fun x z _ => ?_⟩
30  have h1 := hM (x - z) x
31  rw [sub_sub_cancel_left, norm_neg] at h1
32  exact h1.trans (mul_le_mul_of_nonneg_right (le_abs_self M) (Real.rpow_nonneg (norm_nonneg _) _))

34theorem integrable_convGradConv_integrand_of_holder_proof :
35    type_of% @HRS.integrable_convGradConv_integrand_of_holder := by
36  intro d s α hd hs0 hsd hα0 hα1 hα ρ hρ x
37  obtain ⟨C, hG⟩ := KernelGradD1.goodKernel_periodicRieszGrad_of_one_le hd hs0 hsd
38  obtain ⟨B, Λ, hB0, hΛ0, hB, hH⟩ := ch_hol_data hρ
39  exact Duality.integrable_conv_form hG hρ.1.measurable hB hB0 hΛ0 hα0 hα hH x
