19/-- Variation budget of a bounded-variation path. -/
20def q17_Wf {E : Type*} [PseudoEMetricSpace E] (F : ℝ → E) (x y : ℝ) : ℝ :=
21  (eVariationOn F (Set.Icc x y)).toReal

23theorem q17_W_props {E : Type*} [NormedAddCommGroup E] (T : ℝ) (F : ℝ → E)
24    (hF : BoundedVariationOn F (Set.Icc 0 T)) :
25    (∀ x y, 0 ≤ x → x ≤ y → y ≤ T → ‖F y - F x‖ ≤ q17_Wf F x y) ∧
26    (∀ x u y, 0 ≤ x → x ≤ u → u ≤ y → y ≤ T →
27      q17_Wf F x u + q17_Wf F u y ≤ q17_Wf F x y) := by
28  constructor
29  · intro x y hx hxy hy
30    have hfin : eVariationOn F (Set.Icc x y) ≠ ⊤ :=
31      ne_top_of_le_ne_top hF (eVariationOn.mono F (Set.Icc_subset_Icc hx hy))
32    have h1 : edist (F y) (F x) ≤ eVariationOn F (Set.Icc x y) := by
33      rw [edist_comm]
34      exact eVariationOn.edist_le F ⟨le_rfl, hxy⟩ ⟨hxy, le_rfl⟩
35    rw [← dist_eq_norm, dist_edist]
36    exact ENNReal.toReal_mono hfin h1
37  · intro x u y hx hxu huy hy
38    have hu0 : 0 ≤ u := le_trans hx hxu
39    have huT : u ≤ T := le_trans huy hy
40    have hadd := eVariationOn.Icc_add_Icc F (s := Set.Icc 0 T) hxu huy ⟨hu0, huT⟩
41    have e1 : Set.Icc (0 : ℝ) T ∩ Set.Icc x u = Set.Icc x u :=
42      Set.inter_eq_right.2 (Set.Icc_subset_Icc hx huT)
43    have e2 : Set.Icc (0 : ℝ) T ∩ Set.Icc u y = Set.Icc u y :=
44      Set.inter_eq_right.2 (Set.Icc_subset_Icc hu0 hy)
45    have e3 : Set.Icc (0 : ℝ) T ∩ Set.Icc x y = Set.Icc x y :=
46      Set.inter_eq_right.2 (Set.Icc_subset_Icc hx hy)
47    rw [e1, e2, e3] at hadd
48    have hfin1 : eVariationOn F (Set.Icc x u) ≠ ⊤ :=
49      ne_top_of_le_ne_top hF (eVariationOn.mono F (Set.Icc_subset_Icc hx huT))
50    have hfin2 : eVariationOn F (Set.Icc u y) ≠ ⊤ :=
51      ne_top_of_le_ne_top hF (eVariationOn.mono F (Set.Icc_subset_Icc hu0 hy))
52    unfold q17_Wf
53    rw [← ENNReal.toReal_add hfin1 hfin2, hadd]

63/-- Convergence of the projected sums, one term at a time. -/
64theorem q17_term_tendsto {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
65    (Pj : ℕ → (E →L[ℝ] E)) (K : ℝ) (hK : ∀ L, ‖Pj L‖ ≤ K)
66    (hconv : ∀ e, Tendsto (fun L => Pj L e) atTop (𝓝 e))
67    (F : ℝ → E) (A : ℝ → (E →L[ℝ] ℝ)) (AL : ℕ → ℝ → (E →L[ℝ] ℝ)) (T : ℝ)
68    (hAu : ∀ ε : ℝ, 0 < ε → ∃ L₀ : ℕ, ∀ L ≥ L₀, ∀ u ∈ Set.Icc (0 : ℝ) T, ‖AL L u - A u‖ ≤ ε)
69    (τ a b : ℝ) (hτ : τ ∈ Set.Icc (0 : ℝ) T) :
70    Tendsto (fun L => AL L τ (Pj L (F b - F a)) - A τ (F b - F a)) atTop (𝓝 0) := by
71  have hn : Tendsto (fun L => ‖AL L τ - A τ‖) atTop (𝓝 0) := by
72    rw [Metric.tendsto_atTop]
73    intro ε hε
74    obtain ⟨L0, hL0⟩ := hAu (ε / 2) (by positivity)
75    refine ⟨L0, fun L hL => ?_⟩
76    rw [Real.dist_eq, sub_zero, abs_norm]
77    linarith [hL0 L hL τ hτ]
78  obtain ⟨v, hv⟩ : ∃ v : E, v = F b - F a := ⟨_, rfl⟩
79  rw [← hv]
80  have hX : Tendsto (fun L => Pj L v - v) atTop (𝓝 0) :=
81    tendsto_sub_nhds_zero_iff.2 (hconv v)
82  have hnX : Tendsto (fun L => ‖Pj L v - v‖) atTop (𝓝 0) := by
83    simpa using hX.norm
84  have hg : Tendsto (fun L => ‖AL L τ - A τ‖ * (‖v‖ + ‖Pj L v - v‖) + ‖A τ‖ * ‖Pj L v - v‖)
85      atTop (𝓝 0) := by
86    have := (hn.mul ((tendsto_const_nhds : Tendsto (fun _ : ℕ => ‖v‖) atTop (𝓝 ‖v‖)).add hnX)).add
87      (((tendsto_const_nhds : Tendsto (fun _ : ℕ => ‖A τ‖) atTop (𝓝 ‖A τ‖))).mul hnX)
88    simpa using this
89  have hid : ∀ L, AL L τ (Pj L v) - A τ v
90      = (AL L τ - A τ) (v + (Pj L v - v)) + A τ (Pj L v - v) := by
91    intro L
92    simp only [ContinuousLinearMap.sub_apply, map_add, map_sub]
93    abel
94  refine squeeze_zero_norm (fun L => ?_) hg
95  rw [hid L]
96  calc ‖(AL L τ - A τ) (v + (Pj L v - v)) + A τ (Pj L v - v)‖
97      ≤ ‖(AL L τ - A τ) (v + (Pj L v - v))‖ + ‖A τ (Pj L v - v)‖ := norm_add_le _ _
98    _ ≤ ‖AL L τ - A τ‖ * ‖v + (Pj L v - v)‖ + ‖A τ‖ * ‖Pj L v - v‖ := by
99        gcongr
100        · exact ContinuousLinearMap.le_opNorm _ _
101        · exact ContinuousLinearMap.le_opNorm _ _
102    _ ≤ ‖AL L τ - A τ‖ * (‖v‖ + ‖Pj L v - v‖) + ‖A τ‖ * ‖Pj L v - v‖ := by
103        gcongr
104        exact norm_add_le _ _

106theorem measure_projection_proof {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [CompleteSpace E]
107    (Pj : ℕ → (E →L[ℝ] E)) (K : ℝ) (hK : ∀ L, ‖Pj L‖ ≤ K)
108    (hconv : ∀ e, Tendsto (fun L => Pj L e) atTop (𝓝 e))
109    (T : ℝ) (hT : 0 < T) (F : ℝ → E) (hF : BoundedVariationOn F (Set.Icc 0 T)) (hF0 : F 0 = 0)
110    (A : ℝ → (E →L[ℝ] ℝ)) (AL : ℕ → ℝ → (E →L[ℝ] ℝ))
111    (hA : ContinuousOn A (Set.Icc 0 T)) (hAL : ∀ L, ContinuousOn (AL L) (Set.Icc 0 T))
112    (hAu : ∀ ε : ℝ, 0 < ε → ∃ L₀ : ℕ, ∀ L ≥ L₀, ∀ u ∈ Set.Icc (0 : ℝ) T, ‖AL L u - A u‖ ≤ ε) :
113    ∃ I : ℝ, IsRS T (fun τ a b => A τ (F b - F a)) I ∧
114      ∃ IL : ℕ → ℝ, (∀ L, IsRS T (fun τ a b => AL L τ (Pj L (F b - F a))) (IL L)) ∧
115        Tendsto IL atTop (𝓝 I) := by
116  obtain ⟨hW0, hWadd⟩ := q17_W_props T F hF
117  have hK0 : 0 ≤ K := le_trans (norm_nonneg _) (hK 0)
118  have hW00 : 0 ≤ q17_Wf F 0 T := le_trans (norm_nonneg _) (hW0 0 T le_rfl hT.le le_rfl)
119  have hWaddK : ∀ x u y, 0 ≤ x → x ≤ u → u ≤ y → y ≤ T →
120      K * q17_Wf F x u + K * q17_Wf F u y ≤ K * q17_Wf F x y := fun x u y hx hxu huy hy => by
121    rw [← mul_add]
122    exact mul_le_mul_of_nonneg_left (hWadd x u y hx hxu huy hy) hK0
123  have hWG : ∀ L, ∀ x y, 0 ≤ x → x ≤ y → y ≤ T →
124      ‖Pj L (F y) - Pj L (F x)‖ ≤ K * q17_Wf F x y := fun L x y hx hxy hy => by
125    rw [← map_sub]
126    calc ‖Pj L (F y - F x)‖ ≤ ‖Pj L‖ * ‖F y - F x‖ := (Pj L).le_opNorm _
127      _ ≤ K * q17_Wf F x y := mul_le_mul (hK L) (hW0 x y hx hxy hy) (norm_nonneg _) hK0
128  have hconvG : ∀ L (τ a b : ℝ), AL L τ (Pj L (F b) - Pj L (F a)) = AL L τ (Pj L (F b - F a)) :=
129    fun L τ a b => by rw [map_sub (Pj L) (F b) (F a)]
130  obtain ⟨I, hI⟩ := q17_exists T hT A F (q17_Wf F) hW0 hWadd hA
131  have hILex : ∀ L, ∃ x : ℝ, IsRS T (fun τ a b => AL L τ (Pj L (F b - F a))) x := by
132    intro L
133    obtain ⟨x, hx⟩ := q17_exists T hT (AL L) (fun t => Pj L (F t)) (fun x y => K * q17_Wf F x y)
134      (hWG L) hWaddK (hAL L)
135    refine ⟨x, ?_⟩
136    have hfun : (fun τ a b => AL L τ (Pj L (F b - F a))) =
137        (fun τ a b => AL L τ ((fun t => Pj L (F t)) b - (fun t => Pj L (F t)) a)) := by
138      funext τ a b
139      exact (hconvG L τ a b).symm
140    rw [hfun]
141    exact hx
142  choose IL hIL using hILex
143  refine ⟨I, hI, IL, hIL, ?_⟩
144  rw [Metric.tendsto_atTop]
145  intro ε hε
146  have hKW : 0 ≤ K * q17_Wf F 0 T := mul_nonneg hK0 hW00
147  set η₁ : ℝ := ε / (6 * (K * q17_Wf F 0 T + 1)) with hη₁def
148  have hη₁ : 0 < η₁ := div_pos hε (by linarith)
149  have hη₁eq : η₁ * (K * q17_Wf F 0 T + 1) = ε / 6 := by
150    rw [hη₁def]
151    field_simp
152  obtain ⟨δA, hδA, hmodA⟩ := q17_modulus T A hA η₁ hη₁
153  obtain ⟨L₀, hL₀⟩ := hAu η₁ hη₁
154  obtain ⟨δI, hδI, hI'⟩ := hI (ε / 6) (by positivity)
155  set δR : ℝ := δA / 4 with hδRdef
156  have hδR : 0 < δR := by positivity
157  have hmin0 : 0 < min δI δR / 2 := half_pos (lt_min hδI hδR)
158  have hmin1 : min δI δR / 2 < δI := by linarith [min_le_left δI δR]
159  have hmin2 : min δI δR / 2 ≤ δR := by linarith [min_le_right δI δR]
160  obtain ⟨M, hM⟩ : ∃ M : ℕ, T / (min δI δR / 2) < M := exists_nat_gt _
161  have hPmesh : T / ((M + 1 : ℕ) : ℝ) ≤ min δI δR / 2 :=
162    q17_small T _ hT hmin0 M hM (M + 1) (by omega)
163  obtain ⟨hPv1, hPv2, hPv3, hPv4, hPv5⟩ := q17_uf_valid T hT.le M
164  have h0P : ∀ j : Fin (M + 1 + 1), 0 ≤ q17_uf T (M + 1) j := fun j => by
165    have := hPv1 (Fin.zero_le j)
166    rw [hPv4] at this
167    exact this
168  have hTP : ∀ j : Fin (M + 1 + 1), q17_uf T (M + 1) j ≤ T := fun j => by
169    have := hPv1 (Fin.le_last j)
170    rw [hPv5] at this
171    exact this
172  have hτI : ∀ i : Fin (M + 1), q17_ut T (M + 1) i ∈ Set.Icc (0 : ℝ) T := fun i =>
173    ⟨le_trans (h0P _) (hPv3 i).1, le_trans (hPv3 i).2 (hTP _)⟩
174  have hPI : |∑ i : Fin (M + 1), A (q17_ut T (M + 1) i)
175      (F (q17_uf T (M + 1) i.succ) - F (q17_uf T (M + 1) i.castSucc)) - I| < ε / 6 :=
176    hI' (M + 1) (q17_uf T (M + 1)) (q17_ut T (M + 1)) (q17_uf_strict T hT (M + 1) (by omega))
177      hPv4 hPv5 hPv3 (fun i => by rw [hPv2 i]; exact lt_of_le_of_lt hPmesh hmin1)
178  have hlim : Tendsto (fun L => ∑ i : Fin (M + 1), AL L (q17_ut T (M + 1) i)
179      (Pj L (F (q17_uf T (M + 1) i.succ) - F (q17_uf T (M + 1) i.castSucc)))
180      - ∑ i : Fin (M + 1), A (q17_ut T (M + 1) i)
181      (F (q17_uf T (M + 1) i.succ) - F (q17_uf T (M + 1) i.castSucc))) atTop (𝓝 0) := by
182    have h := tendsto_finset_sum (Finset.univ : Finset (Fin (M + 1))) (fun i _ =>
183      q17_term_tendsto Pj K hK hconv F A AL T hAu (q17_ut T (M + 1) i)
184        (q17_uf T (M + 1) i.castSucc) (q17_uf T (M + 1) i.succ) (hτI i))
185    simpa only [Finset.sum_sub_distrib, Finset.sum_const_zero] using h
186  obtain ⟨L₁, hL₁⟩ := Metric.tendsto_atTop.1 hlim (ε / 6) (by positivity)
187  -- per-`L` comparison of the integrals with the projected sums
188  have hILbound : ∀ L, L₀ ≤ L → |IL L - ∑ i : Fin (M + 1), AL L (q17_ut T (M + 1) i)
189      (Pj L (F (q17_uf T (M + 1) i.succ) - F (q17_uf T (M + 1) i.castSucc)))|
190      < ε / 12 + 3 * η₁ * (K * q17_Wf F 0 T) := by
191    intro L hL
192    have hBL : ∀ s t, 0 ≤ s → s ≤ T → 0 ≤ t → t ≤ T → |s - t| ≤ 2 * δR →
193        ‖AL L s - AL L t‖ ≤ 3 * η₁ := by
194      intro s t hs hsT ht htT hst
195      have e1 := hL₀ L hL s ⟨hs, hsT⟩
196      have e3 := hL₀ L hL t ⟨ht, htT⟩
197      have e2 := hmodA s t hs hsT ht htT (by rw [hδRdef] at hst; linarith)
198      calc ‖AL L s - AL L t‖
199          = ‖((AL L s - A s) + (A s - A t)) - (AL L t - A t)‖ := by congr 1; abel
200        _ ≤ ‖(AL L s - A s) + (A s - A t)‖ + ‖AL L t - A t‖ := norm_sub_le _ _
201        _ ≤ ‖AL L s - A s‖ + ‖A s - A t‖ + ‖AL L t - A t‖ := by
202            linarith [norm_add_le (AL L s - A s) (A s - A t)]
