234/-- Proposition 5.2 (`C:prop:adjoint`), in the form of `HRS.PDE.adjoint_decay_comparison`: the
235uniform adjoint decay and comparison estimates, for the propagator of the solution `ρ^t` of
236Proposition 5.1. -/
237theorem adjoint_decay_comparison_wiring (hd : 1 ≤ d) (s βm βp : ℝ) (K : ℕ) (V ρ₀ : Torus d → ℝ)
238    (H : FiniteRegHyp d s βm βp K V ρ₀) (j : ℕ) (_hjeven : Even j) (hj : (d : ℝ) / 2 + 4 < j)
239    (hK : 2 * j + 4 * d + 20 ≤ K) :
240    ∃ C c κ : ℝ, 0 < C ∧ 0 < c ∧ 0 < κ ∧
241      ∀ (β : ℝ) (hβ : β ∈ Set.Icc βm βp) (ρc : ℝ → Coeff d) (_ : IsMFSolutionHK d s V β⁻¹ K ρ₀ ρc)
242        (ρβ : Torus d → ℝ) (_ : IsThermalDensity (periodicRiesz d s) V β ρβ) (_ : Continuous ρβ)
243        (_ : ∀ x, 0 < ρβ x) (φ : Torus d → ℝ) (_ : Continuous φ) (_ : MemSobolev j φ)
244        (r t : ℝ) (_ : 0 ≤ r) (_ : r ≤ t),
245        MemSobolev j (projMean (withDensityReal ρβ)
246          (propagator (mfData d s H.hs0 H.hsd V β (H.hβm.trans_le hβ.1)) (mfPath ρc) r t φ)) ∧
247        sobolevNorm j (projMean (withDensityReal ρβ)
248            (propagator (mfData d s H.hs0 H.hsd V β (H.hβm.trans_le hβ.1)) (mfPath ρc) r t φ)) ≤
249          C * Real.exp (-c * (t - r)) * sobolevNorm j (projMean (withDensityReal ρβ) φ) ∧
250        sobolevNorm j (fun x => projMean (withDensityReal ρβ)
251            (propagator (mfData d s H.hs0 H.hsd V β (H.hβm.trans_le hβ.1)) (mfPath ρc) r t φ) x -
252          equilibriumSemigroup (mfData d s H.hs0 H.hsd V β (H.hβm.trans_le hβ.1)) ρβ (t - r)
253            (projMean (withDensityReal ρβ) φ) x) ≤
254          C * Real.exp (-c * (t - r)) * Real.exp (-κ * r) *
255            sobolevNorm j (projMean (withDensityReal ρβ) φ) := by
256  obtain ⟨R, Λ, Λ₀, c₀, Cd, κ, hc₀, hκ, hall⟩ := finiteReg_hypotheses hd s βm βp K V ρ₀ H j hj hK
257  obtain ⟨C, c, hC, hc, hmain⟩ := exists_adjoint_decay_abstract hd s j hj βp⁻¹ βm⁻¹ c₀ Λ₀
258    ((j : ℝ) + (d : ℝ) / 2 + 2) (by linarith) R 1 Λ (Torus.cbNorm (j + 3) V) Cd κ hκ
259  refine ⟨C, c, κ, hC, hc, hκ, ?_⟩
260  intro β hβ ρc hρc ρβ hρβ hρC hpos φ hφc hφ r t hr hrt
261  obtain ⟨HF, hAD, hdec⟩ := hall β hβ ρc hρc ρβ hρβ hρC hpos
262  exact hmain β V ρβ HF (mfPath ρc) hAD hdec φ hφc hφ r t hr hrt
