14## Representation of quadratic variation (Mathlib has no stochastic calculus)
15
16Mathlib (at the pinned version) has `Martingale` for general index sets, Brownian motion,
17but no stochastic integral, no quadratic variation, no DDS theorem.  A continuous
18square-integrable real martingale `M` and its quadratic variation `A = [M]` are therefore
19represented *axiomatically through their characterising properties*: `M` is a continuous
20martingale with `M_0 = 0`, `A` is continuous, adapted, nondecreasing with `A_0 = 0`, and
21`M² - A` is a martingale.  (For continuous `L²` martingales this characterises `[M]`
22uniquely; the uniqueness is standard and is part of package CLT-01.)  The vector case uses
23the absolutely continuous covariation `[M^a,M^b]_τ = ∫_0^τ Σ^{ab}` with a given process
24`Σ`.  This keeps the lemma statements exactly as in the source without building Itô
25calculus first; when a stochastic-integral library exists (package SI-01), `IsQuadraticVariation`
26is to be *proved* for `∫ Φ dW`, not assumed.
27-/

36variable {Ω : Type*} {mΩ : MeasurableSpace Ω} (P : Measure Ω) [IsProbabilityMeasure P]

38/-- `M` is a continuous square-integrable martingale on `[0,t]` with `M_0 = 0`, with respect
39to the filtration `ℱ`. -/
40structure IsContSqMartingale (ℱ : Filtration ℝ mΩ) (t : ℝ) (M : ℝ → Ω → ℝ) : Prop where
41  martingale : Martingale M ℱ P
42  zero : ∀ᵐ ω ∂P, M 0 ω = 0
43  cont : ∀ᵐ ω ∂P, ContinuousOn (fun τ => M τ ω) (Set.Icc 0 t)
44  sq_int : ∀ τ ∈ Set.Icc 0 t, MemLp (M τ) 2 P
45
46/-- `A` is the quadratic variation `[M]` of `M`: continuous, adapted, nondecreasing, `A_0 = 0`,
47and `M² - A` is a martingale. -/
48structure IsQuadraticVariation (ℱ : Filtration ℝ mΩ) (t : ℝ) (M A : ℝ → Ω → ℝ) : Prop where
49  adapted : Adapted ℱ A
50  zero : ∀ᵐ ω ∂P, A 0 ω = 0
51  cont : ∀ᵐ ω ∂P, ContinuousOn (fun τ => A τ ω) (Set.Icc 0 t)
52  mono : ∀ᵐ ω ∂P, MonotoneOn (fun τ => A τ ω) (Set.Icc 0 t)
53  compensator : Martingale (fun τ ω => M τ ω ^ 2 - A τ ω) ℱ P

55/-- The centered real Gaussian `N(0,v)` (`v ≥ 0`). -/
56def gaussianLaw (v : ℝ) : Measure ℝ := ProbabilityTheory.gaussianReal 0 (Real.toNNReal v)

69/-- The Euclidean matrix norm used as `‖·‖_{S_1}` on symmetric matrices: trace norm via the
70variational formula `sup_{U orthogonal} tr(Qᵀ U)` (`traceNormMatrix`). -/
71abbrev s1Norm (Q : Matrix (Fin p) (Fin p) ℝ) : ℝ := traceNormMatrix Q
72
73/-- `M` is a continuous `ℝ^p`-valued martingale on `[0,t]` with `M_0 = 0` and absolutely
74continuous quadratic covariation `[M^a,M^b]_τ = ∫_0^τ Σ^{ab}_σ dσ`, `Σ_σ ≥ 0` symmetric
75(source lines 10546-10547; the source presupposes that the Lebesgue integrals exist, which
76`sigma_pathint` and `sigma_entry_int` record: without them the Bochner integrals below take the
77junk value `0` on non-integrable paths), represented through: each pair `M^a M^b - ∫_0^· Σ^{ab}` is a
78martingale, `Σ` is adapted and measurable in `(σ, ω)`. -/
79structure IsVectorMartingaleWithCovariation (ℱ : Filtration ℝ mΩ) (t : ℝ)
80    (M : ℝ → Ω → (ℝ^p)) (Sg : ℝ → Ω → Matrix (Fin p) (Fin p) ℝ) : Prop where
81  martingale : ∀ a : Fin p, Martingale (fun τ ω => M τ ω a) ℱ P
82  zero : ∀ᵐ ω ∂P, M 0 ω = 0
83  cont : ∀ᵐ ω ∂P, ContinuousOn (fun τ => M τ ω) (Set.Icc 0 t)
84  sq_int : ∀ τ ∈ Set.Icc 0 t, MemLp (M τ) 2 P
85  sigma_meas : Measurable (Function.uncurry Sg)
86  sigma_adapted : ∀ σ, Measurable[ℱ σ] (Sg σ)
87  sigma_psd : ∀ᵐ ω ∂P, ∀ σ ∈ Set.Icc 0 t, (Sg σ ω).PosSemidef
88  sigma_int : Integrable (fun ω => ∫ σ in Set.Icc 0 t, s1Norm (Sg σ ω)) P
89  sigma_pathint : ∀ᵐ ω ∂P, IntegrableOn (fun σ => s1Norm (Sg σ ω)) (Set.Icc 0 t)
90  sigma_entry_int : ∀ᵐ ω ∂P, ∀ a b : Fin p, IntegrableOn (fun σ => Sg σ ω a b) (Set.Icc 0 t)
91  covariation : ∀ a b : Fin p,
92    Martingale (fun τ ω => M τ ω a * M τ ω b - ∫ σ in Set.Icc 0 τ, Sg σ ω a b) ℱ P
