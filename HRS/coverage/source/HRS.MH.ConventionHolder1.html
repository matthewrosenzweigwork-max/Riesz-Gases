5# `convGradConv_eq_convGradHolder`: cube convention versus Euclidean-ball convention
6
7Relocation of `HRS.convGradConv_eq_convGradHolder` (`HRS/PDE/ConventionHolder.lean`).  The two
8integrands differ only on the region `{‖x-y‖ ≤ 1/4 < tdist x y}` (cube minus ball), where the cube
9form subtracts `ρ x • G (x-y)`; this region is symmetric under `y ↦ 2x - y`-type reflection
10(`z ↦ -z` in the variable `z = x - y`) and `G` is odd, so the subtracted piece integrates to zero.
11-/

24/-- On the far field `tdist z 0 > 1/4` a good kernel is bounded. -/
25theorem ch_far_bound (hG : KernelGrad.GoodKernel d s C G) {z : Torus d}
26    (hz : 1 / 4 < KernelGrad.tdist z 0) :
27    ‖G z‖ ≤ C * ((1 / 4 : ℝ) ^ (-s - 1) + 1) := by
28  have hz0 : z ≠ 0 := by
29    rintro rfl
30    have h0 : Torus.mk (0 : Euc d) = 0 := by funext i; simp [Torus.mk]
31    have := KernelGrad.tdist_mk (d := d) (y := 0) (fun i => by simp)
32    rw [h0] at this
33    simp at this
34    linarith
35  refine (hG.bound z hz0).trans ?_
36  have h1 : KernelGrad.tdist z 0 ^ (-s - 1) ≤ (1 / 4 : ℝ) ^ (-s - 1) :=
37    Real.rpow_le_rpow_of_nonpos (by norm_num) hz.le (by have := hG.hs0; linarith)
38  have := hG.hC0
39  gcongr

69/-- The bounded kernel on the cube minus the ball, as a function of `z = x - y`. -/
70def chRing (G : Torus d → Euc d) (z : Torus d) : Euc d :=
71  if ‖z‖ ≤ 1 / 4 ∧ 1 / 4 < KernelGrad.tdist z 0 then G z else 0

79theorem ch_ring_bound (hG : KernelGrad.GoodKernel d s C G) (z : Torus d) :
80    ‖chRing G z‖ ≤ C * ((1 / 4 : ℝ) ^ (-s - 1) + 1) := by
81  unfold chRing

83  · exact ch_far_bound hG h.2

103/-- The far-field integrand is integrable (bounded measurable function on a probability space). -/
104theorem ch_integrable_far (hG : KernelGrad.GoodKernel d s C G) {ρ : Torus d → ℝ}
105    (hρm : Measurable ρ) {B : ℝ} (hB : ∀ y, |ρ y| ≤ B) (x : Torus d) :
106    Integrable (fun y => if 1 / 4 < Torus.tdist x y then ρ y • G (x - y) else 0) := by
107  have hm : Measurable fun y : Torus d => if 1 / 4 < Torus.tdist x y then ρ y • G (x - y) else 0 :=
108    Measurable.ite (measurableSet_lt measurable_const (ch_measurable_tdist_sub x))
109      (hρm.smul (hG.meas.comp (measurable_const.sub measurable_id))) measurable_const
110  refine Integrable.of_bound hm.aestronglyMeasurable (B * (C * ((1 / 4 : ℝ) ^ (-s - 1) + 1)))
111    (Eventually.of_forall fun y => ?_)
112  by_cases h : 1 / 4 < Torus.tdist x y
113  · rw [if_pos h, norm_smul, Real.norm_eq_abs]
114    have h' : 1 / 4 < KernelGrad.tdist (x - y) 0 := by
115      rw [← KernelGrad.tdist_eq_sub_zero]; exact h
116    exact mul_le_mul (hB y) (ch_far_bound hG h') (norm_nonneg _) ((abs_nonneg _).trans (hB y))
117  · rw [if_neg h]
118    have := hG.hC0
119    have hB0 : 0 ≤ B := (abs_nonneg _).trans (hB x)
120    have := Real.rpow_nonneg (by norm_num : (0 : ℝ) ≤ 1 / 4) (-s - 1)
121    simp only [norm_zero]
122    positivity

130theorem ch_pointwise (_hG : KernelGrad.GoodKernel d s C G) (ρ : Torus d → ℝ) (x y : Torus d) :
131    (if ‖x - y‖ ≤ 1 / 4 then (ρ y - ρ x) • G (x - y) else ρ y • G (x - y)) =
132      (if Torus.tdist x y ≤ 1 / 4 then (ρ y - ρ x) • G (x - y) else 0) +
133      (if 1 / 4 < Torus.tdist x y then ρ y • G (x - y) else 0) - ρ x • chRing G (x - y) := by
134  have hle : Torus.tdist x y ≤ 1 / 4 → ‖x - y‖ ≤ 1 / 4 := fun h =>
135    (KernelGrad.norm_le_tdist (x - y)).trans (by
136      have h' := h
137      rwa [ch_tdist_eq, KernelGrad.tdist_eq_sub_zero] at h')
138  have e : Torus.tdist x y = KernelGrad.tdist (x - y) 0 := by
139    rw [ch_tdist_eq, KernelGrad.tdist_eq_sub_zero]
140  have hn : ‖x - y‖ ≤ KernelGrad.tdist (x - y) 0 := KernelGrad.norm_le_tdist _
141  unfold chRing
142  rw [← e] at *
143  split_ifs <;> first
144    | (exfalso; linarith)
145    | (exfalso; simp_all; done)
146    | (simp [sub_smul]; done)
147    | (simp_all [sub_smul]; done)

149theorem convGradConv_eq_convGradHolder_proof :
150    type_of% @HRS.convGradConv_eq_convGradHolder := by
151  intro d s C G hG ρ hρm hρb x hint
152  obtain ⟨B, hB⟩ := hρb
153  have hf2 := ch_integrable_far hG hρm hB x
154  have hc := ch_integrable_ring hG x
155  have hc0 : ∫ y, chRing G (x - y) = 0 := by
156    rw [integral_sub_left_eq_self (fun z => chRing G z) volume x]
157    exact ch_integral_odd (ch_ring_odd hG)
158  unfold convGradConv convGradHolder
159  simp_rw [ch_pointwise hG ρ x]
160  have h1 : ∫ y, ((if Torus.tdist x y ≤ 1 / 4 then (ρ y - ρ x) • G (x - y) else 0) +
161      (if 1 / 4 < Torus.tdist x y then ρ y • G (x - y) else 0) - ρ x • chRing G (x - y)) =
162      (∫ y, ((if Torus.tdist x y ≤ 1 / 4 then (ρ y - ρ x) • G (x - y) else 0) +
163      (if 1 / 4 < Torus.tdist x y then ρ y • G (x - y) else 0))) -
164      ∫ y, ρ x • chRing G (x - y) := integral_sub (hint.add hf2) (hc.smul (ρ x))
165  have h2 : ∫ y, ((if Torus.tdist x y ≤ 1 / 4 then (ρ y - ρ x) • G (x - y) else 0) +
166      (if 1 / 4 < Torus.tdist x y then ρ y • G (x - y) else 0)) =
167      (∫ y, if Torus.tdist x y ≤ 1 / 4 then (ρ y - ρ x) • G (x - y) else 0) +
168      ∫ y, if 1 / 4 < Torus.tdist x y then ρ y • G (x - y) else 0 := integral_add hint hf2
169  rw [h1, h2, integral_smul, hc0]
170  simp
