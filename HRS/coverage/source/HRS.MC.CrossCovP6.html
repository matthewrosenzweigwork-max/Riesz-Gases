29theorem p_cc6_funRealCoeff_add {ψ ψ' : Torus d → ℝ} (hψ : Torus.IsSmooth ψ)
30    (hψ' : Torus.IsSmooth ψ') (k : NonzeroLattice d) :
31    Torus.funRealCoeff (ψ + ψ') k = Torus.funRealCoeff ψ k + Torus.funRealCoeff ψ' k := by
32  have h1 : Integrable (fun x => Torus.realMode k x * ψ x) :=
33    ((p_cc3_continuous (p_cc1_smooth_realMode k)).mul (p_cc3_continuous hψ)).integrable_of_hasCompactSupport
34      (HasCompactSupport.of_compactSpace _)
35  have h2 : Integrable (fun x => Torus.realMode k x * ψ' x) :=
36    ((p_cc3_continuous (p_cc1_smooth_realMode k)).mul (p_cc3_continuous hψ')).integrable_of_hasCompactSupport
37      (HasCompactSupport.of_compactSpace _)
38  unfold Torus.funRealCoeff
39  rw [← integral_add h1 h2]
40  congr 1
41  funext x
42  simp only [Pi.add_apply]
43  ring

130theorem p_cc6_memLp_eval (S : PreparedData d) (β₀ m T' : ℝ) (ν : Measure (NegSobolev d m))
131    (hν : IsCenteredGaussianWithCov m ν (preparedCov d S.s β₀ S.μ0))
132    {Ω' : Type*} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
133    (Z₀ : Ω' → NegSobolev d m) (Z : ℝ → Ω' → NegSobolev d m) (hZ : IsPreparedLinear S m T' ν P' Z₀ Z)
134    {r : ℝ} (hr : r ∈ Set.Icc (0 : ℝ) T') {ψ : Torus d → ℝ} (hψ : Torus.IsSmooth ψ) :
135    MemLp (fun ω => NegSobolev.eval (Z r ω) ψ) 2 P' := by
136  haveI : IsGaussian ν := hν.1.1
137  have hU : Torus.IsSmooth (propagator S.dyn S.ρ 0 r ψ) :=
138    ccd_propagator_smooth (u := 0) (t := r) S le_rfl hr.1 hψ
139  have hC : MemLp (fun ω => NegSobolev.eval (Z₀ ω) (propagator S.dyn S.ρ 0 r ψ)) 2 P' :=
140    p_cc6_memLp_init P' Z₀ ν hZ.meas hZ.law_init hU
141  have hf : Measurable (fun ω => NegSobolev.eval (Z r ω) ψ -
142      NegSobolev.eval (Z₀ ω) (propagator S.dyn S.ρ 0 r ψ)) :=
143    ((p_cc6_measurable_eval hψ).comp (hZ.sol.meas r)).sub
144      ((p_cc6_measurable_eval hU).comp hZ.meas)
145  have hGm : Measurable (fun ω => (WithLp.toLp 2 (fun _ : Fin 1 => NegSobolev.eval (Z r ω) ψ -
146      NegSobolev.eval (Z₀ ω) (propagator S.dyn S.ρ 0 r ψ)) : EuclideanSpace ℝ (Fin 1))) :=
147    (PiLp.continuous_toLp 2 (fun _ : Fin 1 => ℝ)).measurable.comp (Measurable.of_eval fun _ => hf)
148  obtain ⟨-, hlaw⟩ := hZ.sol.gaussian_part 1 (fun _ => r) (fun _ => ψ) (fun _ => hr) (fun _ => hψ)
149  have hD : MemLp (fun ω => NegSobolev.eval (Z r ω) ψ -
150      NegSobolev.eval (Z₀ ω) (propagator S.dyn S.ρ 0 r ψ)) 2 P' :=
151    p_cc6_memLp_coord P' _ hGm hlaw
152  convert hD.add hC using 1
153  funext ω
154  simp

166theorem p_cc6_abs_add_mul (A B k a b : ℝ) (hk : 0 ≤ k) (hA : |A| ≤ a) (hB : |B| ≤ b) :
167    |A + k * B| ≤ a + k * b := by
168  calc |A + k * B| ≤ |A| + |k * B| := abs_add_le _ _
169    _ = |A| + k * |B| := by rw [abs_mul, abs_of_nonneg hk]

174/-- **CCD6a** (`ccd_cov_linear`, glue).  For `r, t ∈ [0,T']` and smooth `φ`, the functional
175`ψ ↦ E[Z^r(ψ) Z^t(φ)]` is additive and homogeneous on smooth functions and vanishes on constants (`eval` drops the zero
176mode).  Additivity and homogeneity need integrability of the products: `Z^r(ψ), Z^t(φ) ∈ L²(P')` (from the decomposition of
177CCD2: a centered Gaussian plus `Z₀(U^{0,r}ψ)` with `Z₀ ~ ν` centered Gaussian) and linearity of `eval` in smooth tests
178(summability of the series). -/
179theorem ccd_cov_linear (S : PreparedData d) (β₀ m T' : ℝ) (ν : Measure (NegSobolev d m))
180    (hν : IsCenteredGaussianWithCov m ν (preparedCov d S.s β₀ S.μ0))
181    {Ω' : Type*} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
182    (Z₀ : Ω' → NegSobolev d m) (Z : ℝ → Ω' → NegSobolev d m) (hZ : IsPreparedLinear S m T' ν P' Z₀ Z)
183    {r t : ℝ} (hr : r ∈ Set.Icc (0 : ℝ) T') (ht : t ∈ Set.Icc (0 : ℝ) T')
184    {φ : Torus d → ℝ} (hφ : Torus.IsSmooth φ) :
185    (∀ ψ ψ' : Torus d → ℝ, Torus.IsSmooth ψ → Torus.IsSmooth ψ' →
186      ∫ ω, NegSobolev.eval (Z r ω) (ψ + ψ') * NegSobolev.eval (Z t ω) φ ∂P' =
187        ∫ ω, NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ ∂P' +
188        ∫ ω, NegSobolev.eval (Z r ω) ψ' * NegSobolev.eval (Z t ω) φ ∂P') ∧
189    (∀ (c : ℝ) (ψ : Torus d → ℝ), Torus.IsSmooth ψ →
190      ∫ ω, NegSobolev.eval (Z r ω) (c • ψ) * NegSobolev.eval (Z t ω) φ ∂P' =
191        c * ∫ ω, NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ ∂P') ∧
192    (∀ c : ℝ, ∫ ω, NegSobolev.eval (Z r ω) (fun _ => c) * NegSobolev.eval (Z t ω) φ ∂P' = 0) := by
193  refine ⟨fun ψ ψ' hψ hψ' => ?_, fun c ψ hψ => ?_, fun c => ?_⟩
194  · have e : ∀ ω, NegSobolev.eval (Z r ω) (ψ + ψ') * NegSobolev.eval (Z t ω) φ =
195        NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ +
196          NegSobolev.eval (Z r ω) ψ' * NegSobolev.eval (Z t ω) φ := by
197      intro ω
198      rw [p_cc6_eval_add _ hψ hψ']
199      ring
200    simp only [e]
201    exact integral_add (p_cc6_integrable S β₀ m T' ν hν P' Z₀ Z hZ hr ht hψ hφ)
202      (p_cc6_integrable S β₀ m T' ν hν P' Z₀ Z hZ hr ht hψ' hφ)
203  · have e : ∀ ω, NegSobolev.eval (Z r ω) (c • ψ) * NegSobolev.eval (Z t ω) φ =
204        c * (NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ) := by
205      intro ω
206      rw [p_cc6_eval_smul]
207      ring
208    simp only [e]
209    exact integral_const_mul c _
210  · have e : ∀ ω, NegSobolev.eval (Z r ω) (fun _ => c) * NegSobolev.eval (Z t ω) φ = 0 := by
211      intro ω
212      rw [p_cc6_eval_const, zero_mul]
213    simp only [e, integral_zero]

215/-- **CCD6b** (`ccd_cov_bound`, the `H^{-N}` bound).  For `r, t ∈ [0,T']`, `0 ≤ β₀`, smooth `φ` and every `N`, there is `C`
216with `|E[Z^r(ψ) Z^t(φ)]| ≤ C ‖ψ‖_{H^{-N}}` for all smooth `ψ`.  Route: `ccd_cross_cov` splits the covariance into the
217initial term, bounded through `ccd_preparedCov_density` (`g = U^{0,t}φ`, smooth by `ccd_propagator_smooth`) and
218`ccd_dual_pairing_bound` (`u = 0`, mean-zero `w`), plus the noise term `ccd_noise_bound`. -/
219theorem ccd_cov_bound (S : PreparedData d) (β₀ m T' : ℝ) (hβ₀ : 0 ≤ β₀) (ν : Measure (NegSobolev d m))
220    (hν : IsCenteredGaussianWithCov m ν (preparedCov d S.s β₀ S.μ0))
221    {Ω' : Type*} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
222    (Z₀ : Ω' → NegSobolev d m) (Z : ℝ → Ω' → NegSobolev d m) (hZ : IsPreparedLinear S m T' ν P' Z₀ Z)
223    {r t : ℝ} (hr : r ∈ Set.Icc (0 : ℝ) T') (ht : t ∈ Set.Icc (0 : ℝ) T')
224    {φ : Torus d → ℝ} (hφ : Torus.IsSmooth φ) (N : ℕ) :
225    ∃ C : ℝ, ∀ ψ : Torus d → ℝ, Torus.IsSmooth ψ →
226      |∫ ω, NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ ∂P'| ≤
227        C * HRS.sobolevNorm (-(N : ℝ)) ψ := by
228  have hgφ : Torus.IsSmooth (propagator S.dyn S.ρ 0 t φ) :=
229    ccd_propagator_smooth (u := 0) (t := t) S le_rfl ht.1 hφ
230  obtain ⟨w, hwS, hwmean, hw⟩ := ccd_preparedCov_density S β₀ hβ₀ hgφ
231  obtain ⟨C0, hC0, hdual⟩ := ccd_dual_pairing_bound S N
232  obtain ⟨C1, hC1, hnoise⟩ := ccd_noise_bound S T' N ht hφ
233  refine ⟨C0 * HRS.sobolevNorm (N : ℝ) w + 2 * S.β⁻¹ * C1, fun ψ hψ => ?_⟩
234  have hk : 0 ≤ 2 * S.β⁻¹ := mul_nonneg (by norm_num) (inv_nonneg.2 S.hβ.le)
235  have hU : Torus.IsSmooth (propagator S.dyn S.ρ 0 r ψ) :=
236    ccd_propagator_smooth (u := 0) (t := r) S le_rfl hr.1 hψ
237  have h1 : |preparedCov d S.s β₀ S.μ0 (propagator S.dyn S.ρ 0 r ψ) (propagator S.dyn S.ρ 0 t φ)|
238      ≤ C0 * HRS.sobolevNorm (N : ℝ) w * HRS.sobolevNorm (-(N : ℝ)) ψ := by
239    rw [hw _ hU]
240    exact hdual 0 r le_rfl hr.1 w ψ hwS hwmean hψ
241  have h2 := hnoise r hr ψ hψ
242  rw [ccd_cross_cov S β₀ m T' ν hν P' Z₀ Z hZ hr ht hψ hφ]
243  refine le_trans (p_cc6_abs_add_mul _ _ _ _ _ hk h1 h2) (le_of_eq ?_)
244  ring

246/-- **CCD6c** (`ccd_single`, note §3 CCD6, one observation).  For one observation `(t, φ)` with `t ∈ [0,T']` and smooth
247`φ`, there are smooth densities `χ^r` (`r ∈ [0,T']`) with `∫ ψ χ^r = Cov(Z^r(ψ), Z^t(φ))` for all smooth `ψ`.  Route: for
248each `r ∈ [0,T']` apply `ccd_smooth_density_of_bound` to `L ψ = E[Z^r(ψ)Z^t(φ)]` (hypotheses from `ccd_cov_linear` and
249`ccd_cov_bound`), choose the densities with `Classical.choose` and put `χ r = 0` for `r ∉ [0,T']`. -/
250theorem ccd_single (S : PreparedData d) (β₀ m T' : ℝ) (hβ₀ : 0 ≤ β₀) {t : ℝ} (ht : t ∈ Set.Icc (0 : ℝ) T')
251    {φ : Torus d → ℝ} (hφ : Torus.IsSmooth φ)
252    (ν : Measure (NegSobolev d m)) (hν : IsCenteredGaussianWithCov m ν (preparedCov d S.s β₀ S.μ0))
253    {Ω' : Type*} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
254    (Z₀ : Ω' → NegSobolev d m) (Z : ℝ → Ω' → NegSobolev d m) (hZ : IsPreparedLinear S m T' ν P' Z₀ Z) :
255    ∃ χ : ℝ → Torus d → ℝ, (∀ r, Torus.IsSmooth (χ r)) ∧
256      ∀ r ∈ Set.Icc (0 : ℝ) T', ∀ ψ : Torus d → ℝ, Torus.IsSmooth ψ →
257        ∫ x, ψ x * χ r x = ∫ ω, NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ ∂P' := by
258  have key : ∀ r : ℝ, ∃ χr : Torus d → ℝ, Torus.IsSmooth χr ∧
259      (r ∈ Set.Icc (0 : ℝ) T' → ∀ ψ : Torus d → ℝ, Torus.IsSmooth ψ →
260        ∫ x, ψ x * χr x = ∫ ω, NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ ∂P') := by
261    intro r
262    by_cases hr : r ∈ Set.Icc (0 : ℝ) T'
263    · have hlin := ccd_cov_linear S β₀ m T' ν hν P' Z₀ Z hZ hr ht hφ
264      obtain ⟨χr, hχr, hL⟩ := ccd_smooth_density_of_bound
265        (fun ψ => ∫ ω, NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z t ω) φ ∂P')
266        hlin.1 (fun c ψ hψ => hlin.2.1 c ψ hψ) (fun c => hlin.2.2 c)
267        (ccd_cov_bound S β₀ m T' hβ₀ ν hν P' Z₀ Z hZ hr ht hφ)
268      exact ⟨χr, hχr, fun _ ψ hψ => hL ψ hψ⟩
269    · exact ⟨fun _ => 0, p_cc1_smooth_const 0, fun h => absurd h hr⟩
270  choose χ hχs hχL using key
271  exact ⟨χ, hχs, fun r hr ψ hψ => hχL r hr ψ hψ⟩

273/-- **CCD6d** (`p_assembly`, the frozen target `HRS.Main.cross_cov_densities_exist`, copied verbatim; TeX L1296-1299).
274Route: `β₀ ≥ 0` from `hE.β₀_nonneg`, `t i ∈ [a, a+T] ⊆ [0, a+T]` since `a ≥ 0`, `T' = a + T`, then `ccd_single` for
275each `i : Fin 3` and `choose`. -/
276theorem p_assembly (S : PreparedData d) (β₀ m T a : ℝ) (hE : EdgeworthHyp S β₀ m)
277    (hT : 0 < T) (ha : 0 ≤ a) : CrossCovDensitiesExist S β₀ m T a := by
278  unfold CrossCovDensitiesExist
279  intro t φ ht hφ ν hν Ω' _ P' _ Z₀ Z hZ
280  have h : ∀ i : Fin 3, ∃ χi : ℝ → Torus d → ℝ, (∀ r, Torus.IsSmooth (χi r)) ∧
281      ∀ r ∈ Set.Icc (0 : ℝ) (a + T), ∀ ψ : Torus d → ℝ, Torus.IsSmooth ψ →
282        ∫ x, ψ x * χi r x = ∫ ω, NegSobolev.eval (Z r ω) ψ * NegSobolev.eval (Z (t i) ω) (φ i) ∂P' := by
283    intro i
284    exact ccd_single (t := t i) S β₀ m (a + T) hE.β₀_nonneg ⟨by linarith [(ht i).1], (ht i).2⟩
285      (hφ i) ν hν P' Z₀ Z hZ
286  choose χ hχs hχL using h
287  exact ⟨χ, fun i r => hχs i r, fun i r hr ψ hψ => hχL i r hr ψ hψ⟩
