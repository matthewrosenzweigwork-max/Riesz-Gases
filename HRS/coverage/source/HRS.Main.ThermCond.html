26theorem therm_ratio_bounds {B Cb δ : ℝ} {N : ℕ} (hN : 1 ≤ N) (hδ : 0 < δ) (hB1 : 0 < B)
27    (hB : B ≤ Cb * (N * δ)) :
28    B / Real.sqrt N ≤ Cb * (Real.sqrt N * δ) ∧ B / N ≤ Cb * δ := by
29  have hN0 : (0 : ℝ) < N := by exact_mod_cast hN
30  have hs0 : 0 < Real.sqrt N := Real.sqrt_pos.2 hN0
31  have hsq : (N : ℝ) = Real.sqrt N * Real.sqrt N := (Real.mul_self_sqrt hN0.le).symm
32  constructor
33  · rw [div_le_iff₀ hs0]
34    calc B ≤ Cb * (N * δ) := hB
35      _ = Cb * (Real.sqrt N * δ) * Real.sqrt N := by linear_combination (Cb * δ) * hsq
36  · rw [div_le_iff₀ hN0]
37    calc B ≤ Cb * (N * δ) := hB
38      _ = Cb * δ * N := by ring
