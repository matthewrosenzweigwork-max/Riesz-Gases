28/-- `K_{N,γ}(μ) = ∫ e^{-γ N F_N(X_N, μ)} dμ^{⊗N}` in `ℝ≥0∞` (`H:eq:modulated-gibbs`, line 450),
29valid for every real `γ`. -/
30def modNormalizerE (g : Torus d → ℝ) (N : ℕ) (γ : ℝ) (μ : Measure (Torus d)) : ℝ≥0∞ :=
31  ∫⁻ X : Config d N, ENNReal.ofReal (Real.exp (-γ * N * modulatedEnergyKernel g X μ))
32    ∂(Measure.pi fun _ => μ)

48/-- Proposition 4.25 `M:O:prop:log-prep` (source lines 5588-5593), `O:eq:small-negative`: for
49the periodic logarithm and `μ` with a bounded density there is `a_* > 0` such that for every
50finite `B₀`, `sup_{N ≥ 2} sup_{-a_* ≤ γ ≤ B₀} K_{N,γ}(μ) < ∞`. (External: DGR Theorem 2.6 and
51Remark 2.1 for the attractive side, Theorem 2.2 for the repulsive side.) -/
52theorem log_prep_small_negative (hd : 2 ≤ d) (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ)
53    (hΛ : ∀ x, ρ x ≤ Λ) :
54    ∃ a : ℝ, 0 < a ∧ ∀ B₀ : ℝ, ∃ C : ℝ, ∀ N : ℕ, 2 ≤ N → ∀ γ ∈ Set.Icc (-a) B₀,
55      modNormalizerE (periodicRiesz d 0) N γ (withDensityReal ρ) ≤ ENNReal.ofReal C := by
56  obtain ⟨a, ha, C₁, hC₁⟩ := log_prep_attractive hd ρ hρ Λ hΛ
57  refine ⟨a, ha, fun B₀ => ?_⟩
58  obtain ⟨C₂, hC₂⟩ := log_prep_repulsive hd ρ hρ Λ hΛ B₀
59  refine ⟨max C₁ C₂, fun N hN γ hγ => ?_⟩
60  rcases le_total γ 0 with h | h
61  · exact (hC₁ N hN γ ⟨hγ.1, h⟩).trans (ENNReal.ofReal_le_ofReal (le_max_left _ _))
62  · exact (hC₂ N hN γ ⟨h, hγ.2⟩).trans (ENNReal.ofReal_le_ofReal (le_max_right _ _))

64/-- Proposition 4.25, `O:eq:prep-uniform` (source lines 5594-5600): for each finite `B, B₀`
65there is `p > 1` with
66`sup_{N≥2} sup_{0≤β≤B} sup_{0≤β₀≤B₀} ∫ (dQ_{N,β₀}(μ)/dQ_{N,β}(μ))^p dQ_{N,β}(μ) < ∞`. -/
67theorem log_prep_uniform (hd : 2 ≤ d) (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ)
68    (hΛ : ∀ x, ρ x ≤ Λ) (B B₀ : ℝ) :
69    ∃ p : ℝ, 1 < p ∧ ∃ C : ℝ, ∀ N : ℕ, 2 ≤ N → ∀ β ∈ Set.Icc (0 : ℝ) B,
70      ∀ β₀ ∈ Set.Icc (0 : ℝ) B₀,
71        modulatedGibbs (periodicRiesz d 0) N β₀ (withDensityReal ρ) ≪
72            modulatedGibbs (periodicRiesz d 0) N β (withDensityReal ρ) ∧
73          ∫⁻ X, ((modulatedGibbs (periodicRiesz d 0) N β₀ (withDensityReal ρ)).rnDeriv
74              (modulatedGibbs (periodicRiesz d 0) N β (withDensityReal ρ)) X) ^ p
75              ∂(modulatedGibbs (periodicRiesz d 0) N β (withDensityReal ρ)) ≤
76            ENNReal.ofReal C := by
77  have hs0 : (0 : ℝ) ≤ 0 := le_rfl
78  have hsd : (0 : ℝ) < d := by exact_mod_cast (by omega : 0 < d)
79  have hg : Measurable (periodicRiesz d 0) := PoleCavityAux.measurable_periodicRiesz hd hs0 hsd
80  have hgi : Integrable (periodicRiesz d 0) := KernelAux.integrable_periodicRiesz hd hs0 hsd
81  haveI := isProb_withDensityReal hρ
82  have hI := PreparationAux.integrable_pair_withDensity hg hgi ρ Λ hΛ (ν := withDensityReal ρ) rfl
83  exact PreparationAux.uniform_of_small_negative hg hI (log_prep_small_negative hd ρ hρ Λ hΛ) B B₀

85/-- Proposition 4.26 `M:O:prop:entropy-endpoint` (source lines 5634-5639),
86`O:eq:entropy-endpoint`: fix `B ≥ 0` and a reference probability measure `μ` with
87`∬ |g| dμ^{⊗2} < ∞` (so `|I| < ∞`), and suppose `log K_{N,γ}(μ) ≤ C N δ_N` for
88`γ ∈ [0, B + 1]` and `N ≥ 2`. Then `H(Q_{N,β₀}(μ) | Q_{N,β}(μ)) ≤ C' N δ_N` uniformly for
89`β, β₀ ∈ [0, B]`, including `β₀ = 0`. (`N δ_N ≥ 1` holds for the `δ_N` of `deltaN`.) -/
90theorem entropy_endpoint {s : ℝ} (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d)
91    (μ : Measure (Torus d)) [IsProbabilityMeasure μ] (B : ℝ) (hB : 0 ≤ B)
92    (hI : Integrable (fun p : Torus d × Torus d => periodicRiesz d s (p.1 - p.2)) (μ.prod μ))
93    (Cp : ℝ) (hpress : ∀ N : ℕ, 2 ≤ N → ∀ γ ∈ Set.Icc (0 : ℝ) (B + 1),
94      modNormalizerE (periodicRiesz d s) N γ μ ≤ ENNReal.ofReal (Real.exp (Cp * N * deltaN d s N))) :
95    ∃ C' : ℝ, ∀ N : ℕ, 2 ≤ N → ∀ β ∈ Set.Icc (0 : ℝ) B, ∀ β₀ ∈ Set.Icc (0 : ℝ) B,
96      relEntropy (modulatedGibbs (periodicRiesz d s) N β₀ μ)
97          (modulatedGibbs (periodicRiesz d s) N β μ) ≤
98        ENNReal.ofReal (C' * N * deltaN d s N) := by
99  have hg : Measurable (periodicRiesz d s) := PoleCavityAux.measurable_periodicRiesz hd hs0 hsd
100  set A := |∫ p : Torus d × Torus d, periodicRiesz d s (p.1 - p.2) ∂(μ.prod μ)| with hA
101  refine ⟨(B * (A / 2 + B * (A / 2) + |Cp|) + |Cp| + B * (A / 2)), fun N hN β hβ β₀ hβ₀ => ?_⟩
102  have hL := PreparationAux.one_le_N_deltaN (d := d) hs0 N (by omega)
103  have hpress' : ∀ γ ∈ Set.Icc (0 : ℝ) (B + 1),
104      PreparationAux.KE (periodicRiesz d s) N γ μ ≤
105        ENNReal.ofReal (Real.exp (Cp * ((N : ℝ) * deltaN d s N))) := by
106    intro γ hγ
107    have := hpress N hN γ hγ
108    rwa [← mul_assoc]
109  have := PreparationAux.entropy_bound_aux hg hI (N := N) (by omega) B Cp ((N : ℝ) * deltaN d s N)
110    hB hL hpress' β hβ β₀ hβ₀
111  rwa [mul_assoc]

113/-- The exact entropy identity in the proof of Proposition 4.26 (source lines 5644-5647):
114`H(Q_{N,β₀} | Q_{N,β}) = (β - β₀) E_{Q_{N,β₀}}[N F_N] + f(β) - f(β₀)`, `f = log K_{N,·}`,
115for `0 ≤ β, β₀`, given `∬|g| dμ^{⊗2} < ∞` and finite partition functions at `β` and at
116`β₀ + 1` (the latter gives the exponential moment of the negative part of `N F_N` used in the
117source to differentiate at `β₀`). -/
118theorem relEntropy_modulatedGibbs_eq {s : ℝ} (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d)
119    (μ : Measure (Torus d)) [IsProbabilityMeasure μ]
120    (hI : Integrable (fun p : Torus d × Torus d => periodicRiesz d s (p.1 - p.2)) (μ.prod μ))
121    (N : ℕ) (hN : 2 ≤ N) (β β₀ : ℝ) (hβ : 0 ≤ β) (hβ₀ : 0 ≤ β₀)
122    (hfin : modNormalizerE (periodicRiesz d s) N β μ < ∞)
123    (hfin₀ : modNormalizerE (periodicRiesz d s) N (β₀ + 1) μ < ∞) :
124    Integrable (fun X : Config d N => (N : ℝ) * modulatedEnergyKernel (periodicRiesz d s) X μ)
125        (modulatedGibbs (periodicRiesz d s) N β₀ μ) ∧
126      relEntropy (modulatedGibbs (periodicRiesz d s) N β₀ μ)
127          (modulatedGibbs (periodicRiesz d s) N β μ) =
128        ENNReal.ofReal ((β - β₀) * ∫ X, (N : ℝ) * modulatedEnergyKernel (periodicRiesz d s) X μ
129            ∂(modulatedGibbs (periodicRiesz d s) N β₀ μ) +
130          Real.log (modNormalizerE (periodicRiesz d s) N β μ).toReal -
131          Real.log (modNormalizerE (periodicRiesz d s) N β₀ μ).toReal) := by
132  have hg : Measurable (periodicRiesz d s) := PoleCavityAux.measurable_periodicRiesz hd hs0 hsd
133  exact PreparationAux.relEntropy_eq_aux hg hI β β₀ hβ₀ hfin.ne hfin₀.ne
