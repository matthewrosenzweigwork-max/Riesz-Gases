7# ASM-15, part 2: clause 2 (ii-b) of `lp_positive_order_product`
8
9For `1 < θ < 2`, `β = θ - 1 ∈ (0,1)`, `Z` real Schwartz let `G = 𝒥^{β} Z` (the complex Schwartz function with
10`𝓕 G = (1+4π²|ξ|²)^{β/2} 𝓕 Z`, `HRS.MS.MovOsc.mova3b2_rot`); then `besselNormE d β p Z = ‖G‖_p`
11(`asm15_besselNormE_eq`) and

14  `‖Z‖_p ≤ C ‖G‖_p` (`asm15_Z_bound`);

19(`HRS.CZ.ElemA.elema_positive_order_bound`) and `pvOdd (aZ) = [pvOdd,a]Z + a pvOdd Z`.  (`eLpNorm_add_le` needs no measurability here.)

36theorem asm15_mik_Lp (hd : 1 ≤ d) {p : ℝ} (hp : 1 < p) :
37    ∃ Cm : ℝ, 0 ≤ Cm ∧ ∀ (A : ℝ) (m : 𝔼 → ℂ), 0 ≤ A → mik1_IsMikhlin d A m → ∀ u : 𝓢(𝔼, ℂ),
38      eLpNorm (mik2_T m u) (ENNReal.ofReal p) volume ≤
39        ENNReal.ofReal (Cm * A) * eLpNorm (⇑u) (ENNReal.ofReal p) volume := by
40  obtain ⟨C, hCtop, hC⟩ := mikhlin_Lp_schwartz hd hp
41  have hp0 : 0 < p := by linarith
42  refine ⟨C.toReal ^ (1 / p), Real.rpow_nonneg ENNReal.toReal_nonneg _, fun A m hA hm u => ?_⟩
43  have hpt : (ENNReal.ofReal p).toReal = p := ENNReal.toReal_ofReal hp0.le
44  have hp1 : ENNReal.ofReal p ≠ 0 := by simpa using hp0
45  have hp2 : ENNReal.ofReal p ≠ ⊤ := ENNReal.ofReal_ne_top
46  rw [eLpNorm_eq_lintegral_rpow_enorm_toReal hp1 hp2 (mik2_T_continuous hd hm u).aestronglyMeasurable,
47    eLpNorm_eq_lintegral_rpow_enorm_toReal hp1 hp2 u.continuous.aestronglyMeasurable, hpt]
48  have hpinv : 0 ≤ 1 / p := by positivity
49  calc (∫⁻ x, ‖mik2_T m u x‖ₑ ^ p) ^ (1 / p)
50      ≤ (C * ENNReal.ofReal A ^ p * ∫⁻ x, ‖u x‖ₑ ^ p) ^ (1 / p) := by
51        gcongr
52        exact hC A m hA hm u
53    _ = ENNReal.ofReal (C.toReal ^ (1 / p) * A) * (∫⁻ x, ‖u x‖ₑ ^ p) ^ (1 / p) := by
54        rw [ENNReal.mul_rpow_of_nonneg _ _ hpinv, ENNReal.mul_rpow_of_nonneg _ _ hpinv,
55          ← ENNReal.rpow_mul, mul_one_div_cancel hp0.ne', ENNReal.rpow_one,
56          ENNReal.ofReal_mul (Real.rpow_nonneg ENNReal.toReal_nonneg _),
57          ← ENNReal.ofReal_rpow_of_nonneg ENNReal.toReal_nonneg hpinv, ENNReal.ofReal_toReal hCtop]

214    (i : Fin d) :

216      (∀ W : SchwartzMap 𝔼 ℝ,

219            (ENNReal.ofReal p) volume ≤
220          ENNReal.ofReal (C * HRS.cbNorm 1 a) * eLpNorm W (ENNReal.ofReal p) volume) ∧
221      ∀ Z : SchwartzMap 𝔼 ℝ,
222        eLpNorm (HRS.Whole.AppendixA.pvOdd d θ (fun u : 𝔼 => u i) fun y => a y * Z y)
223            (ENNReal.ofReal p) volume ≤
224          ENNReal.ofReal (C * HRS.cbNorm 1 a) * HRS.Whole.AppendixA.besselNormE d (θ - 1) p Z := by
225  have hp1 : 1 < p := by linarith
226  have hK := isPvKernel_coord (d := d) i
227  obtain ⟨C0, hC0, hb⟩ := elema_positive_order_bound hd hθ1 hθ2 hK
228  obtain ⟨CZ, hCZ, hZ⟩ := asm15_Z_bound hd (β := θ - 1) (p := p) (by linarith) hp1
229  obtain ⟨CT, hCT, hT⟩ := asm15_T_bound hd hθ1 hθ2 hp1 i
230  have hCnn : 0 ≤ C0 * (1 + CZ) + CT := add_nonneg (mul_nonneg hC0 (by linarith)) hCT

232  obtain ⟨Ba, La, hBa, hLa, hcb, hab, hal⟩ := elema_cb1_data ha
233  have hac : Continuous a := ha.1.continuous
234  rw [hcb]

237      eLpNorm (fun x => HRS.Whole.AppendixA.pvOdd d θ (fun u : 𝔼 => u i) (fun y => a y * W y) x -
238          a x * HRS.Whole.AppendixA.pvOdd d θ (fun u : 𝔼 => u i) W x) (ENNReal.ofReal p) volume ≤
239        ENNReal.ofReal (C0 * (Ba + La)) * eLpNorm W (ENNReal.ofReal p) volume :=
240    fun W => hb p hp1.le a Ba La hBa hLa hac hab hal W
241  refine ⟨fun W => (hcomm W).trans (mul_le_mul_left ?_ _), fun Z => ?_⟩
242  · refine ENNReal.ofReal_le_ofReal ?_
243    nlinarith [mul_nonneg hC0 (mul_nonneg hCZ hBL), mul_nonneg hCT hBL]
244  set T : 𝔼 → ℝ := HRS.Whole.AppendixA.pvOdd d θ (fun u : 𝔼 => u i) (⇑Z) with hTdef
245  have hsplit : (HRS.Whole.AppendixA.pvOdd d θ (fun u : 𝔼 => u i) fun y => a y * Z y) =
246      (fun x => HRS.Whole.AppendixA.pvOdd d θ (fun u : 𝔼 => u i) (fun y => a y * Z y) x - a x * T x) +
247        (fun x => a x * T x) := by
248    funext x; simp
249  have hp1' : 1 ≤ ENNReal.ofReal p := ENNReal.one_le_ofReal.2 hp1.le
250  have h1 := eLpNorm_add_le (f := fun x => HRS.Whole.AppendixA.pvOdd d θ (fun u : 𝔼 => u i) (fun y => a y * Z y) x - a x * T x) (g := fun x => a x * T x) (μ := volume) hp1'
251  have hTc : Continuous T := asm15_pvOdd_continuous hd hθ1 hθ2 i Z
252  have hpt : ∀ᵐ x ∂(volume : Measure 𝔼), ‖a x * T x‖₊ ≤ Real.toNNReal Ba * ‖T x‖₊ := by
253    refine ae_of_all _ fun x => ?_
254    rw [← NNReal.coe_le_coe]
255    simp only [coe_nnnorm, NNReal.coe_mul, Real.coe_toNNReal _ hBa, Real.norm_eq_abs, abs_mul]
256    exact mul_le_mul_of_nonneg_right (hab x) (abs_nonneg _)
257  have hg : eLpNorm (fun x => a x * T x) (ENNReal.ofReal p) volume ≤
258      ENNReal.ofReal Ba * eLpNorm T (ENNReal.ofReal p) volume := by
259    have h := eLpNorm_le_nnreal_smul_eLpNorm_of_ae_le_mul (g := T) (c := Real.toNNReal Ba)
260      (hac.mul hTc).aestronglyMeasurable hpt (ENNReal.ofReal p)
261    rw [ENNReal.smul_def, smul_eq_mul] at h
262    exact h
263  have hZb := hZ Z
264  have hTb := hT Z
265  rw [hsplit]
266  refine h1.trans ?_
267  calc _ ≤ ENNReal.ofReal (C0 * (Ba + La)) * eLpNorm (⇑Z) (ENNReal.ofReal p) volume +
268        ENNReal.ofReal Ba * eLpNorm T (ENNReal.ofReal p) volume := add_le_add (hcomm Z) hg
269    _ ≤ ENNReal.ofReal (C0 * (Ba + La)) * (ENNReal.ofReal CZ *
270          HRS.Whole.AppendixA.besselNormE d (θ - 1) p Z) +
271        ENNReal.ofReal Ba * (ENNReal.ofReal CT * HRS.Whole.AppendixA.besselNormE d (θ - 1) p Z) :=
272        add_le_add (mul_le_mul' le_rfl hZb) (mul_le_mul' le_rfl hTb)
273    _ = ENNReal.ofReal (C0 * (Ba + La) * CZ + Ba * CT) *
274          HRS.Whole.AppendixA.besselNormE d (θ - 1) p Z := by
