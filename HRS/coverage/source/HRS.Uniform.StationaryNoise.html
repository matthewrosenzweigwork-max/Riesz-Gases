31theorem q48d_evolution_input (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (β : ℝ) (hβ : 0 < β)
32    (V : Torus d → ℝ) (_ : Torus.IsSmooth V) (ρβ : Torus d → ℝ)
33    (_ : IsThermalDensity (periodicRiesz d s) V β ρβ) (_ : Torus.IsSmooth ρβ) (_ : ∀ x, 0 < ρβ x) :
34    Q48d.EvoInput.{u} d s hs0 hsd β hβ V ρβ := by
35  rename_i hVs hth hρs hpos
36  have hprob : IsProbDensity ρβ := hth.1
37  have hd0 : (0 : ℝ) ≤ d := Nat.cast_nonneg d
38  refine ⟨(PDE.mStar d : ℝ) + d, fun m hm => ?_⟩
39  have hplain : ∀ Vev : ℝ → ℝ → NegSobolev d m →L[ℝ] NegSobolev d m,
40      IsDualEvolution (PDE.mfData d s hs0 hsd V β hβ) (fun _ => ρβ) m Vev →
41      ∀ T : ℝ, 0 ≤ T → IsPlainEvolution Vev T := fun Vev hdual T hT =>
42    ⟨fun t ht => Q48e.vev_self hd s hs0 hsd β hβ V hVs ρβ hprob hρs hpos hdual t ht.1,
43      fun a r t ha har hrt _ => Q48e.vev_comp hd s hs0 hsd β hβ V hVs ρβ hprob hρs hpos hdual a r t ha har hrt,
44      fun x => Q48e.vev_cont hd s hs0 hsd β hβ V hVs ρβ hprob hρs hpos m hm hdual T hT x⟩
45  refine ⟨fun T hT => ?_, hplain⟩
46  obtain ⟨c, hc⟩ := Q48e.vev_quasi hd s hs0 hsd β hβ V hVs ρβ hprob hρs hpos m (by linarith)
47  refine ⟨Kot.CD 2 (2 * c) T, Kot.CD_nonneg _ _ _, fun Vev hdual => ?_⟩
48  refine Q48e.winmax_of_quasi (hplain Vev hdual) (K := 2 * c) (fun u t hu hut => ?_) hT.le

52theorem stationary_noise_proof (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (β : ℝ) (hβ : 0 < β)
53    (V : Torus d → ℝ) (_ : Torus.IsSmooth V) (ρβ : Torus d → ℝ)
54    (_ : IsThermalDensity (periodicRiesz d s) V β ρβ) (_ : Torus.IsSmooth ρβ) (_ : ∀ x, 0 < ρβ x) :
55    ∃ m₀ : ℝ, ∀ m : ℝ, m₀ ≤ m → ∀ Cv : ℝ, ∀ T : ℝ, 0 < T → ∃ C : ℝ, 0 ≤ C ∧
56      ∀ {Ω : Type u} {mΩ : MeasurableSpace Ω} (P : Measure Ω) [IsProbabilityMeasure P]
57        (ℱ : Filtration ℝ mΩ) (𝒢 : InitialSigma ℱ) (N : ℕ) (_ : 1 ≤ N) (δ : ℝ) (_ : 0 ≤ δ)
58        (X : ℝ → Ω → Config d N) (Q : Measure (Config d N)) [IsProbabilityMeasure Q] (_ : ∀ t : ℝ, 0 ≤ t → P.map (X t) = Q)
59        (_ : ∀ f : Torus d → ℝ, Measurable f → MemLp f 2 (withDensityReal ρβ) →
60          evariance (fun x : Config d N => pair (empirical x) f) Q ≤
61            ENNReal.ofReal (Cv * (N : ℝ)⁻¹ * ∫ x, f x ^ 2 ∂(withDensityReal ρβ)))
62        (g : Torus d → ℝ) (_ : Measurable g)
63        (_ : (N : ℝ≥0∞)⁻¹ • ∑ i, Q.map (fun x : Config d N => x i) =
64          (withDensityReal ρβ).withDensity fun x => ENNReal.ofReal (g x))
65        (_ : ∫⁻ x, ‖g x - 1‖ₑ ^ 2 ∂(withDensityReal ρβ) ≤ ENNReal.ofReal ((Cv * δ) ^ 2))
66        (Vev : ℝ → ℝ → NegSobolev d m →L[ℝ] NegSobolev d m)
67        (_ : IsDualEvolution (PDE.mfData d s hs0 hsd V β hβ) (fun _ => ρβ) m Vev)
68        (M : ℝ → Ω → NegSobolev d m)
69        (_ : ∀ S : ℝ, 0 ≤ S →
70          IsHilbertMartingaleWithCovariation P ℱ S M
71            (fun r ω => covForm m β⁻¹ (empirical (X r ω)))),
72        ∃ (Ω' : Type u) (mΩ' : MeasurableSpace Ω') (P' : ProbabilityMeasure Ω') (π : Ω' → Ω)
73          (ℱ' : Filtration ℝ mΩ') (Gb : ℝ → Ω' → NegSobolev d m)
74          (Sd : ℝ → Ω' → (NegSobolev d m →L[ℝ] NegSobolev d m →L[ℝ] ℝ))
75          (Z : ℝ → Ω' → NegSobolev d m),
76          IsAdaptedGaussianCoupling P ℱ ⊤ 𝒢 M (fun r ω => covForm m β⁻¹ (empirical (X r ω)))
77            (fun r => covForm m β⁻¹ (withDensityReal ρβ)) (fun _ _ => True) (fun _ _ => 0)
78            P' π ℱ' Gb Sd ∧
79          (∀ t : ℝ, 0 ≤ t → IsStochConv (P' : Measure Ω') Vev (ContinuousLinearMap.id ℝ (NegSobolev d m))
80            (fun τ ω' => M τ (π ω') - Gb τ ω') 0 t (Z t)) ∧
81          (∀ᵐ ω' ∂(P' : Measure Ω'), ContinuousOn (fun t => Z t ω') (Set.Ici 0)) ∧
82          ∀ a : ℝ, 0 ≤ a →
83            (∫⁻ ω', (⨆ t ∈ Set.Icc a (a + T), ‖Z t ω'‖ₑ) ^ 2 ∂(P' : Measure Ω')) ^ (1 / 2 : ℝ) ≤
84              ENNReal.ofReal (C * ((N : ℝ) ^ (-(1 : ℝ) / 2) + δ))
85    := by
86  rename_i hVs hth hρs hpos
87  exact Q48d.stationary_noise_of_evo hd s hs0 hsd β hβ V hVs ρβ hth hρs hpos
88    (q48d_evolution_input.{u} hd s hs0 hsd β hβ V hVs ρβ hth hρs hpos)
