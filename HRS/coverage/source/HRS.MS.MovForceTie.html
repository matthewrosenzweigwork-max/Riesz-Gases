45lemma movb2_GA_mul {K : ℕ} {ψ v : Euc d → ℝ} (hψ : ContDiff ℝ (K : WithTop ℕ∞) ψ)

48    (hvb : ∀ i ≤ K, ∀ y, ‖iteratedFDeriv ℝ i v y‖ ≤ M) :

50  intro n hn y
51  have h1 := norm_iteratedFDeriv_mul_le (𝕜 := ℝ) (N := (K : WithTop ℕ∞)) hψ hv y (n := n)
52    (by exact_mod_cast hn)
53  refine h1.trans ?_
54  have hterm : ∀ i ∈ Finset.range (n + 1), (n.choose i : ℝ) * ‖iteratedFDeriv ℝ i ψ y‖ *
55      ‖iteratedFDeriv ℝ (n - i) v y‖ ≤ (n.choose i : ℝ) * (Cψ * M * Real.exp (-c * ‖y‖ ^ 2)) := by
56    intro i hi
57    have hik : i ≤ n := by simpa [Nat.lt_succ_iff] using hi
