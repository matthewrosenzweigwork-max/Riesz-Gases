25theorem MFctx.integrable_energy (c : MFctx D T Λ ρ) (hN : 2 ≤ N)
26    (hsol : IsStrongSolution P D T X₀ W X)
27    {h : Torus d → ℝ} (hsm : Torus.IsSmooth h) (hev : ∀ x, h (-x) = h x) {r : ℝ}
28    (hr : r ∈ Set.Icc 0 T) :

34    (Filter.Eventually.of_forall fun ω => ?_)
35  simpa [Real.norm_eq_abs] using hC r hr _

37theorem MFctx.kid_main (c : MFctx D T Λ ρ) (hN : 2 ≤ N) {t : ℝ} (ht0 : 0 < t) (htT : t ≤ T)
38    (hsol : IsStrongSolution P D T X₀ W X) (hF : IsFilteredSolution P ℱ W' W X)
39    {h : Torus d → ℝ} (hsm : Torus.IsSmooth h) (hev : ∀ x, h (-x) = h x)
40    (hmicro : Integrable (fun p : ℝ × Ω => microTerm D h (ρ p.1) (X (Real.toNNReal p.1) p.2))
41      ((volume.restrict (Set.Icc 0 t)).prod P)) :
42    ∫ ω, modulatedEnergyKernel h (X (Real.toNNReal t) ω) (withDensityReal (ρ t)) ∂P -
43        ∫ ω, modulatedEnergyKernel h (X 0 ω) (withDensityReal (ρ 0)) ∂P =
44      ∫ r in Set.Icc 0 t,
45        ((1 / 2) * ∫ ω, commQuantityWith (Torus.grad h) (meanFieldDrift D (ρ r))
46            (X (Real.toNNReal r) ω) (withDensityReal (ρ r)) ∂P +
47          ∫ ω, microTerm D h (ρ r) (X (Real.toNNReal r) ω) ∂P +
48          D.invβ * ∫ ω, offDiagDiffPairing (Torus.laplacian h) (X (Real.toNNReal r) ω)
49            (withDensityReal (ρ r)) ∂P) := by
50  obtain ⟨hΨpp, hΨf, hpath⟩ := kid_pathwise c hN ht0 htT hsol hF hsm hev
51  have hJ := c.integrable_Jc hN ht0 htT hsol hF hsm hev hmicro
52  have hEt := c.integrable_energy hN hsol hsm hev (r := t) ⟨ht0.le, htT⟩
53  have hE0 : Integrable (fun ω => modulatedEnergyKernel h (X 0 ω) (withDensityReal (ρ 0))) P := by
54    have := c.integrable_energy hN hsol hsm hev (r := 0) ⟨le_rfl, c.hT⟩
55    simpa using this
56  have hI1 := (memLp_two_itoIntegral hF.brownian hΨpp t).integrable one_le_two
57  have hI2 := (memLp_two_itoIntegral hF.brownian hΨf t).integrable one_le_two
58  have hz1 := integral_itoIntegral hF.brownian hΨpp t
59  have hz2 := integral_itoIntegral hF.brownian hΨf t
60  have hJint : Integrable (fun ω => ∫ r in Set.Icc 0 t, Jc D h ρ T X r ω ∂volume) P :=
61    hJ.integral_prod_right
62  have hcongr : ∀ ω, (∫ r in Set.Icc (0 : ℝ) t, Jfull D h ρ X r ω) =
63      ∫ r in Set.Icc 0 t, Jc D h ρ T X r ω := fun ω =>
64    setIntegral_congr_fun measurableSet_Icc fun r hr =>
65      (Jc_eq (T := T) ⟨hr.1, hr.2.trans htT⟩ ω).symm
66  have hA : ∫ ω, (modulatedEnergyKernel h (X (Real.toNNReal t) ω) (withDensityReal (ρ t)) -
67      modulatedEnergyKernel h (X 0 ω) (withDensityReal (ρ 0))) ∂P =
68      ∫ ω, ((∫ r in Set.Icc 0 t, Jc D h ρ T X r ω ∂volume) +
69        itoIntegral P ℱ W' (fun s ω k => fkeRef D t (Φpp h) X s ω k) t ω -
70        itoIntegral P ℱ W' (empiricalNoiseIntegrand D t (fK h ρ) X) t ω) ∂P := by
71    refine integral_congr_ae ?_
72    filter_upwards [hpath] with ω hω
73    rw [hω, hcongr]
74  have e1 := integral_sub (hJint.add hI1) hI2
75  have e2 := integral_add hJint hI1
76  simp only [Pi.add_apply] at e1 e2
77  rw [← integral_sub hEt hE0, hA, e1, e2, hz1, hz2]
78  simp only [add_zero, sub_zero]
79  have hswap := integral_integral_swap (μ := volume.restrict (Set.Icc (0 : ℝ) t)) (ν := P)
80    (f := fun r ω => Jc D h ρ T X r ω) hJ
81  rw [← hswap]
82  refine setIntegral_congr_ae measurableSet_Icc ?_
83  have h1 : ∀ᵐ r ∂(volume.restrict (Set.Icc (0 : ℝ) t)),
84      Integrable (fun ω => microTerm D h (ρ r) (X (Real.toNNReal r) ω)) P :=
85    hmicro.prod_right_ae
86  filter_upwards [(ae_restrict_iff' measurableSet_Icc).1 h1] with r hr1 hr
87  exact c.per_r hN htT hsol hF hsm hev hr (hr1 hr)
