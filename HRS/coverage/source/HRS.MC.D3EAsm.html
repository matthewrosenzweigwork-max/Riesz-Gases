26    (hψ : IsFDLProfile ψ) {b : ℝ} (hb : IsFDLConst ψ s b) :
27    -- (i)
28    (∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
29      (∀ z : Euc d, z ≠ 0 → 0 ≤ fdlWE ψ s b lam z) ∧
30      (∀ z : Euc d, lam ≤ ‖z‖ →
31        fdlWE ψ s b lam z = 0 ∧ fdlTruncE ψ s b lam z = Whole.rieszKernel d s z) ∧
32      (∀ z : Euc d, z ≠ 0 → (s = 0 → ‖z‖ ≤ 1) →
33        fdlWE ψ s b lam z ≤ Whole.rieszKernel d s z + C * (if s = 0 then 1 else 0)) ∧
34      ∫⁻ z : Euc d, ENNReal.ofReal (fdlWE ψ s b lam z) ≤
35        ENNReal.ofReal (C * lam ^ ((d : ℝ) - s))) ∧
36    -- (ii), `j + s > 0`
37    (∀ j : ℕ, 0 < (j : ℝ) + s → ∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
38      ∀ z : Euc d, z ≠ 0 →
39        ‖iteratedFDeriv ℝ j (fdlWE ψ s b lam) z‖ ≤
40          (if ‖z‖ ≤ lam then C * ‖z‖ ^ (-s - (j : ℝ)) else 0) ∧
41        ‖iteratedFDeriv ℝ j (fdlTruncE ψ s b lam) z‖ ≤ C * (max lam ‖z‖) ^ (-s - (j : ℝ))) ∧
42    -- (ii), `s = j = 0`
43    (s = 0 → ∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d, z ≠ 0 →
44      |fdlWE ψ s b lam z| ≤ (if ‖z‖ ≤ lam then C * Real.log (lam / ‖z‖) else 0) ∧
45      |fdlTruncE ψ s b lam z| ≤ C * (1 + |Real.log (max lam ‖z‖)|)) ∧
46    -- (ii), smoothness and sup bounds ("in particular")
47    (∀ lam : ℝ, 0 < lam → lam < 1 →
48      ContDiff ℝ ((⊤ : ℕ∞) : WithTop ℕ∞) (fdlTruncE ψ s b lam)) ∧
49    (∀ j : ℕ, 0 < (j : ℝ) + s → ∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d,
50      ‖iteratedFDeriv ℝ j (fdlTruncE ψ s b lam) z‖ ≤ C * lam ^ (-s - (j : ℝ))) ∧
51    (∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
52      |fdlTruncE ψ s b lam 0| ≤
53        C * (lam ^ (-s) + (if s = 0 then |Real.log lam| else 0))) ∧
54    -- (iv)
55    (0 < s → ∃ c : ℝ, 0 < c ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d, lam ≤ ‖z‖ →
56      c * ‖z‖ ^ (-s) ≤ fdlTruncE ψ s b lam z) ∧
57    (s = 0 → ∃ c C : ℝ, 0 < c ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d, lam ≤ ‖z‖ →
58      ‖z‖ ≤ 1 / 2 → c * Real.log (1 / ‖z‖) - C ≤ fdlTruncE ψ s b lam z) ∧
59    (∀ K : ℝ, fdlK ≤ K → ∃ c : ℝ, 0 < c ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d, z ≠ 0 →
60      ‖z‖ ≤ 2 * lam → c * (if s = 0 then 1 else lam ^ (-s)) ≤ fdlWE ψ s b (K * lam) z) := by
61  obtain ⟨hiv1, hiv2, hiv3⟩ := FDLE.fdle_part_iv hd hs0 hsd hψ hb
62  refine ⟨FDLE.fdle_part_i hd hs0 hsd hψ hb, ?_, ?_, D3EG.d3eg_smooth hd hs0 hsd hψ hb,
63    D3EG.d3eg_sup hd hs0 hsd hψ hb, D3EG.d3eg_zero hd hs0 hsd hψ hb, hiv1, hiv2, hiv3⟩
