57/-- The Galerkin generator `P_n 𝓛 P_n` (Fourier truncation to the box `|k_i| ≤ n`). -/
58def galerkinGen (D : DynData d) (ρ : Torus d → ℝ) (n : ℕ) (F : Coeff d) : Coeff d :=
59  truncCoeff n (forwardGen D ρ (truncCoeff n F))

61/-- The Galerkin system of level `n`: `F_n(τ) = P_n φ + ∫_0^τ P_n 𝓛^{t-σ} P_n F_n(σ) dσ`, with
62`F_n(τ)` a trigonometric polynomial of degree `≤ n` and continuous coefficients. -/
63structure IsGalerkinSolution (D : DynData d) (μ : ℝ → Torus d → ℝ) (t : ℝ) (n : ℕ) (φc : Coeff d)
64    (F : ℝ → Coeff d) : Prop where
65  trig : ∀ τ, IsTrigPoly n (F τ)
66  cont : ∀ k, ContinuousOn (fun τ => F τ k) (Set.Icc 0 t)
67  eq : ∀ k, ∀ τ ∈ Set.Icc 0 t,
68    F τ k = truncCoeff n φc k + ∫ σ in (0 : ℝ)..τ, galerkinGen D (μ (t - σ)) n (F σ) k

142/-- Proposition 3.9 `B:thm:adjoint-torus` (lines 1739–1751), main statement: let `0 ≤ s < d`,
143`m ≥ 1`, and assume `B:eq:coeff-hypothesis` and one of the regimes (`AdjointData`).  There is
144`C > 0` (depending only on `d, s, m, |M|, Λ, ‖V‖_{C^{m+3}}`, the regime, and `λ₀` in R2b) such
145that for every `t ∈ (0,T]` and `φ ∈ H^m(𝕋^d)` there is a unique `F ∈ C_w([0,t];H^m)` solving
146`B:eq:adjoint-forward` in the distributional sense; it lies in `C([0,t];H^{m'})` for every
147`m' < m`, and `‖F(τ)‖_{H^m} ≤ e^{Λ_m τ}‖φ‖_{H^m}` for `τ ∈ [0,t]` (`B:eq:adjoint-torus-bound`),
148`Λ_m = C` in R1/R2 (uniformly in `β ∈ (0,∞]`) and `C(1+ν₀^{-θ/(2-θ)})` in R3 (`B:eq:LambdaM`). -/
149theorem adjoint_torus (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (m : ℕ) (hm : 1 ≤ m)
150    (R : Regime) (MB Λ VB : ℝ) :
151    ∃ C : ℝ, 0 < C ∧ ∀ (D : DynData d) (ν₀ T : ℝ) (μ : ℝ → Torus d → ℝ)
152      (_ : AdjointData s m R MB Λ VB D ν₀ T μ) (t : ℝ) (_ : t ∈ Set.Ioc 0 T)
153      (φc : Coeff d) (_ : MemHs m φc),
154      (∃ F : ℝ → Coeff d, IsForwardSolution D μ t φc F ∧ WeakContinuousInHs m F t ∧
155        (∀ m' : ℝ, m' < m → ContinuousInHs m' F t) ∧
156        ∀ τ ∈ Set.Icc 0 t, hsNorm m (F τ) ≤
157          Real.exp (R.lambdaM C ν₀ (thetaOrder d s) * τ) * hsNorm m φc) ∧
158      ∀ F₁ F₂ : ℝ → Coeff d, IsForwardSolution D μ t φc F₁ → WeakContinuousInHs m F₁ t →
159        IsForwardSolution D μ t φc F₂ → WeakContinuousInHs m F₂ t →
160        ∀ τ ∈ Set.Icc 0 t, F₁ τ = F₂ τ := by
161  obtain ⟨C, hC, hmain⟩ := galerkin_uniform_bound hd s hs0 hsd m R MB Λ VB
162  refine ⟨C, hC, fun D ν₀ T μ hA t ht φc hφ => ?_⟩
163  set lam := R.lambdaM C ν₀ (thetaOrder d s) with hlam
164  have hex : ∀ n, ∃ F : ℝ → Coeff d, IsGalSol D μ t n φc F ∧
165      ∀ τ ∈ Set.Icc 0 t, hsNorm m (F τ) ≤ Real.exp (lam * τ) * hsNorm m φc := fun n =>
166    galSol_complex_exists hA.hμ hA.hV ht (fun τ => Real.exp (lam * τ))
167      (fun τ => (Real.exp_pos _).le)
168      (fun φ' h1 h2 n F hF τ hτ =>
169        hmain D ν₀ T μ hA t ht n φ' h1 h2 F ⟨hF.trig, hF.cont, hF.eq⟩ τ hτ) n φc hφ
170  choose F hF hFb using hex
171  have hF' : ∀ n, IsGalerkinSolution D μ t n φc (F n) := fun n =>
172    ⟨(hF n).trig, (hF n).cont, (hF n).eq⟩
173  have hB : ∀ n, ∀ τ ∈ Set.Icc 0 t, hsNorm m (F n τ) ≤ Real.exp (|lam| * t) * hsNorm m φc := by
174    intro n τ hτ
175    refine (hFb n τ hτ).trans (mul_le_mul_of_nonneg_right (Real.exp_le_exp.2 ?_)
176      (Real.sqrt_nonneg _))
177    calc lam * τ ≤ |lam| * τ := mul_le_mul_of_nonneg_right (le_abs_self _) hτ.1
178      _ ≤ |lam| * t := mul_le_mul_of_nonneg_left hτ.2 (abs_nonneg _)
179  obtain ⟨ψ, G, hψ, hconv, hGsol, hGw, -⟩ := galerkin_limit hd s hs0 hsd m hm R MB Λ VB D ν₀ T μ
180    hA t ht φc hφ F hF' _ hB
181  have hGb : ∀ τ ∈ Set.Icc 0 t, hsNorm m (G τ) ≤ Real.exp (lam * τ) * hsNorm m φc := by
182    intro τ hτ
183    have hsq : ∀ c : Coeff d, hsNorm m c ^ 2 = hsNormSq m c := fun c =>
184      Real.sq_sqrt (hsNormSq_nonneg _ _)
185    have := (bound_of_coord_limit (m := (m : ℝ)) (Fn := fun n => F (ψ n)) hconv hτ
186      ((Real.exp (lam * τ) * hsNorm m φc) ^ 2) (fun n => isTrigPoly_memHs ((hF _).trig τ) _)
187      (fun n => by
188        rw [← hsq]
189        exact pow_le_pow_left₀ (Real.sqrt_nonneg _) (hFb _ τ hτ) 2)).2
190    rw [← hsq] at this
191    exact (pow_le_pow_iff_left₀ (Real.sqrt_nonneg _)
192      (mul_nonneg (Real.exp_pos _).le (Real.sqrt_nonneg _)) two_ne_zero).1 this
193  refine ⟨⟨G, hGsol, hGw, forward_strong_continuity (m : ℝ) t G hGw, hGb⟩, ?_⟩
194  intro F₁ F₂ h₁ w₁ h₂ w₂
195  have hm1 : (1 : ℝ) ≤ m := by exact_mod_cast hm
196  exact forward_unique_H1 hd s hs0 hsd m hm R MB Λ VB D ν₀ T μ hA t ht φc F₁ F₂ h₁
197    (weakContinuousInHs_mono hm1 w₁) h₂ (weakContinuousInHs_mono hm1 w₂)

249/-- Proposition 3.15 `B:prop:propagator-stability` (lines 1981–1986): for `a ∈ {1,2}` let
250`(ν_a, μ_a)` satisfy the hypotheses of Proposition 3.9 at level `m+2` with a common regime and
251common constants (same `R, MB, Λ, VB, ν₀, T`, same `M` and `V`), with propagators `U_a^{r,t}`.
252Then for `φ ∈ H^{m+2}` and `0 ≤ r ≤ t ≤ T`,
253`‖(U_1^{r,t} - U_2^{r,t})φ‖_{H^m} ≤ C (t-r) e^{Λ_{m+2}(t-r)} (|ν₁-ν₂| + sup_{[0,T]}‖μ₁-μ₂‖_{C^{m+2}}) ‖φ‖_{H^{m+2}}`;
254`C'` is the constant of `Λ_{m+2}`, and `δ ≥ sup_{[0,T]}‖μ₁-μ₂‖_{C^{m+2}}` any bound. -/

259        (_ : AdjointData s (m + 2) R MB Λ VB D₁ ν₀ T μ₁)
260        (_ : AdjointData s (m + 2) R MB Λ VB D₂ ν₀ T μ₂) (_ : D₁.M = D₂.M) (_ : D₁.V = D₂.V)
261        (δ : ℝ) (_ : ∀ r ∈ Set.Icc 0 T, Torus.cbNorm (m + 2) (μ₁ r - μ₂ r) ≤ δ)
262        (φc : Coeff d) (_ : MemHs (m + 2) φc) (r t : ℝ) (_ : 0 ≤ r) (_ : r ≤ t) (_ : t ≤ T),
263        hsNorm m (propagatorCoeff D₁ μ₁ (m + 2) t φc (t - r) -
264            propagatorCoeff D₂ μ₂ (m + 2) t φc (t - r)) ≤
265          C * (t - r) * Real.exp (R.lambdaM C' ν₀ (thetaOrder d s) * (t - r)) *
266            (|D₁.invβ - D₂.invβ| + δ) * hsNorm (m + 2) φc := by
267  obtain ⟨C₂, hC₂, hspec⟩ := propagatorCoeff_spec hd s hs0 hsd (m + 2) (by omega) R MB Λ VB
