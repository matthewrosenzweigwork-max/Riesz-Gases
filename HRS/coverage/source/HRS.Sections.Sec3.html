41/-- The stationary Gaussian field with the equilibrium (susceptibility) covariance (M L925-935,
42L5908-5913, L9096-9115) for the gradient model with potential `V`, thermal density `ρβ` and inverse
43temperature `β`, realised as a measurable, a.s. continuous `E_m = H^{-m}_0`-valued process on
44`[0,∞)` whose finite-dimensional laws against smooth tests `φ_a` at times `τ_a ≥ 0` are the centred
45Gaussians with covariance `PDE.stationaryLagCov` (`Σ_{ab} = C_β(φ_a, S_β(τ_b-τ_a) φ_b)` for
46`τ_a ≤ τ_b`).  Existence is not asserted by the predicate. -/
47structure IsStationaryGaussianField {Ω : Type*} [MeasurableSpace Ω] (P : Measure Ω)
48    (s β : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (hβ : 0 < β) (V ρβ : Torus d → ℝ) (m : ℝ)
49    (Z : ℝ → Ω → NegSobolev d m) : Prop where
50  meas : ∀ t, Measurable (Z t)
51  cont : ∀ᵐ ω ∂P, ContinuousOn (fun t => Z t ω) (Set.Ici 0)
52  gaussian : ∀ (p : ℕ) (τ : Fin p → ℝ) (φ : Fin p → Torus d → ℝ),
53    (∀ a, 0 ≤ τ a) → (∀ a, Torus.IsSmooth (φ a)) →
54    law P (fun ω => (WithLp.toLp 2 fun a => NegSobolev.eval (Z (τ a) ω) (φ a) :
55        EuclideanSpace ℝ (Fin p))) =
56      ProbabilityTheory.multivariateGaussian 0
57        (PDE.stationaryLagCov d s β (PDE.mfData d s hs0 hsd V β hβ) ρβ τ φ)

73/-- Proposition 3.46 (`M:O:prop:beta-zero`, TeX M L3313-3326), first sentence: for the homogeneous
74stationary Gaussian fluctuation (`V = 0`, `ρ_β = 1`), a nonzero normalized real Fourier coordinate
75has the covariance `O:eq:beta-cov`
76`E[Z_β^t Z_β^u] = exp[-(2π|k|)²(β⁻¹ + ĝ(k))|t-u|] / (1 + β ĝ(k))`.
77Technical hypotheses: `2 ≤ d` (standing assumption of Section 3, M L417); `t, u ≥ 0` (the process
78is indexed by `[0,∞)`); `d/2 < m` (the TeX fixes no index: the coordinate is a real Fourier
79coordinate; `m > d/2` is the range in which the stationary field exists in `E_m`, because the equal-time
80variance is bounded below by a constant times `1/(1+βĝ(k))` for every mode, so the second moment of the
81`H^{-m}` norm is `∑⟨k⟩^{-2m}` times a bounded quantity; below it the predicate is unsatisfiable). -/

114  have key := q38b_gauss_prod P _ hWm.aemeasurable _ hS hlaw
115  have e1 : (fun ω => NegSobolev.eval (Z t ω) (Torus.realMode k) *
116      NegSobolev.eval (Z u ω) (Torus.realMode k)) =
117      fun ω => (WithLp.toLp 2 fun a => NegSobolev.eval (Z (τ a) ω) (φ a) :
118        EuclideanSpace ℝ (Fin 2)) 0 *
119        (WithLp.toLp 2 fun a => NegSobolev.eval (Z (τ a) ω) (φ a) :
120          EuclideanSpace ℝ (Fin 2)) 1 := by
121    funext ω
122    rfl

244/-- Proposition 3.46, second sentence (`M:O:prop:beta-zero`, M L3314-3315): the path laws of a
245nonzero normalized real Fourier coordinate of the homogeneous stationary Gaussian fluctuation are
246not tight in `C([0,T];ℝ)` as `β ↓ 0`.  Stated along every sequence `β_n → 0+` (with an arbitrary
247probability space `Ω n` and field `Z n` for each `n`): the family of laws of the coordinate paths
248`t ↦ ⟨Z_{β_n}^t, e_k⟩` is not a tight set of measures (Mathlib `IsTightMeasureSet`).
249Technical hypotheses: `2 ≤ d`, `0 < T`, `d/2 < m` (see `hot_covariance`: the TeX fixes no index and the

293    linarith

410/-- Proposition 3.46, third sentence (`M:O:prop:beta-zero`, M L3315-3322): with physical time
411`t = βτ` the particle equation becomes `O:eq:hot-clock`.  Precisely: if `X` is a collision-free
412strong solution on `[0,T]` of the gradient particle system at inverse temperature `β > 0`
413(`PDE.mfData`, i.e. `M = -I`, `β⁻¹` noise) driven by the vector Brownian motion `W`, then
414`W'_τ = β^{-1/2} W_{βτ}` is again a vector Brownian motion and `τ ↦ X_{βτ}` is a collision-free
415strong solution on `[0,T/β]` of the rescaled dynamics `hotDyn` (drift
416`-β(∇V + N⁻¹∑∇γ)`, noise `√2 dW'`).  (Brownian scaling, M L3327-3328.) -/
417theorem hot_clock {s β : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) (hβ : 0 < β) (V : Torus d → ℝ)
418    {N : ℕ} {Ω : Type*} [MeasurableSpace Ω] (P : Measure Ω) (T : ℝ)
419    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N)
420    (hW : IsVectorBrownian P W)
421    (hX : IsStrongSolution P (PDE.mfData d s hs0 hsd V β hβ) T X₀ W X) :
422    IsVectorBrownian P (fun τ ω i => (Real.sqrt β)⁻¹ • W (⟨β, hβ.le⟩ * τ) ω i) ∧
423      IsStrongSolution P (hotDyn d s hs0 hsd V β) (T / β) X₀
424        (fun τ ω i => (Real.sqrt β)⁻¹ • W (⟨β, hβ.le⟩ * τ) ω i)
425        (fun τ ω => X (⟨β, hβ.le⟩ * τ) ω) := by
426  classical
427  set c : ℝ≥0 := ⟨β, hβ.le⟩ with hc_def
428  have hcR : (c : ℝ) = β := rfl
429  have hc0 : c ≠ 0 := by
430    intro h
431    have h' : (c : ℝ) = 0 := by rw [h]; rfl
432    linarith
433  have hcτ : ∀ τ : ℝ≥0, ((c * τ : ℝ≥0) : ℝ) = β * τ := fun τ => by rw [NNReal.coe_mul, hcR]
434  have hBr : IsVectorBrownian P (fun τ ω i => (Real.sqrt β)⁻¹ • W (c * τ) ω i) := by
435    refine ⟨fun i j => ?_, ?_⟩
436    · have h := (hW.1 i j).smul hc0
437      convert h using 1
438      funext τ ω
439      simp [hcR, smul_eq_mul]
440    · have h := hW.2.comp (fun _ => fun f : ℝ≥0 → ℝ => fun τ => (Real.sqrt β)⁻¹ * f (c * τ))
441        (fun _ => Measurable.of_eval fun τ => (measurable_pi_apply (c * τ)).const_mul _)
442      convert h using 1
443      funext ij ω t
444      simp [smul_eq_mul]
445  refine ⟨hBr, ?_⟩
446  obtain ⟨⟨Y, hY, hXY⟩, hinit, hcont, hadapt, hcoll⟩ := hX
447  subst hXY
448  refine ⟨⟨fun τ ω i => Y (c * τ) ω i, ?_, rfl⟩, ?_, ?_, ?_, ?_⟩
449  · filter_upwards [hY] with ω hω
450    intro τ hτ
451    have hτ' : ((c * τ : ℝ≥0) : ℝ) ≤ T := by
452      rw [hcτ, mul_comm]; exact (le_div_iff₀ hβ).1 hτ
453    obtain ⟨hint, heq⟩ := hω (c * τ) hτ'
454    have hτ0 : (0 : ℝ) ≤ τ := NNReal.coe_nonneg τ
455    have hsq : ∀ (A B : Euc d), Real.sqrt (2 * (hotDyn d s hs0 hsd V β).invβ) •
456        ((Real.sqrt β)⁻¹ • A - (Real.sqrt β)⁻¹ • B) =
457        Real.sqrt (2 * (PDE.mfData d s hs0 hsd V β hβ).invβ) • (A - B) := by
458      intro A B
459      have h1 : (hotDyn d s hs0 hsd V β).invβ = 1 := rfl
460      have h2 : (PDE.mfData d s hs0 hsd V β hβ).invβ = β⁻¹ := rfl
461      rw [h1, h2, ← smul_sub, mul_one, smul_smul]
462      congr 1
463      rw [Real.sqrt_mul (by norm_num : (0 : ℝ) ≤ 2), Real.sqrt_inv]
464    have hpt : ∀ r ∈ Set.Icc (0 : ℝ) (τ : ℝ), ∀ i : Fin N, particleDrift (hotDyn d s hs0 hsd V β)
465          (projectPath (fun (τ : ℝ≥0) ω i => Y (c * τ) ω i) (Real.toNNReal r) ω) i =
466          β • particleDrift (PDE.mfData d s hs0 hsd V β hβ)
467            (projectPath Y (Real.toNNReal (β * r)) ω) i := by
468        intro r hr i
469        rw [q38_proj_eq Y c hcR hβ hr.1 ω, q38_hot_drift]
470    refine ⟨fun i => ?_, fun i => ?_⟩
471    · have hint' : IntegrableOn (fun r => β • particleDrift (PDE.mfData d s hs0 hsd V β hβ)
472          (projectPath Y (Real.toNNReal (β * r)) ω) i) (Set.Icc 0 (τ : ℝ)) := by
473        have hI := hint i
474        rw [hcτ] at hI
475        exact Integrable.smul β (q38_integrableOn_scale
476          (fun u => particleDrift (PDE.mfData d s hs0 hsd V β hβ)
477            (projectPath Y (Real.toNNReal u) ω) i) hβ hτ0 hI)
478      refine hint'.congr_fun ?_ measurableSet_Icc
479      intro r hr
480      exact (hpt r hr i).symm
481    · have hI : ∫ r in Set.Icc (0 : ℝ) τ, particleDrift (hotDyn d s hs0 hsd V β)
482          (projectPath (fun (τ : ℝ≥0) ω i => Y (c * τ) ω i) (Real.toNNReal r) ω) i =
483          ∫ u in Set.Icc 0 (β * τ), particleDrift (PDE.mfData d s hs0 hsd V β hβ)
484            (projectPath Y (Real.toNNReal u) ω) i := by
485        rw [setIntegral_congr_fun measurableSet_Icc (fun r hr => hpt r hr i)]
486        exact q38_cov_scale (fun u => particleDrift (PDE.mfData d s hs0 hsd V β hβ)
487          (projectPath Y (Real.toNNReal u) ω) i) hβ hτ0
488      have heq' := heq i
489      rw [hcτ] at heq'
490      simp only [mul_zero] at ⊢
491      rw [hI, heq', hsq]
492  · filter_upwards [hinit] with ω hω
493    simpa using hω
494  · filter_upwards [hcont] with ω hω i
495    exact (hω i).comp (continuous_const.mul continuous_id)
496  · intro τ
497    exact hadapt (c * τ)
498  · filter_upwards [hcoll] with ω hω t ht
499    have ht' : ((c * t : ℝ≥0) : ℝ) ≤ T := by
500      rw [hcτ, mul_comm]; exact (le_div_iff₀ hβ).1 ht
501    exact hω (c * t) ht'
