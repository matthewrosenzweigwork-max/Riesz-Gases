24theorem screened_pairing_proof : type_of% @HRS.Whole.screened_pairing := by
25  intro d hd s hs0 hsd h2
26  by_cases hs : s < (d : ℝ) - 1
27  · exact screened_pairing_lt_proof hd hs0 hs h2
28  · have hsd1 : (d : ℝ) - 1 ≤ s := not_lt.mp hs
29    have hseq : s = (d : ℝ) - 1 := le_antisymm hsd hsd1
30    have hd1 : (1 : ℝ) ≤ d := by exact_mod_cast hd
31    have hsdd : s < d := by linarith
32    have hk0 : kgOrder d s = 0 := by
33      unfold kgOrder
34      exact max_eq_right (by linarith)
35    have hsi : s + 1 - (d : ℝ) = 0 := by linarith
36    obtain ⟨C0, hC0⟩ := screened_local_global_proof hd hsdd hsd1
37    refine ⟨max C0 0, fun Λ a hA i ψ w u hψ hw hψfin hu hrep => ?_⟩
38    have hΛ : 0 ≤ Λ := by
39      have h00 : 0 ≤ jbracket (0 : Euc d) ^ ((d : ℝ) + 3) * |a 0| :=
40        mul_nonneg (Real.rpow_nonneg (le_trans zero_le_one (p1_one_le_jbracket 0)) _) (abs_nonneg _)
41      linarith [hA 0]
42    have hab : ∀ x, |a x| ≤ Λ := by
43      intro x
44      have h1 : 1 ≤ jbracket x ^ ((d : ℝ) + 3) :=
45        Real.one_le_rpow (p1_one_le_jbracket x) (by positivity)
46      exact (le_mul_of_one_le_left (abs_nonneg _) h1).trans (hA x)
47    have hm : AEStronglyMeasurable (fun y => a y * ψ y) volume := q1_aesm_of_rep hs0 i hrep
48    have hψa : MemLp (fun y => a y * ψ y) 2 volume := by
49      refine MemLp.of_le ((hψ.norm).const_mul Λ) hm (ae_of_all _ fun x => ?_)
50      show ‖a x * ψ x‖ ≤ ‖Λ * ‖ψ x‖‖
51      rw [Real.norm_of_nonneg (mul_nonneg hΛ (norm_nonneg _)), norm_mul, Real.norm_eq_abs]
52      exact mul_le_mul_of_nonneg_right (hab x) (norm_nonneg _)
53    have hnorm_aψ : eLpNorm (fun y => a y * ψ y) 2 volume ≤ ENNReal.ofReal Λ * eLpNorm ψ 2 volume := by
54      have h1 : eLpNorm (fun y => a y * ψ y) 2 volume ≤ eLpNorm (fun y => Λ * ‖ψ y‖) 2 volume :=
55        eLpNorm_mono_real hm (fun x => by
56          show ‖a x * ψ x‖ ≤ Λ * ‖ψ x‖
57          rw [norm_mul, Real.norm_eq_abs]
58          exact mul_le_mul_of_nonneg_right (hab x) (norm_nonneg _))
59      have h2' : eLpNorm (fun y => Λ * ‖ψ y‖) 2 volume = ‖Λ‖ₑ * eLpNorm (fun y => ‖ψ y‖) 2 volume :=
60        eLpNorm_const_smul Λ (fun y => ‖ψ y‖) 2 volume
61      rw [eLpNorm_norm ψ hψ.aestronglyMeasurable] at h2'
62      rw [← ofReal_norm_eq_enorm, Real.norm_of_nonneg hΛ] at h2'
63      rw [h2'] at h1
64      exact h1
65    have hfin : sobolevNormE (s + 1 - d) (fun y => a y * ψ y) < ⊤ := by
66      rw [hsi, l1_sobolev_zero_eq hψa]
67      exact hψa.eLpNorm_lt_top
68    obtain ⟨w', hw'rep, hw'L2, hw'bd⟩ := hC0 i (fun y => a y * ψ y) hfin
69    rw [hsi, l1_sobolev_zero_eq hψa] at hw'bd
70    have hu_ae : u =ᵐ[volume] w' := gk_unique_proof i hrep hw'rep
71    have hbound : eLpNorm w' 2 volume ≤ ENNReal.ofReal (max C0 0) * (ENNReal.ofReal Λ * eLpNorm ψ 2 volume) :=
72      hw'bd.trans (mul_le_mul' (ENNReal.ofReal_le_ofReal (le_max_left _ _)) hnorm_aψ)
73    have hEq : ∫ x, u x * w x = ∫ x, w' x * w x := by
74      apply integral_congr_ae
75      filter_upwards [hu_ae] with x hx
76      simp only [hx]
77    have hint : ‖∫ x, u x * w x‖ₑ ≤ ∫⁻ x, ‖w' x‖ₑ * ‖w x‖ₑ := by
78      rw [hEq]
79      exact (enorm_integral_le_lintegral_enorm _).trans (le_of_eq (lintegral_congr fun x => enorm_mul _ _))
80    have hCS : (∫⁻ x, ‖w' x‖ₑ * ‖w x‖ₑ) ^ 2 ≤ (∫⁻ x, ‖w' x‖ₑ ^ 2) * ∫⁻ x, ‖w x‖ₑ ^ 2 :=
81      p1_cs_aemeas volume (fun x => ‖w' x‖ₑ) (fun x => ‖w x‖ₑ)
82        hw'L2.aestronglyMeasurable.enorm hw.aestronglyMeasurable.enorm
83    have hW : (∫⁻ x, ‖w' x‖ₑ * ‖w x‖ₑ) ≤ eLpNorm w' 2 volume * eLpNorm w 2 volume := by
84      apply p1_le_of_sq_le
85      calc (∫⁻ x, ‖w' x‖ₑ * ‖w x‖ₑ) ^ 2 ≤ (∫⁻ x, ‖w' x‖ₑ ^ 2) * ∫⁻ x, ‖w x‖ₑ ^ 2 := hCS
86        _ = (eLpNorm w' 2 volume) ^ 2 * (eLpNorm w 2 volume) ^ 2 := by
87            rw [← l1_sq_eLpNorm_ae w' hw'L2.aestronglyMeasurable, ← l1_sq_eLpNorm_ae w hw.aestronglyMeasurable]
88        _ = (eLpNorm w' 2 volume * eLpNorm w 2 volume) ^ 2 := by ring
89    rw [hk0, l1_sobolev_zero_eq hψ, ENNReal.ofReal_mul (le_max_right _ _), ← Real.norm_eq_abs,
90      ofReal_norm_eq_enorm]
91    calc ‖∫ x, u x * w x‖ₑ ≤ ∫⁻ x, ‖w' x‖ₑ * ‖w x‖ₑ := hint
92      _ ≤ eLpNorm w' 2 volume * eLpNorm w 2 volume := hW
93      _ ≤ (ENNReal.ofReal (max C0 0) * (ENNReal.ofReal Λ * eLpNorm ψ 2 volume)) * eLpNorm w 2 volume :=
94          mul_le_mul_of_nonneg_right hbound zero_le
95      _ = ENNReal.ofReal (max C0 0) * ENNReal.ofReal Λ * eLpNorm ψ 2 volume * eLpNorm w 2 volume := by ring
