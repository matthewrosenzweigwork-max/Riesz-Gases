9The three conjuncts of `HRS.cosine_tilt_example`:

12* the convergence of the tilted law: `HRS.ME.TiltConv.tilt_convergence`, whose analytic input is the centering clause of
13  Theorem 2.9 (`HRS.MD.EQFrozen29Trans.centering_transition_proof`) and the positivity of `v` (`HRS.ME.SuscPos.susc_pos`);
14* the limit density is not Gaussian: `HRS.ME.TiltBounds.cosineTiltDensity_not_gaussian`.

21theorem gibbs_isProb {d : ℕ} (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {V : Torus d → ℝ}
22    (hV : Torus.IsSmooth V) {β : ℝ} (hβ : 0 < β) (N : ℕ) :
23    IsProbabilityMeasure (canonicalGibbs (periodicRiesz d s) V N β) := by
24  rcases Nat.eq_zero_or_pos N with rfl | hN
25  · have hH : ∀ X : Config d 0, canonicalHamiltonian (periodicRiesz d s) V X = 0 := fun X => by
26      simp [canonicalHamiltonian]
27    have hZ : canonicalPartition (periodicRiesz d s) V 0 β = 1 := by
28      simp [canonicalPartition, hH]
29    have e : canonicalGibbs (periodicRiesz d s) V 0 β = (volume : Measure (Config d 0)) := by
30      simp [canonicalGibbs, hZ, hH]
31    rw [e]
32    infer_instance
33  · exact HRS.MD.EQLawFlow.eqlaw_gibbs_prob hd hs0 hsd hV hβ hN
