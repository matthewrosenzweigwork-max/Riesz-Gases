147theorem fdl_compare_proof (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {ψ : Euc d → ℝ}
148    (hψ : IsFDLProfile ψ) {b : ℝ} (hb : IsFDLConst ψ s b) :
149    ∃ C₀ C : ℝ, 0 ≤ C₀ ∧ 0 < C ∧
150      ∀ Λ : ℝ, 1 ≤ Λ → ∀ N : ℕ, 1 ≤ N → smearScale d N Λ < 1 / 2 →
151      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
152      ∀ X : Config d N, X ∉ collisionSet d N →
153        (∀ η : ℝ, smearScale d N Λ ≤ η → η ≤ 2 * fdlK * smearScale d N Λ →
154          0 ≤ fdlEnergy ψ s b η X (withDensityReal ρ) + oNΛ d s C₀ Λ N) ∧
155        (∀ η : ℝ, smearScale d N Λ ≤ η → η ≤ 2 * fdlK * smearScale d N Λ →
156          0 ≤ fdlPairW ψ s b η X ∧
157          fdlPairW ψ s b η X ≤ C * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N)) ∧
158        (|rieszEnergy d s X (withDensityReal ρ) - fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ)
159            - (1 / 2) * fdlPairW ψ s b (smearScale d N Λ) X|
160          ≤ C * Λ * smearScale d N Λ ^ ((d : ℝ) - s) ∧
161        fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ) + oNΛ d s C₀ Λ N
162          ≤ C * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N) ∧
163        fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ) - C * oNΛ d s C₀ Λ N
164          ≤ rieszEnergy d s X (withDensityReal ρ)) := by
165  obtain ⟨C₀, C, hC₀, hC, h⟩ := d47_master hd hs0 hsd hψ hb d47_two_fdlK
166  refine ⟨C₀, C, hC₀, hC, fun Λ hΛ N hN _ ρ hρ hρΛ X hX => ?_⟩
167  have H := fun η h1 h2 => h Λ hΛ N hN ρ hρ hρΛ X hX η h1 h2
168  refine ⟨fun η h1 h2 => (H η h1 h2).1, fun η h1 h2 => (H η h1 h2).2.1, ?_⟩
169  exact (H _ le_rfl (d47_one_le_mul (d47_lam_pos hN hΛ) d47_two_fdlK)).2.2.1
