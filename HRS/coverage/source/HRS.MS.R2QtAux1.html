35/-- Jensen for a probability measure in `ℝ≥0∞`: `(∫ g)^q ≤ ∫ g^q`. -/
36theorem s92_jensen_enn {α : Type*} [MeasurableSpace α] (ν : Measure α) [IsProbabilityMeasure ν]
37    {g : α → ℝ≥0∞} (hg : Measurable g) {q : ℝ} (hq : 1 < q) :
38    (∫⁻ x, g x ∂ν) ^ q ≤ ∫⁻ x, g x ^ q ∂ν := by
39  have hpq : q.HolderConjugate (Real.conjExponent q) := Real.HolderConjugate.conjExponent hq
40  have h := ENNReal.lintegral_mul_le_Lp_mul_Lq ν hpq hg.aemeasurable
41    (measurable_const (a := (1 : ℝ≥0∞))).aemeasurable
42  simp only [Pi.mul_apply, mul_one, ENNReal.one_rpow, lintegral_const, measure_univ] at h
43  have hq0 : 0 < q := by linarith
44  calc (∫⁻ x, g x ∂ν) ^ q ≤ ((∫⁻ x, g x ^ q ∂ν) ^ (1 / q)) ^ q :=
45        ENNReal.rpow_le_rpow (by simpa using h) hq0.le
46    _ = ∫⁻ x, g x ^ q ∂ν := by
47        rw [← ENNReal.rpow_mul, one_div, inv_mul_cancel₀ hq0.ne', ENNReal.rpow_one]

49/-- Duality inequality for the push-forward of a density along an invariant Markov kernel. -/
50theorem s92_dual {Ω : Type*} [MeasurableSpace Ω] (Q : Measure Ω) [IsProbabilityMeasure Q]
51    (κ : Kernel Ω Ω) [IsMarkovKernel κ] (hinv : Q.bind κ = Q) {f : Ω → ℝ≥0∞} (hf : Measurable f)
52    {p : ℝ} (hp : 1 < p) {g : Ω → ℝ≥0∞} (hg : Measurable g) :
53    ∫⁻ x, g x ∂((Q.withDensity f).bind κ) ≤
54      (∫⁻ x, f x ^ p ∂Q) ^ (1 / p) * (∫⁻ x, g x ^ Real.conjExponent p ∂Q) ^ (1 / Real.conjExponent p) := by
55  have hpq : p.HolderConjugate (Real.conjExponent p) := Real.HolderConjugate.conjExponent hp
56  set q := Real.conjExponent p with hqdef
57  have hq1 : 1 < q := hpq.symm.lt
58  have hKg : Measurable (fun a => ∫⁻ x, g x ∂(κ a)) := hg.lintegral_kernel
59  rw [Measure.lintegral_bind κ.measurable.aemeasurable hg.aemeasurable,
60    lintegral_withDensity_eq_lintegral_mul Q hf hKg]
61  refine (ENNReal.lintegral_mul_le_Lp_mul_Lq Q hpq hf.aemeasurable hKg.aemeasurable).trans ?_
62  refine mul_le_mul' le_rfl (ENNReal.rpow_le_rpow ?_ (by positivity))
63  calc ∫⁻ a, (∫⁻ x, g x ∂(κ a)) ^ q ∂Q ≤ ∫⁻ a, ∫⁻ x, g x ^ q ∂(κ a) ∂Q :=
64        lintegral_mono fun a => s92_jensen_enn (κ a) hg hq1
65    _ = ∫⁻ x, g x ^ q ∂(Q.bind κ) :=
66        (Measure.lintegral_bind κ.measurable.aemeasurable (hg.pow_const q).aemeasurable).symm
67    _ = ∫⁻ x, g x ^ q ∂Q := by rw [hinv]

90/-- **`L^p` contraction of an invariant Markov kernel on densities** (Jensen/duality): if `Q` is a probability

92`‖d(Rκ)/dQ‖_{L^p(Q)} ≤ ‖f‖_{L^p(Q)}`. -/

112      filter_upwards [h1] with a ha
113      simp [Pi.mul_apply, ha]

119  have hpq : p.HolderConjugate (Real.conjExponent p) := Real.HolderConjugate.conjExponent hp

123  have hn : ∀ n : ℕ, ∫⁻ x, (min (u x) (n : ℝ≥0∞)) ^ p ∂Q ≤ F := by

138      rw [← hRu]; exact lintegral_withDensity_eq_lintegral_mul Q hum hgm

155      calc A ≤ ∫⁻ x, (u * g) x ∂Q := lintegral_mono hle

157        _ ≤ F ^ (1 / p) * (∫⁻ x, g x ^ q ∂Q) ^ (1 / q) := hdual
158        _ = F ^ (1 / p) * A ^ (1 / q) := by rw [lintegral_congr hgq]

165    intro n m hnm x
166    exact ENNReal.rpow_le_rpow (min_le_min le_rfl (by exact_mod_cast hnm)) hp0.le
167  have hsup : ∀ x, ⨆ n : ℕ, (min (u x) (n : ℝ≥0∞)) ^ p = (u x) ^ p := by
168    intro x
169    have h1 : Tendsto (fun n : ℕ => min (u x) (n : ℝ≥0∞)) atTop (𝓝 (u x)) := by
170      have := (tendsto_const_nhds (x := u x)).min ENNReal.tendsto_nat_nhds_top
171      simpa using this
172    have h2 : Tendsto (fun n : ℕ => (min (u x) (n : ℝ≥0∞)) ^ p) atTop (𝓝 ((u x) ^ p)) :=
173      (ENNReal.continuous_rpow_const.tendsto _).comp h1
174    exact (tendsto_nhds_unique (tendsto_atTop_iSup (fun n m hnm => hmono hnm x)) h2)
175  calc ∫⁻ x, u x ^ p ∂Q = ∫⁻ x, ⨆ n : ℕ, (min (u x) (n : ℝ≥0∞)) ^ p ∂Q := by simp_rw [hsup]
176    _ = ⨆ n : ℕ, ∫⁻ x, (min (u x) (n : ℝ≥0∞)) ^ p ∂Q :=
177        lintegral_iSup (fun n => (hum.min measurable_const).pow_const _) hmono
178    _ ≤ F := iSup_le hn
