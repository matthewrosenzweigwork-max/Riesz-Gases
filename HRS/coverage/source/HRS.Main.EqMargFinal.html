35theorem eqmarg_marginal_unif (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d) (βm βp : ℝ) (hβm : 0 < βm)
36    (hββ : βm ≤ βp) (Λ0 : ℝ) :
37    ∃ C : ℝ, 0 ≤ C ∧ ∀ β : ℝ, βm ≤ β → β ≤ βp → ∀ ρ : Torus d → ℝ, SmoothPosDensity ρ →
38      (∀ x, ρ x ≤ Λ0) → ∀ N : ℕ, 2 ≤ N →
39      firstMarginal (modulatedGibbs (periodicRiesz d s) N β (withDensityReal ρ)) ≪
40          withDensityReal ρ ∧
41        eLpNorm (fun y => ((firstMarginal (modulatedGibbs (periodicRiesz d s) N β
42            (withDensityReal ρ))).rnDeriv (withDensityReal ρ) y).toReal - 1) 2
43            (withDensityReal ρ) ≤ ENNReal.ofReal (C * deltaN d s N) := by
44  have hβp : 0 < βp := lt_of_lt_of_le hβm hββ
45  obtain ⟨C, hC0, hCU⟩ := eqmarg_L2_all hd hs0 hsd βm βp hβm hββ Λ0
46  obtain ⟨Mp, hMp1, hMpU⟩ := eqmarg_pDens_bounds hd hs0 hsd βm βp hβm hββ Λ0
47  refine ⟨C, hC0, fun β hβ1 hβ2 ρ hρ hΛ N hN => ?_⟩
48  have hβ : 0 < β := lt_of_lt_of_le hβm hβ1
49  have hμ := isProbabilityMeasure_mu hρ
50  have hC := hCU β hβ1 hβ2 ρ hρ hΛ
51  have hMp := hMpU β hβ1 hβ2 ρ hρ hΛ
52  obtain ⟨n, rfl⟩ : ∃ n, N = n + 1 := ⟨N - 1, by omega⟩
53  have hn : 0 < n := by omega
54  obtain ⟨hlam, hln, -⟩ := lam_facts hd hs0 hsd hβ hρ hn (lam := β / ((n + 1 : ℕ) : ℝ)) rfl
55  have hae := rnDeriv_pi_eq hd hs0 hsd hρ hlam hln
56  rw [modulatedGibbs_eq_cavity hd hs0 hsd hβ hρ (n + 1) (Nat.succ_pos n)]
57  refine ⟨?_, ?_⟩
58  · rw [firstMarginal_eq_withDensity hd hs0 hsd hρ hlam hln]
59    exact withDensity_absolutelyContinuous _ _
60  · have hN' : 2 ≤ n + 1 := hN
61    have hpm := measurable_pDens hd hs0 hsd hρ hlam hln
62    have hp0 := pDens_pos hd hs0 hsd hρ hlam hln
63    have hz : ∀ y, |pDens (periodicRiesz d s) (withDensityReal ρ) n (β / ((n + 1 : ℕ) : ℝ)) β y - 1| ≤ Mp := by
64      intro y
65      have h1 := hp0 y
66      have h2 := (hMp (n + 1) hN' y).2
67      simp only [Nat.add_sub_cancel] at h2
68      rw [abs_le]; constructor <;> linarith
69    have hmem : MemLp (fun y => pDens (periodicRiesz d s) (withDensityReal ρ) n (β / ((n + 1 : ℕ) : ℝ)) β y - 1)
70        2 (withDensityReal ρ) :=
71      MemLp.of_bound (hpm.sub measurable_const).aestronglyMeasurable Mp (Filter.Eventually.of_forall fun y => by
72        rw [Real.norm_eq_abs]; exact hz y)
73    rw [eLpNorm_congr_ae (g := fun y => pDens (periodicRiesz d s) (withDensityReal ρ) n
74      (β / ((n + 1 : ℕ) : ℝ)) β y - 1) (hae.mono fun y hy => by simp only [hy])]
75    rw [hmem.eLpNorm_eq_integral_rpow_norm (by norm_num) (by norm_num)]
76    refine ENNReal.ofReal_le_ofReal ?_
77    have hA := hC n hn (β / ((n + 1 : ℕ) : ℝ)) rfl
78    have hδ0 : 0 ≤ deltaN d s (n + 1) := by
79      have hN0 : (0 : ℝ) < ((n + 1 : ℕ) : ℝ) := by positivity
80      have := inv_le_deltaN (d := d) hs0 (n + 1) (by omega)
81      exact (inv_nonneg.2 hN0.le).trans this
82    have e : ∫ y, ‖pDens (periodicRiesz d s) (withDensityReal ρ) n (β / ((n + 1 : ℕ) : ℝ)) β y - 1‖ ^
83        (2 : ENNReal).toReal ∂(withDensityReal ρ) =
84        ∫ y, (pDens (periodicRiesz d s) (withDensityReal ρ) n (β / ((n + 1 : ℕ) : ℝ)) β y - 1) ^ 2
85          ∂(withDensityReal ρ) := by
86      refine integral_congr_ae (Filter.Eventually.of_forall fun y => ?_)
87      simp [Real.norm_eq_abs, Real.rpow_two, sq_abs]
88    rw [e]
89    have h3 : (2 : ENNReal).toReal⁻¹ = 1 / 2 := by simp
90    rw [h3, ← Real.sqrt_eq_rpow]
91    exact Real.sqrt_le_iff.2 ⟨mul_nonneg hC0 hδ0, hA⟩

99theorem equilibrium_marginal_L2_proof (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d)
100    (V : Torus d → ℝ) (hV : Torus.IsSmooth V) (βm βp : ℝ) (hβm : 0 < βm) (hββ : βm ≤ βp) :
101    ∃ C : ℝ, 0 < C ∧ ∀ β : ℝ, β ∈ Set.Icc βm βp → ∀ ρβ : Torus d → ℝ,
102      IsThermalDensity (periodicRiesz d s) V β ρβ → ∀ N : ℕ, 2 ≤ N →
103      firstMarginal (canonicalGibbs (periodicRiesz d s) V N β) ≪ withDensityReal ρβ ∧
104        eLpNorm (fun y => ((firstMarginal (canonicalGibbs (periodicRiesz d s) V N β)).rnDeriv
105            (withDensityReal ρβ) y).toReal - 1) 2 (withDensityReal ρβ) ≤
106          ENNReal.ofReal (C * deltaN d s N) := by
107  have hVc : Continuous V :=
108    HRS.PDE.MFAux.continuous_of_isCk (ContDiff.of_le
109      (show ContDiff ℝ ((⊤ : ℕ∞) : WithTop ℕ∞) (Torus.lift V) from hV) (by exact_mod_cast le_top))
110  obtain ⟨lam, Lam, hlam, hb⟩ :=
111    PDE.MFDecay.exists_thermal_bounds (by omega : 1 ≤ d) s hs0 hsd βm βp hβm hββ V hVc
112  obtain ⟨C, hC0, hCU⟩ := EqMarg.eqmarg_marginal_unif hd hs0 hsd βm βp hβm hββ Lam
113  refine ⟨C + 1, by linarith, fun β hβ ρβ hρ N hN => ?_⟩
114  have hβ0 : 0 < β := lt_of_lt_of_le hβm hβ.1
115  obtain ⟨hpos, hsm⟩ := Q41.q41_thermal_regular hd hs0 hsd hβ0 hV hρ
116  have hρs : SmoothPosDensity ρβ := ⟨hρ.1, hsm, hpos⟩
117  have hΛ : ∀ x, ρβ x ≤ Lam :=
118    fun x => (hb β hβ ρβ hρ (PoleCavityAux.continuous_of_smoothPos hρs) hpos x).2
119  have hP : canonicalGibbs (periodicRiesz d s) V N β =
120      modulatedGibbs (periodicRiesz d s) N β (withDensityReal ρβ) := by
121    obtain ⟨Λ, hΛ0, hΛ'⟩ := PoleCavityAux.exists_density_bound hρs
122    refine canonicalGibbs_eq_modulatedGibbs (periodicRiesz d s) V β ρβ hρ
123      (fun x => periodicRiesz_neg x) ?_
124    intro x
125    exact (CoercAux.J_bounds (by omega : 1 ≤ d) hs0 hsd one_pos hρs.density hΛ0.le hΛ' x).1
126  rw [hP]
127  obtain ⟨h1, h2⟩ := hCU β hβ.1 hβ.2 ρβ hρs hΛ N hN
128  refine ⟨h1, h2.trans (ENNReal.ofReal_le_ofReal ?_)⟩
129  have hδ0 : 0 ≤ deltaN d s N := by
130    have hN0 : (0 : ℝ) < N := by exact_mod_cast (by omega : 0 < N)
131    exact (inv_nonneg.2 hN0.le).trans (MarginalAux.inv_le_deltaN (d := d) hs0 N (by omega))
132  nlinarith
