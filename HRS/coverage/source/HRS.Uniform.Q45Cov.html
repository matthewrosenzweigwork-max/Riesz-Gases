27theorem covariance_measure_proof (T K : ℝ) (hT : 0 ≤ T) :
28    ∃ C : ℝ, 0 ≤ C ∧
29      ∀ (H E : Type) [NormedAddCommGroup H] [InnerProductSpace ℝ H] [CompleteSpace H]
30        [MeasurableSpace H] [BorelSpace H] [SecondCountableTopology H]
31        [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]
32        (S : EvolData H E) (_ : S.MaxHyp T K),
33      ∃ J : (ℝ → H →L[ℝ] H →L[ℝ] ℝ) → VectorMeasure ℝ (ProjTensor2 (PathSp T E)),
34        (∀ D, IsTraceDensityL1 T D →
35          (∀ I, MeasurableSet I → J D I = J D (I ∩ Set.Icc 0 T)) ∧
36          (∀ I, MeasurableSet I → ∀ Ψ : PathSp T E →L[ℝ] PathSp T E →L[ℝ] ℝ,
37            tensorPairing Ψ (J D I) = tensorPairing Ψ.flip (J D I)) ∧
38          (J D).variation (Set.Icc 0 T) ≤
39            ENNReal.ofReal C * ∫⁻ r in Set.Icc (0 : ℝ) T, ENNReal.ofReal (traceNormForm (D r)) ∧
40          (∀ (t₁ t₂ : Set.Icc (0 : ℝ) T) (ℓ₁ ℓ₂ : E →L[ℝ] ℝ) (I : Set ℝ), MeasurableSet I →
41            tensorPairing (obsForm t₁ t₂ ℓ₁ ℓ₂) (J D I) =
42              ∫ r in I ∩ Set.Icc 0 (min (t₁ : ℝ) t₂),
43                D r (pullbackVec S t₁ r ℓ₁) (pullbackVec S t₂ r ℓ₂)) ∧
44          ContinuousOn (fun t => J D (Set.Icc 0 t)) (Set.Icc 0 T)) ∧
45        (∀ D D', IsTraceDensityL1 T D → IsTraceDensityL1 T D' → ∀ a b : ℝ, ∀ I, MeasurableSet I →
46          J (fun r => a • D r + b • D' r) I = a • J D I + b • J D' I) := by
47  rcases hT.eq_or_lt with h0 | hpos
48  · -- `T = 0`: the zero measure
49    subst h0
50    refine ⟨0, le_rfl, ?_⟩
51    intro H E _ _ _ _ _ _ _ _ _ S hS
52    refine ⟨fun _ => 0, fun D hD => ⟨?_, ?_, ?_, ?_, ?_⟩, ?_⟩
53    · intro I hI; rfl
54    · intro I hI Ψ; simp [HRS.Uniform.Q45.tensorPairing_eq]
55    · simp [VectorMeasure.variation_zero]
56    · intro t₁ t₂ ℓ₁ ℓ₂ I hI
57      have hmin : (min (t₁ : ℝ) t₂) ≤ 0 := (min_le_left _ _).trans t₁.2.2
58      have hz : volume (I ∩ Set.Icc (0 : ℝ) (min (t₁ : ℝ) t₂)) = 0 := by
59        refine measure_mono_null Set.inter_subset_right ?_
60        rw [Real.volume_Icc, sub_zero, ENNReal.ofReal_eq_zero]
61        exact hmin
62      rw [setIntegral_measure_zero _ hz]
63      simp [HRS.Uniform.Q45.tensorPairing_eq]
64    · exact continuousOn_const
65    · intro D D' hD hD' a b I hI; simp
66  · obtain ⟨C, hC0, hC⟩ := HRS.Uniform.Q45.cov_transport T K hpos
67    refine ⟨C, hC0, ?_⟩
68    intro H E _ _ _ _ _ _ _ _ _ S hS
69    obtain ⟨hev, h | h⟩ := hS
70    · -- two levels: pass to the closed span of the range of `j`
71      haveI : CompleteSpace (HRS.Uniform.Q45.E0 S) := (HRS.Uniform.Q45.E0_closed S).completeSpace_coe
72      haveI : SecondCountableTopology (HRS.Uniform.Q45.E0 S) := by
73        have hsep : TopologicalSpace.IsSeparable (HRS.Uniform.Q45.E0 S : Set E) := by
74          have := (TopologicalSpace.isSeparable_range S.j.continuous).closure
75          simpa [HRS.Uniform.Q45.E0, Submodule.topologicalClosure_coe, LinearMap.coe_range] using this
76        haveI := hsep.separableSpace
77        exact UniformSpace.secondCountable_of_separable _
78      letI : MeasurableSpace (HRS.Uniform.Q45.E0 S) := borel _
79      haveI : BorelSpace (HRS.Uniform.Q45.E0 S) := ⟨rfl⟩
80      refine hC H E (HRS.Uniform.Q45.E0 S) S (HRS.Uniform.Q45.subData S)
81        (HRS.Uniform.Q45.subData_maxHyp_twoLevel hpos hev h) (HRS.Uniform.Q45.E0 S).subtypeₗᵢ
82        (fun x => rfl) (fun r t hr hrt htT y => ?_)
83      exact HRS.Uniform.Q45.VE0_coe hev hr hrt htT y
84    · -- one dissipative level: `j` is onto, so `E` itself is separable
85      haveI : SecondCountableTopology E := by
86        have hsep : TopologicalSpace.IsSeparable (Set.univ : Set E) := by
87          have := TopologicalSpace.isSeparable_range S.j.continuous
88          have hr : Set.range S.j = Set.univ := Set.range_eq_univ.2 h.2.1
89          rwa [hr] at this
90        haveI := TopologicalSpace.isSeparable_univ_iff.1 hsep
91        exact UniformSpace.secondCountable_of_separable _
92      letI : MeasurableSpace E := borel _
93      haveI : BorelSpace E := ⟨rfl⟩
94      exact hC H E E S S ⟨hev, Or.inr h⟩ (LinearIsometry.id) (fun x => rfl) (fun r t hr hrt htT y => rfl)
