11## Representation decisions
12
13* The cutoff `χ₁` is the existing `HRS.PDE.IsFriedrichsCutoff` (reused, not redefined).
14* `mollifiedRiesz d s χ δ` is the trigonometric series `∑_k ĝ(k) χ₁(δk)² cos(2π k·x)`, i.e. the function
15  with Fourier coefficients `ĝ(k) χ₁(δk)²` (`ĝ` even, `rieszCoeff`).  Because `χ₁(δk) = 0` for `|δk| ≥ 2`
16  the series is a finite trigonometric sum, so the `tsum` is not a junk value.
17* The TeX range `1 ≤ n ≤ m + 2` (with `m` the regularity level of the surrounding section, the constant
18  `C_n` not depending on `m`) is stated as "for every `n ≥ 1`".
19* `|x|` on the torus is the quotient distance `KernelGrad.tdist x 0` (norm of the centered representative);
20  derivatives are taken on the covering space (`iteratedFDeriv` of the periodic lift) at a point `y` with
21  `mk y ≠ 0` (the bounds are meaningless at `x = 0`, where `|x|^{-s-n} = ∞`).
22* Constants `C_n`, `C` depend on `d, s, n` and the cutoff `χ₁` and are chosen before `δ` and `x`.

34/-- The mollified periodic Riesz/log kernel `g_δ` with `ĝ_δ(k) = ĝ(k) χ₁(δk)²`
35(Step 1 of `B:sec:adjoint-existence`, TeX M L1957): `g_δ(x) = ∑_k ĝ(k) χ₁(δk)² cos(2π k·x)`
36(a finite sum since `χ₁` has compact support). -/
37def mollifiedRiesz (d : ℕ) (s : ℝ) (χ : Euc d → ℝ) (δ : ℝ) (x : Torus d) : ℝ :=
38  ∑' k : Lattice d, rieszCoeff d s k * χ (δ • Lattice.toEuc k) ^ 2 * Torus.cosMode k x

61/-- the coefficients of `g_δ`: `fcoef g_δ k = ĝ(k) χ(δk)²` -/
62theorem fcoef_mollifiedRiesz (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {χ : Euc d → ℝ}
63    (hχ : PDE.IsFriedrichsCutoff χ) {δ : ℝ} (hδ : 0 < δ) (k : Lattice d) :
64    Torus.fcoef (mollifiedRiesz d s χ δ) k =
65      ((rieszCoeff d s k * χ (δ • Lattice.toEuc k) ^ 2 : ℝ) : ℂ) := by
66  have h2 : ∀ ξ : Euc d, 2 ≤ ‖ξ‖ → χ ξ = 0 := hχ.support
67  set b : Lattice d → ℂ := fun k => ((HDB.bC d s χ δ k : ℝ) : ℂ) with hb
68  have hT : PDE.IsTrigPoly ⌈2 / δ⌉₊ b :=
69    HDB.isTrigPoly_of_enorm_lt (fun k hk => HDB.bC_support h2 hδ k (by simpa [hb] using hk))
70  have hbeven : ∀ k, b (-k) = b k := by
71    intro k
72    simp only [hb, HDB.bC]
73    have h1 : rieszCoeff d s (-k) = rieszCoeff d s k := rieszCoeff_neg k
74    have h3 : Lattice.toEuc (-k) = -Lattice.toEuc k := by ext i; simp [Lattice.toEuc]
75    have h4 : ‖δ • Lattice.toEuc k‖ = ‖δ • Lattice.toEuc (-k)‖ := by rw [h3, smul_neg, norm_neg]
76    rw [h1, hχ.radial _ _ h4.symm]
77  -- the function is the trigonometric polynomial `trig b`
78  have hfun : ∀ x : Torus d, ((mollifiedRiesz d s χ δ x : ℝ) : ℂ) = PDE.Kern.trig b x := by
79    intro x
80    have hs1 : Summable fun k => b k * Torus.mode k x := PDE.Kern.summable_trig hT x
81    have hs2 : Summable fun k => b k * Torus.mode (-k) x := by
82      refine summable_of_ne_finset_zero (s := PDE.Kern.boxF d ⌈2 / δ⌉₊) fun k hk => ?_
83      have : b k = 0 := by
84        by_contra h
85        exact hk (PDE.Kern.mem_boxF.2 (hT k h))
86      simp [this]
87    have hneg : PDE.Kern.trig b x = ∑' k, b k * Torus.mode (-k) x := by
88      unfold PDE.Kern.trig
89      rw [← (Equiv.neg (Lattice d)).tsum_eq (fun k => b k * Torus.mode k x)]
90      exact tsum_congr fun k => by simp [hbeven]
91    have h2x : 2 * PDE.Kern.trig b x = ∑' k, b k * (2 * (Torus.cosMode k x : ℂ)) := by
92      have : ∀ k, b k * (2 * (Torus.cosMode k x : ℂ)) = b k * Torus.mode k x + b k * Torus.mode (-k) x := by
93        intro k
94        rw [Torus.mode_neg, Torus.cosMode, ← mul_add, Complex.add_conj]
95        push_cast; ring
96      simp_rw [this]
97      rw [hs1.tsum_add hs2, ← hneg]
98      unfold PDE.Kern.trig; ring
99    have : PDE.Kern.trig b x = ∑' k, b k * (Torus.cosMode k x : ℂ) := by
100      have h3 : ∑' k, b k * (2 * (Torus.cosMode k x : ℂ)) = 2 * ∑' k, b k * (Torus.cosMode k x : ℂ) := by
101        rw [← tsum_mul_left]
102        exact tsum_congr fun k => by ring
103      rw [h3] at h2x
104      exact mul_left_cancel₀ two_ne_zero h2x
105    rw [this]
106    unfold mollifiedRiesz
107    rw [Complex.ofReal_tsum]
108    exact tsum_congr fun k => by simp [hb, HDB.bC]
109  have hcoef : Torus.fcoef (mollifiedRiesz d s χ δ) k = PDE.coeffOfC (PDE.Kern.trig b) k := by
110    unfold Torus.fcoef
111    rw [PDE.coeffOfC_eq_integral']
112    refine integral_congr_ae (Filter.Eventually.of_forall fun x => ?_)
113    simp only
114    rw [Torus.mode_neg, hfun x]
115  rw [hcoef, PDE.Fried.coeffOfC_trig hT]
116  simp [hb, HDB.bC]

118/-- Lemma C.2 (`M:B:lem:mollified-kernels`, TeX M L12235-12237).  Let `γ_δ` be the mollified kernel,
119`0 ≤ s < d`.  Then
120* for `n ≥ 1` and `0 < δ ≤ 1`, `|∇^{⊗n} g_δ(x)| ≤ C_n (|x|^{-s-n} + 1)`, uniformly in `δ`;
121* for `n = 0` the same bound `|g_δ(x)| ≤ C (|x|^{-s} + 1)` holds when `s > 0`, while in the logarithmic
122  case `s = 0`, `|g_δ(x)| ≤ C (1 + |log(δ ∨ |x|)|)`;
123* in all cases `g_δ` is even and `ĝ_δ ≥ 0` (`ĝ_δ(k) = ĝ(k) χ₁(δk)² ≥ 0` is the Fourier coefficient of
124  `g_δ`, stated through `Torus.fcoef`).
125Pointwise bounds are for `x ≠ 0`.  Technical hypotheses: `1 ≤ d` (`d = 0` is degenerate). -/
126theorem mollified_kernels (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d)
127    {χ : Euc d → ℝ} (hχ : PDE.IsFriedrichsCutoff χ) :
128    (∀ n : ℕ, 1 ≤ n → ∃ C : ℝ, 0 ≤ C ∧ ∀ δ : ℝ, 0 < δ → δ ≤ 1 → ∀ y : Euc d, Torus.mk y ≠ 0 →
129        ‖iteratedFDeriv ℝ n (Torus.lift (mollifiedRiesz d s χ δ)) y‖ ≤
130          C * (KernelGrad.tdist (Torus.mk y) 0 ^ (-s - (n : ℝ)) + 1)) ∧
131    (0 < s → ∃ C : ℝ, 0 ≤ C ∧ ∀ δ : ℝ, 0 < δ → δ ≤ 1 → ∀ x : Torus d, x ≠ 0 →
132        |mollifiedRiesz d s χ δ x| ≤ C * (KernelGrad.tdist x 0 ^ (-s) + 1)) ∧
133    (s = 0 → ∃ C : ℝ, 0 ≤ C ∧ ∀ δ : ℝ, 0 < δ → δ ≤ 1 → ∀ x : Torus d, x ≠ 0 →
134        |mollifiedRiesz d s χ δ x| ≤ C * (1 + |Real.log (max δ (KernelGrad.tdist x 0))|)) ∧
135    (∀ δ : ℝ, 0 < δ → δ ≤ 1 →
136      (∀ x : Torus d, mollifiedRiesz d s χ δ (-x) = mollifiedRiesz d s χ δ x) ∧
137      ∀ k : Lattice d,
138        Torus.fcoef (mollifiedRiesz d s χ δ) k =
139            ((rieszCoeff d s k * χ (δ • Lattice.toEuc k) ^ 2 : ℝ) : ℂ) ∧
140          0 ≤ rieszCoeff d s k * χ (δ • Lattice.toEuc k) ^ 2) := by
141  have hcut := hχ
142  have hs : ContDiff ℝ ((⊤ : ℕ∞) : WithTop ℕ∞) χ := hχ.smooth
143  have h0 : ∀ ξ, 0 ≤ χ ξ ∧ χ ξ ≤ 1 := hχ.range
144  have h1 : ∀ ξ : Euc d, ‖ξ‖ ≤ 1 → χ ξ = 1 := hχ.one
145  have h2 : ∀ ξ : Euc d, 2 ≤ ‖ξ‖ → χ ξ = 0 := hχ.support
146  refine ⟨?_, ?_, ?_, ?_⟩
147  · intro n hn
148    obtain ⟨C, hC0, hC⟩ := HDB.deriv_bound_pos hd hsd hs h0 h1 h2 n (by
149      have : (1 : ℝ) ≤ n := by exact_mod_cast hn
150      linarith)
151    exact ⟨C, hC0, fun δ hδ hδ1 y hy => hC δ hδ hδ1 y hy⟩
152  · intro hs'
153    obtain ⟨C, hC0, hC⟩ := HDB.zeroth_bound_pos hd hsd hs h0 h1 h2 hs'
154    exact ⟨C, hC0, fun δ hδ hδ1 x hx => hC δ hδ hδ1 x hx⟩
155  · intro hs'
156    subst hs'
157    obtain ⟨C, hC0, hC⟩ := HDB.zeroth_bound_log hd hsd hs h0 h1 h2
158    exact ⟨C, hC0, fun δ hδ hδ1 x hx => hC δ hδ hδ1 x hx⟩
159  · intro δ hδ hδ1
160    refine ⟨mollifiedRiesz_neg s χ δ, fun k => ⟨fcoef_mollifiedRiesz hd hs0 hsd hχ hδ k, ?_⟩⟩
161    exact mul_nonneg (rieszCoeff_nonneg_of_one_le hd hs0 hsd k) (sq_nonneg _)
