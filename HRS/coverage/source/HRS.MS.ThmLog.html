313  refine ⟨⟨hρ.1, hfin, ?_⟩, t67_thermalDensity_of_fixed hS ha hβ hM hρ hfix⟩
314  intro ρ' hρ' hfin'
315  have hfae : ρ =ᵐ[volume] Tmap (t67K d) (t67gammaBar d a β) β ρ := ae_of_all _ hfix
316  have hlρ' := t67_lrho_integrable ha hρ' hfin'.2.1
317  have hmin := (minimizer_of_fixedpoint hS hβ hM hρ hfae).1 ρ' hρ'
318    (t67_ent_integrable ha hβ hρ' hfin'.1 hfin'.2.1) (t67_K_integrable_of_g hρ' hlρ' hfin'.2.2)
319  rw [t67_freeEnergy_eq ha hβ hρ.1 hfin, t67_freeEnergy_eq ha hβ hρ' hfin']
320  linarith

322/-- Every free-energy minimiser equals the fixed point a.e. -/
323theorem t67_minimizer_ae_eq (hS : ThmSetting (t67K d) (t67gammaBar d a β) B) (ha : 0 < a)
324    (hβ : 0 < β) (hM : 1 ≤ M) (hρ : ρ ∈ Cset (t67gammaBar d a β) M)
325    (hfix : ∀ x, ρ x = Tmap (t67K d) (t67gammaBar d a β) β ρ x) {l : Euc d → ℝ}
326    (hl : HRS.Whole.IsFreeEnergyMinimizer (rieszKernel d 0) (confinement a) β l) :
327    l =ᵐ[volume] ρ := by
328  have hfae : ρ =ᵐ[volume] Tmap (t67K d) (t67gammaBar d a β) β ρ := ae_of_all _ hfix
329  have hfin := t67_hasFiniteFreeEnergy_of_le hS ha hβ hM hρ
330  obtain ⟨hlp, hlfin, hlmin⟩ := hl
331  have h1 := hlmin ρ hρ.1 hfin
332  rw [t67_freeEnergy_eq ha hβ hρ.1 hfin, t67_freeEnergy_eq ha hβ hlp hlfin] at h1
333  have hlρ' := t67_lrho_integrable ha hlp hlfin.2.1
334  have hent := t67_ent_integrable ha hβ hlp hlfin.1 hlfin.2.1
335  have hKl := t67_K_integrable_of_g hlp hlρ' hlfin.2.2
336  obtain ⟨hA, hB'⟩ := minimizer_of_fixedpoint hS hβ hM hρ hfae
337  have h2 := hA l hlp hent hKl
338  have h3 : Gfun (t67K d) (t67gammaBar d a β) β l = Gfun (t67K d) (t67gammaBar d a β) β ρ := by
339    exact le_antisymm (by linarith) h2
340  exact hB' l hlp hent hKl h3

344/-- The data produced by THM-3/THM-4 for `K̃` and `γ̄`: an a.e. Euler fixed point in `C_M`. -/
345theorem t67_exists_fixed (hd : 1 ≤ d) {a β : ℝ} (ha : 0 < a) (hβ : 0 < β) :
346    ∃ (B M : ℝ) (μ : Euc d → ℝ), ThmSetting (t67K d) (t67gammaBar d a β) B ∧ 1 ≤ M ∧
347      Real.exp (β * B) ≤ M ∧ μ ∈ Cset (t67gammaBar d a β) M ∧
348      μ =ᵐ[volume] Tmap (t67K d) (t67gammaBar d a β) β μ := by
349  obtain ⟨B, hS⟩ := t67_setting hd ha hβ
350  have hM : 1 ≤ max 1 (Real.exp (β * B)) := le_max_left _ _
351  have hexp : Real.exp (β * B) ≤ max 1 (Real.exp (β * B)) := le_max_right _ _
352  obtain ⟨μ, hμ, hmin⟩ := exists_min_Cset hS hM hβ
353  exact ⟨B, _, μ, hS, hM, hexp, hμ, fixedpoint_of_min hS hβ hM hexp hμ hmin⟩

355/-- Everything about the `s = 0` thermal equilibrium: the pointwise representative `ρ₀ = thermalRep μ` is a bounded
356thermal equilibrium and every free-energy minimiser equals it a.e. -/
357theorem t67_spec_data (hd : 1 ≤ d) {a β : ℝ} (ha : 0 < a) (hβ : 0 < β) :
358    ∃ ρ₀ : Euc d → ℝ, HRS.Whole.IsThermalEquilibrium d 0 a β ρ₀ ∧ (∃ C, ∀ x, ρ₀ x ≤ C) ∧
359      ∀ l : Euc d → ℝ, HRS.Whole.IsFreeEnergyMinimizer (rieszKernel d 0) (confinement a) β l →
360        l =ᵐ[volume] ρ₀ := by
361  obtain ⟨B, M, μ, hS, hM, hexp, hμ, hfix⟩ := t67_exists_fixed hd ha hβ
362  have hρC := thermalRep_mem_Cset hS hβ.le hexp hμ.1
363  have hρfix := thermalRep_fixed hfix
364  have hEq := t67_equilibrium_of_fixed hS ha hβ hM hρC hρfix
365  refine ⟨_, hEq, ⟨M * (Real.exp (β * (a / 2 - 1 - Real.log (a / 2)) / 2) / t67cbar d a β), fun x => ?_⟩,
366    fun l hl => t67_minimizer_ae_eq hS ha hβ hM hρC hρfix hl⟩
367  exact (hρC.2 x).trans (mul_le_mul_of_nonneg_left (t67gammaBar_le ha hβ x) (by linarith))

370minimiser (a.e.), and is `thermalDensity d 0 a β` (a.e.). -/
