51/-- Threshold arithmetic (M L3152-3153): for `s = d - 2`, `Λ ≥ 1`, `C₀ > 0`:
52`(δ (NΛ)^{2/d})⁻¹ ≤ (δ C₀)⁻¹ o_N(Λ)`.  PROVED here (it is elementary). -/
53theorem lt9_inv_le_oN (hd : 2 ≤ d) {Λ : ℝ} (hΛ : 1 ≤ Λ) {N : ℕ} (hN : 1 ≤ N) {δ C₀ : ℝ} (hδ : 0 < δ)
54    (hC₀ : 0 < C₀) :
55    (δ * ((N : ℝ) * Λ) ^ (2 / (d : ℝ)))⁻¹ ≤ (δ * C₀)⁻¹ * oNΛ d ((d : ℝ) - 2) C₀ Λ N := by
56  have hdpos : (0 : ℝ) < d := by exact_mod_cast (by omega : 0 < d)
57  have hNpos : (0 : ℝ) < N := by exact_mod_cast hN
58  have hΛpos : 0 < Λ := by linarith
59  have hNΛ1 : (1 : ℝ) ≤ (N : ℝ) * Λ := by
60    have : (1 : ℝ) ≤ N := by exact_mod_cast hN
61    nlinarith
62  set x : ℝ := 2 / (d : ℝ) with hx
63  have hx0 : 0 ≤ x := by positivity
64  have hsd : ((d : ℝ) - 2) / d = 1 - x := by rw [hx]; field_simp
65  have hlog : 0 ≤ (if ((d : ℝ) - 2) = 0 then Real.log (N * Λ) / (2 * d * N) else 0) := by
66    split_ifs
67    · exact div_nonneg (Real.log_nonneg hNΛ1) (by positivity)
68    · exact le_rfl
69  unfold oNΛ
70  rw [hsd]
71  have h1 : (1 - x) - 1 = -x := by ring
72  rw [h1]
73  have hL : (δ * ((N : ℝ) * Λ) ^ x)⁻¹ = δ⁻¹ * ((N : ℝ) ^ (-x) * Λ ^ (-x)) := by
74    rw [mul_inv, Real.mul_rpow hNpos.le hΛpos.le, mul_inv, Real.rpow_neg hNpos.le, Real.rpow_neg hΛpos.le]
75  have hR : C₀ * Λ ^ (1 - x) * (N : ℝ) ^ (-x) ≥ C₀ * Λ ^ (-x) * (N : ℝ) ^ (-x) := by
76    have : Λ ^ (-x) ≤ Λ ^ (1 - x) := Real.rpow_le_rpow_of_exponent_le hΛ (by linarith)
77    have hN' : 0 ≤ (N : ℝ) ^ (-x) := Real.rpow_nonneg hNpos.le _
78    exact mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_left this hC₀.le) hN'
79  have hK : 0 ≤ (δ * C₀)⁻¹ := by positivity
80  rw [hL]
81  calc δ⁻¹ * ((N : ℝ) ^ (-x) * Λ ^ (-x))
82      = (δ * C₀)⁻¹ * (C₀ * Λ ^ (-x) * (N : ℝ) ^ (-x)) := by
83        field_simp
84    _ ≤ (δ * C₀)⁻¹ * (C₀ * Λ ^ (1 - x) * (N : ℝ) ^ (-x)) := mul_le_mul_of_nonneg_left hR hK
85    _ ≤ (δ * C₀)⁻¹ * ((if ((d : ℝ) - 2) = 0 then Real.log (N * Λ) / (2 * d * N) else 0) +
86          C₀ * Λ ^ (1 - x) * (N : ℝ) ^ (-x)) := mul_le_mul_of_nonneg_left (by linarith) hK

91theorem lowtemp_coulomb_bound_mono (hd : 2 ≤ d) (T Λ ΛV Mb : ℝ) (hT : 0 ≤ T) :
92    ∃ C₀min C : ℝ, 0 ≤ C₀min ∧ 0 < C ∧ ∀ C₀ : ℝ, C₀min ≤ C₀ →
93      ∀ (D : DynData d) (N : ℕ), 2 ≤ N →
94      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
95        (ρ : ℝ → Torus d → ℝ) (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d))
96        (X : ℝ≥0 → Ω → Config d N),
97        LowTempHyp P D N T Λ ΛV Mb ρ X₀ W X →
98        Integrable (fun ω => correctedEnergy D C₀ Λ ρ (X 0 ω) 0) P →
99        0 ≤ ∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P ∧
100        ∀ t ∈ Set.Icc 0 T,
101          ∫⁻ ω, ENNReal.ofReal (correctedEnergy D C₀ Λ ρ (X (Real.toNNReal t) ω) t) ∂P ≤
102            ENNReal.ofReal (C * (∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P + D.invβ)) := by
103  obtain ⟨C₀min, κ, κ', hC₀min, hκ, hκ', h6⟩ := lt6_drift_bound_proof hd T Λ ΛV Mb
104  obtain ⟨C, hC, h8⟩ := lt8_gronwall_fatou_proof (d := d) T κ κ' hT hκ hκ'
105  refine ⟨C₀min, C, hC₀min, hC, fun C₀ hC₀ D N hN Ω _ P _ ρ X₀ W X h hint => ?_⟩
106  have hcol : ∀ᵐ ω ∂P, ∀ s ∈ Set.Icc (0 : ℝ) T, X (Real.toNNReal s) ω ∉ collisionSet d N := by
107    filter_upwards [h.sol.no_collision] with ω hω s hs
108    exact hω _ (by rw [Real.coe_toNNReal _ hs.1]; exact hs.2)
109  have hpath : ∀ᵐ ω ∂P, ∀ s ∈ Set.Icc (0 : ℝ) T,
110      0 ≤ correctedEnergy D C₀ Λ ρ (X (Real.toNNReal s) ω) s ∧
111      ltDrift D ρ s (X (Real.toNNReal s) ω) ≤
112        κ * correctedEnergy D C₀ Λ ρ (X (Real.toNNReal s) ω) s + κ' * D.invβ := by
113    filter_upwards [hcol] with ω hω s hs
114    exact h6 C₀ hC₀ D N hN ρ (lowTempHyp_det h) s hs _ (hω s hs)
115  have hnn : 0 ≤ ∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P := by
116    refine integral_nonneg_of_ae ?_
117    filter_upwards [hpath] with ω hω
118    have := (hω 0 ⟨le_rfl, hT⟩).1
119    simpa using this
120  refine ⟨hnn, fun t ht => ?_⟩
121  obtain ⟨ℱ, W', hF⟩ := h.filtered
122  obtain ⟨τ, hτ⟩ := lt4_stopping_proof hd h hF
123  have hid := lt5_stopped_identity_proof hd hN hT h hF hτ C₀ hint
124  exact h8 P D N Λ C₀ ρ X ℱ τ hτ hid hpath hint t ht

143theorem lowtemp_coulomb_threshold_proof_aux (hd : 2 ≤ d) (T Λ ΛV Mb δ : ℝ) (hT : 0 ≤ T) (hδ : 0 < δ) :

145      ∀ (D : DynData d) (N : ℕ), 2 ≤ N →
146      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
147        (ρ : ℝ → Torus d → ℝ) (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d))
148        (X : ℝ≥0 → Ω → Config d N),
149        LowTempHyp P D N T Λ ΛV Mb ρ X₀ W X →
150        Integrable (fun ω => correctedEnergy D C₀ Λ ρ (X 0 ω) 0) P →
151        D.invβ ≤ (δ * (N * Λ) ^ (2 / (d : ℝ)))⁻¹ →
152        ∀ t ∈ Set.Icc 0 T,
153          ∫⁻ ω, ENNReal.ofReal (correctedEnergy D C₀ Λ ρ (X (Real.toNNReal t) ω) t) ∂P ≤
154            ENNReal.ofReal (Cδ * (∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P +
155              oNΛ d D.s C₀ Λ N)) := by
156  obtain ⟨C₀min, C, h0, hC, hm⟩ := lowtemp_coulomb_bound_mono hd T Λ ΛV Mb hT
157  set C₀ : ℝ := max C₀min 1 with hC₀def
158  have hC₀pos : 0 < C₀ := lt_of_lt_of_le one_pos (le_max_right _ _)
159  have hC₀min : C₀min ≤ C₀ := le_max_left _ _
160  set K : ℝ := (δ * C₀)⁻¹ with hK
161  have hK0 : 0 ≤ K := by positivity
162  refine ⟨C₀, C * max 1 K, hC₀pos.le, by positivity, fun D N hN Ω _ P _ ρ X₀ W X h hint hβ t ht => ?_⟩
163  obtain ⟨hnn, hb⟩ := hm C₀ hC₀min D N hN P ρ X₀ W X h hint
164  refine (hb t ht).trans (ENNReal.ofReal_le_ofReal ?_)
165  have hΛ : 1 ≤ Λ := h.Λ_ge
166  have hs : D.s = (d : ℝ) - 2 := h.coulomb
167  have hβK : D.invβ ≤ K * oNΛ d D.s C₀ Λ N := by
168    have := lt9_inv_le_oN hd hΛ (by omega : 1 ≤ N) hδ hC₀pos
169    rw [hs]
170    exact hβ.trans this
171  have ho : 0 ≤ oNΛ d D.s C₀ Λ N := by
172    rw [hs]; exact lt9_oN_nonneg hd hΛ (by omega : 1 ≤ N) hC₀pos.le
173  have hm1 : (1 : ℝ) ≤ max 1 K := le_max_left _ _
174  have hmK : K ≤ max 1 K := le_max_right _ _
175  have hCo : 0 < C := hC
176  have h1 : ∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P + D.invβ ≤
177      max 1 K * (∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P + oNΛ d D.s C₀ Λ N) := by
178    nlinarith [mul_nonneg (sub_nonneg.2 hm1) hnn, mul_nonneg (sub_nonneg.2 hmK) ho]
179  calc C * (∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P + D.invβ)
180      ≤ C * (max 1 K * (∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P + oNΛ d D.s C₀ Λ N)) :=
181        mul_le_mul_of_nonneg_left h1 hCo.le
