19lemma rpa_increment_mono {Cf : WholeCoeff d} {T : ℝ} {q m₀ : ℕ} {C C' : ℝ}
20    (h : IncrementBoundRd Cf T q m₀ C) (hCC : C ≤ C') : IncrementBoundRd Cf T q m₀ C' := by
21  intro r s' t hr hrs hst htT φ hφ l hl idx x
22  refine (h r s' t hr hrs hst htT φ hφ l hl idx x).trans ?_
23  have h1 : 0 ≤ |t - s'| * jbracket x * HRS.Whole.sobolevNorm (q : ℝ) φ :=
24    mul_nonneg (mul_nonneg (abs_nonneg _) (by unfold jbracket; positivity)) (Real.sqrt_nonneg _)
25  nlinarith

27lemma rpa_kernel_mono {Cf : WholeCoeff d} {T : ℝ} {q : ℕ} {C C' : ℝ}
28    (h : KernelBoundRd Cf T q C) (hCC : C ≤ C') : KernelBoundRd Cf T q C' := by
29  refine ⟨h.1, fun Kern hK j r s' t hr hrs hst htT x => ?_⟩
30  obtain ⟨h1, h2⟩ := h.2 Kern hK j r s' t hr hrs hst htT x
31  refine ⟨h1.trans ?_, h2.trans hCC⟩
32  have : 0 ≤ |t - s'| * (1 + ‖x‖) := mul_nonneg (abs_nonneg _) (by positivity)
33  nlinarith

35/-- The statement of `path_propagator_increment_rd` without the (unused) hypothesis `1 ≤ m₀`. -/
36theorem rpa_core (s : ℝ) (fd : FieldDensityRd d s)
37    (ν₀ MB Λ ΛV G₀ νbar T Kadj Cw : ℝ) (m₀ : ℕ) (hm₀' : m₀ ≤ mStar d) :
38    ∃ C : ℝ, ∀ (S : WholeMainData d), S.s = s → S.T = T → (∀ N, S.invβ N ≤ νbar) →
39      fd.Holds S ν₀ → PDE.MatBound MB S.M → MeanFieldRegularityRd S (fd.Gamma + 5) Λ →
40      VHypRd (fd.Gamma + 5) ΛV G₀ S.V → PathAdjointInput S fd.Gamma m₀ Kadj Cw →
41      (∀ N : ℕ, 1 ≤ N →
42        IncrementBoundRd (S.coeff N) T (fd.Gamma + 2) m₀ C ∧
43        KernelBoundRd (S.coeff N) T (fd.Gamma + 2) C) ∧
44      ∀ L : LimitCoeffsRd d, LimitPathHyp S L fd ν₀ Λ Kadj Cw m₀ → L.invβ ≤ νbar →
45        IncrementBoundRd (L.coeff S) T (fd.Gamma + 2) m₀ C ∧
46        KernelBoundRd (L.coeff S) T (fd.Gamma + 2) C := by
47  classical
48  by_cases hex : ∃ S0 : WholeMainData d, S0.s = s ∧ fd.Holds S0 ν₀
49  · obtain ⟨S0, hS0s, hS0⟩ := hex
50    have hΓ : (mStar d : ℝ) + d / 2 + 4 < ((fd.Gamma : ℝ) + 2) := by
51      have h := hS0 1 le_rfl
52      rw [hS0s] at h
53      exact rpi_hGamma fd h
54    obtain ⟨Cgen, hCgen0, hgen⟩ :=
55      rpi_increment (s := s) S0.hd (hS0s ▸ S0.hs0) (hS0s ▸ S0.hsd) fd.Gamma m₀ hm₀' hΓ MB Λ ΛV G₀ νbar
56    set Cinc : ℝ := max Cw 0 * Real.exp (max Cw 0 * T) * Cgen with hCinc
57    set Ck : ℝ := max (max Kadj Cinc) 0 with hCk
58    refine ⟨max Cinc Ck, fun S hs hT hν hHold hMB hMF hV hP => ?_⟩
59    have hd := S.hd
60    -- the increment and kernel bounds for one coefficient, from the pieces
61    have key : ∀ (D : WholeDyn d) (ρ : ℝ → Euc d → ℝ), D.s = s → 0 ≤ D.invβ → D.invβ ≤ νbar →
62        VHypRd (fd.Gamma + 5) ΛV G₀ D.V → HRS.PDE.MatBound MB D.M →
63        IsRegularMFRd s (fd.Gamma + 2 + 3) Λ T ρ →
64        AdjointInputOneD (((fd.Gamma + 2 : ℕ) : ℝ)) (m₀ + 1) (D.coeff ρ) T Kadj →
65        (∃ U, IsWeightedExtension (D.coeff ρ) (m₀ + 1) (fd.Gamma + 2) T Cw U) →
66        IncrementBoundRd (D.coeff ρ) T (fd.Gamma + 2) m₀ (max Cinc Ck) ∧
67        KernelBoundRd (D.coeff ρ) T (fd.Gamma + 2) (max Cinc Ck) := by
68      intro D ρ hDs hν0 hνD hVD hMD hreg hadj ⟨U, hU⟩
69      have hinc := hgen D ρ T Kadj Cw U hDs hMD hVD hreg hν0 hνD hadj hU
70      refine ⟨rpa_increment_mono hinc (le_max_left _ _), rpa_kernel_mono ?_ (le_max_right _ _)⟩
71      exact rpk_kernel fd.Gamma m₀ (D.coeff ρ) T Kadj Cinc hadj hinc
72    refine ⟨fun N hN => ?_, fun L hL hLν => ?_⟩
73    · have hDs : (S.dyn N).s = s := hs
74      have hreg : IsRegularMFRd s (fd.Gamma + 2 + 3) Λ T (S.ρ N) := by
75        have h : IsRegularMFRd S.s (fd.Gamma + 5) Λ S.T (S.ρ N) := hMF.regular N hN
76        rw [hs, hT] at h
77        exact h
78      exact key (S.dyn N) (S.ρ N) hDs (S.hinvβ N) (hν N) hV hMB
79        hreg (hT ▸ hP.adjoint N hN) (hT ▸ hP.weighted N hN)
80    · have hDs : (L.dyn S).s = s := hs
81      have hLreg : IsRegularMFRd s (fd.Gamma + 2 + 3) Λ T L.ρ := by
82        have h : IsRegularMFRd S.s (fd.Gamma + 2 + 3) Λ S.T L.ρ := hL.coeff.regular
83        rw [hs, hT] at h
84        exact h
85      exact key (L.dyn S) L.ρ hDs L.hinvβ hLν hV hMB hLreg (hT ▸ hL.adjoint) (hT ▸ hL.weighted)
86  · refine ⟨0, fun S hs hT hν hHold => ?_⟩
87    exact absurd ⟨S, hs, hHold⟩ hex

89theorem path_propagator_increment_rd_proof (s : ℝ) (fd : FieldDensityRd d s)
90    (ν₀ MB Λ ΛV G₀ νbar T Kadj Cw : ℝ) (m₀ : ℕ) (_hm₀ : 1 ≤ m₀) (hm₀' : m₀ ≤ mStar d) :
91    ∃ C : ℝ, ∀ (S : WholeMainData d), S.s = s → S.T = T → (∀ N, S.invβ N ≤ νbar) →
92      fd.Holds S ν₀ → PDE.MatBound MB S.M → MeanFieldRegularityRd S (fd.Gamma + 5) Λ →
93      VHypRd (fd.Gamma + 5) ΛV G₀ S.V → PathAdjointInput S fd.Gamma m₀ Kadj Cw →
94      (∀ N : ℕ, 1 ≤ N →
95        IncrementBoundRd (S.coeff N) T (fd.Gamma + 2) m₀ C ∧
96        KernelBoundRd (S.coeff N) T (fd.Gamma + 2) C) ∧
97      ∀ L : LimitCoeffsRd d, LimitPathHyp S L fd ν₀ Λ Kadj Cw m₀ → L.invβ ≤ νbar →
98        IncrementBoundRd (L.coeff S) T (fd.Gamma + 2) m₀ C ∧
99        KernelBoundRd (L.coeff S) T (fd.Gamma + 2) C :=
100  rpa_core s fd ν₀ MB Λ ΛV G₀ νbar T Kadj Cw m₀ hm₀'
