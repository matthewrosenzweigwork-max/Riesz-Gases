22/-- The interaction integral is bounded uniformly over bounded probability densities:
23`|∬ g(x - y) dμ dμ| ≤ max Λ 0 · ∫ |g|` for `μ = ρ dx` with `0 ≤ ρ ≤ Λ`, `∫ρ = 1`. -/
24theorem abs_interaction_le {g : Torus d → ℝ} (hg : Measurable g) (hgi : Integrable g)
25    (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ) (hΛ : ∀ x, ρ x ≤ Λ) :
26    |∫ z : Torus d × Torus d, g (z.1 - z.2) ∂((withDensityReal ρ).prod (withDensityReal ρ))| ≤
27      max Λ 0 * ∫ y, ‖g y‖ := by
28  have := isProb_withDensityReal hρ
29  have hm : Measurable (fun z : Torus d × Torus d => ‖g (z.1 - z.2)‖ₑ) :=
30    (hg.comp (measurable_fst.sub measurable_snd)).enorm
31  have hL : ∫⁻ y, ‖g y‖ₑ < ∞ := hgi.2
32  have h1 := norm_integral_le_lintegral_norm
33    (μ := (withDensityReal ρ).prod (withDensityReal ρ)) (fun z : Torus d × Torus d => g (z.1 - z.2))
34  simp only [ofReal_norm] at h1
35  have h2 : ∫⁻ z : Torus d × Torus d, ‖g (z.1 - z.2)‖ₑ ∂((withDensityReal ρ).prod (withDensityReal ρ))
36      ≤ ENNReal.ofReal (max Λ 0) * ∫⁻ y, ‖g y‖ₑ := by
37    rw [lintegral_prod _ hm.aemeasurable]
38    calc ∫⁻ x, ∫⁻ y, ‖g (x - y)‖ₑ ∂withDensityReal ρ ∂withDensityReal ρ
39        ≤ ∫⁻ _x, (ENNReal.ofReal (max Λ 0) * ∫⁻ y, ‖g y‖ₑ) ∂withDensityReal ρ :=
40          lintegral_mono fun x => (KernelAux2.lintegral_conv_le ρ Λ hΛ (fun z => ‖g z‖ₑ) x).trans
41            (mul_le_mul_of_nonneg_right (ENNReal.ofReal_le_ofReal (le_max_left Λ 0)) zero_le)
42      _ = ENNReal.ofReal (max Λ 0) * ∫⁻ y, ‖g y‖ₑ := by rw [lintegral_const, measure_univ, mul_one]
43  rw [← Real.norm_eq_abs]
44  refine h1.trans ?_
45  rw [integral_norm_eq_lintegral_enorm hgi.aestronglyMeasurable]
46  calc _ ≤ (ENNReal.ofReal (max Λ 0) * ∫⁻ y, ‖g y‖ₑ).toReal :=
47        ENNReal.toReal_mono (ENNReal.mul_ne_top ENNReal.ofReal_ne_top hL.ne) h2
48    _ = max Λ 0 * (∫⁻ y, ‖g y‖ₑ).toReal := by
49        rw [ENNReal.toReal_mul, ENNReal.toReal_ofReal (le_max_right _ _)]

55/-- Family version of `PreparationAux.uniform_of_small_negative`. The exponent `p` (from `a` and `B`)
56and the constant `C` (from `CK`, `L`, `a`, `B`, `B₀`) are chosen before the reference measure `μ`.
57The partition bound `CK` is assumed on `[-a, (1 + a)|B₀| + |B|] ⊇ [-a, p|B₀| + |B|]`, and the
58interaction integral is assumed bounded by `L`. -/
59theorem uniform_of_partition_bound {g : Torus d → ℝ} (hg : Measurable g) (a : ℝ) (ha : 0 < a)
60    (CK L B B₀ : ℝ) :
61    ∃ p : ℝ, 1 < p ∧ ∃ C : ℝ, ∀ (μ : Measure (Torus d)) [IsProbabilityMeasure μ],
62      Integrable (fun z : Torus d × Torus d => g (z.1 - z.2)) (μ.prod μ) →
63      (∀ N : ℕ, 2 ≤ N → ∀ γ ∈ Set.Icc (-a) ((1 + a) * |B₀| + |B|), KE g N γ μ ≤ ENNReal.ofReal CK) →
64      |∫ z : Torus d × Torus d, g (z.1 - z.2) ∂(μ.prod μ)| ≤ L →
65      ∀ N : ℕ, 2 ≤ N → ∀ β ∈ Set.Icc (0 : ℝ) B, ∀ β₀ ∈ Set.Icc (0 : ℝ) B₀,
66        modulatedGibbs g N β₀ μ ≪ modulatedGibbs g N β μ ∧
67          ∫⁻ X, ((modulatedGibbs g N β₀ μ).rnDeriv (modulatedGibbs g N β μ) X) ^ p
68              ∂(modulatedGibbs g N β μ) ≤ ENNReal.ofReal C := by
69  set q : ℝ := a / (2 * (|B| + 1)) with hq
70  have hq0 : 0 < q := by positivity
71  have hqa : q ≤ a / 2 := by
72    rw [hq, div_le_div_iff₀ (by positivity) (by norm_num)]
73    nlinarith [abs_nonneg B]
74  set p : ℝ := 1 + q with hp
75  have hp1 : 1 < p := by linarith
76  have hp0 : 0 ≤ p := by linarith
77  have hB0abs : 0 ≤ |B₀| := abs_nonneg _
78  have hBabs : 0 ≤ |B| := abs_nonneg _
79  have hpB : 0 ≤ p * |B₀| := mul_nonneg hp0 hB0abs
80  set B₁ : ℝ := p * |B₀| + |B| with hB₁
81  have hB12 : B₁ ≤ (1 + a) * |B₀| + |B| := by
82    have : p * |B₀| ≤ (1 + a) * |B₀| := mul_le_mul_of_nonneg_right (by linarith) hB0abs
83    linarith
84  set Cp : ℝ := max CK 1 with hCp
85  have hCp1 : 1 ≤ Cp := le_max_right _ _
86  set m₀ : ℝ := Real.exp (-(|B₀| * L / 2)) with hm₀
87  have hm0 : 0 < m₀ := Real.exp_pos _
88  refine ⟨p, hp1, Cp ^ (p - 1) * Cp / m₀ ^ p, fun μ _ hI hT hGL N hN β hβ β₀ hβ₀ => ?_⟩
89  set G := ∫ z : Torus d × Torus d, g (z.1 - z.2) ∂(μ.prod μ) with hG
90  have hC₁ : ∀ N : ℕ, 2 ≤ N → ∀ γ ∈ Set.Icc (-a) B₁, KE g N γ μ ≤ ENNReal.ofReal CK :=
91    fun N hN γ hγ => hT N hN γ ⟨hγ.1, hγ.2.trans hB12⟩
92  obtain ⟨hβ1, hβ2⟩ := hβ
93  obtain ⟨hβ₀1, hβ₀2⟩ := hβ₀
94  -- range facts
95  have hβB1 : β ∈ Set.Icc (-a) B₁ := ⟨by linarith, by have := le_abs_self B; linarith⟩
96  have hβ₀B1 : β₀ ∈ Set.Icc (-a) B₁ := ⟨by linarith, by
97    have h1 := le_abs_self B₀
98    have h2 : |B₀| ≤ p * |B₀| := by nlinarith
99    linarith⟩
100  have hγ : β - p * (β - β₀) ∈ Set.Icc (-a) B₁ := by
101    have e : β - p * (β - β₀) = p * β₀ - q * β := by rw [hp]; ring
102    rw [e]
103    constructor
104    · have h1 : q * β ≤ q * |B| := mul_le_mul_of_nonneg_left (hβ2.trans (le_abs_self B)) hq0.le
105      have h2 : q * |B| ≤ a := by
106        rw [hq]
107        rw [div_mul_eq_mul_div, div_le_iff₀ (by positivity)]
108        nlinarith
109      have h3 : 0 ≤ p * β₀ := by positivity
110      linarith
111    · have h1 : p * β₀ ≤ p * |B₀| := mul_le_mul_of_nonneg_left (hβ₀2.trans (le_abs_self B₀)) hp0
112      have h2 : 0 ≤ q * β := by positivity
113      linarith
114  have hfin : ∀ γ ∈ Set.Icc (-a) B₁, KE g N γ μ ≠ ∞ := fun γ hγ =>
115    ne_top_of_le_ne_top ENNReal.ofReal_ne_top (hC₁ N hN γ hγ)
116  have hZle : ∀ γ ∈ Set.Icc (-a) B₁, Zf (Measure.pi fun _ : Fin N => μ) (Wen g N μ) γ ≤ Cp := by
117    intro γ hγ
118    rw [← toReal_KE hg]
119    calc (KE g N γ μ).toReal ≤ (ENNReal.ofReal CK).toReal :=
120          ENNReal.toReal_mono ENNReal.ofReal_ne_top (hC₁ N hN γ hγ)
121      _ = max CK 0 := ENNReal.toReal_ofReal'
122      _ ≤ Cp := max_le_max le_rfl zero_le_one
123  have hf0 := log_Zf_ge hg hI (by omega : 1 ≤ N) (hfin β₀ hβ₀B1)
124  have hZ0ge : m₀ ≤ Zf (Measure.pi fun _ : Fin N => μ) (Wen g N μ) β₀ := by
125    have hZpos := Zf_pos hg (hfin β₀ hβ₀B1)
126    have h1 : -(|B₀| * L / 2) ≤ Real.log (Zf (Measure.pi fun _ : Fin N => μ) (Wen g N μ) β₀) := by
127      have hb : β₀ ≤ |B₀| := hβ₀2.trans (le_abs_self B₀)
128      have e1 : β₀ * (-|G|) ≤ β₀ * G := mul_le_mul_of_nonneg_left (neg_abs_le G) hβ₀1
129      have e2 : β₀ * |G| ≤ |B₀| * L := mul_le_mul hb hGL (abs_nonneg _) hB0abs
130      nlinarith
131    calc m₀ = Real.exp (-(|B₀| * L / 2)) := rfl
132      _ ≤ Real.exp (Real.log _) := Real.exp_le_exp.2 h1
133      _ = _ := Real.exp_log hZpos
134  obtain ⟨hac, hint⟩ := lintegral_rnDeriv_rpow (N := N) hg hp1.le (hfin β hβB1) (hfin β₀ hβ₀B1)
135  refine ⟨hac, ?_⟩
136  rw [hint, ← KE_eq]
137  have hKγ : KE g N (β - p * (β - β₀)) μ =
138      ENNReal.ofReal (Zf (Measure.pi fun _ : Fin N => μ) (Wen g N μ) (β - p * (β - β₀))) := by
139    rw [← toReal_KE hg, ENNReal.ofReal_toReal (hfin _ hγ)]
140  have hZb := Zf_pos hg (hfin β hβB1)
141  have hZg := Zf_pos hg (hfin _ hγ)
142  have hZ0pos : 0 < Zf (Measure.pi fun _ : Fin N => μ) (Wen g N μ) β₀ := lt_of_lt_of_le hm0 hZ0ge
143  have hr : 0 ≤ (Zf (Measure.pi fun _ : Fin N => μ) (Wen g N μ) β /
144      Zf (Measure.pi fun _ : Fin N => μ) (Wen g N μ) β₀) ^ p :=
145    Real.rpow_nonneg (div_nonneg hZb.le hZ0pos.le) p
146  rw [hKγ, ← ENNReal.ofReal_inv_of_pos hZb, ← ENNReal.ofReal_mul hr,
147    ← ENNReal.ofReal_mul (inv_nonneg.2 hZb.le)]
148  exact ENNReal.ofReal_le_ofReal
149    (real_ineq hZb hm0 hZ0ge hp1.le (hZle β hβB1) hZg (hZle _ hγ))

153/-- Proposition 4.25, `O:eq:prep-uniform`, uniformly over all bounded probability densities that satisfy
154the same density bound `Λ` and the same partition bound `CK` on `[-a, (1 + a)|B₀| + |B|]`. -/
155theorem log_prep_uniform_family (hd : 2 ≤ d) (Λ a CK B B₀ : ℝ) (ha : 0 < a) :
156    ∃ p : ℝ, 1 < p ∧ ∃ C : ℝ, ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
157      (∀ N : ℕ, 2 ≤ N → ∀ γ ∈ Set.Icc (-a) ((1 + a) * |B₀| + |B|),
158        modNormalizerE (periodicRiesz d 0) N γ (withDensityReal ρ) ≤ ENNReal.ofReal CK) →
159      ∀ N : ℕ, 2 ≤ N → ∀ β ∈ Set.Icc (0 : ℝ) B, ∀ β₀ ∈ Set.Icc (0 : ℝ) B₀,
160        modulatedGibbs (periodicRiesz d 0) N β₀ (withDensityReal ρ) ≪
161            modulatedGibbs (periodicRiesz d 0) N β (withDensityReal ρ) ∧
162          ∫⁻ X, ((modulatedGibbs (periodicRiesz d 0) N β₀ (withDensityReal ρ)).rnDeriv
163              (modulatedGibbs (periodicRiesz d 0) N β (withDensityReal ρ)) X) ^ p
164              ∂(modulatedGibbs (periodicRiesz d 0) N β (withDensityReal ρ)) ≤
165            ENNReal.ofReal C := by
166  have hs0 : (0 : ℝ) ≤ 0 := le_rfl
167  have hsd : (0 : ℝ) < d := by exact_mod_cast (by omega : 0 < d)
168  have hg : Measurable (periodicRiesz d 0) := PoleCavityAux.measurable_periodicRiesz hd hs0 hsd
169  have hgi : Integrable (periodicRiesz d 0) := KernelAux.integrable_periodicRiesz hd hs0 hsd
170  obtain ⟨p, hp, C, hC⟩ := uniform_of_partition_bound (g := periodicRiesz d 0) hg a ha CK
171    (max Λ 0 * ∫ y, ‖periodicRiesz d 0 y‖) B B₀
172  refine ⟨p, hp, C, fun ρ hρ hΛ hPart => ?_⟩
173  have := isProb_withDensityReal hρ
174  have hI := PreparationAux.integrable_pair_withDensity hg hgi ρ Λ hΛ (ν := withDensityReal ρ) rfl
175  exact hC (withDensityReal ρ) hI hPart (abs_interaction_le hg hgi ρ hρ Λ hΛ)
