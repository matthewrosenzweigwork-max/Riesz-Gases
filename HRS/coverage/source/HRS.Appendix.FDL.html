20* **The profile `ψ`** (L12360) is fixed but arbitrary: `IsFDLProfile ψ` (radial, `C^∞`, `≥ 0`,
21  `supp ψ ⊂ B(0,1/2)`, `ψ > 0` on `B(0,1/4)`), `φ = ψ * ψ` (`fdlPhi`, convolution), `φ_t = t^{-d} φ(·/t)`
22  (`fdlPhiScale`).  "Constants `C = C(d,s,j,φ)`" are `∃ C` after `ψ`, `s`, `b`, `j`, before the scale `λ`
23  and the point.
24* **The constant `b_{d,s}`** is not given a closed form: `IsFDLConst ψ s b` says that `b > 0` makes the
25  representations of Lemma D.1 hold (Euclidean and torus).  Lemma D.1 asserts `∃ b, IsFDLConst ψ s b`
26  ("there is `b_{d,s} > 0`"); `b` is then unique (it is determined by the normalisation
27  `g = |x|^{-s}` resp. `-log|x|`), and the later statements take it as a hypothesis `hb`.
28* **Two settings.**  The M-manuscript works on `𝔻 = 𝕋^d` (statements D.2-D.6 on the torus, as the rest of
29  `Energy/Coercivity.lean`); Lemma D.1 and the real-space part of Lemma D.3 are also stated on `ℝ^d`
30  because the TeX says "on `ℝ^d`" explicitly (Euclidean `g` is `HRS.Whole.rieszKernel`).  NOT stated:
31  the Euclidean Fourier items of D.3(iii) (they need distributional Fourier transforms of the
32  non-integrable `g` on `ℝ^d`) and `ℝ^d` versions of D.4-D.6.
33* **Periodization** `fdlPer f x = ∑_{n ∈ ℤ^d} f(rep x + n)` (a finite sum for compactly supported `f`, so no
34  `tsum` junk); `w_λ` on the torus periodizes the short-scale kernels `φ_t` *without* subtracting their
35  average (L12376-12377): `fdlWT`.
36* **Values at the origin.**  `w_λ(0) = +∞`, which Lean's Bochner integral turns into the junk `0`; every
37  statement about `w_λ` excludes `z = 0`.  `g_λ(0)` is defined by continuous extension
38  (`limUnder` over the punctured neighbourhood in `fdlTruncE`, `fdlTruncT`); Lemma D.3(ii) asserts the
39  smoothness of `g_λ` (so the extension exists), and the bounds at `0` refer to this value.
40* **Definition D.2** says `λ ∈ (0,1)`; the formulas make sense for every `λ > 0` and are defined so
41  (needed for `η ≤ 2Kλ` in Lemma D.4).  See SUSPECTED ISSUE in `fdl_compare`.
42* **The correction `o_N`.**  Lemma D.4 says "after enlarging `C_0` in `H:eq:energy-completion`".  We use the
43  exact correction of Proposition 3.34, `oNΛ d s C₀ Λ N` (`o_N(Λ)`, `B:eq:hrs-coercivity-exact-correction`),
44  as `Energy/Coercivity.lean` does, with `C₀` chosen *before* `Λ` (depending only on `d, s, φ`); this implies
45  the energy-completion form (`Λ`-dependent `C₀`).  `oNΛ` is monotone in `C₀`, so "enlarging" is automatic.
46* `F_N = rieszEnergy d s`, `F_{N,η} = fdlEnergy` (modulated energy with the kernel `g_η`).

62/-! ### The profile, `φ = ψ * ψ`, the scaled kernels -/

64/-- The profile of L12360: `ψ ∈ C_c^∞(B(0,1/2); [0,∞))` radial with `ψ > 0` on `B(0,1/4)`. -/
65structure IsFDLProfile (ψ : Euc d → ℝ) : Prop where
66  smooth : ContDiff ℝ ((⊤ : ℕ∞) : WithTop ℕ∞) ψ
67  nonneg : ∀ x, 0 ≤ ψ x
68  support : tsupport ψ ⊆ Metric.ball (0 : Euc d) (1 / 2)
69  radial : ∀ x y : Euc d, ‖x‖ = ‖y‖ → ψ x = ψ y
70  pos : ∀ x : Euc d, ‖x‖ < 1 / 4 → 0 < ψ x
71
72/-- `φ := ψ * ψ` (convolution on `ℝ^d`; L12360). -/
73def fdlPhi (ψ : Euc d → ℝ) : Euc d → ℝ :=
74  convolution ψ ψ (ContinuousLinearMap.mul ℝ ℝ) volume
75
76/-- `φ_t := t^{-d} φ(·/t)` (L12360). -/
77def fdlPhiScale (ψ : Euc d → ℝ) (t : ℝ) (x : Euc d) : ℝ := (t ^ d)⁻¹ * fdlPhi ψ (t⁻¹ • x)
78
79/-- The Fourier transform `ψ̂(ξ) = ∫ ψ(x) e^{-2πi ξ·x} dx`; real because `ψ` is radial (even), so it is the
80cosine transform `∫ ψ(x) cos(2π ξ·x) dx`.  Then `φ̂ = ψ̂²` (L12360). -/
81def fdlPsiHat (ψ : Euc d → ℝ) (ξ : Euc d) : ℝ :=
82  ∫ x, ψ x * Real.cos (2 * Real.pi * inner ℝ ξ x)
83
84/-- Periodization `P f (x) = ∑_{n ∈ ℤ^d} f(x + n)` of a function on `ℝ^d` to the torus
85(`P φ_t` of L12369; a finite sum for compactly supported `f`). -/
86def fdlPer (f : Euc d → ℝ) (x : Torus d) : ℝ :=
87  ∑' n : Lattice d, f (Torus.rep x + Lattice.toEuc n)

89/-! ### Definition D.2: the smooth truncation -/

91/-- Definition D.2 (`M:B:def:fdl-truncation`, TeX M L12396-12401), short-scale kernel on `ℝ^d`:
92`w_λ = b ∫_0^λ t^{d-s} φ_t dt/t ≥ 0`.  At `x = 0` the integral diverges and is the Lean junk `0`. -/
93def fdlWE (ψ : Euc d → ℝ) (s b lam : ℝ) (x : Euc d) : ℝ :=
94  b * ∫ t in Set.Ioc (0 : ℝ) lam, t ^ ((d : ℝ) - s) * fdlPhiScale ψ t x / t
95
96/-- Definition D.2, short-scale kernel on `𝕋^d`: the integrand `t^{d-s} φ_t` is periodized *without*
97subtracting its average (L12376-12377). -/
98def fdlWT (ψ : Euc d → ℝ) (s b lam : ℝ) (x : Torus d) : ℝ :=
99  b * ∫ t in Set.Ioc (0 : ℝ) lam, t ^ ((d : ℝ) - s) * fdlPer (fdlPhiScale ψ t) x / t
100
101/-- Definition D.2, the truncated kernel `g_λ := g - w_λ` on `ℝ^d` (`g = rieszKernel`), with the value
102at `0` given by continuous extension (junk if the limit does not exist; Lemma D.3 asserts smoothness). -/
103def fdlTruncE (ψ : Euc d → ℝ) (s b lam : ℝ) (z : Euc d) : ℝ :=
104  if z = 0 then
105    limUnder (𝓝[≠] (0 : Euc d)) (fun z' => Whole.rieszKernel d s z' - fdlWE ψ s b lam z')
106  else Whole.rieszKernel d s z - fdlWE ψ s b lam z
107
108/-- Definition D.2, the truncated kernel `g_λ := g - w_λ` on `𝕋^d` (`g = periodicRiesz d s`), with the
109value at `0` given by continuous extension. -/
110def fdlTruncT (ψ : Euc d → ℝ) (s b lam : ℝ) (x : Torus d) : ℝ :=
111  if x = 0 then
112    limUnder (𝓝[≠] (0 : Torus d)) (fun x' => periodicRiesz d s x' - fdlWT ψ s b lam x')
113  else periodicRiesz d s x - fdlWT ψ s b lam x

115/-- `F_{N,η}(X_N, μ)`: the modulated energy built from `g_η` (Lemma D.4). -/
116def fdlEnergy (ψ : Euc d → ℝ) (s b η : ℝ) {N : ℕ} (X : Config d N) (μ : Measure (Torus d)) : ℝ :=
117  modulatedEnergyKernel (fdlTruncT ψ s b η) X μ

119/-- The small-scale pair sum `N⁻² ∑_{i≠j} w_η(x_i - x_j)` (Lemma D.4(ii)). -/
120def fdlPairW (ψ : Euc d → ℝ) (s b η : ℝ) {N : ℕ} (X : Config d N) : ℝ :=
121  (1 / (N : ℝ) ^ 2) * ∑ i, ∑ j ∈ Finset.univ.erase i, fdlWT ψ s b η (X i - X j)

134/-! ### Lemma D.1: the smooth scale representation -/

136/-- `b` is a constant `b_{d,s}` for which Lemma D.1 holds: `b > 0` and
137* (`0 < s < d`) `g(x) = b ∫_0^∞ t^{d-s} φ_t(x) dt/t` for `x ≠ 0` on `ℝ^d` (absolutely convergent);
138* (`s = 0`) `g(x) = b lim_{R→∞} (∫_0^R t^d φ_t(x) dt/t - c_φ log R) + C_d`, `c_φ = φ(0)`, with `C_d`
139  independent of `x`;
140* (torus) `g(x) = b ∫_0^∞ t^{d-s} (Pφ_t(x) - ∫_{ℝ^d} φ) dt/t` for `x ≠ 0`, pointwise.
141Here `g = Whole.rieszKernel` on `ℝ^d` and `g = periodicRiesz d s` on `𝕋^d` (L12363-12373). -/
142def IsFDLConst (ψ : Euc d → ℝ) (s b : ℝ) : Prop :=
143  0 < b ∧
144  (0 < s → ∀ x : Euc d, x ≠ 0 →
145    IntegrableOn (fun t : ℝ => t ^ ((d : ℝ) - s) * fdlPhiScale ψ t x / t) (Set.Ioi 0) ∧
146    Whole.rieszKernel d s x =
147      b * ∫ t in Set.Ioi (0 : ℝ), t ^ ((d : ℝ) - s) * fdlPhiScale ψ t x / t) ∧
148  (s = 0 → ∃ C : ℝ, ∀ x : Euc d, x ≠ 0 →
149    (∀ R : ℝ, 0 < R →
150      IntegrableOn (fun t : ℝ => t ^ ((d : ℝ) - s) * fdlPhiScale ψ t x / t) (Set.Ioc 0 R)) ∧
151    Tendsto (fun R : ℝ =>
152        (∫ t in Set.Ioc (0 : ℝ) R, t ^ ((d : ℝ) - s) * fdlPhiScale ψ t x / t)
153          - fdlPhi ψ 0 * Real.log R) atTop
154      (𝓝 ((Whole.rieszKernel d s x - C) / b))) ∧
155  (∀ x : Torus d, x ≠ 0 →
156    IntegrableOn (fun t : ℝ =>
157      t ^ ((d : ℝ) - s) * (fdlPer (fdlPhiScale ψ t) x - ∫ y, fdlPhi ψ y) / t) (Set.Ioi 0) ∧
158    periodicRiesz d s x =
159      b * ∫ t in Set.Ioi (0 : ℝ),
160        t ^ ((d : ℝ) - s) * (fdlPer (fdlPhiScale ψ t) x - ∫ y, fdlPhi ψ y) / t)
