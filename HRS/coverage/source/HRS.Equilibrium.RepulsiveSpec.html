36/-- Uniform exponential moment at fixed `β ≥ 0` (DGR Theorem 2.2(i) on the torus): there is `C`
37with `∫ exp(-β N F_N) dμ^{⊗N} ≤ C` for all `N ≥ 1`, together with integrability. -/
38theorem exponential_moment_uniform (hd : 2 ≤ d) (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ)
39    (hΛ : ∀ x, ρ x ≤ Λ) {β : ℝ} (hβ : 0 ≤ β) :
40    ∃ C : ℝ, ∀ N : ℕ, 1 ≤ N →
41      Integrable (fun X => Real.exp (exponent d β N (withDensityReal ρ) X))
42        (Measure.pi fun _ : Fin N => withDensityReal ρ) ∧
43      (∫ X, Real.exp (exponent d β N (withDensityReal ρ) X)
44        ∂(Measure.pi fun _ : Fin N => withDensityReal ρ)) ≤ C := by
45  have : IsProbabilityMeasure (withDensityReal ρ) := isProb_withDensityReal hρ
46  have : MeasurableSpace.CountablyGenerated (Torus d) := countablyGenerated_torus d
47  obtain ⟨δ, u₀, A, hδ, hδ2, hu₀, hA, hdata⟩ :=
48    exists_masterData (μ := withDensityReal ρ) hd rfl hρ hΛ hβ
49  choose D hD using hdata
50  obtain ⟨K, hK⟩ := masterCutoffSize_le (μ := withDensityReal ρ) hd hβ hδ (by linarith) hA hu₀
51  have hSize : ∀ (_θ : Unit) (n : ℕ) (hn : 1 ≤ n),
52      DGRHS.Concentration.masterCutoffSize (withDensityReal ρ) (D n hn).kernel (D n hn).cap u₀
53        ≤ 0 + K * (n : ℝ) ^ ((2 : ℝ) / 3) := by
54    intro _θ n hn
55    rw [(hD n hn).1, (hD n hn).2, zero_add]
56    exact (hK n hn).2
57  obtain ⟨M, hM⟩ := DGRHS.Concentration.master_uniform_sublinear_exponential_envelope
58    (I := Unit) (Ω := Torus d) (withDensityReal ρ)
59    (fun _ n => exponent d β n (withDensityReal ρ))
60    (fun _ n _ => (measurable_exponent hd β n (withDensityReal ρ)).aemeasurable)
61    u₀ hu₀ (fun _ n hn => D n hn) 0 K ((2 : ℝ) / 3) (by norm_num) hSize
62  exact ⟨M, fun N hN => hM () N hN⟩

64/-- The repulsive half of Proposition 4.25: `sup_{N ≥ 2, γ ∈ [0, B₀]} K_{N,γ}(μ) < ∞`. -/
65theorem log_prep_repulsive_aux (hd : 2 ≤ d) (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ)
66    (hΛ : ∀ x, ρ x ≤ Λ) (B₀ : ℝ) :
67    ∃ C : ℝ, ∀ N : ℕ, 2 ≤ N → ∀ γ ∈ Set.Icc (0 : ℝ) B₀,
68      PreparationAux.KE (periodicRiesz d 0) N γ (withDensityReal ρ) ≤ ENNReal.ofReal C := by
69  have : IsProbabilityMeasure (withDensityReal ρ) := isProb_withDensityReal hρ
70  obtain ⟨C, hC⟩ := exponential_moment_uniform hd ρ hρ Λ hΛ (β := max B₀ 0) (le_max_right B₀ 0)
71  refine ⟨1 + |C|, fun N hN γ hγ => ?_⟩
72  obtain ⟨hint, hle⟩ := hC N (by omega)
73  have h1 := KE_le_one_add (periodicRiesz d 0) N hγ.1 (hγ.2.trans (le_max_left B₀ 0))
74    (withDensityReal ρ)
75  rw [KE_eq_ofReal_integral (max B₀ 0) N (withDensityReal ρ) hint] at h1
76  refine le_trans h1 ?_
77  calc 1 + ENNReal.ofReal (∫ X, Real.exp (exponent d (max B₀ 0) N (withDensityReal ρ) X)
78          ∂(Measure.pi fun _ : Fin N => withDensityReal ρ))
79      ≤ 1 + ENNReal.ofReal |C| :=
80        add_le_add le_rfl (ENNReal.ofReal_le_ofReal (hle.trans (le_abs_self C)))
81    _ = ENNReal.ofReal (1 + |C|) := by
82        rw [ENNReal.ofReal_add zero_le_one (abs_nonneg C), ENNReal.ofReal_one]
