28theorem expanding_tori_kernel_proof (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (m : ℕ) :
29    (∀ L : ℝ, 1 ≤ L → ∃ γ : Euc d → ℝ, IsPeriodicRieszKernel d s L γ) ∧
30    (∃ C : ℝ, ∀ L : ℝ, 1 ≤ L → ∀ γ : Euc d → ℝ, IsPeriodicRieszKernel d s L γ →
31      (∀ n : ℕ, n ≤ m + 2 → 0 < (n : ℝ) + s → ∀ x : Euc d, x ≠ 0 → ‖x‖ ≤ L / 2 →
32        ‖iteratedFDeriv ℝ n γ x‖ ≤ C * (‖x‖ ^ (-s - n) + (if 0 < s then L ^ (-s - n) else 0) +
33          (if s = 0 then L ^ (-(n : ℝ)) else 0))) ∧
34      (s = 0 → ∀ x : Euc d, x ≠ 0 → ‖x‖ ≤ L / 2 →
35        |γ x - Real.log L| ≤ C * (1 + |Real.log ‖x‖|))) ∧
36    ∀ K : Set (Euc d), IsCompact K → (∀ x ∈ K, x ≠ 0) → ∀ n : ℕ, n ≤ m + 2 → ∀ ε : ℝ, 0 < ε →
37      ∃ L₀ : ℝ, ∀ L : ℝ, L₀ ≤ L → ∀ γ : Euc d → ℝ, IsPeriodicRieszKernel d s L γ →
38        ((0 < (n : ℝ) + s → ∀ x ∈ K,
39            ‖iteratedFDeriv ℝ n γ x - iteratedFDeriv ℝ n (rieszKernel d s) x‖ ≤ ε) ∧
40          (n = 0 → s = 0 → ∃ c : ℝ, ∀ x ∈ K, |γ x - rieszKernel d s x - c| ≤ ε)) := by
41  refine ⟨fun L hL => ⟨ker1_gam d s L, ker1_gam_isKernel hd hs0 hsd (by linarith)⟩, ?_, ?_⟩
42  · -- the uniform constant
43    have hCn : ∀ n : ℕ, ∃ C : ℝ, 0 ≤ C ∧ (0 < (n : ℝ) + s → ∀ L : ℝ, 0 < L → ∀ x : Euc d, x ≠ 0 →
44        ‖x‖ ≤ L / 2 → ‖iteratedFDeriv ℝ n (ker1_gam d s L) x‖ ≤ C * ‖x‖ ^ (-s - n)) := by
45      intro n
46      by_cases hn : 0 < (n : ℝ) + s
47      · obtain ⟨C, hC0, hC⟩ := ker1_gam_bound_a hd hs0 hsd n hn
48        exact ⟨C, hC0, fun _ => hC⟩
49      · exact ⟨0, le_rfl, fun h => absurd h hn⟩
50    choose Cn hCn0 hCn using hCn
51    have hClog : ∃ C : ℝ, 0 ≤ C ∧ (s = 0 → ∀ L : ℝ, 0 < L → ∀ x : Euc d, x ≠ 0 → ‖x‖ ≤ L / 2 →
52        |ker1_gam d s L x - Real.log L| ≤ C + |Real.log ‖x‖|) := by
53      by_cases hs : s = 0
54      · subst hs
55        obtain ⟨C, hC0, hC⟩ := ker1_gam_bound_b hd
56        exact ⟨C, hC0, fun _ => hC⟩
57      · exact ⟨0, le_rfl, fun h => absurd h hs⟩
58    obtain ⟨Cl, hCl0, hCl⟩ := hClog
59    refine ⟨1 + ∑ n ∈ Finset.range (m + 3), Cn n + Cl, fun L hL γ hγ => ⟨fun n hn hns x hx0 hx => ?_, fun hs x hx0 hx => ?_⟩⟩
60    · have hLp : 0 < L := by linarith
61      have hoff := ker1_offLattice_of_half hLp hx0 hx
62      have heq := (ker1_eventually_eq_gam hd hs0 hsd hLp hγ hoff).iteratedFDeriv ℝ n
63      rw [heq.eq_of_nhds]
64      refine (hCn n hns L hLp x hx0 hx).trans ?_
65      have hle : Cn n ≤ 1 + ∑ n ∈ Finset.range (m + 3), Cn n + Cl := by
66        have : Cn n ≤ ∑ n ∈ Finset.range (m + 3), Cn n :=
67          Finset.single_le_sum (f := Cn) (fun i _ => hCn0 i) (Finset.mem_range.2 (by omega))
68        linarith
69      have h1 : 0 ≤ ‖x‖ ^ (-s - (n : ℝ)) := by positivity
70      have h2 : 0 ≤ (if 0 < s then L ^ (-s - (n : ℝ)) else 0) := by
71        split_ifs
72        · exact Real.rpow_nonneg hLp.le _
73        · exact le_rfl
74      have h3 : 0 ≤ (if s = 0 then L ^ (-(n : ℝ)) else 0) := by
75        split_ifs
76        · exact Real.rpow_nonneg hLp.le _
77        · exact le_rfl
78      have hCn0' := hCn0 n
79      nlinarith
80    · have hLp : 0 < L := by linarith
81      have hoff := ker1_offLattice_of_half hLp hx0 hx
82      have heq := (ker1_eventually_eq_gam hd hs0 hsd hLp hγ hoff).self_of_nhds
83      rw [heq]
84      refine (hCl hs L hLp x hx0 hx).trans ?_
85      have h1 : 0 ≤ |Real.log ‖x‖| := abs_nonneg _
86      have hS : 0 ≤ ∑ n ∈ Finset.range (m + 3), Cn n := Finset.sum_nonneg fun i _ => hCn0 i
87      nlinarith
88  · intro K hK hK0 n _ ε hε
89    obtain ⟨q, δ, hδ, hq, hloc⟩ := ker1_decomp hd hs0 hsd
90    obtain ⟨L₁, hL₁pos, hL₁⟩ := ker1_conv_aux hd hs0 hδ hq hloc hK hK0 n hε
91    obtain ⟨R, hR⟩ := hK.isBounded.exists_norm_le
92    refine ⟨max L₁ (2 * max R 1), fun L hL γ hγ => ?_⟩
93    have hL1 : L₁ ≤ L := (le_max_left _ _).trans hL
94    have hL2 : 2 * max R 1 ≤ L := (le_max_right _ _).trans hL
95    have hRpos : 0 < max R 1 := lt_of_lt_of_le one_pos (le_max_right _ _)
96    have hLp : 0 < L := by linarith
97    have hoff : ∀ x ∈ K, ∀ z : Fin d → ℤ, x ≠ L • latticePt z := fun x hx =>
98      ker1_offLattice_of_half hLp (hK0 x hx) (by
99        have := (hR x hx).trans (le_max_left R 1); linarith)
100    refine ⟨fun hns x hx => ?_, fun hn hs => ?_⟩
101    · have heq := (ker1_eventually_eq_gam hd hs0 hsd hLp hγ (hoff x hx)).iteratedFDeriv ℝ n
102      rw [heq.eq_of_nhds]
103      refine (hL₁ L hL1 x hx).1 ?_
104      rcases Nat.eq_zero_or_pos n with h0 | h0
105      · right
106        intro hs0'
107        subst h0; subst hs0'; simp at hns
108      · left; omega
109    · refine ⟨Real.log L + q 0, fun x hx => ?_⟩
110      have heq := (ker1_eventually_eq_gam hd hs0 hsd hLp hγ (hoff x hx)).self_of_nhds
111      rw [heq]
112      exact (hL₁ L hL1 x hx).2 hn hs
