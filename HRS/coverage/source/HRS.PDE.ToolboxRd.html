35/-- Lemma C.3(i) on `ℝ^d` (line 12256): `B` is symmetric, positive semidefinite, and
36`B(u,u) ≤ C ‖u‖²_{H^{θ/2}}` (all forms finite for `u, v ∈ H^{θ/2}`). -/
37theorem quadratic_toolbox_basic (θ : ℝ) (hθ0 : 0 < θ) (hθ2 : θ < 2) (m : Euc d → ℝ)
38    (hm : IsHomogSymbol θ m) :
39    ∃ C : ℝ, ∀ u v : Euc d → ℝ, MemSobolev (θ / 2) u → MemSobolev (θ / 2) v →
40      Integrable (fun ξ => m ξ * ‖fourierL2 u ξ‖ ^ 2) ∧ Integrable (fun ξ => m ξ * ‖fourierL2 v ξ‖ ^ 2) ∧
41      Integrable (fun ξ => ((m ξ : ℝ) : ℂ) * (fourierL2 u ξ * conj (fourierL2 v ξ))) ∧
42      symbolForm m u v = symbolForm m v u ∧ 0 ≤ symbolForm m u u ∧
43      symbolForm m u u ≤ C * sobolevNormSq (θ / 2) u :=
44  FourierAux.form_basic hθ0 hm

54/-- Lemma C.3(iii) on `ℝ^d` (lines 12262–12264), weighted symmetrization: for `0 ≤ a ∈ C^{1,1}_b`
55with `m = √a` Lipschitz and `w ∈ H¹ ∩ C^{0,1}`,
56`B(aw,w) = B(mw,mw) + E`, `E = -½ ∬ 𝔨(z) w(x) w(x-z) (δ_z m(x))² dz dx`, and
57`|E| ≤ C (‖∇m‖²_∞ + ‖m‖²_∞) ‖w‖²_{L²}` with `C` depending on the kernel only. -/
58theorem quadratic_toolbox_weighted (hd : 1 ≤ d) (θ : ℝ) (hθ0 : 0 < θ) (hθ2 : θ < 2)
59    (m𝔪 : Euc d → ℝ) (hm𝔪 : IsHomogSymbol θ m𝔪) (𝔨 : Euc d → ℝ) (h𝔨 : IsLevyKernel θ m𝔪 𝔨) :
60    ∃ C : ℝ, ∀ (L La Lm Lw : ℝ) (a w : Euc d → ℝ), (∀ x, 0 ≤ a x) → IsC11 L a →
61      IsLipschitz La a → IsLipschitz Lm (fun x => Real.sqrt (a x)) →
62      IsLipschitz Lw w → MemSobolev 1 w →
63      let m := fun x => Real.sqrt (a x)
64      let E := symbolForm m𝔪 (fun x => a x * w x) w -
65        symbolForm m𝔪 (fun x => m x * w x) (fun x => m x * w x)
66      Integrable (fun p : Euc d × Euc d => 𝔨 p.2 * w p.1 * w (p.1 - p.2) * (m p.1 - m (p.1 - p.2)) ^ 2) ∧
67      E = -(1 / 2) * ∫ p : Euc d × Euc d, 𝔨 p.2 * w p.1 * w (p.1 - p.2) * (m p.1 - m (p.1 - p.2)) ^ 2 ∧
68      |E| ≤ C * (Lm ^ 2 + (⨆ x, |m x|) ^ 2) * l2NormSq w := by
69  exact WeightAux.weighted_core hd θ hθ0 hθ2 m𝔪 𝔨 h𝔨
70    (fun w _ hw h2 => FourierAux.bdd_of_lipschitz_memLp hw h2)
71    (fun _ _ _ hb hbb hh => FourierAux.memSobolev_one_mul hb hbb hh)

73/-- Lemma C.3(iv) on `ℝ^d` (line 12265), weighted comparison: if moreover `a ≥ λ₀ > 0`, then
74`B(w,w) ≤ λ₀⁻¹ B(mw,mw) + C(1+λ₀⁻¹)(‖√a‖²_{C^{0,1}} + ‖√(a/λ₀-1)‖²_{C^{0,1}} + 1) ‖w‖²_{L²}`. -/
75theorem quadratic_toolbox_compare (hd : 1 ≤ d) (θ : ℝ) (hθ0 : 0 < θ) (hθ2 : θ < 2)
76    (m𝔪 : Euc d → ℝ) (hm𝔪 : IsHomogSymbol θ m𝔪) :
77    ∃ C : ℝ, ∀ (L lam0 Lm Lm' Lw : ℝ) (a w : Euc d → ℝ), 0 < lam0 → (∀ x, lam0 ≤ a x) → IsC11 L a →
78      IsLipschitz Lm (fun x => Real.sqrt (a x)) →
79      IsLipschitz Lm' (fun x => Real.sqrt (a x / lam0 - 1)) → IsLipschitz Lw w → MemSobolev 1 w →
80      let m := fun x => Real.sqrt (a x)
81      symbolForm m𝔪 w w ≤ lam0⁻¹ * symbolForm m𝔪 (fun x => m x * w x) (fun x => m x * w x) +
82        C * (1 + lam0⁻¹) * (c01Norm m ^ 2 +
83          c01Norm (fun x => Real.sqrt (a x / lam0 - 1)) ^ 2 + 1) * l2NormSq w := by
84  obtain ⟨𝔨, h𝔨⟩ := quadratic_toolbox_kernel hd θ hθ0 hθ2 m𝔪 hm𝔪
85  exact WeightAux.compare_core hd θ hθ0 hθ2 m𝔪 hm𝔪 𝔨 h𝔨
86    (fun w _ hw h2 => FourierAux.bdd_of_lipschitz_memLp hw h2)
87    (fun _ _ _ hb hbb hh => FourierAux.memSobolev_one_mul hb hbb hh)
