28/-- Theorem 2.1, general-matrix clause, transport improvement (TeX M L609-619: "The transport improvement still
29applies with the positive density bound"; the improvement itself is `H:eq:general-transport-improvement`,
30L598-604).  For a constant matrix `M` with `Sym M ≤ 0`, under the adjoint input `AdjointInput M K`
31(`H:eq:matrix-adjoint-input`) and the positive density bound `PosDensityWith b.c`, the covariance contribution in the
32two Wasserstein estimates on `𝒳_{γ+2,T}` can be replaced by
33`C √q_N min{𝔈_N^{1/4} + N^{-1/d}, 𝔈_N^{1/max{2,d-s}} + N^{-1/d}}`.
34All other hypotheses are those of `HRS.GeneralSetup.general_fluctuation_matrix`; the conclusion is the
35matrix-clause analogue of `HRS.GeneralSetup.general_fluctuation_transport` (paths compared in `E_{γ+2}`). -/
36theorem general_fluctuation_matrix_transport (b : GeneralBounds d) (M : Matrix (Fin d) (Fin d) ℝ)
37    (hM : Torus.SymNonpos M) (K : ℝ) :
38    ∃ C₀min : ℝ, ∀ C₀ : ℝ, C₀min ≤ C₀ → ∃ C : ℝ, 0 < C ∧
39      ∀ S : GeneralSetup d, S.Within { b with C₀ := C₀ } → S.MFHypM M → S.AdjointInput M K →
40      S.PosDensityWith b.c →
41      ∀ (N : ℕ) (_ : 2 ≤ N) {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω)
42      [IsProbabilityMeasure P]
43      (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (_ : IsVectorBrownian P W)
44      (X : ℝ≥0 → Ω → Config d N) (_ : IsStrongSolution P (S.dynM M N) S.T X₀ W X)
45      (_ : ∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
46        (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X)
47      (_ : S.EnergyMomentFinite P N X)
48      (η₀ : Ω → NegSobolev d S.γ) (_ : Measurable η₀)
49      (_ : ProbabilityTheory.IndepFun (fun ω => (X₀ ω, η₀ ω)) (fun ω => fun t => W t ω) P)
50      (_ : S.InitialErrorFinite P N 1 X₀ η₀)
51      (Y : ℝ → Ω → NegSobolev d ((S.γ : ℝ) + 2))
52      (_ : IsLinearFluctuationSolution P (S.dynM M N) (S.ρ N) ((S.γ : ℝ) + 2)
53        (qScale N (S.κ N) (S.invβ N)) S.T
54        (fun ω => Ascale N (S.κ N) (S.invβ N) • NegSobolev.embed ((S.γ : ℝ) + 2) (η₀ ω)) Y),
55      let σ := sigmaScale N (S.κ N) (S.invβ N)
56      let A := Ascale N (S.κ N) (S.invβ N)
57      let q := qScale N (S.κ N) (S.invβ N)
58      let 𝔈 := S.energyMoment P N X
59      let cov := Real.sqrt q * min (𝔈 ^ ((1 : ℝ) / 4) + (N : ℝ) ^ (-(1 : ℝ) / d))
60        (𝔈 ^ ((1 : ℝ) / max 2 ((d : ℝ) - S.s)) + (N : ℝ) ^ (-(1 : ℝ) / d))
61      let Ξ := law P (flucPath ((S.γ : ℝ) + 2) S.T σ X (S.ρ N))
62      let YN := law P (toPath ((S.γ : ℝ) + 2) S.T Y)
63      wassersteinE 1 Ξ YN ≤ ENNReal.ofReal
64          (C * (A * S.initialError P N 1 X₀ η₀ + σ * 𝔈 + cov)) ∧
65      (S.EnergyMoment2Finite P N X →
66        Integrable (fun ω => ‖flucPath ((S.γ : ℝ) + 2) S.T σ X (S.ρ N) ω‖ ^ 2) P →
67        S.InitialErrorFinite P N 2 X₀ η₀ →
68        wassersteinE 2 Ξ YN ≤ ENNReal.ofReal
69          (C * (A * S.initialError P N 2 X₀ η₀ + σ * S.energyMoment2 P N X + cov))) := by
70  sorry

72/-- Higher heat-filter orders, gradient dynamics (TeX M L629-632 and L3283-3293).  For an integer `L ≥ 1` and
73`γ > max{d+4, d/2 + 1 + max(2L, ⌈(d-s)/2⌉ + 1)}` (`U:eq:gamma-condition`), under the hypotheses of
74`HRS.GeneralSetup.general_fluctuation` (same coefficient hypotheses, positive density bound), the covariance terms
75`N^{-2/d}` and `N^{-1/d}` of `H:eq:general-smooth`-`H:eq:general-W2` are replaced by `N^{-2L/d}` and `N^{-L/d}`,
76"without changing the nonlinear energy condition".  The constant depends on `(b, L)`.  (The transport alternative
77"retains its stated `N^{-1/d}` term", i.e. it is `HRS.GeneralSetup.general_fluctuation_transport` unchanged, and is
78not restated.)  For `L = 1` the condition on `γ` is implied by `2d + 10 < γ` and the statement is
79`general_fluctuation`. -/
80theorem general_fluctuation_filtered (b : GeneralBounds d) (L : ℕ) (hL : 1 ≤ L)
81    (hγL : max ((d : ℝ) + 4) ((d : ℝ) / 2 + 1 +
82        max (2 * (L : ℝ)) ((⌈((d : ℝ) - b.s) / 2⌉ : ℝ) + 1)) < (b.γ : ℝ)) :
83    ∃ C₀min : ℝ, ∀ C₀ : ℝ, C₀min ≤ C₀ → ∃ C : ℝ, 0 < C ∧
84      ∀ S : GeneralSetup d, S.Within { b with C₀ := C₀ } → S.MFHyp → S.PosDensityWith b.c →
85      ∀ (N : ℕ) (_ : 2 ≤ N) {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω)
86      [IsProbabilityMeasure P]
87      (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (_ : IsVectorBrownian P W)
88      (X : ℝ≥0 → Ω → Config d N) (_ : IsStrongSolution P (S.dyn N) S.T X₀ W X)
89      (_ : ∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
90        (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X)
91      (_ : S.EnergyMomentFinite P N X)
92      (η₀ : Ω → NegSobolev d S.γ) (_ : Measurable η₀)
93      (_ : ProbabilityTheory.IndepFun (fun ω => (X₀ ω, η₀ ω)) (fun ω => fun t => W t ω) P)
94      (_ : S.InitialErrorFinite P N 1 X₀ η₀)
95      (Y : ℝ → Ω → NegSobolev d S.γ)
96      (_ : IsLinearFluctuationSolution P (S.dyn N) (S.ρ N) S.γ (qScale N (S.κ N) (S.invβ N)) S.T
97        (fun ω => Ascale N (S.κ N) (S.invβ N) • η₀ ω) Y),
98      let σ := sigmaScale N (S.κ N) (S.invβ N)
99      let A := Ascale N (S.κ N) (S.invβ N)
100      let q := qScale N (S.κ N) (S.invβ N)
101      let 𝔈 := S.energyMoment P N X
102      let Ξ := law P (flucPath S.γ S.T σ X (S.ρ N))
103      let YN := law P (toPath S.γ S.T Y)
104      smoothDist 2 Ξ YN ≤
105          C * (A * S.initialError P N 1 X₀ η₀ + σ * 𝔈 +
106            q * (Real.sqrt 𝔈 + (N : ℝ) ^ (-(2 * (L : ℝ)) / d))) ∧
107      wassersteinE 1 Ξ YN ≤ ENNReal.ofReal
108          (C * (A * S.initialError P N 1 X₀ η₀ + σ * 𝔈 +
109            Real.sqrt q * (𝔈 ^ ((1 : ℝ) / 4) + (N : ℝ) ^ (-(L : ℝ) / d)))) ∧
110      (S.EnergyMoment2Finite P N X →
111        Integrable (fun ω => ‖flucPath S.γ S.T σ X (S.ρ N) ω‖ ^ 2) P →
112        S.InitialErrorFinite P N 2 X₀ η₀ →
113        wassersteinE 2 Ξ YN ≤ ENNReal.ofReal
114          (C * (A * S.initialError P N 2 X₀ η₀ + σ * S.energyMoment2 P N X +
115            Real.sqrt q * (𝔈 ^ ((1 : ℝ) / 4) + (N : ℝ) ^ (-(L : ℝ) / d))))) := by
116  sorry
