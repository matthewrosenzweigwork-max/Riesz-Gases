20/-!
21Route for `duhamel_coeff` (coefficientwise, no Hilbert structure, no Fubini).  Write `m = j - 2`,
22`U(x)ψ := frozenCoeff D ρβ m ψ x`, `L = forwardGen D ρβ`, `F(σ)` the `μ`-propagator path (level `m+2`),
23`G(σ) = forwardGen D (μ (t-σ)) (F σ)`, `E(σ) = G(σ) - L F(σ)`.  Fix `τ` and `k`, and put
24`h(σ) = (U(τ-σ) F(σ)) k` on `[0,τ]`.
25* `h` is continuous (`F ∈ C([0,t];H^m)` strongly, `U` uniformly bounded and Lipschitz in the datum,
26  modes of `U(·)ψ` continuous);
27* `h` has right derivative `(U(τ-σ) E(σ)) k` at `σ ∈ (0,τ)`: with `δ = y - σ`,
28  `h(y) - h(σ) = U(τ-y)[F(y) - U(δ)F(σ)] k` by the semigroup law `frozenCoeff_shift`, and
29  `δ⁻¹[F(y) - U(δ)F(σ)] → G(σ) - L F(σ)` in `H^m` (time regularity of `F` and of the frozen path
30  with `H^{m+2}` datum `F(σ)`), `U(τ-y)` being bounded on `H^m` and linear (`frozenCoeff_sub/smul`);
31* the derivative is measurable (`measurable_derivWithin_Ici`) and bounded, hence integrable, and
32  `integral_eq_sub_of_hasDeriv_right_of_le` gives `h(τ) - h(0) = ∫_0^τ (U(τ-σ)E(σ)) k dσ`.
33-/

51theorem frozenCoeff_smul (hd : 1 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d) {m : ℕ} (hm : 1 ≤ m)

59  · have ht : τ ∈ Set.Ioc 0 T := ⟨hpos, hτ.2⟩
60    have hm1 : (1 : ℝ) ≤ m := by exact_mod_cast hm
61    obtain ⟨C, -, hspec⟩ := propagatorCoeff_spec hd s hs0 hsd m hm R MB Λ VB
62    obtain ⟨h1, w1, -⟩ := hspec D ν₀ T (fun _ => ρ) hA τ ht ψc hψ
63    have w1' := weakContinuousInHs_mono hm1 w1
64    have hsol : IsForwardSolution D (fun _ => ρ) τ (z • ψc)
65        (fun σ => z • propagatorCoeff D (fun _ => ρ) m τ ψc σ) := by
66      refine ⟨fun σ hσ => memHs_smul_coeff (h1.h1 σ hσ) z, fun k => ?_, fun k σ hσ => ?_⟩
67      · have : IntegrableOn (fun σ => z * forwardGen D ρ (propagatorCoeff D (fun _ => ρ) m τ ψc σ) k)
68            (Set.Icc 0 τ) := Integrable.const_mul (h1.integrable k) z
69        refine this.congr_fun (fun σ _ => ?_) measurableSet_Icc
70        simp only [forwardGen_smul, Pi.smul_apply, smul_eq_mul]
71      · have hi : IntervalIntegrable (fun x => forwardGen D ρ (propagatorCoeff D (fun _ => ρ) m τ ψc x) k)
72            volume 0 σ := by
73          rw [intervalIntegrable_iff_integrableOn_Icc_of_le hσ.1]
74          exact (h1.integrable k).mono_set (Set.Icc_subset_Icc_right hσ.2)
75        have e := h1.eq k σ hσ
76        simp only [forwardGen_smul, Pi.smul_apply, smul_eq_mul]
77        rw [intervalIntegral.integral_const_mul, e]
78        ring
79    have hw : WeakContinuousInHs 1 (fun σ => z • propagatorCoeff D (fun _ => ρ) m τ ψc σ) τ := by
80      refine ⟨fun σ hσ => memHs_smul_coeff (w1'.1 σ hσ) z, fun G hG => ?_⟩

84    exact this

145  obtain ⟨C, -, hspec⟩ := propagatorCoeff_spec hd s hs0 hsd (m + 2) (by omega) R MB Λ VB

161  rw [hP0, e1]
