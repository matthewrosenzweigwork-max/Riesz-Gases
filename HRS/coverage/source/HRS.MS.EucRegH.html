4/-!
5# C-REG-h: `euc_thermal_regularity` in the hard regime `d - 1 ≤ s < d`, and the full frozen theorem
6
7Namespace `HRS.MS.Thm`, helper prefix `regh_`.
8
9Route (Hölder gain, no Littlewood-Paley).  `U = ∫ ‖x-y‖^{-s} ρ(y) dy = T ρ` and `ρ = e^{-β(a‖x‖²/2+U)}/Z`.
10* `regh_Tk_hold`/`regh_T_hold` (Aux2): for `F` bounded, rapidly decaying and `θ`-Hölder, `T F` is
11  `(θ + a)`-Hölder, `a = d - σ`, `σ = (s+d)/2` (the localised kernel `k = ‖z‖^{-s} - (1+‖z‖²)^{-s/2}` is integrable and
12  `∫ (k(z+h) - k(z)) dz = 0`).
13* `regh_cross` (Aux3): if moreover `θ > 1 - (d - s)`, `T F` is `C¹` with a Hölder derivative (second-order Taylor
14  estimate of `k` for the remainder, Hölder-interpolation `[Δ_h F]_{θ'} ≲ |h|^{θ-θ'}` for the derivative).
15* The recursive class `regh_C` (`C^k` with Hölder `D^k`) is closed under sums, products and `exp` (Aux3); `regh_gain_rec`,
16  `regh_cross_rec` lift both facts to all orders (the derivative of `T F` is `T (DF)`).
17* `regh_bootstrap`: `U ∈ C^k_{b}` with Hölder `D^k U` for every `k`, by induction, alternating gain and crossing.
18* `euc_thermal_regularity_proof`: cases `s < d - 1` (`euc_thermal_regularity_easy`) and `d - 1 ≤ s` (this file).
19
20No statement was adjusted.
21-/

34lemma regh_U_eq (s : ℝ) (F : Euc d → ℝ) : rege_U s F = regh_T s F := by
35  funext x
36  have := integral_sub_left_eq_self (fun z : Euc d => ‖z‖ ^ (-s) * F (x - z)) volume x
37  simp only [sub_sub_cancel] at this
38  unfold rege_U regh_T
39  simp only [smul_eq_mul]
40  rw [← this]

42/-- From the class for `U` to the class for `ρ` (and decay of all its derivatives). -/
43lemma regh_rho_of_U {s a β Z C0 θ : ℝ} (ha : 0 < a) (hβ : 0 < β) (hZ : 0 < Z) {ρ : Euc d → ℝ}
44    (heq : ∀ x, ρ x = Real.exp (-β * (a * ‖x‖ ^ 2 / 2 + rege_U s ρ x)) / Z)
45    (hup : ∀ x, ρ x ≤ C0 * Real.exp (-β * a * ‖x‖ ^ 2 / 2)) (hθ : 0 < θ) (hθ1 : θ ≤ 1) (k : ℕ)
46    (hU : regh_Cl θ k (rege_U s ρ)) : regh_Cl θ k ρ ∧ regh_Rl k ρ := by
47  have hb2 : 0 < β * a / 2 := by positivity
48  obtain ⟨hUk, M, hM⟩ := regh_Cl_contDiff hθ k hU
49  have hM0 : 0 ≤ max M 0 := le_max_right _ _
50  obtain ⟨C, hC, hD⟩ := rege_exponent_bound a β hUk hM0 (fun i _ hik x => (hM i hik x).trans (le_max_left _ _))
51  have hφ : ContDiff ℝ k (fun x : Euc d => -β * (a * ‖x‖ ^ 2 / 2 + rege_U s ρ x)) :=
52    contDiff_const.mul (((contDiff_const.mul (contDiff_norm_sq ℝ)).div_const 2).add hUk)
53  obtain ⟨-, hρD⟩ := rege_rho_deriv hZ heq hup hφ hC hD
54  refine ⟨?_, regh_R_of_iter k fun j hj => ?_⟩
55  · have e : ρ = fun x => (Real.exp (-β * (a * ‖x‖ ^ 2 / 2 + 0)) / Z) *
56        Real.exp ((-β) • rege_U s ρ x) := funext fun x => by
57      rw [heq x, smul_eq_mul, show -β * (a * ‖x‖ ^ 2 / 2 + rege_U s ρ x) =
58        -β * (a * ‖x‖ ^ 2 / 2 + 0) + -β * rege_U s ρ x by ring, Real.exp_add]
59      ring
60    rw [e]
61    exact regh_Cl_prod hθ.le hθ1 k (ContinuousLinearMap.mul ℝ ℝ) (regh_gauss_Cl ha hβ hZ hθ.le hθ1 k)
62      (regh_Cl_exp hθ.le hθ1 k (regh_Cl_smul (-β) k hU))
63  · obtain ⟨K, hK, hKb⟩ := hρD j hj
64    exact rege_rdec_of_gauss (j := j) hb2 hK (fun x => by rw [← rege_exp_form]; exact hKb x)

66/-- **Bootstrap in the hard regime**: `U = ∫ ‖x-y‖^{-s} ρ(y) dy` is `C^∞` with bounded derivatives of every
67positive order (`d - 1 ≤ s < d`). -/
68theorem regh_bootstrap {s a β Z C0 : ℝ} (hd1 : 1 ≤ d) (hs : 0 < s) (hsd : s < d)
69    (hs1 : (d : ℝ) - 1 ≤ s) (ha : 0 < a) (hβ : 0 < β) (hZ : 0 < Z) {ρ : Euc d → ℝ}
70    (hm : Measurable ρ) (heq : ∀ x, ρ x = Real.exp (-β * (a * ‖x‖ ^ 2 / 2 + rege_U s ρ x)) / Z)
71    (hup : ∀ x, ρ x ≤ C0 * Real.exp (-β * a * ‖x‖ ^ 2 / 2)) :
72    ∀ k : ℕ, ContDiff ℝ k (rege_U s ρ) ∧ ∃ M : ℝ, 0 ≤ M ∧ ∀ i, 1 ≤ i → i ≤ k → ∀ x,
73      ‖iteratedFDeriv ℝ i (rege_U s ρ) x‖ ≤ M := by
74  obtain ⟨α, hα⟩ : ∃ α : ℝ, α = (d : ℝ) - s := ⟨_, rfl⟩
75  have hα0 : 0 < α := by rw [hα]; linarith
76  have hα1 : α ≤ 1 := by rw [hα]; linarith
77  obtain ⟨T, hT⟩ : ∃ T : ℝ, T = 1 - α / 4 := ⟨_, rfl⟩
78  have hT0 : 0 < T := by rw [hT]; linarith
79  have hT1 : T < 1 := by rw [hT]; linarith
80  have hs2 : (d : ℝ) < s + 2 := by linarith
81  have hσe : (d : ℝ) - (s + α / 2) = α / 2 := by rw [hα]; ring
82  have hUeq : rege_U s ρ = regh_T s ρ := regh_U_eq s ρ
83  have hb2 : 0 < β * a / 2 := by positivity
84  have hdec : lreg_RDec ρ := by
85    have hρpos : ∀ x, 0 ≤ ρ x := fun x => by rw [heq x]; positivity
86    refine rege_rdec_of_gauss (j := 0) hb2 (K := max C0 0) (le_max_right _ _) (fun x => ?_)
87    rw [Real.norm_of_nonneg (hρpos x), pow_zero, mul_one, ← rege_exp_form]
88    exact (hup x).trans (mul_le_mul_of_nonneg_right (le_max_left _ _) (Real.exp_pos _).le)
89  have hgain : ∀ (k : ℕ) (θ θ' : ℝ), 0 < θ → regh_Cl θ k ρ → regh_Rl k ρ → 0 ≤ θ' → θ' ≤ θ →
90      θ' + α / 2 < 1 → regh_Cl (θ' + α / 2) k (regh_T s ρ) := by
91    intro k θ θ' hθ hρC hρR h0 h1 h2
92    have := regh_gain_rec hd1 hs hsd hs2 (σ := s + α / 2) (by linarith) (by linarith) (by rw [hα] at *; linarith)
93      hθ k hρC hρR θ' h0 h1 (by rw [hσe]; exact h2)
94    rwa [hσe] at this
95  -- raising the Hölder exponent
96  have hlift : ∀ (k : ℕ) (θ : ℝ), 0 < θ → θ ≤ T → regh_Cl θ k (rege_U s ρ) →
97      regh_Cl (min (θ + α / 2) T) k (rege_U s ρ) := by
98    intro k θ hθ hθT hU
99    obtain ⟨hρC, hρR⟩ := regh_rho_of_U ha hβ hZ heq hup hθ (by linarith) k hU
100    have := hgain k θ (min θ (T - α / 2)) hθ hρC hρR (le_min hθ.le (by rw [hT]; linarith))
101      (min_le_left _ _) (by have := min_le_right θ (T - α / 2); linarith)
102    rw [← hUeq] at this
103    have e : min θ (T - α / 2) + α / 2 = min (θ + α / 2) T := by
104      simp only [min_def]; split_ifs <;> linarith
105    rwa [e] at this
106  have hloopn : ∀ (k n : ℕ) (θ : ℝ), 0 < θ → θ ≤ T → regh_Cl θ k (rege_U s ρ) →
107      T ≤ θ + n * (α / 2) → regh_Cl T k (rege_U s ρ) := by
108    intro k n
109    induction n with
110    | zero =>
111      intro θ hθ hθT hU hle
112      have : θ = T := le_antisymm hθT (by simpa using hle)
113      rwa [this] at hU
114    | succ n ih =>
115      intro θ hθ hθT hU hle
116      have h1 := hlift k θ hθ hθT hU
117      by_cases hc : T ≤ θ + α / 2
118      · rwa [min_eq_right hc] at h1
119      · push_neg at hc
120        rw [min_eq_left hc.le] at h1
121        exact ih (θ + α / 2) (by linarith) hc.le h1 (by push_cast at hle; linarith)
122  have hloop : ∀ (k : ℕ) (θ : ℝ), 0 < θ → θ ≤ T → regh_Cl θ k (rege_U s ρ) →
123      regh_Cl T k (rege_U s ρ) := by
124    intro k θ hθ hθT hU
125    obtain ⟨n, hn⟩ := exists_nat_ge (T / (α / 2))
126    refine hloopn k n θ hθ hθT hU ?_
127    have : T ≤ n * (α / 2) := by rwa [div_le_iff₀ (by positivity)] at hn
128    linarith
129  -- base: level 0
130  have hbase : regh_Cl (α / 2) 0 (rege_U s ρ) := by
131    obtain ⟨C1, hC1, h1⟩ := lreg_bound_of_rdec hdec 0
132    have hB : ∀ x, ‖ρ x‖ ≤ C1 := fun x =>
133      (by simpa using h1 x : ‖ρ x‖ ≤ C1 * lreg_phi d x).trans (by nlinarith [lreg_phi_le_one (d := d) x])
134    have hK : ∀ x h, ‖ρ (x + h) - ρ x‖ ≤ (2 * C1) * ‖h‖ ^ (0 : ℝ) := fun x h => by
135      rw [Real.rpow_zero, mul_one]
136      calc _ ≤ ‖ρ (x + h)‖ + ‖ρ x‖ := norm_sub_le _ _
137        _ ≤ C1 + C1 := add_le_add (hB _) (hB _)
138        _ = _ := by ring
139    obtain ⟨⟨Bt, hBt⟩, K', hK'0, hK'⟩ := regh_T_hold hd1 hs hsd hs2 (σ := s + α / 2) (by linarith)
140      (by linarith) (by rw [hα] at *; linarith) le_rfl (by rw [hσe]; linarith) hm.aestronglyMeasurable hdec
141      (by positivity) hK
142    rw [hσe, zero_add] at hK'
143    rw [hUeq]
144    exact ⟨(regh_hol_cont (by positivity) hK').aestronglyMeasurable, ⟨Bt, hBt⟩, K', hK'0, hK'⟩
145  have hcross : ∀ k : ℕ, regh_Cl T k (rege_U s ρ) → regh_Cl (α / 4) (k + 1) (rege_U s ρ) := by
146    intro k hU
147    obtain ⟨hρC, hρR⟩ := regh_rho_of_U ha hβ hZ heq hup hT0 hT1.le k hU
148    have := regh_cross_rec hd1 hs hsd hs2 (θ := T) (θ' := 1 - α / 2) hT0 hT1.le (by linarith)
149      (by rw [hT]; linarith) (by rw [← hα]; linarith) (by rw [← hα]; linarith) k hρC hρR
150    rw [← hUeq] at this
151    have e : T - (1 - α / 2) = α / 4 := by rw [hT]; ring
152    rwa [e] at this
153  have hmain : ∀ k : ℕ, regh_Cl T k (rege_U s ρ) := by
154    intro k
155    induction k with
156    | zero => exact hloop 0 (α / 2) (by positivity) (by rw [hT]; linarith) hbase
157    | succ k ih => exact hloop (k + 1) (α / 4) (by positivity) (by rw [hT]; linarith) (hcross k ih)
158  intro k
159  obtain ⟨hc, M, hM⟩ := regh_Cl_contDiff hT0 k (hmain k)
160  exact ⟨hc, max M 0, le_max_right _ _, fun i _ hik x => (hM i hik x).trans (le_max_left _ _)⟩

162/-- The thermal regularity theorem for `d - 1 ≤ s < d`. -/
163theorem regh_euc_thermal_regularity_hard {d : ℕ} {s : ℝ} (hd : 2 ≤ d) (hs0 : 0 < s) (hsd : s < d)
164    (hs1 : (d : ℝ) - 1 ≤ s) (a β : ℝ) (ha : 0 < a) (hβ : 0 < β) :
165    ContDiff ℝ ∞ (thermalDensity d s a β) ∧
166    ContDiff ℝ ∞ (HRS.Whole.convMeasure (rieszKernel d s) (thermalMeasure d s a β)) ∧
167    (∀ k : ℕ, ∃ C : ℝ, ∀ x,
168      ‖iteratedFDeriv ℝ k (HRS.Whole.convMeasure (rieszKernel d s) (thermalMeasure d s a β)) x‖ ≤ C) ∧
169    (∀ k : ℕ, ∃ C : ℝ, ∀ x, ‖iteratedFDeriv ℝ k (thermalDensity d s a β) x‖ ≤
170      C * (1 + ‖x‖) ^ k * Real.exp (-β * a * ‖x‖ ^ 2 / 2)) ∧
171    (∀ k : ℕ, Integrable (fun x => ‖iteratedFDeriv ℝ k (thermalDensity d s a β) x‖) ∧
172      ∃ C : ℝ, ∀ x, ‖iteratedFDeriv ℝ k (thermalDensity d s a β) x‖ ≤ C) ∧
173    ∀ (k : ℕ) (p : ℝ), 0 ≤ p →
174      Integrable (fun x => ‖x‖ ^ p * ‖iteratedFDeriv ℝ k (thermalDensity d s a β) x‖) := by
175  have hd1 : 1 ≤ d := by omega
176  obtain ⟨Z, C0, hZ, hC0, hρ0, hρm, heq, hup, hU⟩ := rege_data hd hs0 hsd ha hβ
177  set ρ := thermalDensity d s a β with hρdef
178  have hb2 : 0 < β * a / 2 := by positivity
179  have boot := regh_bootstrap hd1 hs0 hsd hs1 ha hβ hZ hρm heq hup
180  have hfun : HRS.Whole.convMeasure (rieszKernel d s) (thermalMeasure d s a β) = rege_U s ρ := funext hU
181  have hρn : ∀ n : ℕ, ContDiff ℝ n ρ ∧ ∀ j ≤ n, ∃ K : ℝ, 0 ≤ K ∧ ∀ x,
182      ‖iteratedFDeriv ℝ j ρ x‖ ≤ K * (1 + ‖x‖) ^ j * Real.exp (-β * a * ‖x‖ ^ 2 / 2) := by
183    intro n
184    obtain ⟨hUn, M, hM0, hM⟩ := boot n
185    obtain ⟨C, hC, hD⟩ := rege_exponent_bound a β hUn hM0 hM
186    have hφ : ContDiff ℝ n (fun x : Euc d => -β * (a * ‖x‖ ^ 2 / 2 + rege_U s ρ x)) :=
187      contDiff_const.mul (((contDiff_const.mul (contDiff_norm_sq ℝ)).div_const 2).add hUn)
188    exact rege_rho_deriv hZ heq hup hφ hC hD
189  have hρinf : ContDiff ℝ ∞ ρ := contDiff_infty.2 fun n => (hρn n).1
190  -- Gaussian envelope for every `k`
191  have henv : ∀ k : ℕ, ∃ K : ℝ, 0 ≤ K ∧ ∀ x, ‖iteratedFDeriv ℝ k ρ x‖ ≤
192      K * (1 + ‖x‖) ^ k * Real.exp (-(β * a / 2) * ‖x‖ ^ 2) := by
193    intro k
194    obtain ⟨K, hK, hKb⟩ := (hρn k).2 k le_rfl
195    exact ⟨K, hK, fun x => by rw [← rege_exp_form]; exact hKb x⟩
196  rw [hfun]
197  refine ⟨hρinf, contDiff_infty.2 fun k => (boot k).1, fun k => ?_, fun k => ?_, fun k => ?_,
198    fun k p hp => ?_⟩
199  · rcases Nat.eq_zero_or_pos k with rfl | hk
200    · obtain ⟨Cb, hCb, h0⟩ := lreg_bound_of_rdec
201        (rege_rdec_of_gauss (j := 0) hb2 (K := max C0 0) (le_max_right _ _) (fun x => by
202          rw [Real.norm_of_nonneg (hρ0 x), pow_zero, mul_one, ← rege_exp_form]
203          exact (hup x).trans (mul_le_mul_of_nonneg_right (le_max_left _ _) (Real.exp_pos _).le)))
204        0
205      refine ⟨Cb * lreg_A d s, fun x => ?_⟩
206      rw [norm_iteratedFDeriv_zero]
207      exact rege_U_bound hd1 hs0 hsd hρm.aestronglyMeasurable hCb (fun y => by simpa using h0 y) x
208    · obtain ⟨_, M, _, hM⟩ := boot k
209      exact ⟨M, fun x => hM k hk le_rfl x⟩
210  · obtain ⟨K, hK, hKb⟩ := (hρn k).2 k le_rfl
211    exact ⟨K, hKb⟩
212  · obtain ⟨K, hK, hKb⟩ := henv k
213    have hcont : Continuous (iteratedFDeriv ℝ k ρ) := (hρn k).1.continuous_iteratedFDeriv le_rfl
214    refine ⟨?_, ?_⟩
215    · have := rege_weighted_integrable hb2 (G := iteratedFDeriv ℝ k ρ) (j := k) (K := K)
216        hcont.aestronglyMeasurable hKb 0 le_rfl
217      simpa using this
218    · obtain ⟨C, hC, hCb⟩ := rege_poly_gauss_bdd hb2 k
219      refine ⟨K * C, fun x => ?_⟩
220      calc ‖iteratedFDeriv ℝ k ρ x‖ ≤ K * (1 + ‖x‖) ^ k * Real.exp (-(β * a / 2) * ‖x‖ ^ 2) := hKb x
221        _ = K * ((1 + ‖x‖) ^ k * Real.exp (-(β * a / 2) * ‖x‖ ^ 2)) := by ring
222        _ ≤ K * C := mul_le_mul_of_nonneg_left (hCb _ (norm_nonneg x)) hK
223  · obtain ⟨K, hK, hKb⟩ := henv k
224    have hcont : Continuous (iteratedFDeriv ℝ k ρ) := (hρn k).1.continuous_iteratedFDeriv le_rfl
225    exact rege_weighted_integrable hb2 (G := iteratedFDeriv ℝ k ρ) (j := k) (K := K)
226      hcont.aestronglyMeasurable hKb p hp

228/-- **C-REG.**  `euc_thermal_regularity`, all `0 < s < d`: the easy regime `s < d - 1` and the hard regime
229`d - 1 ≤ s` (Hölder gain). -/
230theorem euc_thermal_regularity_proof : type_of% @HRS.Whole.euc_thermal_regularity := by
231  intro d s hd hs0 hsd a β ha hβ
232  by_cases hs1 : s < (d : ℝ) - 1
233  · exact euc_thermal_regularity_easy hd hs0 hsd hs1 a β ha hβ
234  · exact regh_euc_thermal_regularity_hard hd hs0 hsd (not_lt.1 hs1) a β ha hβ
