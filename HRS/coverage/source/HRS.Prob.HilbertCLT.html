2# HRS.Prob.HilbertCLT — the i.i.d. central limit theorem in `W₂` on a separable Hilbert space
3
4Work package CLT-H (wave 2). This file states, with `sorry` proofs, the probabilistic inputs
5that the static field limits (Proposition 4.21, `M:O:prop:negative`, source lines 5513-5557)
6and the i.i.d. initial-data lemmas (Lemmas A.36 and A.38, `M:B:lem:iid-init`,
7`M:B:lem:iid-init-field`) use, in exactly the form their proofs need:
8
9* existence of a centered Gaussian measure with a prescribed trace-class covariance on a
10  separable real Hilbert space (Mathlib at the pin has `IsGaussian` and `covarianceBilin`
11  but no such existence theorem in infinite dimension);
12* the i.i.d. central limit theorem in `W₂`: for i.i.d. centered `H`-valued summands with a
13  finite second moment (possibly degenerate covariance) the normalised sums converge in
14  `W₂` to the centered Gaussian with the same covariance (source lines 5521-5557: finite-
15  dimensional CLT, uniformly small tails, and "weak convergence together with convergence of
16  second moments implies `W₂` convergence in a separable Hilbert space");
17* the uniform (triangular-array) version over families with uniformly small Hilbert tails
18  and uniformly bounded second moments, which the proof of Proposition 4.21 uses to get
19  uniformity in the reference density `μ`;
20* the fourth-moment bound `O:eq:fourth-sum` (source line 5540 region).
21
22Conventions. All statements are in `ℝ≥0∞` (`wassersteinE`, `lintegral`) or carry explicit
23`MemLp` hypotheses, so no junk value of a real `iSup`/`tsum`/Bochner integral can make a
24statement vacuous.

42/-- The normalised partial sum `N^{-1/2} ∑_{i<N} X_i`. -/
43def normalizedSum {Ω : Type*} (X : ℕ → Ω → H) (N : ℕ) (ω : Ω) : H :=
44  (Real.sqrt N)⁻¹ • ∑ i ∈ Finset.range N, X i ω

46/-- A trace-class covariance form on `H`: a bounded bilinear form which is symmetric,
47positive semidefinite, and has finite trace along some Hilbert basis. (The trace of a
48nonnegative operator does not depend on the basis; one basis suffices.) -/
49structure IsTraceClassCov (C : H →L[ℝ] H →L[ℝ] ℝ) : Prop where
50  symm : ∀ x y, C x y = C y x
51  nonneg : ∀ x, 0 ≤ C x x
52  trace : ∃ (ι : Type) (b : HilbertBasis ι ℝ H), Summable fun i => C (b i) (b i)

61lemma countable_of_orthonormal {ι : Type*} {v : ι → H} (hon : Orthonormal ℝ v) : Countable ι := by
62  have := Set.PairwiseDisjoint.countable_of_isOpen (a := (Set.univ : Set ι))
63    (s := fun i : ι => Metric.ball (v i) (1 / 2))
64    (fun i _ j _ hij => ?_) (fun i _ => Metric.isOpen_ball)
65    (fun i _ => ⟨v i, Metric.mem_ball_self (by norm_num)⟩)
66  · exact Set.countable_univ_iff.mp this
67  rw [Function.onFun, Set.disjoint_left]
68  intro x hx hx'
69  have h1 : dist (v i) (v j) ≤ dist (v i) x + dist x (v j) := dist_triangle _ _ _
70  rw [Metric.mem_ball] at hx hx'
71  have hd : dist (v i) (v j) ^ 2 = 2 := by
72    rw [dist_eq_norm, norm_sub_sq_real, hon.inner_eq_zero hij, hon.norm_eq_one, hon.norm_eq_one]
73    norm_num
74  have : dist (v i) (v j) < 1 := by linarith [dist_comm x (v i)]
75  nlinarith [dist_nonneg (x := v i) (y := v j)]

78lemma exists_hilbertBasis_type : ∃ (ι : Type) (_ : Countable ι) (_ : HilbertBasis ι ℝ H), True := by
79  obtain ⟨w, b, hb⟩ := exists_hilbertBasis ℝ H
80  have hon : Orthonormal ℝ ((↑) : w → H) := hb ▸ b.orthonormal
81  have hcount : Countable w := countable_of_orthonormal hon
82  obtain ⟨e, he⟩ := Countable.exists_injective_nat w
83  let ι := Set.range e
84  let eq : w ≃ ι := Equiv.ofInjective e he
85  have hv : Orthonormal ℝ (fun j : ι => ((eq.symm j : w) : H)) :=
86    hon.comp _ eq.symm.injective
87  have hrange : Set.range (fun j : ι => ((eq.symm j : w) : H)) = Set.range ((↑) : w → H) := by
88    ext x; constructor
89    · rintro ⟨j, rfl⟩; exact ⟨_, rfl⟩
90    · rintro ⟨i, rfl⟩; exact ⟨eq i, by simp⟩

92  rw [hrange, ← hb]
93  rw [← Submodule.orthogonal_closure, b.dense_span]
