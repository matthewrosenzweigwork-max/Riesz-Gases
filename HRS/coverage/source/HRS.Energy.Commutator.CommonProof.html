23theorem commutator_coercivity_common_proof (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) :
24    ∃ C₀ C : ℝ, 0 ≤ C₀ ∧ 0 < C ∧
25      ∀ (Λ : ℝ), 1 ≤ Λ → ∀ (N : ℕ), 2 ≤ N →
26      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
27      ∀ X : Config d N, X ∉ collisionSet d N →
28        0 ≤ rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N ∧
29        (∀ (v : Torus d → Euc d) (K : ℝ), HasTransportBound d s v K →
30          |commQuantity d s v X (withDensityReal ρ)| ≤
31            C * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N)) ∧
32        (∀ (ψ : Torus d → ℝ) (L : ℝ), 0 ≤ L → KernelGrad.IsLipschitz L ψ →
33          HasFiniteHomSobolev (((d : ℝ) - s) / 2) ψ →
34          |∫ x, ψ x ∂(empirical X) - ∫ x, ψ x ∂(withDensityReal ρ)| ≤
35            C * L * smearScale d N Λ +
36              C * homSobolev (((d : ℝ) - s) / 2) ψ *
37                Real.sqrt (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N)) := by
38  obtain ⟨A₀, A, hA₀, hA, hcomm⟩ := HRS.Commutator.commutator_estimate_of_packets hd hs0 hsd
39  obtain ⟨B₀, B, hB₀, hB, hcoer⟩ := periodic_coercivity hd hs0 hsd
40  refine ⟨max A₀ B₀, max A B, le_max_of_le_left hA₀, lt_max_of_lt_left hA, ?_⟩
41  intro Λ hΛ N hN ρ hρ hρΛ X hX
42  have hΛ0 : 0 ≤ Λ := by linarith
43  obtain ⟨hlow, htest⟩ := hcoer Λ hΛ N hN ρ hρ hρΛ X hX
44  have hoB : oNΛ d s B₀ Λ N ≤ oNΛ d s (max A₀ B₀) Λ N :=
45    oNΛ_mono_C₀ (le_max_right _ _) hΛ0 N
46  have hoA : oNΛ d s A₀ Λ N ≤ oNΛ d s (max A₀ B₀) Λ N :=
47    oNΛ_mono_C₀ (le_max_left _ _) hΛ0 N
48  have hlow' : 0 ≤ rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max A₀ B₀) Λ N := by
49    linarith
50  refine ⟨hlow', ?_, ?_⟩
51  · intro v K hK
52    have hK0 := hK.nonneg
53    have h1 := (hcomm Λ hΛ N hN ρ hρ hρΛ X hX v K hK).2
54    refine h1.trans ?_
55    have hAK : 0 ≤ A * K := mul_nonneg hA.le hK0
56    calc A * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s A₀ Λ N)
57        ≤ A * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max A₀ B₀) Λ N) :=
58          mul_le_mul_of_nonneg_left (by linarith) hAK
59      _ ≤ max A B * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max A₀ B₀) Λ N) := by
60          apply mul_le_mul_of_nonneg_right _ hlow'
61          exact mul_le_mul_of_nonneg_right (le_max_left _ _) hK0
62  · intro ψ L hL hψ hfin
63    refine (htest ψ L hL hψ hfin).trans ?_
64    have hsm : 0 ≤ smearScale d N Λ :=
65      Real.rpow_nonneg (mul_nonneg (Nat.cast_nonneg N) hΛ0) _
66    have hhs : 0 ≤ homSobolev (((d : ℝ) - s) / 2) ψ := Real.sqrt_nonneg _
67    have hsq : Real.sqrt (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s B₀ Λ N) ≤
68        Real.sqrt (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max A₀ B₀) Λ N) :=
69      Real.sqrt_le_sqrt (by linarith)
70    have hAB : B ≤ max A B := le_max_right _ _
71    have hsq0 := Real.sqrt_nonneg (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s B₀ Λ N)
72    have e1 : B * smearScale d N Λ * L ≤ max A B * L * smearScale d N Λ := by
73      nlinarith [mul_nonneg hsm hL]
74    have e2 : B * homSobolev (((d : ℝ) - s) / 2) ψ *
75        Real.sqrt (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s B₀ Λ N) ≤
76        max A B * homSobolev (((d : ℝ) - s) / 2) ψ *
77        Real.sqrt (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max A₀ B₀) Λ N) := by
78      apply mul_le_mul _ hsq hsq0 (mul_nonneg (hA.le.trans (le_max_left _ _)) hhs)
79      exact mul_le_mul_of_nonneg_right hAB hhs
80    linarith
