3/-!
4# WPC: proof of `HRS.Main.SpecSW2.weak_reference_pinned_cavity`
5
6Route.  A pinned weight on `n` free coordinates is `pairWeight` on `n + 1` coordinates evaluated at
7`Fin.insertNth p y Z`; its one-site kernels are those of the unpinned ensemble at the configuration
8`Fin.insertNth p y x` and the site `p.succAbove i`, so the unpinned estimates
9`WeakKernel.siteKernel_le`, `WeakKernel.siteKernel_coeff` apply with `N = n + 1`.  The cavity weight is the
10`fieldWeight` of the abstract weak specification (`HRS/External/WeakSpec.lean`) with the empty pin, so
11`weak_siteKernel_le`, `weak_dobrushin` apply.  The witness matrix is the constant matrix `κ`.
12-/

23lemma wpc_measurable_insertNth {n : ℕ} (p : Fin (n + 1)) (y : Torus d) :
24    Measurable (fun Z : Config d n => Fin.insertNth (α := fun _ => Torus d) p y Z) := by

26  rcases Fin.eq_self_or_eq_succAbove p j with rfl | ⟨k, rfl⟩
27  · simp only [Fin.insertNth_apply_same]; exact measurable_const
28  · simp only [Fin.insertNth_apply_succAbove]; exact measurable_pi_apply k

30lemma wpc_siteKernel_pinned {μ : Measure (Torus d)} [IsProbabilityMeasure μ]
31    {q : Torus d → Torus d → ℝ} {b : ℝ} {n : ℕ} (hW : Measurable (pairWeight q (n + 1) b μ))
32    (p : Fin (n + 1)) (y : Torus d) (i : Fin n) (x : Config d n) :
33    External.siteKernel μ (SpecSW2.pinnedWeight q μ b n p y) i x
34      = External.siteKernel μ (pairWeight q (n + 1) b μ) (p.succAbove i)
35          (Fin.insertNth (α := fun _ => Torus d) p y x) := by
36  have hP : Measurable (SpecSW2.pinnedWeight q μ b n p y) :=
37    hW.comp (wpc_measurable_insertNth p y)
38  have hupd : ∀ (x : Config d n) (z : Torus d),
39      Fin.insertNth (α := fun _ => Torus d) p y (Function.update x i z)
40        = Function.update (Fin.insertNth (α := fun _ => Torus d) p y x) (p.succAbove i) z :=
41    fun x z => Fin.insertNth_update (α := fun _ => Torus d) p y i z x
42  rw [External.siteKernel_apply hP, External.siteKernel_apply hW]
43  have hn : External.normalizer μ (SpecSW2.pinnedWeight q μ b n p y) i x
44      = External.normalizer μ (pairWeight q (n + 1) b μ) (p.succAbove i)
45          (Fin.insertNth (α := fun _ => Torus d) p y x) := by
46    unfold External.normalizer
47    simp only [SpecSW2.pinnedWeight, hupd]
48  rw [hn]
49  simp only [SpecSW2.pinnedWeight, hupd]

51lemma wpc_insertNth_agree {n : ℕ} (p : Fin (n + 1)) (y : Torus d) (j : Fin n) {x x' : Config d n}
52    (hxy : ∀ k, k ≠ j → x k = x' k) :
53    ∀ k, k ≠ p.succAbove j → Fin.insertNth (α := fun _ => Torus d) p y x k
54      = Fin.insertNth (α := fun _ => Torus d) p y x' k := by
55  intro k hk
56  rcases Fin.eq_self_or_eq_succAbove p k with rfl | ⟨k', rfl⟩
57  · simp
58  · simp only [Fin.insertNth_apply_succAbove]
59    exact hxy k' fun h => hk (by rw [h])

61lemma wpc_bounds_of {μ : Measure (Torus d)} [IsProbabilityMeasure μ] {n : ℕ}
62    {W : Config d n → ℝ≥0∞} {C κ : ℝ}
63    (h1 : ∀ (i : Fin n) (x : Config d n), External.siteKernel μ W i x ≤ ENNReal.ofReal C • μ)
64    (h2 : ∀ i j : Fin n, i ≠ j → ∀ x y : Config d n, (∀ k, k ≠ j → x k = y k) →
65      ∀ S : Set (Torus d), MeasurableSet S →
66        |(External.siteKernel μ W i x).real S - (External.siteKernel μ W i y).real S| ≤ κ)
67    (hκ : 0 ≤ κ) (hsum : (n : ℝ) * κ < 1 / 2) :
68    SpecSW2.WeakConditionalBounds μ W C κ := by
69  refine ⟨h1, fun _ _ => κ, fun _ _ => ⟨hκ, le_rfl⟩, ?_, ?_, h2⟩
70  · intro i; simpa using hsum
71  · intro j; simpa using hsum

75/-- Cavity law: one-site kernels, via the abstract weak specification with empty pin. -/
76lemma wpc_cavity_bounds {μ : Measure (Torus d)} [IsProbabilityMeasure μ] {a : ℝ}
77    {q : Torus d → Torus d → ℝ} (h : WeakKernel μ a q) (ha : 0 ≤ a) {b : ℝ} (hb : 0 ≤ b) (n : ℕ) :
78    (∀ (i : Fin n) (x : Config d n),
79      External.siteKernel μ (SpecSW2.cavityWeight q μ b (n + 1) (n := n)) i x
80        ≤ ENNReal.ofReal (Real.exp (b * a + 0 + b * a)) • μ) ∧
81    (∀ i j : Fin n, i ≠ j → ∀ x y : Config d n, (∀ k, k ≠ j → x k = y k) →
82      ∀ S : Set (Torus d), MeasurableSet S →
83        |(External.siteKernel μ (SpecSW2.cavityWeight q μ b (n + 1) (n := n)) i x).real S
84          - (External.siteKernel μ (SpecSW2.cavityWeight q μ b (n + 1) (n := n)) i y).real S|
85        ≤ 4 * Real.exp (b * a + 0 + b * a) * (b * a) / ((n + 1 : ℕ) : ℝ)) := by
86  have iH : Integrable (rowInt q μ) μ :=
87    integrable_of_bdd' μ h.measurable_rowInt (M := a) fun z => by
88      rw [abs_of_nonneg (h.rowInt_nonneg z)]; exact h.rowInt_le ha z
89  let V : External.WeakPotential μ :=
90    { φ := fun z => b * rowInt q μ z
91      q := q
92      M := b * a
93      M' := 0
94      γ := a
95      measurable_φ := h.measurable_rowInt.const_mul b
96      measurable_q := h.meas
97      q_nonneg := h.nonneg
98      q_symm := h.symm
99      γ_nonneg := ha
100      φ_le := fun x => mul_le_mul_of_nonneg_left (h.rowInt_le ha x) hb
101      integrable_φ := iH.const_mul b
102      neg_le_integral_φ := by
103        have : 0 ≤ ∫ x, b * rowInt q μ x ∂μ :=
104          integral_nonneg fun x => mul_nonneg hb (h.rowInt_nonneg x)
105        linarith
106      integrable_q := h.integrable_row
107      row_le := fun x => h.rowInt_le ha x }
108  let a0 : Empty → Torus d := fun e => e.elim
109  have hlam : 0 ≤ b / ((n + 1 : ℕ) : ℝ) := div_nonneg hb (Nat.cast_nonneg _)
110  have hcav : SpecSW2.cavityWeight q μ b (n + 1) (n := n)
111      = fun Z => External.fieldWeight V (External.pinField V (b / ((n + 1 : ℕ) : ℝ)) a0)
112          (b / ((n + 1 : ℕ) : ℝ)) Z := by
113    funext Z
114    unfold SpecSW2.cavityWeight External.fieldWeight External.pairEnergy External.pinField
115    congr 2
116    simp [V, rowInt, Finset.mul_sum]
117  have hψ := External.isField_pinField V hlam a0
118  have hK0 := (External.fieldNorm_pos hψ.measurable V (b / ((n + 1 : ℕ) : ℝ)) n).ne'
119  have hKt := (External.fieldNorm_lt_top (V := V) hψ hlam n).ne
120  have hker : ∀ i : Fin n, External.siteKernel μ (SpecSW2.cavityWeight q μ b (n + 1) (n := n)) i
121      = External.siteKernel μ (External.pinnedDensity V (b / ((n + 1 : ℕ) : ℝ)) n a0) i := by
122    intro i
123    rw [hcav]
124    unfold External.pinnedDensity External.fieldDensity
125    simp only [← div_eq_mul_inv]
126    exact (siteKernel_div_const μ _ _ hK0 hKt i).symm
127  have hN0 : 0 < n + 1 := Nat.succ_pos n
128  have hN : n + Fintype.card Empty ≤ n + 1 := by simp
129  refine ⟨fun i x => ?_, fun i j hij x y hxy S hS => ?_⟩
130  · rw [hker i]
131    exact External.weak_siteKernel_le V hb hN0 hN a0 i x
132  · rw [hker i]
133    have := External.weak_dobrushin V hb hN0 hN a0 i j hij x y hxy S hS
134    refine this.trans (le_of_eq ?_)
135    simp only [External.weakKappa, V]

143theorem weak_reference_pinned_cavity_proof (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d)
144    (cstar : ℝ) (hw : ∀ z : Torus d, z ≠ 0 → 1 ≤ periodicRiesz d s z + cstar)
145    (βm βp : ℝ) (hβm : 0 < βm) (c₀ : ℝ) (hc₀ : 0 < c₀) (Λ : ℕ → ℝ) (A : ℝ) :
146    ∃ a₀ C : ℝ, 0 < a₀ ∧
147      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → ∀ [IsProbabilityMeasure (withDensityReal ρ)],
148        Torus.IsSmooth ρ → (∀ x, c₀ ≤ ρ x) →
149        (∀ k : ℕ, Torus.cbNorm k ρ ≤ Λ k) →
150      ∀ q : Torus d → Torus d → ℝ, Measurable (Function.uncurry q) → (∀ x y, q x y = q y x) →
151        (∀ x y, 0 ≤ q x y) → (∀ x y, q x y ≤ A * (periodicRiesz d s (x - y) + cstar)) →
152      ∀ a : ℝ, 0 ≤ a → a < a₀ →
153        (∀ x, ∫⁻ y, ENNReal.ofReal (q x y) ∂(withDensityReal ρ) ≤ ENNReal.ofReal a) →
154      ∀ n : ℕ, 1 ≤ n → ∀ β ∈ Set.Icc βm βp, ∀ τ ∈ Set.Icc (0 : ℝ) 1,
155        (∀ (p : Fin (n + 1)) (y : Torus d),
156          WeakConditionalBounds (withDensityReal ρ)
157            (pinnedWeight q (withDensityReal ρ) (τ * β) n p y) C (C * τ * β * a / ((n + 1 : ℕ) : ℝ))) ∧
158        WeakConditionalBounds (withDensityReal ρ)
159          (cavityWeight q (withDensityReal ρ) (τ * β) (n + 1) (n := n)) C
160          (C * τ * β * a / ((n + 1 : ℕ) : ℝ)) := by
161  set P : ℝ := max βp 0 with hP
162  have hP0 : 0 ≤ P := le_max_right _ _
163  set C : ℝ := 4 * Real.exp (2 * P) with hC
164  have hCpos : 0 < C := by positivity
165  have hpos : 0 < 4 * C * (P + 1) := by positivity
166  refine ⟨min 1 (1 / (4 * C * (P + 1))), C, lt_min one_pos (by positivity), ?_⟩
167  intro ρ hρ hprob hsm hc0 hΛ q hqm hqs hq0 hqA a ha0 ha hrow n hn β hβ τ hτ
168  have h : WeakRef.WeakKernel (withDensityReal ρ) a q := ⟨hqm, hqs, hq0, hrow⟩
169  have ha1 : a < 1 := lt_of_lt_of_le ha (min_le_left _ _)
170  have ha2 : a < 1 / (4 * C * (P + 1)) := lt_of_lt_of_le ha (min_le_right _ _)
171  have hβpos : 0 < β := lt_of_lt_of_le hβm hβ.1
172  have hb0 : 0 ≤ τ * β := mul_nonneg hτ.1 hβpos.le
173  have hbP : τ * β ≤ P := by
174    have h1 : τ * β ≤ β := by nlinarith [hτ.2]
175    have h2 : β ≤ P := hβ.2.trans (le_max_left _ _)
176    linarith
177  have hba : τ * β * a ≤ P := by nlinarith
178  have hexp : Real.exp (2 * (τ * β * a)) ≤ C := by
179    have h1 : Real.exp (2 * (τ * β * a)) ≤ Real.exp (2 * P) :=
180      Real.exp_le_exp.2 (by linarith)
181    have := Real.exp_pos (2 * P)
182    rw [hC]; linarith
183  have hN0 : (0 : ℝ) < ((n + 1 : ℕ) : ℝ) := by positivity
184  set κ : ℝ := C * τ * β * a / ((n + 1 : ℕ) : ℝ) with hκ
185  have hκ0 : 0 ≤ κ := by
186    rw [hκ]
187    refine div_nonneg ?_ hN0.le
188    have : C * τ * β * a = C * (τ * β * a) := by ring
189    rw [this]
190    exact mul_nonneg hCpos.le (mul_nonneg hb0 ha0)
191  have hcoeff : 4 * Real.exp (2 * (τ * β * a)) * (τ * β) * a / ((n + 1 : ℕ) : ℝ) ≤ κ := by
192    rw [hκ]
193    refine div_le_div_of_nonneg_right ?_ hN0.le
194    have h1 : 4 * Real.exp (2 * (τ * β * a)) ≤ C := by
195      have := Real.exp_pos (2 * (τ * β * a))
196      rw [hC]
197      have h2 : Real.exp (2 * (τ * β * a)) ≤ Real.exp (2 * P) :=
198        Real.exp_le_exp.2 (by linarith)
199      linarith
200    have h3 : 0 ≤ τ * β * a := mul_nonneg hb0 ha0
201    calc 4 * Real.exp (2 * (τ * β * a)) * (τ * β) * a
202        = (4 * Real.exp (2 * (τ * β * a))) * (τ * β * a) := by ring
203      _ ≤ C * (τ * β * a) := mul_le_mul_of_nonneg_right h1 h3
204      _ = C * τ * β * a := by ring
205  have hsum : (n : ℝ) * κ < 1 / 2 := by
206    have h1 : (n : ℝ) * κ ≤ C * τ * β * a := by
207      rw [hκ, ← mul_div_assoc, div_le_iff₀ hN0]
208      have h3 : 0 ≤ C * τ * β * a := by
209        have := mul_nonneg hb0 ha0
210        have : C * τ * β * a = C * (τ * β * a) := by ring
211        rw [this]; positivity
212      push_cast
213      nlinarith
214    have h2 : C * τ * β * a ≤ C * (P + 1) * a := by
215      have : τ * β ≤ P + 1 := by linarith
216      have h4 : C * τ * β * a = C * (τ * β) * a := by ring
217      rw [h4]
218      exact mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_left this hCpos.le) ha0
219    have h5 : a * (4 * C * (P + 1)) < 1 := by
220      rwa [lt_div_iff₀ hpos] at ha2
221    nlinarith
222  have hW := h.measurable_pairWeight (n + 1) (τ * β)
223  refine ⟨fun p y => ?_, ?_⟩
224  · refine WPC.wpc_bounds_of (fun i x => ?_) (fun i j hij x' y' hxy S hS => ?_) hκ0 hsum
225    · rw [WPC.wpc_siteKernel_pinned hW p y i x]
226      exact h.siteKernel_le ha0 hb0 (Nat.succ_pos n) _ _ hexp
227    · rw [WPC.wpc_siteKernel_pinned hW p y i x', WPC.wpc_siteKernel_pinned hW p y i y']
228      exact (h.siteKernel_coeff ha0 hb0 (Nat.succ_pos n) (p.succAbove i) (p.succAbove j)
229        ((Fin.succAbove_right_injective (p := p)).ne hij) _ _
230        (WPC.wpc_insertNth_agree p y j hxy) S hS).trans hcoeff
231  · obtain ⟨hc1, hc2⟩ := WPC.wpc_cavity_bounds h ha0 hb0 n
232    have e : τ * β * a + 0 + τ * β * a = 2 * (τ * β * a) := by ring
233    simp only [e] at hc1 hc2
234    refine WPC.wpc_bounds_of (fun i x => ?_) (fun i j hij x' y' hxy S hS => ?_) hκ0 hsum
235    · refine (hc1 i x).trans ?_
236      refine Measure.le_iff.2 fun S hS => ?_
237      simp only [Measure.smul_apply, smul_eq_mul]
238      gcongr
239    · refine (hc2 i j hij x' y' hxy S hS).trans ?_
240      calc 4 * Real.exp (2 * (τ * β * a)) * (τ * β * a) / ((n + 1 : ℕ) : ℝ)
241          = 4 * Real.exp (2 * (τ * β * a)) * (τ * β) * a / ((n + 1 : ℕ) : ℝ) := by ring
242        _ ≤ κ := hcoeff
