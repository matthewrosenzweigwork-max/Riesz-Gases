21theorem q32d_chernoff {Ω : Type*} [MeasurableSpace Ω] (μ : Measure Ω) [IsProbabilityMeasure μ]
22    (Z : Ω → ℝ) (hint : ∀ t : ℝ, Integrable (fun ω => Real.exp (t * Z ω)) μ)
23    (D : ℝ) (hD : 0 < D) (x : ℝ) (hx : 0 ≤ x)
24    (hmgf : ∀ t : ℝ, 0 ≤ t → t ≤ x / D → ∫ ω, Real.exp (t * Z ω) ∂μ ≤ Real.exp (D * t ^ 2 / 2)) :
25    μ {ω | x < Z ω} ≤ ENNReal.ofReal (Real.exp (-x ^ 2 / (2 * D))) := by
26  have ht0 : 0 ≤ x / D := div_nonneg hx hD.le
27  have hmain : μ.real {ω | x ≤ Z ω} ≤ Real.exp (-(x / D) * x) * mgf Z μ (x / D) :=
28    measure_ge_le_exp_mul_mgf x ht0 (hint _)
29  have hmgf' : mgf Z μ (x / D) ≤ Real.exp (D * (x / D) ^ 2 / 2) := hmgf _ ht0 le_rfl
30  have hexp : Real.exp (-(x / D) * x) * Real.exp (D * (x / D) ^ 2 / 2) =
31      Real.exp (-x ^ 2 / (2 * D)) := by
32    rw [← Real.exp_add]
33    congr 1
34    field_simp
35    ring
36  have hle : μ.real {ω | x < Z ω} ≤ Real.exp (-x ^ 2 / (2 * D)) := by
37    calc μ.real {ω | x < Z ω} ≤ μ.real {ω | x ≤ Z ω} :=
38          measureReal_mono (μ := μ) (s₁ := {ω | x < Z ω}) (s₂ := {ω | x ≤ Z ω})
39            (by intro ω hω; exact le_of_lt (show x < Z ω from hω)) (measure_ne_top _ _)
40      _ ≤ Real.exp (-(x / D) * x) * mgf Z μ (x / D) := hmain
41      _ ≤ Real.exp (-(x / D) * x) * Real.exp (D * (x / D) ^ 2 / 2) :=
42          mul_le_mul_of_nonneg_left hmgf' (Real.exp_pos _).le
43      _ = Real.exp (-x ^ 2 / (2 * D)) := hexp
44  calc μ {ω | x < Z ω} = ENNReal.ofReal (μ.real {ω | x < Z ω}) := by
45        rw [measureReal_def, ENNReal.ofReal_toReal (measure_ne_top _ _)]
46    _ ≤ ENNReal.ofReal (Real.exp (-x ^ 2 / (2 * D))) := ENNReal.ofReal_le_ofReal hle

78/-- Two-sided version: `|f'| ≤ D |u|` on `[-T, T]` gives `f t ≤ D t²/2` for `|t| ≤ T`. -/
79theorem q32d_quad_bound (f f' : ℝ → ℝ) (D T : ℝ) (h0 : f 0 = 0)
80    (hf : ∀ u, |u| ≤ T → HasDerivAt f (f' u) u)
81    (hb : ∀ u, |u| ≤ T → |f' u| ≤ D * |u|) (t : ℝ) (ht : |t| ≤ T) :
82    f t ≤ D * t ^ 2 / 2 := by
83  rcases le_total 0 t with ht0 | ht0
84  · have htT : t ≤ T := le_trans (le_abs_self t) ht
85    refine q32d_quad_nonneg f f' D T h0 (fun u h1 h2 => hf u ?_) (fun u h1 h2 => ?_) t ht0 htT
86    · rw [abs_of_nonneg h1]; exact h2
87    · have h3 := hb u (by rw [abs_of_nonneg h1]; exact h2)
88      rw [abs_of_nonneg h1] at h3
89      exact le_trans (le_abs_self _) h3
90  · have hnt : -t ≤ T := le_trans (neg_le_abs t) ht
91    have hg : ∀ u, 0 ≤ u → u ≤ T →
92        HasDerivAt (fun u => f (-u)) (f' (-u) * (-1)) u := by
93      intro u h1 h2
94      have h := (hf (-u) (by rw [abs_neg, abs_of_nonneg h1]; exact h2)).comp u (hasDerivAt_neg u)
95      exact h
96    have hbg : ∀ u, 0 ≤ u → u ≤ T → f' (-u) * (-1) ≤ D * u := by
97      intro u h1 h2
98      have h3 := hb (-u) (by rw [abs_neg, abs_of_nonneg h1]; exact h2)
99      rw [abs_neg, abs_of_nonneg h1] at h3
100      have h4 := neg_le_abs (f' (-u))
101      linarith
102    have key := q32d_quad_nonneg (fun u => f (-u)) (fun u => f' (-u) * (-1)) D T
103      (by simpa using h0) hg hbg (-t) (by linarith) hnt
104    simpa using key

186/-- Gaussian tails: the two one-sided bounds for `N(0, v)` with `v ≤ D`. -/
187theorem q32d_gauss_tail (v : NNReal) (D : ℝ) (hD : 0 < D) (hvD : (v : ℝ) ≤ D) (x : ℝ)
188    (hx : 0 ≤ x) :
189    gaussianReal 0 v {z | x < z} ≤ ENNReal.ofReal (Real.exp (-x ^ 2 / (2 * D))) ∧
190      gaussianReal 0 v {z | x < -z} ≤ ENNReal.ofReal (Real.exp (-x ^ 2 / (2 * D))) := by
191  have hmgf : ∀ t : ℝ, ∫ z, Real.exp (t * z) ∂(gaussianReal 0 v) ≤ Real.exp (D * t ^ 2 / 2) := by
192    intro t
193    have h := congrFun (mgf_id_gaussianReal (μ := 0) (v := v)) t
194    have h2 : ∫ z, Real.exp (t * z) ∂(gaussianReal 0 v) =
195        Real.exp (0 * t + (v : ℝ) * t ^ 2 / 2) := h
196    rw [h2]
197    apply Real.exp_le_exp.2
198    have : (v : ℝ) * t ^ 2 ≤ D * t ^ 2 := mul_le_mul_of_nonneg_right hvD (sq_nonneg t)
199    linarith
200  constructor
201  · exact q32d_chernoff (gaussianReal 0 v) (fun z => z)
202      (fun t => integrable_exp_mul_gaussianReal t) D hD x hx (fun t _ _ => hmgf t)
203  · refine q32d_chernoff (gaussianReal 0 v) (fun z => -z) (fun t => ?_) D hD x hx
204      (fun t _ _ => ?_)
205    · have h := integrable_exp_mul_gaussianReal (μ := 0) (v := v) (-t)
206      simpa [mul_neg, neg_mul] using h
207    · have h := hmgf (-t)
208      simp only [neg_mul, mul_neg, neg_sq] at h ⊢
209      exact h

339theorem q32d_rateY_abs_le (gm : Torus d → ℝ) (hgm : Measurable gm)
340    (hgm_ae : ∀ᵐ x ∂(volume : Measure (Torus d)), gm x = periodicRiesz d s x)
341    {β : ℝ} (hβ : 0 < β) {c : ℝ} (hc : 0 ≤ c) (hgc : ∀ x, -c ≤ gm x) {N : ℕ} (hN : 0 < N)
342    (k : Lattice d) (θ : ℝ) [IsProbabilityMeasure (rateGibbs d s k β θ N)]
343    (X : Fin N → Torus d) : |rateY d s k β N θ X| ≤ 2 * Real.sqrt N := by
344  have hNr : (0 : ℝ) < N := by exact_mod_cast hN
345  have hS : |(1 / (N : ℝ)) * respS k X| ≤ 1 := by
346    have h1 := respS_abs_le k X
347    rw [abs_mul, abs_of_pos (by positivity : (0 : ℝ) < 1 / N)]
348    calc 1 / (N : ℝ) * |respS k X| ≤ 1 / N * N := mul_le_mul_of_nonneg_left h1 (by positivity)
349      _ = 1 := by field_simp
350  have hM := q32d_mean_abs_le gm hgm hgm_ae hβ hc hgc hN k θ
351  have hY : rateY d s k β N θ X =
352      Real.sqrt N * ((1 / (N : ℝ)) * respS k X - rateMeanN d s k β N θ) := rfl
353  rw [hY, abs_mul, abs_of_nonneg (Real.sqrt_nonneg _)]
354  have htri : |(1 / (N : ℝ)) * respS k X - rateMeanN d s k β N θ| ≤ 2 := by
355    calc |(1 / (N : ℝ)) * respS k X - rateMeanN d s k β N θ|
356        ≤ |(1 / (N : ℝ)) * respS k X| + |rateMeanN d s k β N θ| := abs_sub _ _
357      _ ≤ 1 + 1 := add_le_add hS hM
358      _ = 2 := by norm_num
359  calc Real.sqrt N * |(1 / (N : ℝ)) * respS k X - rateMeanN d s k β N θ| ≤ Real.sqrt N * 2 :=
360        mul_le_mul_of_nonneg_left htri (Real.sqrt_nonneg _)
361    _ = 2 * Real.sqrt N := by ring

381theorem q32d_mgf_identity (gm : Torus d → ℝ) (hgm : Measurable gm)
382    (hgm_ae : ∀ᵐ x ∂(volume : Measure (Torus d)), gm x = periodicRiesz d s x)
383    {β : ℝ} (hβ : 0 < β) {c : ℝ} (hc : 0 ≤ c) (hgc : ∀ x, -c ≤ gm x) {N : ℕ} (hN : 0 < N)
384    (k : Lattice d) (θ t : ℝ) :
385    ∫ X, Real.exp (t * rateY d s k β N θ X) ∂(rateGibbs d s k β θ N) =
386      respA k gm β N 0 (θ + t / Real.sqrt N) / respA k gm β N 0 θ *
387        Real.exp (-(t * Real.sqrt N * rateMeanN d s k β N θ)) := by
388  have hNr : (0 : ℝ) < N := by exact_mod_cast hN
389  have hr : 0 < Real.sqrt (N : ℝ) := Real.sqrt_pos.2 hNr
390  have hrN : (N : ℝ) = Real.sqrt (N : ℝ) * Real.sqrt (N : ℝ) :=
391    (Real.mul_self_sqrt hNr.le).symm
392  rw [q32d_expect gm hgm hgm_ae hβ k θ (fun X => Real.exp (t * rateY d s k β N θ X))]
393  beta_reduce
394  have hpt : ∀ X : Fin N → Torus d,
395      Real.exp (θ * respS k X) * respW gm β X * Real.exp (t * rateY d s k β N θ X) =
396        Real.exp (-(t * Real.sqrt N * rateMeanN d s k β N θ)) *
397          (Real.exp ((θ + t / Real.sqrt N) * respS k X) * respW gm β X) := by
398    intro X
399    have key : θ * respS k X + t * rateY d s k β N θ X =
400        (θ + t / Real.sqrt N) * respS k X + -(t * Real.sqrt N * rateMeanN d s k β N θ) :=
401      q32d_alg (Real.sqrt N) (N : ℝ) hr hrN t (respS k X) (rateMeanN d s k β N θ) θ
402    calc Real.exp (θ * respS k X) * respW gm β X * Real.exp (t * rateY d s k β N θ X)
403        = Real.exp (θ * respS k X + t * rateY d s k β N θ X) * respW gm β X := by
404          rw [Real.exp_add]; ring
405      _ = Real.exp ((θ + t / Real.sqrt N) * respS k X +
406            -(t * Real.sqrt N * rateMeanN d s k β N θ)) * respW gm β X := by rw [key]
407      _ = _ := by rw [Real.exp_add]; ring
408  have hint : ∫ X : Fin N → Torus d,
409      Real.exp (θ * respS k X) * respW gm β X * Real.exp (t * rateY d s k β N θ X)
410        ∂(volume : Measure (Fin N → Torus d)) =
411      Real.exp (-(t * Real.sqrt N * rateMeanN d s k β N θ)) *
412        ∫ X : Fin N → Torus d, Real.exp ((θ + t / Real.sqrt N) * respS k X) * respW gm β X
413          ∂(volume : Measure (Fin N → Torus d)) := by
414    rw [← integral_const_mul]
415    exact integral_congr_ae (Filter.Eventually.of_forall hpt)
416  rw [hint, resp_respA_zero_eq, resp_respA_zero_eq]
417  ring
