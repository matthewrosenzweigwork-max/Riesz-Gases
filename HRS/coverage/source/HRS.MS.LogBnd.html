4# C-LB: proof of `HRS.Whole.logThermal_bounds`

6For `s = 0`, `thermalDensity d 0 a β = thermalRep K̃ γ̄ β μ` pointwise (`clb_thermalDensity_eq_rep`), an everywhere Euler
7fixed point in `C_M` for `K̃ = g + ½ l(x) + ½ l(y) + ½ log 2`, `γ̄ = (1+|x|²)^{β/2} e^{-βa|x|²/2} / c̄`
8(`c̄ = t67cbar d a β`).  Since `0 ≤ K̃ * ρ ≤ M B` (K1 for `K̃`; the `x`-dependent terms of `K̃` are what produce the
9factor `(1+|x|²)^{β/2}` inside `γ̄`), `clb_rep_bounds` gives `e^{-β M B} γ̄ ≤ ρ ≤ e^{β B} γ̄`, i.e. the claim with
10`c = e^{-β M B} / c̄` and `C = e^{β B} / c̄`.  No statement was adjusted.

26/-- Closed form of the reference density `γ̄`. -/
27lemma clb_gammaBar_eq {a β : ℝ} (x : Euc d) :
28    t67gammaBar d a β x =
29      (1 + ‖x‖ ^ 2) ^ (β / 2) * Real.exp (-β * a * ‖x‖ ^ 2 / 2) / t67cbar d a β := by
30  unfold t67gammaBar t67w t67l confinement
31  congr 1
32  have hpos : (0 : ℝ) < 1 + ‖x‖ ^ 2 := by positivity
33  rw [Real.rpow_def_of_pos hpos, ← Real.exp_add]
34  congr 1
35  ring

37/-- **C-LB.**  The two-sided bound for the logarithmic thermal density. -/
38theorem logThermal_bounds_proof : type_of% @HRS.Whole.logThermal_bounds := by
39  intro d hd a β ha hβ
40  have hd1 : 1 ≤ d := by omega
41  obtain ⟨B, M, μ, hS, hM, hρC, hρfix, hpt⟩ := clb_thermalDensity_eq_rep hd1 ha hβ
42  have hcb := t67cbar_pos (d := d) ha hβ
43  refine ⟨Real.exp (-β * (M * B)) / t67cbar d a β, div_pos (Real.exp_pos _) hcb,
44    Real.exp (β * B) / t67cbar d a β, fun x => ?_⟩
45  obtain ⟨hlo, hhi⟩ := clb_rep_bounds hS hβ.le (by linarith) hρC hρfix x
46  rw [hpt x]
47  rw [clb_gammaBar_eq] at hlo hhi
48  constructor
49  · calc Real.exp (-β * (M * B)) / t67cbar d a β * (1 + ‖x‖ ^ 2) ^ (β / 2) *
50          Real.exp (-β * a * ‖x‖ ^ 2 / 2)
51        = Real.exp (-β * (M * B)) *
52          ((1 + ‖x‖ ^ 2) ^ (β / 2) * Real.exp (-β * a * ‖x‖ ^ 2 / 2) / t67cbar d a β) := by ring
53      _ ≤ _ := hlo
54  · calc _ ≤ Real.exp (β * B) *
55          ((1 + ‖x‖ ^ 2) ^ (β / 2) * Real.exp (-β * a * ‖x‖ ^ 2 / 2) / t67cbar d a β) := hhi
56      _ = _ := by ring
