103/-- The high-frequency transport bound, from `high_identity_T`, `high_particles_T`, `high_background_T`. -/
104theorem high_bound_T (hd : 3 ≤ d) (hs0 : 0 ≤ s) (hsd : s < (d : ℝ) - 2) :
105    ∃ η₀ C : ℝ, 0 < η₀ ∧ η₀ ≤ 1 ∧ 0 < C ∧ ∀ p : ℝ≥0∞,
106      ENNReal.ofReal ((d : ℝ) / (d - s)) < p →
107      ∃ Cp0 : ℝ, 0 < Cp0 ∧ ∀ Cp : ℝ, Cp0 ≤ Cp → ∀ η : ℝ, 0 < η → η ≤ η₀ →
108      ∀ ρ : Torus d → ℝ, BddDensityT ρ → ∀ (N : ℕ) (X : Config d N), 1 ≤ N → X ∉ collisionSet d N →
109      ∀ (v : Torus d → Euc d) (K : ℝ), TorusSubTransport d s v K →
110        |CcommT (gloForceT s η) (PgtT η v) (chargeT X ρ)| ≤ C * K * scaleQT s Cp p η X ρ := by
111  obtain ⟨e₁, C₁, he₁, he₁1, hC₁, H₁⟩ := high_particles_T hd hs0 hsd
112  obtain ⟨e₂, C₂, he₂, he₂1, hC₂, H₂⟩ := high_background_T hd hs0 hsd
113  refine ⟨min e₁ e₂, 2 * C₁ + 2 * C₂, lt_min he₁ he₂, le_trans (min_le_left _ _) he₁1, by positivity,
114    fun p hp => ?_⟩
115  obtain ⟨c₁, hc₁, G₁⟩ := H₁ p hp
116  obtain ⟨c₂, hc₂, G₂⟩ := H₂ p hp
117  refine ⟨max c₁ c₂, lt_max_of_lt_left hc₁, fun Cp hCp η hη hηe ρ hρ N X hN hX v K hv => ?_⟩
118  have hη1 : η ≤ 1 := le_trans hηe (le_trans (min_le_left _ _) he₁1)
119  obtain ⟨L, hL0, hLip⟩ : ∃ L : ℝ, 0 ≤ L ∧ IsLipschitzVF L v := by
120    obtain ⟨L, F, hL, -, hLip, -, -⟩ := hv
121    exact ⟨L, hL, hLip⟩
122  have e1 := G₁ Cp (le_trans (le_max_left _ _) hCp) η hη (le_trans hηe (min_le_left _ _)) ρ hρ N X hN hX v K hv
123  have e2 := G₂ Cp (le_trans (le_max_right _ _) hCp) η hη (le_trans hηe (min_le_right _ _)) ρ hρ N X hN hX v K hv
124  rw [high_identity_T hd hs0 hsd η hη hη1 ρ hρ hN X hX v L hL0 hLip]
125  have hNpos : (0 : ℝ) < N := by exact_mod_cast hN
126  set A := ∑ i, inner ℝ (PgtT η v (X i)) ((chargeT X ρ).convForce (gloForceT s η) (X i)) with hA
127  set B := ∫ x, ρ x * inner ℝ (PgtT η v x) ((chargeT X ρ).convForce (gloForceT s η) x) with hB
128  have hA' : |A| ≤ ∑ i, |inner ℝ (PgtT η v (X i)) ((chargeT X ρ).convForce (gloForceT s η) (X i))| :=
129    Finset.abs_sum_le_sum_abs _ _
130  have h1 : |(2 / (N : ℝ)) * A| ≤ 2 * (C₁ * K * scaleQT s Cp p η X ρ) := by
131    rw [abs_mul, abs_of_pos (by positivity : (0 : ℝ) < 2 / N)]
132    calc 2 / (N : ℝ) * |A|
133        ≤ 2 / (N : ℝ) * ∑ i, |inner ℝ (PgtT η v (X i))
134            ((chargeT X ρ).convForce (gloForceT s η) (X i))| :=
135          mul_le_mul_of_nonneg_left hA' (by positivity)
136      _ = 2 * ((1 / (N : ℝ)) * ∑ i, |inner ℝ (PgtT η v (X i))
137            ((chargeT X ρ).convForce (gloForceT s η) (X i))|) := by ring
138      _ ≤ 2 * (C₁ * K * scaleQT s Cp p η X ρ) := by gcongr
139  have h2 : |2 * B| ≤ 2 * (C₂ * K * scaleQT s Cp p η X ρ) := by
140    rw [abs_mul, abs_two]
141    linarith
142  calc |(2 / (N : ℝ)) * A - 2 * B|
143      ≤ |(2 / (N : ℝ)) * A| + |2 * B| := abs_sub _ _
144    _ ≤ 2 * (C₁ * K * scaleQT s Cp p η X ρ) + 2 * (C₂ * K * scaleQT s Cp p η X ρ) := add_le_add h1 h2
145    _ = (2 * C₁ + 2 * C₂) * K * scaleQT s Cp p η X ρ := by ring

147/-- The unit-scale estimate `(5.unitestimate)` on the torus, original variables: for `η ≤ η₀`,
148`|𝖠₁[X,μ,v]| ≤ C·C(v)·Q_η` (`C_p` large).  Proved from the three bounds and the decomposition. -/
149theorem unit_estimate_T (hd : 3 ≤ d) (hs0 : 0 ≤ s) (hsd : s < (d : ℝ) - 2) :
150    ∃ η₀ C : ℝ, 0 < η₀ ∧ η₀ ≤ 1 ∧ 0 < C ∧ ∀ p : ℝ≥0∞,
151      ENNReal.ofReal ((d : ℝ) / (d - s)) < p →
152      ∃ Cp0 : ℝ, 0 < Cp0 ∧ ∀ Cp : ℝ, Cp0 ≤ Cp → ∀ η : ℝ, 0 < η → η ≤ η₀ →
153      ∀ ρ : Torus d → ℝ, BddDensityT ρ → ∀ (N : ℕ) (X : Config d N), 1 ≤ N → X ∉ collisionSet d N →
154      ∀ (v : Torus d → Euc d) (K : ℝ), TorusSubTransport d s v K →
155        |commAT s X ρ v| ≤ C * K * scaleQT s Cp p η X ρ := by
156  obtain ⟨e₁, C₁, he₁, he₁1, hC₁, H₁⟩ := remainder_bound_T hd hs0 hsd
157  obtain ⟨e₂, C₂, he₂, he₂1, hC₂, H₂⟩ := low_bound_T hd hs0 hsd
158  obtain ⟨e₃, C₃, he₃, he₃1, hC₃, H₃⟩ := high_bound_T hd hs0 hsd
159  refine ⟨min (min e₁ e₂) e₃, C₁ + C₂ + C₃, lt_min (lt_min he₁ he₂) he₃,
160    le_trans (min_le_right _ _) he₃1, by positivity, fun p hp => ?_⟩
161  obtain ⟨c₁, hc₁, G₁⟩ := H₁ p hp
162  obtain ⟨c₂, hc₂, G₂⟩ := H₂ p hp
163  obtain ⟨c₃, hc₃, G₃⟩ := H₃ p hp
164  refine ⟨max (max c₁ c₂) c₃, lt_max_of_lt_left (lt_max_of_lt_left hc₁),
165    fun Cp hCp η hη hηe ρ hρ N X hN hX v K hv => ?_⟩
166  have hηa : η ≤ e₁ := le_trans hηe (le_trans (min_le_left _ _) (min_le_left _ _))
167  have hηb : η ≤ e₂ := le_trans hηe (le_trans (min_le_left _ _) (min_le_right _ _))
168  have hηc : η ≤ e₃ := le_trans hηe (min_le_right _ _)
169  have hη1 : η ≤ 1 := le_trans hηc he₃1
170  obtain ⟨L, hL0, hLip⟩ : ∃ L : ℝ, 0 ≤ L ∧ IsLipschitzVF L v := by
171    obtain ⟨L, F, hL, -, hLip, -, -⟩ := hv
172    exact ⟨L, hL, hLip⟩
173  have x1 := G₁ Cp (le_trans (le_trans (le_max_left _ _) (le_max_left _ _)) hCp) η hη hηa ρ hρ N X hN hX v K hv
174  have x2 := G₂ Cp (le_trans (le_trans (le_max_right _ _) (le_max_left _ _)) hCp) η hη hηb ρ hρ N X hN hX v K hv
175  have x3 := G₃ Cp (le_trans (le_max_right _ _) hCp) η hη hηc ρ hρ N X hN hX v K hv
176  rw [commAT_eq_CcommT hd hs0 hsd ρ hρ hN X hX v L hL0 hLip,
177    CcommT_decomp hd hs0 hsd η hη hη1 ρ hρ hN X hX v L hL0 hLip]
178  calc |CcommT (ghiForceT s η) v (chargeT X ρ) + CcommT (gloForceT s η) (PleqT η v) (chargeT X ρ) +
179          CcommT (gloForceT s η) (PgtT η v) (chargeT X ρ)|
180      ≤ |CcommT (ghiForceT s η) v (chargeT X ρ) + CcommT (gloForceT s η) (PleqT η v) (chargeT X ρ)| +
181          |CcommT (gloForceT s η) (PgtT η v) (chargeT X ρ)| := abs_add_le _ _
182    _ ≤ |CcommT (ghiForceT s η) v (chargeT X ρ)| + |CcommT (gloForceT s η) (PleqT η v) (chargeT X ρ)| +
183          |CcommT (gloForceT s η) (PgtT η v) (chargeT X ρ)| := by
184        gcongr
185        exact abs_add_le _ _
186    _ ≤ C₁ * K * scaleQT s Cp p η X ρ + C₂ * K * scaleQT s Cp p η X ρ + C₃ * K * scaleQT s Cp p η X ρ := by
187        gcongr
188    _ = (C₁ + C₂ + C₃) * K * scaleQT s Cp p η X ρ := by ring

190/-- The exponent `d(1-1/p) - s` is positive for `p > d/(d-s)`. -/
191theorem exp_pos_T (hs0 : 0 ≤ s) (hsd : s < (d : ℝ)) (p : ℝ≥0∞)
192    (hp : ENNReal.ofReal ((d : ℝ) / (d - s)) < p) :
193    0 ≤ (d : ℝ) * (1 - (p⁻¹).toReal) - s := by
194  have hds : (0 : ℝ) < d - s := by linarith
195  have hd : (0 : ℝ) < d := by linarith
196  by_cases hpt : p = ⊤
197  · subst hpt
198    simp only [ENNReal.inv_top, ENNReal.toReal_zero, sub_zero, mul_one]
199    linarith
200  · have h := (ENNReal.ofReal_lt_iff_lt_toReal (by positivity) hpt).1 hp
201    have hq : 0 < p.toReal := lt_of_le_of_lt (by positivity) h
202    rw [ENNReal.toReal_inv]
203    rw [div_lt_iff₀ hds] at h
204    have h2 : (d : ℝ) / p.toReal < d - s := by
205      rw [div_lt_iff₀ hq]; linarith
206    have e : (d : ℝ) * (1 - p.toReal⁻¹) - s = (d - s) - d / p.toReal := by
207      rw [div_eq_mul_inv]; ring
208    linarith

210/-- Monotonicity in the scale (the range extension from `η ≤ η₂` to `η ≤ 1`, explicit algebra): for
211`η₂ ≤ η ≤ 1` and `C_p ≥ C_p' + |log η₂|/2 + C_p' η₂^{-s}`, `Q_{η₂}(C_p') ≤ Q_η(C_p)`. -/
212theorem scaleQT_mono (hs0 : 0 ≤ s) (p : ℝ≥0∞) (hexp : 0 ≤ (d : ℝ) * (1 - (p⁻¹).toReal) - s)
213    {N : ℕ} (hN : 1 ≤ N) (X : Config d N) (ρ : Torus d → ℝ) {Cp' Cp η₂ η : ℝ} (hCp' : 0 ≤ Cp')
214    (hη₂ : 0 < η₂) (h21 : η₂ ≤ η) (hη1 : η ≤ 1)
215    (hCp : Cp' + |Real.log η₂| / 2 + Cp' * η₂ ^ (-s) ≤ Cp) :
216    scaleQT s Cp' p η₂ X ρ ≤ scaleQT s Cp p η X ρ := by
217  have hη : 0 < η := lt_of_lt_of_le hη₂ h21
218  have hNpos : (0 : ℝ) < N := by exact_mod_cast hN
219  have hM : 0 ≤ normT p ρ := ENNReal.toReal_nonneg
220  have h1 : η₂ ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s) ≤ η ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s) :=
221    Real.rpow_le_rpow hη₂.le h21 hexp
222  have h2 : 1 ≤ η ^ (-s) := Real.one_le_rpow_of_pos_of_le_one_of_nonpos hη hη1 (by linarith)
223  have hlog : Real.log η ≤ 0 := Real.log_nonpos hη.le hη1
224  have hlog2 : (if s = 0 then Real.log η / (2 * (N : ℝ)) else 0) -
225      (if s = 0 then Real.log η₂ / (2 * (N : ℝ)) else 0) ≤ (|Real.log η₂| / 2) / N := by
226    split_ifs
227    · have : Real.log η₂ ≥ -|Real.log η₂| := neg_abs_le _
228      have h3 : Real.log η / (2 * (N : ℝ)) - Real.log η₂ / (2 * (N : ℝ)) ≤ (|Real.log η₂| / 2) / N := by
229        rw [← sub_div]
230        rw [div_le_div_iff₀ (by positivity) hNpos]
231        nlinarith
232      exact h3
233    · simp only [sub_zero]
234      positivity
235  unfold scaleQT
236  have hMη : 0 ≤ normT p ρ * η ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s) := by positivity
237  have hMη2 : normT p ρ * η₂ ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s) ≤
238      normT p ρ * η ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s) := mul_le_mul_of_nonneg_left h1 hM
239  have hCpCp' : Cp' ≤ Cp := by
240    have : 0 ≤ |Real.log η₂| / 2 + Cp' * η₂ ^ (-s) := by positivity
241    linarith
242  have h4 : Cp' * (η₂ ^ (-s) / (N : ℝ)) + (|Real.log η₂| / 2) / N ≤ Cp / N := by
243    rw [← mul_div_assoc, ← add_div]
244    exact div_le_div_of_nonneg_right (by linarith) hNpos.le
245  have h5 : Cp / N ≤ Cp * (η ^ (-s) / (N : ℝ)) := by
246    have : 0 ≤ Cp := le_trans hCp' hCpCp'
247    rw [← mul_div_assoc]
248    exact div_le_div_of_nonneg_right (by nlinarith) hNpos.le
249  have h6 : Cp' * (normT p ρ * η₂ ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s)) ≤
250      Cp * (normT p ρ * η ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s)) :=
251    mul_le_mul hCpCp' hMη2 (by positivity) (le_trans hCp' hCpCp')

535    have : (3 : ℝ) ≤ d := by exact_mod_cast hd
536    linarith
537  have hNM : 0 < (N : ℝ) * normT ⊤ ρ := mul_pos hNpos hM
538  have hlam : 0 < lamT d N ρ := Real.rpow_pos_of_pos hNM _
539  have hNM1 : 1 ≤ (N : ℝ) * normT ⊤ ρ := by
540    have hN1 : (1 : ℝ) ≤ N := by exact_mod_cast hN
541    nlinarith
542  have hlam1 : lamT d N ρ ≤ 1 := by
543    unfold lamT
544    exact Real.rpow_le_one_of_one_le_of_nonpos hNM1 (by
545      have : (0 : ℝ) < d := by
546        have : (3 : ℝ) ≤ d := by exact_mod_cast hd
547        linarith
548      have : (0 : ℝ) ≤ 1 / d := by positivity
549      rw [neg_div]; linarith)
550  have hlamd : lamT d N ρ ^ (d : ℝ) = ((N : ℝ) * normT ⊤ ρ)⁻¹ := by
551    unfold lamT
552    rw [← Real.rpow_mul hNM.le, show (-(1 : ℝ) / d) * d = -1 by field_simp, Real.rpow_neg_one]
553  have hkey : lamT d N ρ ^ (-s) / (N : ℝ) = normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s) := by
554    rw [sub_eq_add_neg, Real.rpow_add hlam, hlamd]
555    field_simp
556  obtain ⟨hE, hCI, hQ, hA⟩ := S' ρ h1 hinf hmass v K hv N X hN hX (lamT d N ρ) hlam hlam1
557  have hQe : scaleQT s Cp ⊤ (lamT d N ρ) X ρ =
558      modEnergyT s X ρ - (if s = 0 then Real.log (lamT d N ρ) / (2 * (N : ℝ)) else 0) +
559        2 * Cp * normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s) := by
560    unfold scaleQT
561    simp only [ENNReal.inv_top, ENNReal.toReal_zero, sub_zero, mul_one]
562    rw [hkey]
563    ring
564  rw [hQe] at hQ hA
565  exact ⟨hE, hCI, hQ, hA⟩
