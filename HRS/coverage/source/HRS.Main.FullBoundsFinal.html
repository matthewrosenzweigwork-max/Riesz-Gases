51    ∃ C : ℝ, 0 < C ∧
52      ∀ a : ℝ, 0 ≤ a → ∀ N : ℕ, 2 ≤ N →
53      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
54        (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N),
55        SelectedFlow P S.dyn (a + T) X₀ W X →
56        (∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
57          (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X) →
58        law P X₀ = modulatedGibbs (periodicRiesz d S.s) N β₀ S.μ0 →
59      ∀ (ν : Measure (NegSobolev d m)),
60        IsCenteredGaussianWithCov m ν (preparedCov d S.s β₀ S.μ0) →
61      ∀ {Ω' : Type} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
62        (Y₀ : Ω' → NegSobolev d m) (Y : ℝ → Ω' → NegSobolev d m),
63        IsPreparedLinear S m (a + T) ν P' Y₀ Y →
64        (β₀ = 0 →
65          smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
66            C * (Real.sqrt N * deltaN d S.s N + iidRateSmooth d m N) ∧
67          wassersteinE 1 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
68            ENNReal.ofReal (C * (Real.sqrt N * deltaN d S.s N +
69              deltaN d S.s N ^ ((1 : ℝ) / 4) + iidRateW d m N))) ∧
70        (0 < β₀ →
71          smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
72            C * (Real.sqrt N * deltaN d S.s N) ∧
73          wassersteinE 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
74            ENNReal.ofReal (C * (Real.sqrt N * deltaN d S.s N +
75              deltaN d S.s N ^ ((1 : ℝ) / 4)))) ∧
76        (β₀ = 0 → S.s = 0 →
77          wassersteinE 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
