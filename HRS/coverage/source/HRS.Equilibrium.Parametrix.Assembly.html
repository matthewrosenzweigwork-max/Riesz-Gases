29theorem pair_poisson_gen (H : ParxHyp d s V β ρ) (J : ℕ) (hspec : PairInverseSpec H J) (K : ℕ)
30    (hK : parxK d J ≤ K) :
31    ∃ (Ψ : (Torus d → ℝ) → Torus d → Torus d → ℝ) (C : ℝ),
32      (∀ (f f' : Torus d → ℝ) (a b : ℝ), Torus.IsSmooth f → Torus.IsSmooth f' →
33        Ψ (a • f + b • f') = a • Ψ f + b • Ψ f') ∧
34      ∀ f : Torus d → ℝ, Torus.IsSmooth f →
35        IsPairCorrector d s V β ρ f (C * posSeminorm K f) (Ψ f) ∧
36        (∃ M : ℝ, ∀ x, |aCoef d s ρ f x| ≤ M) ∧
37        eLpNorm (aCoef d s ρ f) 2 (withDensityReal ρ) ≤ ENNReal.ofReal (C * posSeminorm K f) := by
38  obtain ⟨C₁, G⟩ := parx_gap_proof H J
39  have hspec' := hspec
40  obtain ⟨Cinv, hCinv0, hinv, hlin, hsym, -⟩ := hspec
41  obtain ⟨CJ, hCJ0, hCJ⟩ := jpair_smooth H
42  obtain ⟨Ca, hCa0, hCa⟩ := aCoef_bound H
43  have : IsProbabilityMeasure (withDensityReal ρ) :=
44    PDE.MFSemigroup.isProb_withDensityReal H.rho_smooth.density
45  have hC₁ : 0 ≤ C₁ := G.C_nonneg
46  set L := parxL d J with hL
47  set Cst : ℝ := C₁ * (1 + Cinv) * (1 + CJ) + Ca with hCst
48  have hK15 : 2 ≤ K := by unfold parxK at hK; omega
49  refine ⟨fun f x y => parxPsi d s V β ρ f L x y +
50      PDE.pairInverse H.dyn ρ (diagExt (parxR d s V β ρ f L)) x y, Cst, ?_, ?_⟩
51  · -- linearity
52    intro f f' a b hf hf'
53    funext x y
54    have h1 := congrFun (congrFun (G.linear_psi f f' a b hf hf') x) y
55    have h2 : PDE.pairInverse H.dyn ρ (diagExt (parxR d s V β ρ (a • f + b • f') L)) =
56        fun x y => a * PDE.pairInverse H.dyn ρ (diagExt (parxR d s V β ρ f L)) x y +
57          b * PDE.pairInverse H.dyn ρ (diagExt (parxR d s V β ρ f' L)) x y := by
58      rw [G.linear_R f f' a b hf hf']
59      exact hlin _ _ a b (G.remainder_ck f hf).1 (G.remainder_ck f' hf').1
60    have h3 := congrFun (congrFun h2 x) y
61    simp only [Pi.add_apply, Pi.smul_apply, smul_eq_mul]
62    rw [h1, h3]
63    ring
64  · intro f hf
65    obtain ⟨hck, hdc, hnorm, hRsym⟩ := G.remainder_ck f hf
66    set Rf := diagExt (parxR d s V β ρ f L) with hRf
67    set I := PDE.pairInverse H.dyn ρ Rf with hI
68    obtain ⟨hI_ck, hI_dc, hI_norm, hI_eq⟩ := hinv Rf hck hdc
69    set P := posSeminorm (parxK d J) f with hP
70    set P' := posSeminorm K f with hP'
71    have hP0 : 0 ≤ P := posSeminorm_nonneg' _ _
72    have hPP : P ≤ P' := posSeminorm_mono' hK f
73    have hI_bd : PDE.cbNorm2 2 I ≤ Cinv * (C₁ * P) :=
74      hI_norm.trans (mul_le_mul_of_nonneg_left hnorm hCinv0)
75    have hIcont : Continuous fun p : Torus d × Torus d => I p.1 p.2 :=
76      PDE.MFPair.continuous_pair_of_continuous_lift2 hI_ck.continuous
77    have hIslice : ∀ x, Continuous fun y => I x y := fun x =>
78      hIcont.comp (continuous_const.prodMk continuous_id)
79    have hCst1 : C₁ * (1 + Cinv) ≤ Cst := by
80      have : 0 ≤ C₁ * (1 + Cinv) * CJ := by positivity
81      nlinarith
82    have hCst2 : C₁ + CJ * (Cinv * C₁) ≤ Cst := by
83      have : 0 ≤ C₁ * Cinv * (1 + CJ) := by positivity
84      nlinarith
85    have hP'0 : 0 ≤ P' := posSeminorm_nonneg' _ _
86    have hCst0 : 0 ≤ Cst := by rw [hCst]; positivity
87    have hsmooth := parx_psi_smooth_proof H J hspec' f hf
88    refine ⟨⟨?_, ?_, ?_, ?_, ?_, ?_, ?_⟩, ?_, ?_⟩
89    · -- measurable
90      exact (G.measurable f hf).add hIcont.measurable
91    · -- symmetric
92      intro x y
93      have h2 : I x y = I y x := hsym Rf hck hRsym x y
94      exact congrArg₂ (· + ·) (G.symm f hf x y) h2
95    · -- canonical
96      intro x
97      obtain ⟨hi, h0⟩ := G.canonical f hf x
98      have hi2 : Integrable (fun y => I x y) (withDensityReal ρ) :=
99        (hIslice x).integrable_of_hasCompactSupport (HasCompactSupport.of_compactSpace _)
100      have h02 : ∫ y, I x y ∂(withDensityReal ρ) = 0 := hI_dc.2 x
101      exact ⟨hi.add hi2, by rw [integral_add hi hi2, h0, h02]; ring⟩
102    · -- size
103      have hs1 : PsiSize s (C₁ * P) (parxPsi d s V β ρ f L) := G.size f hf
104      have hs2 := PsiSize.add_ck2 hs1 hI_ck hI_bd hsmooth
105      refine PsiSize.mono hs2 ?_
106      have : C₁ * P + Cinv * (C₁ * P) ≤ Cst * P' := by
107        have h1 : C₁ * P + Cinv * (C₁ * P) = C₁ * (1 + Cinv) * P := by ring
108        rw [h1]
109        calc C₁ * (1 + Cinv) * P ≤ Cst * P := mul_le_mul_of_nonneg_right hCst1 hP0
110          _ ≤ Cst * P' := mul_le_mul_of_nonneg_left hPP hCst0
111      exact this
112    · -- Poisson equation
113      intro x y hxy
114      obtain ⟨hA1, hA2, -⟩ := G.add_smooth f hf I hI_ck x y hxy
115      have htel := G.telescope f hf x y hxy
116      have hRxy : Rf x y = parxR d s V β ρ f L x y := by simp [hRf, diagExt, hxy]
117      have hg1 : genFst d s V β ρ I x y = PDE.centeredGen H.dyn ρ (fun x' => I x' y) x := by
118        have := congrFun (projGen_eq_centeredGen H (fun z => I z y)
119          (by simpa using PDE.MFPair.isCk_slice 2 I hI_ck y)) x
120        exact this
121      have hg2 : genSnd d s V β ρ I x y = PDE.centeredGen H.dyn ρ (fun y' => I x y') y := by
122        have := congrFun (projGen_eq_centeredGen H (fun z => I x z)
123          (by simpa using PDE.MFPair.isCk_slice2 2 I hI_ck x)) y
124        exact this
125      have heq := hI_eq x y
126      show -(genFst d s V β ρ (fun x y => parxPsi d s V β ρ f L x y + I x y) x y +
127        genSnd d s V β ρ (fun x y => parxPsi d s V β ρ f L x y + I x y) x y) = Hsource d s ρ f x y
128      rw [hA1, hA2, hg1, hg2]
129      linarith
130    · -- 𝒥ψ canonical
131      intro x
132      obtain ⟨hi, h0⟩ := G.jpair_canonical f hf x
133      obtain ⟨hcI, -⟩ := hCJ I hI_ck hI_dc
134      obtain ⟨hiI, h0I⟩ := hcI x
135      have hae : ∀ᵐ y ∂(withDensityReal ρ),
136          Jpair d s ρ (fun x y => parxPsi d s V β ρ f L x y + I x y) x y =
137            Jpair d s ρ (parxPsi d s V β ρ f L) x y + Jpair d s ρ I x y := by
138        have hne : ∀ᵐ y : Torus d, y ≠ x := by
139          have := PDE.MFSchauder.ae_add_ne_zero (d := d) (by have := H.two_le; omega) (-x)
140          filter_upwards [this] with y hy
141          intro hyx; apply hy; rw [hyx]; simp
142        have hne' := (withDensity_absolutelyContinuous (volume : Measure (Torus d))
143          (fun x => ENNReal.ofReal (ρ x))).ae_le hne
144        filter_upwards [hne'] with y hy
145        exact (G.add_smooth f hf I hI_ck x y (Ne.symm hy)).2.2
146      refine ⟨(integrable_congr hae).mpr (hi.add hiI), ?_⟩
147      rw [integral_congr_ae hae, integral_add hi hiI, h0, h0I]; ring
148    · -- 𝒥ψ bound
149      intro x y hxy
150      have h1 := G.jpair_bound f hf x y hxy
151      have h2 := (hCJ I hI_ck hI_dc).2 x y
152      have hw : 1 ≤ wEnv s x y := PairCorrAux.one_le_wEnv s x y
153      have h3 : Jpair d s ρ (fun x y => parxPsi d s V β ρ f L x y + I x y) x y =
154          Jpair d s ρ (parxPsi d s V β ρ f L) x y + Jpair d s ρ I x y :=
155        (G.add_smooth f hf I hI_ck x y hxy).2.2
156      rw [h3]
157      have h4 : |Jpair d s ρ I x y| ≤ CJ * (Cinv * (C₁ * P)) :=
158        h2.trans (mul_le_mul_of_nonneg_left hI_bd hCJ0)
159      have h5 : 0 ≤ P := hP0
160      calc |Jpair d s ρ (parxPsi d s V β ρ f L) x y + Jpair d s ρ I x y|
161          ≤ |Jpair d s ρ (parxPsi d s V β ρ f L) x y| + |Jpair d s ρ I x y| := abs_add_le _ _
162        _ ≤ C₁ * P * wEnv s x y + CJ * (Cinv * (C₁ * P)) * wEnv s x y :=
163            add_le_add h1 (h4.trans (le_mul_of_one_le_right (by positivity) hw))
164        _ = (C₁ + CJ * (Cinv * C₁)) * P * wEnv s x y := by ring
165        _ ≤ Cst * P' * wEnv s x y := by
166            have hw0 : 0 ≤ wEnv s x y := by linarith
167            have hc0 : 0 ≤ C₁ + CJ * (Cinv * C₁) := by positivity
168            have : (C₁ + CJ * (Cinv * C₁)) * P ≤ Cst * P' :=
169              mul_le_mul hCst2 hPP hP0 (hc0.trans hCst2)
170            exact mul_le_mul_of_nonneg_right this hw0
171    · -- a_f bounded
172      obtain ⟨-, hb⟩ := hCa f hf
173      exact ⟨Ca * posSeminorm 2 f, hb⟩
174    · -- a_f in L²
175      obtain ⟨hm, hb⟩ := hCa f hf
176      have hP2 : posSeminorm 2 f ≤ P' := posSeminorm_mono' hK15 f
177      have hae : ∀ᵐ x ∂(withDensityReal ρ), ‖aCoef d s ρ f x‖ ≤ Cst * P' :=
178        Filter.Eventually.of_forall fun x => by
179          rw [Real.norm_eq_abs]
180          calc |aCoef d s ρ f x| ≤ Ca * posSeminorm 2 f := hb x
181            _ ≤ Ca * P' := mul_le_mul_of_nonneg_left hP2 hCa0
182            _ ≤ Cst * P' := by
183                refine mul_le_mul_of_nonneg_right ?_ hP'0
184                have : 0 ≤ C₁ * (1 + Cinv) * (1 + CJ) := by positivity
185                linarith
186      calc eLpNorm (aCoef d s ρ f) 2 (withDensityReal ρ)
187          ≤ (withDensityReal ρ) Set.univ ^ (2 : ℝ≥0∞).toReal⁻¹ * ENNReal.ofReal (Cst * P') :=
188            eLpNorm_le_of_ae_bound hm hae
189        _ = ENNReal.ofReal (Cst * P') := by simp

197/-- Proposition 4.13 `M:Q:prop:poisson`: `pair_poisson` verbatim, proved from the gap lemma
198`Parx.parx_gap`, the smoothness bootstrap `Parx.parx_psi_smooth`, the pair inverse
199`Parx.pair_inverse_small` and the glue lemmas of `Parametrix/Glue.lean` (all `sorry` packets). -/
200theorem pair_poisson_proof (hd : 2 ≤ d) (hcrit : s = (d : ℝ) / 2) (β : ℝ) (hβ : 0 < β)
201    (V : Torus d → ℝ) (hV : Torus.IsSmooth V) (ρ : Torus d → ℝ)
202    (hρ : IsThermalDensity (periodicRiesz d s) V β ρ) (hρs : SmoothPosDensity ρ) :
203    ∃ (Ψ : (Torus d → ℝ) → Torus d → Torus d → ℝ) (C : ℝ),
204      (∀ (f f' : Torus d → ℝ) (a b : ℝ), Torus.IsSmooth f → Torus.IsSmooth f' →
205        Ψ (a • f + b • f') = a • Ψ f + b • Ψ f') ∧
206      ∀ f : Torus d → ℝ, Torus.IsSmooth f →
207        IsPairCorrector d s V β ρ f (C * posSeminorm (8 * d + 80) f) (Ψ f) ∧
208        (∃ M : ℝ, ∀ x, |aCoef d s ρ f x| ≤ M) ∧
209        eLpNorm (aCoef d s ρ f) 2 (withDensityReal ρ) ≤
210          ENNReal.ofReal (C * posSeminorm (8 * d + 80) f) := by
211  have H : Parx.ParxHyp d s V β ρ := ⟨hd, hcrit, hβ, hV, hρ, hρs⟩
212  exact Parx.pair_poisson_gen H (Parx.parxJ d) (Parx.pair_inverse_small H) (8 * d + 80)
213    (by unfold Parx.parxK Parx.parxJ; omega)
