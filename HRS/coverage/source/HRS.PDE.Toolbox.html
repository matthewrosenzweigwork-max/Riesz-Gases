26## Scope and representation (SPEC_NOTES §1, §4)
27
28* C.3 is stated in the source "on `𝕋^d` or `ℝ^d`"; **only the torus versions are specified here**;
29  the `ℝ^d` versions belong to the whole-space packages and are recorded as not covered.
30* A Fourier multiplier `K` with symbol `𝔪` acts on functions through `coeffOf`; its quadratic form
31  is `symbolForm 𝔪 u v = Re ∑_k 𝔪(k) û(k) \overline{v̂(k)}` (meaningful under the stated summability,
32  which is part of every conclusion).  `multiplierForm` of `HRS/Kernel/Riesz.lean` (real basis) is
33  the same form (`symbolForm_eq_multiplierForm`).
34* C.4 defines `⟨κ*(aw), w⟩` for a non-integrable odd kernel by symmetric truncation; the statement
35  is given in that form (symmetrized double integral + convergence of the truncations).  The
36  identification of this real-space pairing with the Fourier-side pairing for `κ = v·∇g` is the
37  **required kernel fact** `rieszGrad_pairing_symmetrized` (extends KERN-01).
38* All real suprema (`cbNorm`, `supNorm`, `holderSeminorm`) are applied only under hypotheses
39  (`IsCk`, `IsLipschitz`) that make them honest; on the compact torus a `C^k` function has bounded
40  derivatives.

54/-- `B(u,v) = ⟨Ku, v⟩_{L²} = Re ∑_k 𝔪(k) û(k) \overline{v̂(k)}` for a real multiplier. -/
55def symbolForm (𝔪 : Lattice d → ℝ) (u v : Torus d → ℝ) : ℝ :=
56  (bilinForm 𝔪 (coeffOf u) (coeffOf v)).re

58/-- The quadratic form `∑_k 𝔪(k)|û(k)|²` is finite. -/
59def SymbolFormFinite (𝔪 : Lattice d → ℝ) (u : Torus d → ℝ) : Prop :=
60  Summable fun k => 𝔪 k * ‖coeffOf u k‖ ^ 2
61
62/-- `‖w‖²_{L²(𝕋^d)}`. -/
63def l2NormSq (w : Torus d → ℝ) : ℝ := ∫ x, w x ^ 2

65/-- `u ∈ C^{1,1}_b(𝕋^d)` with Lipschitz constant `L` of the gradient. -/
66def IsC11 (L : ℝ) (u : Torus d → ℝ) : Prop :=
67  Torus.IsCk 1 u ∧ LipschitzWith (Real.toNNReal L) (fun x => fderiv ℝ (Torus.lift u) x)

69/-- `C^k` functions of two torus variables, through the lift to `ℝ^d × ℝ^d`. -/
70def lift2 (R : Torus d → Torus d → ℝ) : Euc d × Euc d → ℝ := fun p => R (Torus.mk p.1) (Torus.mk p.2)

72/-- `R ∈ C^k(𝕋^d × 𝕋^d)`. -/
73def IsCk2 (k : ℕ) (R : Torus d → Torus d → ℝ) : Prop := ContDiff ℝ k (lift2 R)

75/-- The `C^k` norm (sum over orders) of a function of two torus variables. -/
76def cbNorm2 (k : ℕ) (R : Torus d → Torus d → ℝ) : ℝ :=
77  ∑ j ∈ Finset.range (k + 1), ⨆ p : Euc d × Euc d, ‖iteratedFDeriv ℝ j (lift2 R) p‖

79/-- The symbol class of Lemma C.3: `𝔪` is the restriction to `ℤ^d` of a real, even, nonnegative
80function on `ℝ^d`, smooth away from `0` and homogeneous of degree `θ` there, with `𝔪(0) = 0`
81(lines 12254–12255). -/
82structure IsHomogSymbol (θ : ℝ) (𝔪 : Lattice d → ℝ) : Prop where
83  exists_ext : ∃ m₀ : Euc d → ℝ,
84    ContDiffOn ℝ ((⊤ : ℕ∞) : WithTop ℕ∞) m₀ {ξ | ξ ≠ 0} ∧ (∀ ξ, m₀ (-ξ) = m₀ ξ) ∧ (∀ ξ, 0 ≤ m₀ ξ) ∧
85      (∀ c : ℝ, 0 < c → ∀ ξ, m₀ (c • ξ) = c ^ θ * m₀ ξ) ∧ (∀ k : Lattice d, k ≠ 0 → 𝔪 k = m₀ (Lattice.toEuc k))
86  zero : 𝔪 0 = 0

88/-- The Lévy-kernel representation of Lemma C.3(ii) (lines 12257–12261): an even kernel `𝔨`,
89smooth away from `0`, with `|𝔨(z)| ≤ C(|z|^{-d-θ} + 1)`, such that
90`∫ 𝔨(z)(1 - cos 2πk·z) dz = 𝔪(k)` (absolutely convergent), the principal-value formula
91`Ku(x) = lim_{ε→0} ∫_{|z|>ε} 𝔨(z)(u(x) - u(x-z)) dz` for `u ∈ C^{1,1}_b` (pointwise, where `Ku`
92is the function with coefficients `𝔪 û`), and the bilinear identity
93`B(u,v) = ½ ∬ 𝔨(z) δ_z u(x) δ_z v(x) dz dx` (absolutely convergent) for Lipschitz `u, v`. -/
94structure IsLevyKernel (θ : ℝ) (𝔪 : Lattice d → ℝ) (𝔨 : Torus d → ℝ) : Prop where
95  even : ∀ z, 𝔨 (-z) = 𝔨 z
96  smooth : Torus.IsSmoothOffZero 𝔨
97  bound : ∃ C : ℝ, ∀ z : Torus d, z ≠ 0 → |𝔨 z| ≤ C * (Torus.tdist z 0 ^ (-(d : ℝ) - θ) + 1)
98  symbol : ∀ k : Lattice d, Integrable (fun z => 𝔨 z * (1 - Torus.cosMode k z)) ∧
99    ∫ z, 𝔨 z * (1 - Torus.cosMode k z) = 𝔪 k
100  pv : ∀ (L : ℝ) (u : Torus d → ℝ), IsC11 L u → ∃ Ku : Torus d → ℝ,
101    MemLp Ku 2 volume ∧ coeffOf Ku = (fun k => ((𝔪 k : ℝ) : ℂ) * coeffOf u k) ∧
102    ∀ x, Tendsto (fun ε : ℝ => ∫ z in {z | ε < Torus.tdist z 0}, 𝔨 z * (u x - u (x - z)))
103      (𝓝[>] 0) (𝓝 (Ku x))
104  bilinear : ∀ (Lu Lv : ℝ) (u v : Torus d → ℝ), Torus.IsLipschitz Lu u → Torus.IsLipschitz Lv v →
105    Integrable (fun p : Torus d × Torus d => 𝔨 p.2 * (u p.1 - u (p.1 - p.2)) * (v p.1 - v (p.1 - p.2))) ∧
106    symbolForm 𝔪 u v =
107      (1 / 2) * ∫ p : Torus d × Torus d, 𝔨 p.2 * (u p.1 - u (p.1 - p.2)) * (v p.1 - v (p.1 - p.2))

195/-- Lemma C.3(iii) (lines 12262–12264), weighted symmetrization: for `0 ≤ a ∈ C^{1,1}_b` with
196`m = √a` (Lipschitz by C.6) and Lipschitz `w`,
197`B(aw,w) = B(mw,mw) + E`, `E = -½ ∬ 𝔨(z) w(x) w(x-z) (δ_z m(x))² dz dx`, and
198`|E| ≤ C (‖∇m‖²_∞ + ‖m‖²_∞) ‖w‖²_{L²}` with `C` depending on the kernel only. -/

207      E = -(1 / 2) * ∫ p : Torus d × Torus d, 𝔨 p.2 * w p.1 * w (p.1 - p.2) * (m p.1 - m (p.1 - p.2)) ^ 2 ∧

338/-- A Friedrichs cutoff `χ₁ ∈ C_c^∞(ℝ^d;[0,1])`, radial, `≡ 1` on `B(0,1)`, supported in `B(0,2)`
339(Step 1 of the proof of Prop 3.9, line 1957). -/
340structure IsFriedrichsCutoff (χ : Euc d → ℝ) : Prop where
341  smooth : ContDiff ℝ ((⊤ : ℕ∞) : WithTop ℕ∞) χ
342  range : ∀ ξ, 0 ≤ χ ξ ∧ χ ξ ≤ 1
343  radial : ∀ ξ η : Euc d, ‖ξ‖ = ‖η‖ → χ ξ = χ η
344  one : ∀ ξ, ‖ξ‖ ≤ 1 → χ ξ = 1
345  support : ∀ ξ, 2 ≤ ‖ξ‖ → χ ξ = 0
346
347/-- The mollifier `J_ε = χ₁(εD)²` (Fourier multiplier `χ₁(εk)²`), on coefficient sequences. -/
348def mollify (χ : Euc d → ℝ) (ε : ℝ) (c : Coeff d) : Coeff d :=
349  fun k => ((χ (ε • Lattice.toEuc k) ^ 2 : ℝ) : ℂ) * c k
350
351/-- The commutator `[J_ε, a]u = J_ε(au) - a J_ε u` (coefficients). -/
352def mollifyCommutator (χ : Euc d → ℝ) (ε : ℝ) (a : Torus d → ℝ) (u : Coeff d) : Coeff d :=
353  mollify χ ε (mulCoeff (coeffOf a) u) - mulCoeff (coeffOf a) (mollify χ ε u)
354
355/-- The transport commutator `[J_ε, b·∇]u = J_ε(b·∇u) - b·∇(J_ε u)` (coefficients; `b·∇u` is the
356distribution `∑_j b_j ∂_j u` for `u ∈ L²`). -/
357def mollifyTransportCommutator (χ : Euc d → ℝ) (ε : ℝ) (b : Fin d → Torus d → ℝ) (u : Coeff d) :
358    Coeff d :=
359  mollify χ ε (fun k => ∑ j, mulCoeff (coeffOf (b j)) (derivCoeff j u) k) -
360    fun k => ∑ j, mulCoeff (coeffOf (b j)) (derivCoeff j (mollify χ ε u)) k
361
362/-- Lemma C.7 `B:lem:friedrichs` (lines 12319–12321), torus version with `J_ε = χ₁(εD)²`:
363for `a ∈ C¹_b`, `u ∈ L²`, `ε ∈ (0,1]`:
364`‖[J_ε,a]u‖_{L²} ≤ C ε ‖∇a‖_∞ ‖u‖_{L²}` and `‖[J_ε,a]u‖_{H^σ} ≤ C ε^{1-σ} ‖a‖_{C¹} ‖u‖_{L²}` for
365`σ ∈ [0,1]`. -/
366theorem friedrichs_commutator (hd : 1 ≤ d) (χ : Euc d → ℝ) (hχ : IsFriedrichsCutoff χ) :
367    ∃ C : ℝ, ∀ (a : Torus d → ℝ), Torus.IsCk 1 a → ∀ (u : Coeff d), MemHs 0 u →
368      ∀ ε : ℝ, 0 < ε → ε ≤ 1 →
369        MemHs 1 (mollifyCommutator χ ε a u) ∧
370        hsNorm 0 (mollifyCommutator χ ε a u) ≤
371          C * ε * (⨆ x : Euc d, ‖fderiv ℝ (Torus.lift a) x‖) * hsNorm 0 u ∧
372        ∀ σ : ℝ, 0 ≤ σ → σ ≤ 1 →
373          hsNorm σ (mollifyCommutator χ ε a u) ≤ C * ε ^ (1 - σ) * Torus.cbNorm 1 a * hsNorm 0 u := by
374  obtain ⟨C, -, hC⟩ := Fried.friedrichs_main (d := d)
375    (χ := χ) ⟨hχ.smooth, hχ.range, hχ.one, hχ.support⟩
376  exact ⟨C, hC⟩

378/-- Lemma C.7, transport commutator: for `b ∈ C¹_b` vector-valued and `u ∈ L²`,
379`‖[J_ε, b·∇]u‖_{L²} ≤ C ‖∇b‖_∞ ‖u‖_{L²}` uniformly in `ε ∈ (0,1]`, and
380`[J_ε, b·∇]u → 0` in `L²` as `ε → 0` for each fixed `u`. -/
381theorem friedrichs_transport_commutator (hd : 1 ≤ d) (χ : Euc d → ℝ) (hχ : IsFriedrichsCutoff χ) :

387      Tendsto (fun ε => hsNorm 0 (mollifyTransportCommutator χ ε b u)) (𝓝[>] 0) (𝓝 0) := by

390  exact ⟨C, hC⟩
