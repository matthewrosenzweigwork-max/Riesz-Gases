1/-
2# HRS.ME.FamW (packet W4b): the lower bound of `critical_family_sharpness`
3
4`critical_family_sharpness_lower` is the lower conjunct of the frozen `HRS.Main.critical_family_sharpness`
5(`H:eq:critical-family-sharpness`), with constants `c`, `N₀` chosen before the density family `ρ`.
6
7Mechanism.  Let `R = 16 (1 + β ĝ(k))`, `v₀ = 1/(2 (1 + β ĝ(k))) = 8/R`.
8* `rateVarN N 0 → v₀` (`rateVarN_zero_tendsto`, with the uniform density and `susceptibilityCov_one_cos`),
9  so `rateVarN N 0 ≥ v₀/2` for large `N`;
10* `∫₀^R rateVarN N ≤ 2` and `rateVarN' = √N E[Y³]` give `∫₀^R |√N E Y_{N,θ}³| dθ ≥ v₀/2 - 2/R = v₀/4`
11  (`rate_variation_lower`);
12* `|E Y³| ≤ 3 √(32 D²) · W₂(law Y, Gaussian)` pointwise (`third_moment_pointwise`), with `D` the
13  sub-Gaussian constant of Lemma 4.19, independent of `ρ` because the thermal density is unique
14  (`thermal_eq_of_thermal`);
15* integrating, `v₀/4 ≤ √N · K · ∫⁻ W₂`, i.e. the stated bound with `c = v₀/(4K)`, `K = 3 √(32 D²)`.
16-/

33theorem thermal_eq_of_thermal (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d) {β : ℝ} (hβ : 0 < β)
34    (k : Lattice d) (θ : ℝ) (ρ ρ' : Torus d → ℝ)
35    (hρ : IsThermalDensity (periodicRiesz d s) (rateV k β θ) β ρ)
36    (hρ' : IsThermalDensity (periodicRiesz d s) (rateV k β θ) β ρ') : ρ = ρ' := by

40  exact (PDE.MFThermal.thermal_unique hd s hs0 hsd β hβ (rateV k β θ) ρ ρ' hρ hcont hpos
41    hρ' hcont' hpos').symm

43/-- **Pointwise third-moment bound.**  If the finite-`N` law of `Y_{N,θ}` and the Gaussian
44`N(0, v)` satisfy the sub-Gaussian tail bound with constant `D`, then
45`|E Y³| ≤ 3 √(32 D²) · W₂(law Y, N(0,v))`. -/
46theorem third_moment_pointwise (hd : 2 ≤ d) (hs : (d : ℝ) / 2 ≤ s) (hsd : s < d) {β : ℝ}
47    (hβ : 0 < β) (k : Lattice d) {N : ℕ} (hN : 0 < N) (θ v D : ℝ) (hD : 0 < D)
48    (htail : ∀ x : ℝ, 0 ≤ x →
49      rateGibbs d s k β θ N {X | x < |rateY d s k β N θ X|} +
50          gaussianReal 0 v.toNNReal {z | x < |z|} ≤ ENNReal.ofReal (4 * Real.exp (-x ^ 2 / (2 * D)))) :
51    ENNReal.ofReal |∫ X, rateY d s k β N θ X ^ 3 ∂(rateGibbs d s k β θ N)| ≤
52      ENNReal.ofReal (3 * Real.sqrt (32 * D ^ 2)) *
53        wassersteinE 2 ((rateGibbs d s k β θ N).map (rateY d s k β N θ)) (gaussianReal 0 v.toNNReal) := by
54  haveI := rateGibbs_isProb' hd hs hsd hβ k hN θ
55  have hY : Measurable (rateY d s k β N θ) := HRS.ME.FamV.measurable_rateY s k β θ N
56  haveI hP : IsProbabilityMeasure ((rateGibbs d s k β θ N).map (rateY d s k β N θ)) :=
57    (Measure.isProbabilityMeasure_map_iff hY.aemeasurable).2 inferInstance
58  haveI hQ : IsProbabilityMeasure (gaussianReal 0 v.toNNReal) := inferInstance
59  have hPtail : ∀ x : ℝ, 0 ≤ x →
60      ((rateGibbs d s k β θ N).map (rateY d s k β N θ)) {z | x < |z|} ≤
61        ENNReal.ofReal (4 * Real.exp (-x ^ 2 / (2 * D))) := fun x hx => by
62    have hm : MeasurableSet {z : ℝ | x < |z|} := measurableSet_lt measurable_const measurable_abs
63    rw [Measure.map_apply hY hm]
64    exact le_trans (le_self_add) (htail x hx)
65  have hQtail : ∀ x : ℝ, 0 ≤ x → gaussianReal 0 v.toNNReal {z | x < |z|} ≤
66      ENNReal.ofReal (4 * Real.exp (-x ^ 2 / (2 * D))) := fun x hx =>
67    le_trans (le_add_self) (htail x hx)
68  obtain ⟨hP4, hPB⟩ := fourth_moment_of_tail _ D hD hPtail
69  obtain ⟨hQ4, hQB⟩ := fourth_moment_of_tail _ D hD hQtail
70  have h := third_moment_le_wasserstein _ _ (32 * D ^ 2) (by positivity) hP4 hQ4 hPB hQB
71    (gaussianReal_third_moment_zero v.toNNReal)
72  have hint : ∫ z, z ^ 3 ∂((rateGibbs d s k β θ N).map (rateY d s k β N θ)) =
73      ∫ X, rateY d s k β N θ X ^ 3 ∂(rateGibbs d s k β θ N) :=
74    integral_map hY.aemeasurable (by fun_prop : Continuous fun z : ℝ => z ^ 3).aestronglyMeasurable
75  rw [hint] at h
76  exact h

79/-- **W4b.**  The lower bound of the frozen `HRS.Main.critical_family_sharpness`, with the constants `c`
80and `N₀` chosen before the density family `ρ`. -/
81theorem critical_family_sharpness_lower {d : ℕ} (hd : 2 ≤ d) {β : ℝ} (hβ : 0 < β) (k : Lattice d)
82    (hk : k ≠ 0) :
83    ∃ c : ℝ, 0 < c ∧ ∃ N₀ : ℕ, ∀ ρ : ℝ → Torus d → ℝ,
84      (∀ θ, IsThermalDensity (periodicRiesz d ((d : ℝ) / 2)) (rateV k β θ) β (ρ θ)) →
85      ∀ N : ℕ, N₀ ≤ N →
86        ENNReal.ofReal (c * (N : ℝ) ^ (-(1 : ℝ) / 2)) ≤
87          ∫⁻ θ in Set.Icc (0 : ℝ) (16 * (1 + β * rieszCoeff d ((d : ℝ) / 2) k)),
88            wassersteinE 2 (Main.rateLaw d ((d : ℝ) / 2) k β N θ)
89              (Main.rateGaussian d ((d : ℝ) / 2) k β (ρ θ)) := by
90  obtain ⟨s, hsdef⟩ : ∃ s : ℝ, s = (d : ℝ) / 2 := ⟨_, rfl⟩
91  rw [← hsdef]
92  have hd2 : (2 : ℝ) ≤ d := by exact_mod_cast hd
93  have hs0 : 0 ≤ s := by rw [hsdef]; positivity
94  have hsd : s < d := by rw [hsdef]; linarith
95  have hs : (d : ℝ) / 2 ≤ s := hsdef.symm.le
96  have hd1 : 1 ≤ d := by omega
97  obtain ⟨g, hgdef⟩ : ∃ g : ℝ, g = rieszCoeff d s k := ⟨_, rfl⟩
98  rw [← hgdef]
99  have hg : 0 ≤ g := by rw [hgdef]; exact (rieszCoeff_pos hd hs0 hsd k hk).le
100  have h1g : 0 < 1 + β * g := by positivity
101  obtain ⟨R, hRdef⟩ : ∃ R : ℝ, R = 16 * (1 + β * g) := ⟨_, rfl⟩
102  rw [← hRdef]
103  obtain ⟨v0, hv0def⟩ : ∃ v0 : ℝ, v0 = 1 / (2 * (1 + β * g)) := ⟨_, rfl⟩
104  have hRpos : 0 < R := by rw [hRdef]; positivity
105  have hv0 : 0 < v0 := by rw [hv0def]; positivity
106  have hRv : 2 / R = v0 / 4 := by
107    rw [hRdef, hv0def]; field_simp; ring
108  -- a canonical thermal family
109  have hex : ∀ θ : ℝ, ∃ ρ : Torus d → ℝ,
110      IsThermalDensity (periodicRiesz d s) (rateV k β θ) β ρ := fun θ =>
111    let ⟨ρ, h, _, _⟩ := PDE.MFThermalAssembly.thermal_exists_continuous hd1 s hs0 hsd
112      (rateV k β θ) (Q32c.q32c_rateV_continuous k θ) β hβ
113    ⟨ρ, h⟩
114  choose ρs hρs using hex
115  obtain ⟨D, hD2, hDtail⟩ := response_subGaussian_proof hd hs hsd β hβ k hk ρs hρs 0 R
116  have hDpos : 0 < D := by linarith
117  obtain ⟨K, hKdef⟩ : ∃ K : ℝ, K = 3 * Real.sqrt (32 * D ^ 2) := ⟨_, rfl⟩
118  have hKpos : 0 < K := by rw [hKdef]; positivity
119  -- the variance at `θ = 0` converges to `v0`
120  have hlim := rateVarN_zero_tendsto hd hs hsd hβ k hk (ρs 0) (hρs 0)
121  have hcov0 : susceptibilityCov (periodicRiesz d s) (withDensityReal (ρs 0)) β
122      (Torus.cosMode k) (Torus.cosMode k) = v0 := by
123    rw [thermal_rateV_zero hd hs0 hsd hβ k (ρs 0) (hρs 0), hv0def, hgdef]
124    exact susceptibilityCov_one_cos hd hs0 hsd hβ k hk
125  rw [hcov0] at hlim
126  obtain ⟨N1, hN1⟩ := eventually_atTop.1
127    (hlim.eventually (lt_mem_nhds (show v0 / 2 < v0 by linarith)))
128  refine ⟨v0 / (4 * K), by positivity, max N1 2, fun ρ hρ N hN => ?_⟩
129  have hN2 : 2 ≤ N := le_trans (le_max_right _ _) hN
130  have hN1' : N1 ≤ N := le_trans (le_max_left _ _) hN
131  have hN0 : 0 < N := by omega
132  have hNr : (0 : ℝ) < N := by exact_mod_cast hN0
133  have hsqrt : 0 < Real.sqrt N := Real.sqrt_pos.2 hNr
134  have hV : v0 / 2 < rateVarN d s k β N 0 := hN1 N hN1'
135  have hvar := rate_variation_lower hd hs hsd hβ k hk hN2 hRpos
136  obtain ⟨F, hFdef⟩ : ∃ F : ℝ → ℝ, F = fun θ =>
137      |Real.sqrt N * ∫ X, rateY d s k β N θ X ^ 3 ∂(rateGibbs d s k β θ N)| := ⟨_, rfl⟩
138  have hFge : v0 / 4 ≤ ∫ θ in (0 : ℝ)..R, F θ := by
139    rw [hFdef]; rw [hRv] at hvar; linarith
140  have hint : IntegrableOn F (Set.Ioc 0 R) := by
141    by_contra hni
142    have h0 : ∫ θ in (0 : ℝ)..R, F θ = 0 := by
143      rw [intervalIntegral.integral_of_le hRpos.le]; exact integral_undef hni
144    linarith
145  have hlow : ENNReal.ofReal (v0 / 4) ≤ ∫⁻ θ in Set.Icc (0 : ℝ) R, ENNReal.ofReal (F θ) :=
146    calc ENNReal.ofReal (v0 / 4) ≤ ENNReal.ofReal (∫ θ in (0 : ℝ)..R, F θ) :=
147          ENNReal.ofReal_le_ofReal hFge
148      _ = ENNReal.ofReal (∫ θ in Set.Ioc 0 R, F θ) := by
149          rw [intervalIntegral.integral_of_le hRpos.le]
150      _ = ∫⁻ θ in Set.Ioc 0 R, ENNReal.ofReal (F θ) :=
151          ofReal_integral_eq_lintegral_ofReal hint
152            (Eventually.of_forall fun θ => by rw [hFdef]; exact abs_nonneg _)
153      _ = ∫⁻ θ in Set.Icc (0 : ℝ) R, ENNReal.ofReal (F θ) := setLIntegral_congr Ioc_ae_eq_Icc
154  have hpt : ∀ θ ∈ Set.Icc (0 : ℝ) R, ENNReal.ofReal (F θ) ≤
155      ENNReal.ofReal (Real.sqrt N * K) *
156        wassersteinE 2 (Main.rateLaw d s k β N θ) (Main.rateGaussian d s k β (ρ θ)) := by
157    intro θ hθ
158    have hρeq : ρ θ = ρs θ := thermal_eq_of_thermal hd hs0 hsd hβ k θ _ _ (hρ θ) (hρs θ)
159    have hp := third_moment_pointwise hd hs hsd hβ k hN0 θ
160      (susceptibilityCov (periodicRiesz d s) (withDensityReal (ρs θ)) β
161        (Torus.cosMode k) (Torus.cosMode k)) D hDpos (fun x hx => hDtail x hx N hN2 θ hθ)
162    rw [← hKdef] at hp
163    have hG : Main.rateGaussian d s k β (ρ θ) = gaussianReal 0
164        (susceptibilityCov (periodicRiesz d s) (withDensityReal (ρs θ)) β
165          (Torus.cosMode k) (Torus.cosMode k)).toNNReal := by
166      rw [hρeq]; rfl
167    rw [hG]
168    have hL : Main.rateLaw d s k β N θ = (rateGibbs d s k β θ N).map (rateY d s k β N θ) := rfl
169    rw [hL, hFdef]
170    show ENNReal.ofReal |Real.sqrt N * _| ≤ _
171    rw [abs_mul, abs_of_pos hsqrt, ENNReal.ofReal_mul hsqrt.le, ENNReal.ofReal_mul hsqrt.le, mul_assoc]
172    exact mul_le_mul' le_rfl hp
173  have hmono : ∫⁻ θ in Set.Icc (0 : ℝ) R, ENNReal.ofReal (F θ) ≤
174      ENNReal.ofReal (Real.sqrt N * K) *
175        ∫⁻ θ in Set.Icc (0 : ℝ) R,
176          wassersteinE 2 (Main.rateLaw d s k β N θ) (Main.rateGaussian d s k β (ρ θ)) := by
177    rw [← lintegral_const_mul' _ _ ENNReal.ofReal_ne_top]
178    exact setLIntegral_mono' measurableSet_Icc hpt
179  have hsq : (N : ℝ) ^ (-(1 : ℝ) / 2) = 1 / Real.sqrt N := by
180    rw [show -(1 : ℝ) / 2 = -(1 / 2) by ring, Real.rpow_neg hNr.le, ← Real.sqrt_eq_rpow]
181    simp
182  have hak : 0 < Real.sqrt N * K := by positivity
183  have hc : v0 / (4 * K) * (N : ℝ) ^ (-(1 : ℝ) / 2) = v0 / 4 * (1 / (Real.sqrt N * K)) := by
184    rw [hsq]; field_simp
185  calc ENNReal.ofReal (v0 / (4 * K) * (N : ℝ) ^ (-(1 : ℝ) / 2))
186      = ENNReal.ofReal (v0 / 4) * ENNReal.ofReal (1 / (Real.sqrt N * K)) := by
187        rw [hc, ENNReal.ofReal_mul (by positivity)]
188    _ ≤ (ENNReal.ofReal (Real.sqrt N * K) *
189          ∫⁻ θ in Set.Icc (0 : ℝ) R,
190            wassersteinE 2 (Main.rateLaw d s k β N θ) (Main.rateGaussian d s k β (ρ θ))) *
191          ENNReal.ofReal (1 / (Real.sqrt N * K)) := by
192        gcongr
193        exact le_trans hlow hmono
194    _ = (ENNReal.ofReal (1 / (Real.sqrt N * K)) * ENNReal.ofReal (Real.sqrt N * K)) *
195          ∫⁻ θ in Set.Icc (0 : ℝ) R,
196            wassersteinE 2 (Main.rateLaw d s k β N θ) (Main.rateGaussian d s k β (ρ θ)) := by ring
197    _ = ∫⁻ θ in Set.Icc (0 : ℝ) R,
198            wassersteinE 2 (Main.rateLaw d s k β N θ) (Main.rateGaussian d s k β (ρ θ)) := by
199        rw [← ENNReal.ofReal_mul (by positivity), one_div_mul_cancel hak.ne', ENNReal.ofReal_one,
200          one_mul]
