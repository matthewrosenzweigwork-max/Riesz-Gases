24/-- Lemma 4.20 in the form that the manuscript's proof actually gives: a dense `G_δ` set `R ⊆ (a,b)`
25of points `θ` with `f_N θ > q/2` for arbitrarily large `N` (so `limsup_N f_N θ ≥ q/2` in `EReal`).
26This frequent form implies the `EReal` limsup statement `response_category` below. -/
27theorem response_category_freq (a b : ℝ) (hab : a < b) (f : ℕ → ℝ → ℝ)
28    (hcont : ∀ N, ContinuousOn (f N) (Set.Ioo a b))

30    (q : ℝ) (hq : 0 < q)

32      ((q : ℝ) : EReal) ≤ liminf (fun N => (((d - c)⁻¹ * ∫ θ in c..d, f N θ : ℝ) : EReal)) atTop) :
33    ∃ R : Set ℝ, R ⊆ Set.Ioo a b ∧ IsGδ R ∧ Set.Ioo a b ⊆ closure R ∧

35  set O : ℕ → Set ℝ := fun L => ⋃ N ∈ {N : ℕ | L ≤ N}, (Set.Ioo a b ∩ f N ⁻¹' Set.Ioi (q / 2)) with hO

68      exact EReal.coe_lt_coe_iff.1 hN
69    obtain ⟨N, hN, hLN⟩ := (hev.and (eventually_ge_atTop L)).exists
70    by_contra hnone
71    have hle : ∀ x ∈ Set.Icc c d, f N x ≤ q / 2 := by
72      intro x hx
73      by_contra hlt
74      push Not at hlt
75      apply hnone
76      refine ⟨x, (hsub hx).1, ?_⟩
77      simp only [hO, Set.mem_iUnion]

119/-- Lemma 4.20 `M:V:lem:response-category` (lines 5386-5395): let `I = (a,b)` be an open bounded
120interval and `f_N : I → [0,∞)` continuous. If some `q > 0` satisfies
121`liminf_N |J|⁻¹ ∫_J f_N ≥ q` for every closed nondegenerate interval `J ⊂ I`, then a dense
122`G_δ` subset of `I` consists of points where `limsup_N f_N ≥ q/2`.
123
124The `liminf` and `limsup` are taken in `EReal`, as in the source (values in `[0,∞]`); the real
125`liminf`/`limsup` of Mathlib take junk values on unbounded sequences and would change the
126statement (docs/ISSUES.md, item on Lemma 4.20). "Dense in `I`" is `I ⊆ closure R` with `R ⊆ I`;
127a `G_δ` subset of the open set `I` is the same as a `G_δ` subset of `ℝ` contained in `I`. -/
128theorem response_category (a b : ℝ) (hab : a < b) (f : ℕ → ℝ → ℝ)
129    (hcont : ∀ N, ContinuousOn (f N) (Set.Ioo a b))
130    (hnonneg : ∀ N, ∀ θ ∈ Set.Ioo a b, 0 ≤ f N θ)
131    (q : ℝ) (hq : 0 < q)
132    (havg : ∀ c d, a < c → c < d → d < b →
133      ((q : ℝ) : EReal) ≤ liminf (fun N => (((d - c)⁻¹ * ∫ θ in c..d, f N θ : ℝ) : EReal)) atTop) :
134    ∃ R : Set ℝ, R ⊆ Set.Ioo a b ∧ IsGδ R ∧ Set.Ioo a b ⊆ closure R ∧
135      ∀ θ ∈ R, ((q / 2 : ℝ) : EReal) ≤ limsup (fun N => ((f N θ : ℝ) : EReal)) atTop := by
136  obtain ⟨R, h1, h2, h3, h4⟩ := response_category_freq a b hab f hcont hnonneg q hq havg
137  refine ⟨R, h1, h2, h3, fun θ hθ => ?_⟩
138  refine Filter.le_limsup_of_frequently_le ?_
139  rw [Filter.frequently_atTop]
140  intro L
141  obtain ⟨N, hN, hlt⟩ := h4 θ hθ L
142  exact ⟨N, hN, EReal.coe_le_coe_iff.2 hlt.le⟩

149lemma sqrtLip_taylor (a : E → ℝ) (ha : ContDiff ℝ 1 a) (L : ℝ) (hL : 0 ≤ L)
150    (hlip : LipschitzWith (Real.toNNReal L) (fun x => fderiv ℝ a x)) (x h : E) :

152  have hda : Differentiable ℝ a := ha.differentiable (by norm_num)
153  have hd : ∀ t : ℝ, HasDerivAt (fun t : ℝ => a (x + t • h)) (fderiv ℝ a (x + t • h) h) t := by
154    intro t
155    have h1 : HasDerivAt (fun t : ℝ => x + t • h) h t := by
156      simpa using ((hasDerivAt_id t).smul_const h).const_add x
157    exact (hda (x + t • h)).hasFDerivAt.comp_hasDerivAt t h1
158  set φ : ℝ → ℝ := fun t => a (x + t • h) - t * fderiv ℝ a x h - L / 2 * t ^ 2 * ‖h‖ ^ 2 with hφ
159  have hφd : ∀ t : ℝ, HasDerivAt φ
160      (fderiv ℝ a (x + t • h) h - fderiv ℝ a x h - L * t * ‖h‖ ^ 2) t := by
161    intro t
162    have := (((hd t).sub ((hasDerivAt_id t).mul_const (fderiv ℝ a x h))).sub
163      ((((hasDerivAt_id t).pow 2).const_mul (L / 2)).mul_const (‖h‖ ^ 2)))
164    exact this.congr_deriv (by simp only [id, Nat.cast_ofNat, Nat.add_one_sub_one, pow_one, mul_one, one_mul]; ring)
165  have hanti : AntitoneOn φ (Set.Icc 0 1) := by
166    apply antitoneOn_of_deriv_nonpos (convex_Icc 0 1)
167    · exact fun t _ => (hφd t).continuousAt.continuousWithinAt
168    · exact fun t _ => (hφd t).differentiableAt.differentiableWithinAt
169    · intro t ht
170      rw [interior_Icc] at ht
171      rw [(hφd t).deriv]
172      have h0 : 0 ≤ t := ht.1.le
173      have h1 : dist (fderiv ℝ a (x + t • h)) (fderiv ℝ a x) ≤ L * (t * ‖h‖) := by
174        have := hlip.dist_le_mul (x + t • h) x
175        simpa [Real.coe_toNNReal _ hL, dist_eq_norm, norm_smul, abs_of_nonneg h0] using this
176      have h2 : fderiv ℝ a (x + t • h) h - fderiv ℝ a x h ≤ L * (t * ‖h‖) * ‖h‖ := by
177        have := (fderiv ℝ a (x + t • h) - fderiv ℝ a x).le_opNorm h
178        rw [dist_eq_norm] at h1
179        have h3 : (fderiv ℝ a (x + t • h) - fderiv ℝ a x) h = fderiv ℝ a (x + t • h) h - fderiv ℝ a x h := rfl
180        rw [h3] at this
181        calc _ ≤ ‖fderiv ℝ a (x + t • h) h - fderiv ℝ a x h‖ := Real.le_norm_self _
182          _ ≤ ‖fderiv ℝ a (x + t • h) - fderiv ℝ a x‖ * ‖h‖ := this
183          _ ≤ _ := by gcongr
184      nlinarith
185  have := hanti (Set.left_mem_Icc.2 zero_le_one) (Set.right_mem_Icc.2 zero_le_one) zero_le_one
186  simp only [hφ] at this

192    (hlip : LipschitzWith (Real.toNNReal L) (fun x => fderiv ℝ a x)) (x h : E) :
193    (fderiv ℝ a x h) ^ 2 ≤ 2 * L * a x * ‖h‖ ^ 2 := by
194  have key : ∀ s : ℝ, 0 ≤ a x + s * fderiv ℝ a x h + L / 2 * s ^ 2 * ‖h‖ ^ 2 := by
195    intro s
196    have := sqrtLip_taylor a ha L hL hlip x (s • h)

198    simp only [map_smul, smul_eq_mul, norm_smul, Real.norm_eq_abs, mul_pow, sq_abs] at this
199    nlinarith

225/-- Lemma C.6 `M:B:lem:sqrt-lipschitz` (lines 12311-12313), on a finite-dimensional Euclidean
226space (the torus case follows by periodic lifting): if `0 ≤ a ∈ C^{1,1}` with
227`‖∇²a‖_∞ ≤ L`, then `√a` is Lipschitz with constant `(L/2)^{1/2}`. -/
228theorem sqrt_lipschitz (a : E → ℝ) (ha : ContDiff ℝ 1 a) (hnn : ∀ x, 0 ≤ a x) (L : ℝ) (hL : 0 ≤ L)
229    (hlip : LipschitzWith (Real.toNNReal L) (fun x => fderiv ℝ a x)) :
230    LipschitzWith (Real.toNNReal (Real.sqrt (L / 2))) (fun x => Real.sqrt (a x)) := by
231  have step : ∀ x y : E, Real.sqrt (a y) ≤ Real.sqrt (a x) + Real.sqrt (L / 2) * dist y x := by
232    intro x y
233    set h := y - x with hh
234    have hy : y = x + h := by simp [hh]
235    have hdist : dist y x = ‖h‖ := by rw [dist_eq_norm]
236    rw [hdist]
237    set M := Real.sqrt (L / 2)
238    have hM : 0 ≤ M := Real.sqrt_nonneg _
239    have hM2 : M ^ 2 = L / 2 := Real.sq_sqrt (by positivity)
240    have hax := hnn x
241    have hsa := Real.sq_sqrt hax
242    have hsa0 := Real.sqrt_nonneg (a x)
243    have hT := sqrtLip_taylor a ha L hL hlip x h
244    have hG := sqrtLip_grad a ha hnn L hL hlip x h
245    have hD : fderiv ℝ a x h ≤ 2 * Real.sqrt (a x) * M * ‖h‖ := by
246      have h1 : (fderiv ℝ a x h) ^ 2 ≤ (2 * Real.sqrt (a x) * M * ‖h‖) ^ 2 := by
247        calc _ ≤ 2 * L * a x * ‖h‖ ^ 2 := hG
248          _ = _ := by
249            have : (2 * Real.sqrt (a x) * M * ‖h‖) ^ 2 = 4 * (Real.sqrt (a x)) ^ 2 * M ^ 2 * ‖h‖ ^ 2 := by ring
250            rw [this, hsa, hM2]; ring
251      exact (le_abs_self _).trans (abs_le_of_sq_le_sq h1 (by positivity))
252    have : a y ≤ (Real.sqrt (a x) + M * ‖h‖) ^ 2 := by
253      rw [hy]
254      nlinarith
255    calc Real.sqrt (a y) ≤ Real.sqrt ((Real.sqrt (a x) + M * ‖h‖) ^ 2) := Real.sqrt_le_sqrt this
256      _ = _ := Real.sqrt_sq (by positivity)
257  refine LipschitzWith.of_dist_le_mul fun x y => ?_
258  rw [Real.coe_toNNReal _ (Real.sqrt_nonneg _), Real.dist_eq, abs_le]
259  have h1 := step x y
260  have h2 := step y x
261  rw [dist_comm y x] at h1
262  constructor <;> linarith

269/-- The Hadamard growth factor of Proposition 3.10 for frequency `k`:
270`⟨k⟩^{m'-m} exp(4π² c (2π|k|)^{s-d} (kᵀ S k) τ)` with `⟨k⟩ = (1+4π²|k|²)^{1/2}`. -/
271def hadamardFactor (d : ℕ) (s c τ m m' : ℝ) (S : Matrix (Fin d) (Fin d) ℝ) (k : Fin d → ℤ) : ℝ :=
272  let nk : ℝ := Real.sqrt (∑ i, ((k i : ℝ)) ^ 2)
273  let kSk : ℝ := ∑ i, ∑ j, (k i : ℝ) * S i j * (k j : ℝ)
274  (1 + 4 * Real.pi ^ 2 * nk ^ 2) ^ (m' / 2) / (1 + 4 * Real.pi ^ 2 * nk ^ 2) ^ (m / 2) *
275    Real.exp (4 * Real.pi ^ 2 * c * (2 * Real.pi * nk) ^ (s - d) * kSk * τ)
