21/-- `supercriticalBias_pos` (verbatim statement of `HRS.Main.supercriticalBias_pos`). -/
22theorem supercriticalBias_pos_proof (hd : 2 ≤ d) {s : ℝ} (hs1 : (d : ℝ) / 2 < s) (hsd : s < d)
23    (V : Torus d → ℝ) (hV : Torus.IsSmooth V) (β : ℝ) (hβ : 0 < β) (ρβ : Torus d → ℝ)
24    (hρβ : IsThermalDensity (periodicRiesz d s) V β ρβ) (hnc : ∃ x y, ρβ x ≠ ρβ y) :
25    0 < supercriticalBias d s β ρβ ρβ := by
26  obtain ⟨hs0, hp1, hp2⟩ := eqbias_p_range hs1 hsd hd
27  have hρs := eqbias_smoothPos hd hs0.le hsd V hV β hβ ρβ hρβ
28  have hint := eqbias_bias_pos hd hs0.le hsd hρs hβ.le hnc
29  unfold supercriticalBias
30  have hA : 0 < sphereArea d := HRS.StaticLimitsAux.sphereArea_pos hd
31  have hG : 0 < Real.Gamma (2 - d / s) := Real.Gamma_pos_of_pos (by linarith)
32  have hB : 0 < β ^ ((d : ℝ) / s) := Real.rpow_pos_of_pos hβ _
33  have hden : 0 < s * (d / s) * (d / s - 1) := by
34    have : 0 < (d : ℝ) / s - 1 := by linarith
35    have : 0 < (d : ℝ) / s := by linarith
36    positivity
37  have hc : 0 < sphereArea d * Real.Gamma (2 - d / s) / (s * (d / s) * (d / s - 1)) :=
38    div_pos (mul_pos hA hG) hden
39  exact mul_pos (mul_pos hc hB) hint

43/-- `equilibrium_bias_expansion` (verbatim statement of `HRS.Main.equilibrium_bias_expansion`). -/
44theorem equilibrium_bias_expansion_proof (hd : 2 ≤ d) {s : ℝ} (hs1 : (d : ℝ) / 2 < s)
45    (hsd : s < d) (V : Torus d → ℝ) (hV : Torus.IsSmooth V) (β : ℝ) (hβ : 0 < β)
46    (ρβ : Torus d → ℝ) (hρβ : IsThermalDensity (periodicRiesz d s) V β ρβ) :
47    (∀ N : ℕ, 2 ≤ N →
48      firstMarginal (canonicalGibbs (periodicRiesz d s) V N β) ≪ withDensityReal ρβ) ∧
49    Tendsto (fun N : ℕ =>
50      eLpNorm (fun y =>
51        (((firstMarginal (canonicalGibbs (periodicRiesz d s) V N β)).rnDeriv
52            (withDensityReal ρβ) y).toReal - 1) / (N : ℝ) ^ (1 - (d : ℝ) / s) -
53          supercriticalBiasDensity d s β ρβ y) 2 (withDensityReal ρβ)) atTop (𝓝 0) := by
54  obtain ⟨hs0, hp1, hp2⟩ := eqbias_p_range hs1 hsd hd
55  have hρs := eqbias_smoothPos hd hs0.le hsd V hV β hβ ρβ hρβ
56  have hcan := fun N => eqbias_canonical hd hs0.le hsd V β ρβ hρs hρβ N
57  have hD := eqbias_hD hd hs1 hsd hβ hρs
58  obtain ⟨hac, hlim⟩ := eqbias_tendsto_of_D hd hs1 hsd hβ hρs hD
59  have hdens : ∀ y, supercriticalBiasDensity d s β ρβ y =
60      (eqbias_c0 d s * β ^ ((d : ℝ) / s)) *
61        responseInverse (periodicRiesz d s) (withDensityReal ρβ) β ρβ y := by
62    intro y
63    unfold supercriticalBiasDensity eqbias_c0
64    ring
65  refine ⟨fun N hN => ?_, ?_⟩
66  · rw [hcan N]; exact hac N hN
67  · simp only [hcan, hdens]
68    exact hlim
