40/-- Glue two martingale pieces at a time `c ≥ 0`: a process vanishing on `(-∞, 0]` that satisfies
41the martingale identity on `[0, c]` and on `[c, ∞)` is a martingale. -/
42theorem martingale_of_pieces {𝒢 : Filtration ℝ mΩ} {Z : ℝ → Ω → ℝ} {c : ℝ}
43    (hZ : StronglyAdapted 𝒢 Z) (hzero : ∀ t ≤ 0, ∀ ω, Z t ω = 0)
44    (h₁ : ∀ s r, 0 ≤ s → s ≤ r → r ≤ c → P[Z r | 𝒢 s] =ᵐ[P] Z s)
45    (h₂ : ∀ s r, c ≤ s → s ≤ r → P[Z r | 𝒢 s] =ᵐ[P] Z s) : Martingale Z 𝒢 P := by
46  refine martingale_of_nonneg_times hZ hzero fun s r hs hsr => ?_
47  by_cases hrc : r ≤ c
48  · exact h₁ s r hs hsr hrc
49  by_cases hcs : c ≤ s
50  · exact h₂ s r hcs hsr
51  push Not at hrc hcs
52  calc P[Z r | 𝒢 s] =ᵐ[P] P[P[Z r | 𝒢 c] | 𝒢 s] :=
53        (condExp_condExp_of_le (𝒢.mono hcs.le) (𝒢.le c)).symm
54    _ =ᵐ[P] P[Z c | 𝒢 s] := condExp_congr_ae (h₂ c r le_rfl hrc.le)
55    _ =ᵐ[P] Z s := h₁ s c hs hcs.le le_rfl

67/-- Conditional expectation given `𝒜 ⊔ ℬ` of an `𝒜'`-measurable function, with `ℬ` independent of
68`𝒜'` and `𝒜 ≤ 𝒜'`. -/
69theorem condExp_sup_of_indep {𝒜 𝒜' ℬ : MeasurableSpace Ω} (h𝒜' : 𝒜' ≤ mΩ) (hℬ : ℬ ≤ mΩ)
70    (h𝒜 : 𝒜 ≤ 𝒜') (hind : Indep 𝒜' ℬ P) {f : Ω → ℝ} (hfi : Integrable f P)
71    (hf : StronglyMeasurable[𝒜'] f) : P[f | 𝒜 ⊔ ℬ] =ᵐ[P] P[f | 𝒜] := by
72  letI : MeasurableSpace Ω := mΩ
73  have h𝒜m : 𝒜 ≤ mΩ := h𝒜.trans h𝒜'
74  have hle : 𝒜 ⊔ ℬ ≤ mΩ := sup_le h𝒜m hℬ
75  -- rectangle integrals of `𝒜'`-measurable functions factor
76  have key : ∀ {F : Ω → ℝ}, StronglyMeasurable[𝒜'] F → ∀ {a b : Set Ω},
77      MeasurableSet[𝒜] a → MeasurableSet[ℬ] b →
78      ∫ ω in a ∩ b, F ω ∂P = (∫ ω in a, F ω ∂P) * P.real b := by
79    intro F hF a b ha hb
80    have ha' : MeasurableSet[𝒜'] a := h𝒜 _ ha
81    have hX : StronglyMeasurable[𝒜'] (a.indicator F) := hF.indicator ha'
82    have hY : Measurable[ℬ] (b.indicator fun _ : Ω => (1 : ℝ)) :=
83      measurable_const.indicator hb
84    have hXY : IndepFun (a.indicator F) (b.indicator fun _ : Ω => (1 : ℝ)) P :=
85      (IndepFun_iff_Indep _ _ _).2
86        (indep_of_indep_of_le_right (indep_of_indep_of_le_left hind hX.measurable.comap_le)
87          hY.comap_le)
88    have h1 : ∫ ω in a ∩ b, F ω ∂P = ∫ ω, a.indicator F ω * b.indicator (fun _ => (1 : ℝ)) ω ∂P := by
89      rw [← integral_indicator ((h𝒜m _ ha).inter (hℬ _ hb))]
90      refine integral_congr_ae (Eventually.of_forall fun ω => ?_)
91      by_cases h1 : ω ∈ a <;> by_cases h2 : ω ∈ b <;> simp [h1, h2]
92    rw [h1, hXY.integral_fun_mul_eq_mul_integral (hX.mono h𝒜').aestronglyMeasurable
93      (hY.mono hℬ le_rfl).aestronglyMeasurable, integral_indicator (h𝒜m _ ha)]
94    congr 1
95    rw [integral_indicator (hℬ _ hb)]
96    simp [measureReal_def]
97  have hgsm : StronglyMeasurable[𝒜'] (P[f | 𝒜]) := stronglyMeasurable_condExp.mono h𝒜
98  have hgi : Integrable (P[f | 𝒜]) P := integrable_condExp
99  refine (ae_eq_condExp_of_forall_setIntegral_eq hle hfi (fun s _ _ => hgi.integrableOn)
100    (fun s hs _ => ?_) (stronglyMeasurable_condExp.mono (le_sup_left : 𝒜 ≤ 𝒜 ⊔ ℬ)).aestronglyMeasurable).symm
101  -- the π-system of rectangles generates `𝒜 ⊔ ℬ`
102  let p : Set (Set Ω) := {s | ∃ a b, MeasurableSet[𝒜] a ∧ MeasurableSet[ℬ] b ∧ s = a ∩ b}
103  have hπ : IsPiSystem p := by
104    rintro _ ⟨a, b, ha, hb, rfl⟩ _ ⟨a', b', ha', hb', rfl⟩ _
105    exact ⟨a ∩ a', b ∩ b', @MeasurableSet.inter Ω 𝒜 _ _ ha ha',
106      @MeasurableSet.inter Ω ℬ _ _ hb hb', by ext; simp; tauto⟩
107  have hgen : 𝒜 ⊔ ℬ = MeasurableSpace.generateFrom p := by
108    apply le_antisymm
109    · refine sup_le (fun a ha => MeasurableSpace.measurableSet_generateFrom ⟨a, Set.univ, ha,
110        @MeasurableSet.univ Ω ℬ, by simp⟩) (fun b hb => MeasurableSpace.measurableSet_generateFrom
111        ⟨Set.univ, b, @MeasurableSet.univ Ω 𝒜, hb, by simp⟩)
112    · refine MeasurableSpace.generateFrom_le ?_
113      rintro _ ⟨a, b, ha, hb, rfl⟩
114      exact @MeasurableSet.inter Ω (𝒜 ⊔ ℬ) _ _ (le_sup_left (a := 𝒜) (b := ℬ) _ ha)
115        (le_sup_right (a := 𝒜) (b := ℬ) _ hb)
116  refine MeasurableSpace.induction_on_inter (m := 𝒜 ⊔ ℬ)
117    (C := fun s _ => ∫ x in s, (P[f | 𝒜]) x ∂P = ∫ x in s, f x ∂P) hgen hπ ?_ ?_ ?_ ?_ s hs
118  · simp
119  · rintro _ ⟨a, b, ha, hb, rfl⟩
120    rw [key hgsm ha hb, key hf ha hb, setIntegral_condExp h𝒜m hfi ha]

123    have e1 := integral_add_compl hs' hgi
124    have e2 := integral_add_compl hs' hfi

158/-- Everywhere-clean data for the DDS construction: a complete filtration, a martingale `M` and an
159adapted `A` with `M² - A` a martingale, all path properties holding for every `ω`. -/
160structure CleanData (P : Measure Ω) (ℱ : Filtration ℝ mΩ) (t : ℝ) (M A : ℝ → Ω → ℝ) : Prop where
161  complete : IsCompleteFiltration P ℱ
162  mart : Martingale M ℱ P
163  qmart : Martingale (fun s ω => M s ω ^ 2 - A s ω) ℱ P
164  adaptedA : Adapted ℱ A

173theorem exists_clean {ℱ : Filtration ℝ mΩ} {t : ℝ} {M A : ℝ → Ω → ℝ}
174    (hM : IsContSqMartingale P ℱ t M) (hA : IsQuadraticVariation P ℱ t M A)
175    (hint : Integrable (A t) P) :
176    ∃ M₁ A₁ : ℝ → Ω → ℝ, CleanData P (augFiltration P ℱ) t M₁ A₁ ∧
177      (∀ s, M₁ s =ᵐ[P] M s) ∧ ∀ s, A₁ s =ᵐ[P] A s := by
178  classical
179  obtain ⟨N, hNsub, hNm, hN0⟩ : ∃ N : Set Ω, {ω | ¬ (M 0 ω = 0 ∧ A 0 ω = 0 ∧
180      ContinuousOn (fun τ => M τ ω) (Set.Icc 0 t) ∧ ContinuousOn (fun τ => A τ ω) (Set.Icc 0 t) ∧
181      MonotoneOn (fun τ => A τ ω) (Set.Icc 0 t))} ⊆ N ∧ MeasurableSet N ∧ P N = 0 := by
182    have h0 := ae_iff.1 (hM.zero.and (hA.zero.and (hM.cont.and (hA.cont.and hA.mono))))
183    exact ⟨toMeasurable P _, subset_toMeasurable _ _, measurableSet_toMeasurable _ _,
184      by rw [measure_toMeasurable]; exact h0⟩

186      ContinuousOn (fun τ => M τ ω) (Set.Icc 0 t) ∧ ContinuousOn (fun τ => A τ ω) (Set.Icc 0 t) ∧
187      MonotoneOn (fun τ => A τ ω) (Set.Icc 0 t) := by
188    intro ω hω
189    by_contra h
190    exact hω (hNsub h)
191  have hNae : ∀ᵐ ω ∂P, ω ∉ N := by rw [ae_iff]; simpa using hN0
192  refine ⟨fun s ω => if ω ∈ N then 0 else M s ω, fun s ω => if ω ∈ N then 0 else A s ω, ?_, ?_, ?_⟩

195  swap

199  have hae1 : ∀ s, (fun ω => if ω ∈ N then (0 : ℝ) else M s ω) =ᵐ[P] M s := by
200    intro s; filter_upwards [hNae] with ω hω; simp [hω]
201  have hae2 : ∀ s, (fun ω => if ω ∈ N then (0 : ℝ) else A s ω) =ᵐ[P] A s := by
202    intro s; filter_upwards [hNae] with ω hω; simp [hω]
203  have hadM : ∀ s, Measurable[augFiltration P ℱ s] (fun ω => if ω ∈ N then (0 : ℝ) else M s ω) :=
204    fun s => Measurable.ite (hN' s) measurable_const
205      ((hM.martingale.1 s).measurable.mono (le_sup_left : ℱ s ≤ ℱ s ⊔ nullSets P) le_rfl)
206  have hadA : ∀ s, Measurable[augFiltration P ℱ s] (fun ω => if ω ∈ N then (0 : ℝ) else A s ω) :=
207    fun s => Measurable.ite (hN' s) measurable_const
208      ((hA.adapted s).mono (le_sup_left : ℱ s ≤ ℱ s ⊔ nullSets P) le_rfl)
209  refine ⟨augFiltration_isComplete ℱ, ?_, ?_, hadA, ?_, ?_, ?_, ?_, ?_, ?_, ?_⟩
210  · exact martingale_congr_ae (martingale_aug hM.martingale) (fun s => (hadM s).stronglyMeasurable) hae1
211  · refine martingale_congr_ae (martingale_aug hA.compensator) (fun s => ?_) (fun s => ?_)
212    · exact (((hadM s).pow_const 2).sub (hadA s)).stronglyMeasurable
213    · filter_upwards [hae1 s, hae2 s] with ω h1 h2
214      simp only [h1, h2]
215  · intro ω; by_cases hω : ω ∈ N

219    · simp [hω]

223    · simp only [hω, ite_false]; exact (hgood ω hω).2.2.1

473/-- Clamp a time into `[0, t]`. -/
474def clampT (t s : ℝ) : ℝ := max (min s t) 0
475
476/-- The real-time scalar Brownian path. -/
477def bw (W : ℝ≥0 → Ω₂ → Fin 1 → ℝ) (s : ℝ) (ω : Ω₂) : ℝ := ofNNRealTime W s ω 0

479/-- The extension of `M'` beyond `t` by `W`. -/
480def Mext (W : ℝ≥0 → Ω₂ → Fin 1 → ℝ) (M' : ℝ → Ω₂ → ℝ) (t s : ℝ) (ω : Ω₂) : ℝ :=
481  M' (clampT t s) ω + (bw W s ω - bw W (min s t) ω)
482
483/-- The extension of `A'` beyond `t` by a linear drift. -/
484def Aext (A' : ℝ → Ω₂ → ℝ) (t s : ℝ) (ω : Ω₂) : ℝ := A' (clampT t s) ω + (max s t - t)

1000stochastic files. -/
1001theorem dds_coupling_strong {Ω : Type*} {mΩ : MeasurableSpace Ω} (P : Measure Ω)
1002    [IsProbabilityMeasure P] (ℱ : Filtration ℝ mΩ) (t : ℝ) (ht : 0 ≤ t) (M A : ℝ → Ω → ℝ)
1003    (hM : IsContSqMartingale P ℱ t M) (hA : IsQuadraticVariation P ℱ t M A)
1004    (hint : Integrable (A t) P) (v : ℝ) (hv : 0 ≤ v) :
1005    ∃ (Ω' : Type) (_ : MeasurableSpace Ω') (P' : Measure Ω') (_ : IsProbabilityMeasure P')
1006      (M' A' : Ω' → ℝ) (β : ℝ≥0 → Ω' → ℝ),
1007      Measurable M' ∧ Measurable A' ∧
1008      ProbabilityTheory.IsBrownianReal β P' ∧
1009      law P' (fun ω' => (M' ω', A' ω')) = law P (fun ω => (M t ω, A t ω)) ∧
1010      (∀ᵐ ω' ∂P', M' ω' = β (Real.toNNReal (A' ω')) ω') ∧
1011      law P' (β (Real.toNNReal v)) = gaussianLaw v ∧
1012      ∫⁻ ω', ‖β (Real.toNNReal (A' ω')) ω' - β (Real.toNNReal v) ω'‖ₑ ^ 2 ∂P' ≤
1013        ENNReal.ofReal (∫ ω, |A t ω - v| ∂P) := by
1014  obtain ⟨Ω', mΩ', P', hP', ℱ', β, M', A', hβ, hlaw, hstop, hid, hM'm, hA'm⟩ :=
1015    dambis_dubins_schwarz_core P ℱ t ht M A hM hA hint
1016  have hMm : Measurable (M t) := (hM.martingale.1 t).measurable.mono (ℱ.le t) le_rfl
1017  have hAm : Measurable (A t) := (hA.adapted t).mono (ℱ.le t) le_rfl
1018  set f : Ω → ℝ × ℝ := fun ω => (M t ω, A t ω) with hf
1019  set f' : Ω' → ℝ × ℝ := fun ω' => (M' t ω', A' t ω') with hf'
1020  have hfm : Measurable f := hMm.prodMk hAm
1021  have hf'm : Measurable f' := (hM'm t).prodMk (hA'm t)
1022  have hlaw1 : law P' (fun ω' (_ : Fin 1) => (M' t ω', A' t ω')) =
1023      law P (fun ω (_ : Fin 1) => (M t ω, A t ω)) := hlaw 1 (fun _ => t)
1024  have hFm : Measurable (fun ω (_ : Fin 1) => (M t ω, A t ω)) := Measurable.of_eval fun _ => hfm
1025  have hF'm : Measurable (fun ω' (_ : Fin 1) => (M' t ω', A' t ω')) :=
1026    Measurable.of_eval fun _ => hf'm
1027  have hf'ae : AEMeasurable f' P' := hf'm.aemeasurable
1028  have hpair : law P' f' = law P f := by
1029    have e := congrArg (Measure.map (fun F : Fin 1 → ℝ × ℝ => F 0)) hlaw1
1030    unfold law at e ⊢
1031    rw [Measure.map_map (measurable_pi_apply 0) hF'm,
1032      Measure.map_map (measurable_pi_apply 0) hFm] at e
1033    exact e
1034  have hpush : ∀ g : ℝ × ℝ → ℝ, Measurable g →
1035      ∫ ω', g (f' ω') ∂P' = ∫ ω, g (f ω) ∂P := by
1036    intro g hg
1037    have h1 := integral_map (μ := P') hf'ae hg.aestronglyMeasurable
1038    have h2 := integral_map (μ := P) hfm.aemeasurable hg.aestronglyMeasurable
1039    unfold law at hpair
1040    rw [← h1, ← h2, hpair]
1041  have hintg : ∀ g : ℝ × ℝ → ℝ, Measurable g → Integrable (fun ω => g (f ω)) P →
1042      Integrable (fun ω' => g (f' ω')) P' := by
1043    intro g hg hgi
1044    have h1 : Integrable g (law P f) :=
1045      (integrable_map_measure hg.aestronglyMeasurable hfm.aemeasurable).2 hgi
1046    rw [← hpair] at h1
1047    exact (integrable_map_measure hg.aestronglyMeasurable hf'ae).1 h1
1048  have hAnn : ∀ᵐ ω ∂P, 0 ≤ A t ω := by
1049    filter_upwards [hA.zero, hA.mono] with ω h0 hm
1050    have h1 : A 0 ω ≤ A t ω := hm ⟨le_rfl, ht⟩ ⟨ht, le_rfl⟩ ht
1051    rwa [h0] at h1
1052  have hSint : Integrable (fun ω' => max (A' t ω') 0) P' :=
1053    hintg (fun p => max p.2 0) (measurable_snd.max measurable_const) hint.pos_part
1054  have hbound := brownian_stopped_L2_bound hβ 0 (fun ω' => max (A' t ω') 0)
1055    (fun _ => le_max_right _ _) (hstop t ⟨ht, le_rfl⟩) hSint v hv
1056  refine ⟨Ω', mΩ', P', hP', M' t, A' t, fun u ω' => β u ω' 0, hM'm t, hA'm t,
1057    hβ.isBrownianReal 0, hpair, ?_, ?_, ?_⟩
1058  · filter_upwards [hid] with ω' h
1059    exact h t ⟨ht, le_rfl⟩
1060  · exact (hβ.hasLaw_coord (Real.toNNReal v) 0).map_eq
1061  · simp_rw [toNNReal_max_zero] at hbound
1062    refine hbound.trans (ENNReal.ofReal_le_ofReal (le_of_eq ?_))
1063    have h1 := hpush (fun p => |max p.2 0 - v|)
1064      ((measurable_snd.max measurable_const).sub measurable_const).abs
1065    refine h1.trans (integral_congr_ae ?_)
1066    filter_upwards [hAnn] with ω hω
1067    simp only [hf, max_eq_left hω]

1089theorem dds_wasserstein_bound_proof {Ω : Type*} {mΩ : MeasurableSpace Ω} (P : Measure Ω)
1090    [IsProbabilityMeasure P] (ℱ : Filtration ℝ mΩ) (t : ℝ) (ht : 0 ≤ t) (M A : ℝ → Ω → ℝ)
1091    (hM : IsContSqMartingale P ℱ t M) (hA : IsQuadraticVariation P ℱ t M A)
1092    (hint : Integrable (A t) P) (v : ℝ) (hv : 0 ≤ v) :
1093    wassersteinE 2 (law P (M t)) (gaussianLaw v) ≤
1094      ENNReal.ofReal (Real.sqrt (∫ ω, |A t ω - v| ∂P)) := by
1095  obtain ⟨Ω', m, P', hP', M', A', β, hMm, hAm, hB, hlaw, hid, hgauss, hbd⟩ :=
1096    dds_coupling_strong P ℱ t ht M A hM hA hint v hv
1097  have hMt : Measurable (M t) := (hM.martingale.1 t).measurable.mono (ℱ.le t) le_rfl
1098  have hAt : Measurable (A t) := (hA.adapted t).mono (ℱ.le t) le_rfl
1099  have hl1 : law P' M' = law P (M t) := by
1100    have e := congrArg (Measure.map Prod.fst) hlaw
1101    unfold law at e ⊢
1102    rwa [Measure.map_map measurable_fst (hMm.prodMk hAm),
1103      Measure.map_map measurable_fst (hMt.prodMk hAt)] at e

1105  have hc := wassersteinE_le_of_coupling P' 2 (by norm_num) M' (β (Real.toNNReal v))
1106    hMm.aemeasurable hY

1111    refine le_trans (le_of_eq ?_) hbd
1112    refine lintegral_congr_ae ?_
1113    filter_upwards [hid] with ω h
1114    rw [h, ← ofReal_norm_eq_enorm, ← ENNReal.ofReal_pow (norm_nonneg _)]
1115    norm_cast

1117      ≤ (ENNReal.ofReal (∫ ω, |A t ω - v| ∂P)) ^ (1 / (2 : ℝ)) :=
1118        ENNReal.rpow_le_rpow h2 (by norm_num)
1119    _ = ENNReal.ofReal (Real.sqrt (∫ ω, |A t ω - v| ∂P)) := by
