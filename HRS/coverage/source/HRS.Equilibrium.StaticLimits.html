28/-- `h_{φ,ε} = 1_{dist(x,y) > ε} h_φ` (`E:eq:hphi`, source lines 9654-9657), with the periodic
29Euclidean distance (the leading constant for `s > d/2` depends on the shape of the cutoff). -/
30def hCut (d : ℕ) (s : ℝ) (φ : Torus d → ℝ) (ε : ℝ) (x y : Torus d) : ℝ :=
31  if ε < Torus.eucDist x y then hTransport d s φ x y else 0

33/-- `J_{φ,ρ} = s²|𝕊^{d-1}|/(d(d+2)) ∫ ρ² ((Δφ)² + 2|D²φ|_F²) dx` (`E:eq:J`). -/
34def localDivCoeff (d : ℕ) (s : ℝ) (ρ φ : Torus d → ℝ) : ℝ :=
35  s ^ 2 * sphereArea d / (d * (d + 2)) *
36    ∫ x, ρ x ^ 2 * (Torus.laplacian φ x ^ 2 +
37      2 * ∑ i : Fin d, ∑ j : Fin d, Torus.partialDeriv i (Torus.partialDeriv j φ) x ^ 2)
38
39/-- Proposition 7.17, `E:eq:J`: `J_{φ,ρ} > 0` for `0 < s < d`, a smooth positive density `ρ`
40and a smooth nonconstant `φ`. -/
41theorem localDivCoeff_pos {s : ℝ} (hd : 2 ≤ d) (hs0 : 0 < s) (hsd : s < d) (ρ : Torus d → ℝ)
42    (hρ : SmoothPosDensity ρ) (φ : Torus d → ℝ) (hφ : Torus.IsSmooth φ)
43    (hnc : ∃ x y, φ x ≠ φ y) :
44    0 < localDivCoeff d s ρ φ := by
45  have hρc : Continuous ρ := StaticLimitAux.continuous_of_smooth hρ.smooth
46  set g : Torus d → ℝ := fun x => ρ x ^ 2 * (Torus.laplacian φ x ^ 2 +
47      2 * ∑ i : Fin d, ∑ j : Fin d, Torus.partialDeriv i (Torus.partialDeriv j φ) x ^ 2) with hg
48  have gcont : Continuous g := by
49    refine (hρc.pow 2).mul ((( StaticLimitsAux.continuous_laplacian hφ).pow 2).add
50      (continuous_const.mul (continuous_finset_sum _ fun i _ => continuous_finset_sum _ fun j _ => ?_)))
51    exact (StaticLimitsAux.continuous_second hφ i j).pow 2
52  have gnn : 0 ≤ g := fun x => by
53    simp only [hg]
54    positivity
55  have hex : ∃ x0, g x0 ≠ 0 := by
56    by_contra hcon
57    push_neg at hcon
58    obtain ⟨x, y, hxy⟩ := hnc
59    refine hxy (StaticLimitsAux.const_of_second_zero hφ (fun i j z => ?_) x y)
60    have h0 := hcon z
61    have hρz : ρ z ^ 2 ≠ 0 := (pow_pos (hρ.pos z) 2).ne'
62    have h1 : Torus.laplacian φ z ^ 2 +
63        2 * ∑ i : Fin d, ∑ j : Fin d, Torus.partialDeriv i (Torus.partialDeriv j φ) z ^ 2 = 0 := by
64      simp only [hg] at h0
65      exact (mul_eq_zero.mp h0).resolve_left hρz
66    have h2 : 0 ≤ ∑ i : Fin d, ∑ j : Fin d, Torus.partialDeriv i (Torus.partialDeriv j φ) z ^ 2 :=
67      Finset.sum_nonneg fun i _ => Finset.sum_nonneg fun j _ => sq_nonneg _
68    have h3 : ∑ i : Fin d, ∑ j : Fin d, Torus.partialDeriv i (Torus.partialDeriv j φ) z ^ 2 = 0 := by
69      nlinarith [sq_nonneg (Torus.laplacian φ z)]
70    have h4 := (Finset.sum_eq_zero_iff_of_nonneg fun i _ => Finset.sum_nonneg fun j _ =>
71      sq_nonneg (Torus.partialDeriv i (Torus.partialDeriv j φ) z)).mp h3 i (Finset.mem_univ _)
72    have h5 := (Finset.sum_eq_zero_iff_of_nonneg fun j _ =>
73      sq_nonneg (Torus.partialDeriv i (Torus.partialDeriv j φ) z)).mp h4 j (Finset.mem_univ _)
74    exact pow_eq_zero_iff (two_ne_zero) |>.mp h5
75  obtain ⟨x0, hx0⟩ := hex
76  have hpos : 0 < ∫ x, g x :=
77    Continuous.integral_pos_of_hasCompactSupport_nonneg_nonzero gcont
78      (HasCompactSupport.of_compactSpace g) gnn hx0
79  have hK : 0 < s ^ 2 * sphereArea d / (d * (d + 2)) := by
80    have : (0 : ℝ) < d := by
81      have : (2 : ℝ) ≤ d := by exact_mod_cast hd
82      linarith
83    have := StaticLimitsAux.sphereArea_pos hd
84    positivity
85  exact mul_pos hK hpos

129/-- The covariance of a field law `γ` on `H^{-m}_0` is comparable to `‖·‖²_{L²(dx)}` on smooth
130zero-Haar-mean functions (hypothesis of Proposition 2.4, source lines 703-705). -/
131def HasL2ComparableCov (m : ℝ) (γ : Measure (NegSobolev d m)) : Prop :=
132  ∃ c C : ℝ, 0 < c ∧ ∀ φ : Torus d → ℝ, Torus.IsSmooth φ → Torus.mean φ = 0 →
133    c * ∫ x, φ x ^ 2 ≤ ∫ F, NegSobolev.eval F φ ^ 2 ∂γ ∧
134      ∫ F, NegSobolev.eval F φ ^ 2 ∂γ ≤ C * ∫ x, φ x ^ 2

136/-- Proposition 2.4 `M:H:prop:field-singularity` (source lines 701-708): for `m > d/2`, a centred
137Gaussian law `γ` on `H^{-m}_0` with `L²`-comparable covariance is mutually singular with the law
138of `√N(μ_N - ν_N)` for every `N`, every particle law and every probability measure `ν_N`.
139`hd : 1 ≤ d` excludes the point torus `d = 0`, where the statement is false; the source works
140under the standing assumption `d ≥ 2` (M L419), so this is weaker than inherited (`docs/ISSUES.md` §E). -/
141theorem field_singularity (hd : 1 ≤ d) (m : ℝ) (hm : (d : ℝ) / 2 < m) (γ : Measure (NegSobolev d m))
142    [IsProbabilityMeasure γ] (hγ : IsCenteredGaussian γ) (hcov : HasL2ComparableCov m γ)
143    (N : ℕ) (P : Measure (Config d N)) [IsProbabilityMeasure P] (ν : Measure (Torus d))
144    [IsProbabilityMeasure ν] :
145    γ ⟂ₘ law P (fun X => Real.sqrt N • NegSobolev.ofMeasureDiff m (empirical X) ν) := by
146  obtain ⟨c, C, hc, H⟩ := hcov
147  exact FieldSingAux.field_singularity_main hd hm γ hγ c hc (fun φ h1 h2 => (H φ h1 h2).1) N P ν

166/-- `Z_N = ⟨ψ, η_N⟩ = √N(μ_N(ψ) - μ_β(ψ))` (source line 7094). -/
167def linCoord {N : ℕ} (μ : Measure (Torus d)) (ψ : Torus d → ℝ) (X : Config d N) : ℝ :=
168  Real.sqrt N * ((1 / (N : ℝ)) * ∑ i, ψ (X i) - pair μ ψ)
169
170/-- The half-normal density `√(2/(πv)) e^{-z²/(2v)} 1_{z>0}` (`A:eq:half-normal`). -/
171def halfNormalDensity (v z : ℝ) : ℝ :=
172  if 0 < z then Real.sqrt (2 / (Real.pi * v)) * Real.exp (-z ^ 2 / (2 * v)) else 0
173
174/-- The cosine-tilted Gaussian density `(1 + a cos z)/(1 + a e^{-v/2}) · e^{-z²/(2v)}/√(2πv)`
175(`A:eq:cosine-density`). -/
176def cosineTiltDensity (a v z : ℝ) : ℝ :=
177  (1 + a * Real.cos z) / (1 + a * Real.exp (-v / 2)) *
178    (Real.exp (-z ^ 2 / (2 * v)) / Real.sqrt (2 * Real.pi * v))
179

181nonconstant `ψ`, `v = C_β(ψ,ψ) > 0`, `P_{N,β}(Z_N > 0) → 1/2`, and the law of `Z_N` under
182`P_{N,β}(· | Z_N > 0)` converges (against bounded continuous tests) to the half-normal law.
183The source derives the convergence of `Z_N = ⟨ψ, η_N⟩` from Theorem 2.9, which centres at the
184exact marginal; the mean-field centering used here has the same limit only for `s < 2d/3`
185(source lines 989-991), which is therefore assumed (see the report). -/

249/-! ### Lemma 7.5: joint centered quadratic limit (`s < d/2`) -/
250
251/-- `(e_i)` is an orthonormal basis of `L²_0(μ)`: square integrable, centred, orthonormal, and
252complete (a centred `L²` function orthogonal to every `e_i` vanishes a.e.). -/
253structure IsONBL2Zero (μ : Measure (Torus d)) (e : ℕ → Torus d → ℝ) : Prop where
254  memLp : ∀ i, MemLp (e i) 2 μ
255  centered : ∀ i, ∫ x, e i x ∂μ = 0
256  orthonormal : ∀ i j, l2Inner μ (e i) (e j) = if i = j then 1 else 0
257  complete : ∀ f : Torus d → ℝ, MemLp f 2 μ → ∫ x, f x ∂μ = 0 →
258    (∀ i, l2Inner μ f (e i) = 0) → f =ᵐ[μ] 0
259
260/-- Lemma 7.5 `M:N:lem:quadratic-limit` (source lines 7956-7966): `s < d/2`, `Q_{N,β}(μ)` with a
261smooth positive background, `w = g + c_* ≥ 1` off zero, `h` symmetric with `|h| ≤ B w`,
262`B_N(h) = N(F_N^h - E F_N^h)`. Jointly with the exactly centred linear statistics
263`√N(μ_N(φ_a) - m_N(φ_a))` of finitely many bounded tests, `B_N(h)` converges to
264`½ I_2^{R_{β,μ}}(H_μ)`, and mixed expectations against bounded continuous functions converge.
265The limit is described on a probability space carrying `G_i = G(e_i)`, jointly Gaussian with
266covariance `C_{β,μ}(e_i, e_j) = ⟨e_i, R_{β,μ} e_j⟩` along an orthonormal basis `(e_i)` of
267`L²_0(μ)`; `G(φ_a)` is the `L²` limit of `∑_{i<L} ⟨φ_a, e_i⟩ G_i`, and the chaos
268`Q = ½ I_2^R(H_μ)` is the `L²` limit of `½ ∑_{i,j<L} ⟨e_i, H_μ e_j⟩ (G_i G_j - C(e_i,e_j))`
269(this equals `½ I_2(R^{1/2} H_μ R^{1/2})`, `N:eq:second-chaos-definition`). -/
270theorem quadratic_limit {s : ℝ} (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hs : s < (d : ℝ) / 2) (β : ℝ)
271    (hβ : 0 < β) (ρ : Torus d → ℝ) (hρ : SmoothPosDensity ρ) (c : ℝ)
272    (hc : ∀ z, z ≠ 0 → 1 ≤ periodicRiesz d s z + c) (B : ℝ) (h : Torus d → Torus d → ℝ)
273    (hmeas : Measurable (Function.uncurry h)) (hsymm : ∀ x y, h x y = h y x)
274    (hB : ∀ x y, x ≠ y → |h x y| ≤ B * (periodicRiesz d s (x - y) + c))
275    (e : ℕ → Torus d → ℝ) (he : IsONBL2Zero (withDensityReal ρ) e)
276    (p : ℕ) (φ : Fin p → Torus d → ℝ) (hφ : ∀ a, Measurable (φ a) ∧ ∃ M, ∀ x, |φ a x| ≤ M) :
277    (∀ N : ℕ, 1 ≤ N → Integrable (fun X : Config d N => modulatedEnergy h X (withDensityReal ρ))
278      (modulatedGibbs (periodicRiesz d s) N β (withDensityReal ρ))) ∧
279    ∃ (Ω : Type) (_ : MeasurableSpace Ω) (P : Measure Ω) (_ : IsProbabilityMeasure P)
280      (G : ℕ → Ω → ℝ) (Q : Ω → ℝ) (Lin : Fin p → Ω → ℝ),
281      (∀ n : ℕ, law P (fun ω => (WithLp.toLp 2 fun i : Fin n => G i ω : EuclideanSpace ℝ (Fin n))) =
282        multivariateGaussian 0 (Matrix.of fun i j : Fin n =>
283          susceptibilityCov (periodicRiesz d s) (withDensityReal ρ) β (e i) (e j))) ∧
