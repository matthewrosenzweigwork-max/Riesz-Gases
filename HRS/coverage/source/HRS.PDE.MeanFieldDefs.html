31/-- The right-hand side `β⁻¹Δρ + Div(ρ∇(V + g*ρ))` of `O:eq:mf-finite` (line 5774) in coefficients:
32`-4π²|k|² β⁻¹ ρ̂(k) + ∑_j 2πi k_j [ρ̂ * (2πi k_j V̂ + 2πi k_j ĝ ρ̂)](k)`. -/
33def mfRhsCoeff (d : ℕ) (s : ℝ) (V : Torus d → ℝ) (invβ : ℝ) (ρc : Coeff d) : Coeff d :=
34  fun k => (invβ : ℂ) * laplaceCoeff ρc k +
35    ∑ j : Fin d, derivCoeff j
36      (mulCoeff ρc (fun k' => derivCoeff j (coeffOf V) k' + gradRieszCoeff d s j ρc k')) k

38/-- A global solution of `O:eq:mf-finite` in `C([0,∞);H^K)` with initial density `ρ⁰`, at the
39coefficient level: `ρ̂(0) = ρ̂⁰`, `ρ̂ ∈ C([0,∞);H^K)`, and the integrated equation for every mode
40with an integrable integrand. -/
41structure IsMFSolutionHK (d : ℕ) (s : ℝ) (V : Torus d → ℝ) (invβ K : ℝ) (ρ₀ : Torus d → ℝ)
42    (ρc : ℝ → Coeff d) : Prop where
43  init : ρc 0 = coeffOf ρ₀
44  cont : ContinuousInHsOn K ρc (Set.Ici 0)
45  integrable : ∀ k, ∀ t, 0 ≤ t →
46    IntegrableOn (fun σ => mfRhsCoeff d s V invβ (ρc σ) k) (Set.Icc 0 t)
47  eq : ∀ k, ∀ t, 0 ≤ t →
48    ρc t k = coeffOf ρ₀ k + ∫ σ in (0 : ℝ)..t, mfRhsCoeff d s V invβ (ρc σ) k

68/-- The effective velocity `u^t = β⁻¹∇log ρ^t + ∇(V + g*ρ^t)` (line 5787), `∇g*ρ` with the

70def effVelocity (d : ℕ) (s : ℝ) (V : Torus d → ℝ) (β : ℝ) (ρ : Torus d → ℝ) (x : Torus d) :

81/-- The centered equilibrium backward semigroup `S_β(τ) ψ := Q_β U^{0,τ} ψ` for the frozen

87/-- A solution `η` of the linear SPDE `C:eq:linear-spde` (lines 5921–5925)
88`∂_tη = νΔη + Div(η∇(V+g*ρ^t) + ρ^t∇g*η) - √(2ν) Div(√ρ^t ξ)` on `[0,∞)`, in mild form
89(`H:eq:linear-mild`; modelled on `HRS.IsLinearFluctuationSolution`): `η^t - η⁰ ∘ U^{0,t}` is a
90centered Gaussian family independent of `η⁰` with covariance
91`2ν ∫_0^{t∧t'} μ^r(∇U^{r,t}φ·∇U^{r,t'}ψ) dr` (noise level `q = ν = D.invβ`), paths continuous in
92`H^{-j}_0` (zero mass built in).  `D` carries `M = -I`, `ν = β⁻¹`. -/
93structure IsLinearSPDESolution {Ω : Type*} [MeasurableSpace Ω] (P : Measure Ω)
94    (D : DynData d) (ρ : ℝ → Torus d → ℝ) (j : ℝ) (η₀ : Ω → NegSobolev d j)
95    (η : ℝ → Ω → NegSobolev d j) : Prop where
96  init : ∀ᵐ ω ∂P, η 0 ω = η₀ ω
97  cont : ∀ᵐ ω ∂P, ContinuousOn (fun t => η t ω) (Set.Ici 0)
98  meas : ∀ t, Measurable (η t)
99  gaussian_part : ∀ (p : ℕ) (ts : Fin p → ℝ) (φ : Fin p → Torus d → ℝ),
100    (∀ a, 0 ≤ ts a) → (∀ a, Torus.IsSmooth (φ a)) →
101    let G : Ω → EuclideanSpace ℝ (Fin p) := fun ω => WithLp.toLp 2 fun a =>
102      NegSobolev.eval (η (ts a) ω) (φ a) - NegSobolev.eval (η₀ ω) (propagator D ρ 0 (ts a) (φ a))
103    ProbabilityTheory.IndepFun G η₀ P ∧
104    law P G = ProbabilityTheory.multivariateGaussian 0 (Matrix.of fun a b =>
105      2 * D.invβ * ∫ r in Set.Icc 0 (min (ts a) (ts b)), ∫ x,
106        inner ℝ (Torus.grad (propagator D ρ r (ts a) (φ a)) x)
107          (Torus.grad (propagator D ρ r (ts b) (φ b)) x) * ρ r x)

109/-- The covariance of the stationary Gaussian evolution at lags `τ_a` for tests `φ_a`
110(`C:eq:stationary-cov`, lines 5908–5913, with the time structure of the stationary process):
111`Σ_{ab} = 𝒞_β(φ_a, S_β(τ_b - τ_a) φ_b)` for `τ_a ≤ τ_b` (and symmetrically), where
112`𝒞_β(φ,ψ) = ⟨Q_βφ, (I+βK_β)^{-1}Q_βψ⟩_{L²(μ_β)}` is `HRS.susceptibilityCov`. -/
113def stationaryLagCov (d : ℕ) (s β : ℝ) (D : DynData d) (ρβ : Torus d → ℝ) {p : ℕ}
114    (τ : Fin p → ℝ) (φ : Fin p → Torus d → ℝ) : Matrix (Fin p) (Fin p) ℝ :=
115  Matrix.of fun a b =>
116    if τ a ≤ τ b then
117      susceptibilityCov (periodicRiesz d s) (withDensityReal ρβ) β (φ a)
118        (equilibriumSemigroup D ρβ (τ b - τ a) (φ b))
119    else
120      susceptibilityCov (periodicRiesz d s) (withDensityReal ρβ) β
121        (equilibriumSemigroup D ρβ (τ a - τ b) (φ a)) (φ b)

127/-- `R` is doubly centered: `∫ R(x,y) dμ_β(x) = 0` for all `y` and `∫ R(x,y) dμ_β(y) = 0` for all
128`x`. -/
129def DoublyCentered (μ : Measure (Torus d)) (R : Torus d → Torus d → ℝ) : Prop :=
130  (∀ y, pair μ (fun x => R x y) = 0) ∧ ∀ x, pair μ (fun y => R x y) = 0

159/-- `E:eq:kchoice` (lines 7291–7294): "such a choice exists at every positive temperature":
160for `β > 0` there is `k ≠ 0` with `β ĝ(k) ≤ 1/4`. -/
161theorem exists_transient_mode (hd : 2 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (β : ℝ)
162    (hβ : 0 < β) : ∃ k : Lattice d, k ≠ 0 ∧ β * rieszCoeff d s k ≤ 1 / 4 := by
163  have hd0 : 0 < d := by omega
164  set i : Fin d := ⟨0, hd0⟩
165  have h1 : Tendsto (fun x : ℝ => x ^ (s - (d : ℝ))) atTop (𝓝 0) := by
166    simpa [neg_sub] using tendsto_rpow_neg_atTop (y := (d : ℝ) - s) (by linarith)
167  have h2 : Tendsto (fun n : ℕ => β * (rieszConst d s * (2 * Real.pi * (n : ℝ)) ^ (s - (d : ℝ))))
168      atTop (𝓝 0) := by
169    have h3 : Tendsto (fun n : ℕ => 2 * Real.pi * (n : ℝ)) atTop atTop :=
170      tendsto_natCast_atTop_atTop.const_mul_atTop (by positivity)
171    have := ((h1.comp h3).const_mul (rieszConst d s)).const_mul β
172    simpa using this
173  obtain ⟨N, hN, hN1⟩ := ((h2.eventually (Iio_mem_nhds (show (0:ℝ) < 1/4 by norm_num))).and
174    (eventually_ge_atTop 1)).exists
175  have hk : (Pi.single i (N : ℤ) : Lattice d) ≠ 0 := by
176    intro h
177    have := congrFun h i
178    simp at this
179    omega
180  refine ⟨Pi.single i (N : ℤ), hk, ?_⟩
181  unfold rieszCoeff
182  simp only [hk, ite_false]
183  rw [enorm_single_aux]
184  exact le_of_lt hN
