22/-- The candidate compact set. -/
23def ptc_K (δ : ℕ → ℝ) (F : ℕ → Finset (Set.Icc (0 : ℝ) T)) (Kt : ℕ → Set.Icc (0 : ℝ) T → Set E) :
24    Set C(Set.Icc (0 : ℝ) T, E) :=
25  {z | ∀ l : ℕ, (∀ s t : Set.Icc (0 : ℝ) T, |(t : ℝ) - s| ≤ δ l →
26      dist (z t) (z s) ≤ 1 / ((l : ℝ) + 1)) ∧ ∀ t ∈ F l, z t ∈ Kt l t}

28theorem ptc_K_isClosed (δ : ℕ → ℝ) (F : ℕ → Finset (Set.Icc (0 : ℝ) T))
29    (Kt : ℕ → Set.Icc (0 : ℝ) T → Set E) (hKt : ∀ l t, IsClosed (Kt l t)) :
30    IsClosed (ptc_K T δ F Kt) := by
31  have hev : ∀ x : Set.Icc (0 : ℝ) T, Continuous fun z : C(Set.Icc (0 : ℝ) T, E) => z x :=
32    fun x => continuous_eval_const x
33  simp only [ptc_K, Set.ofPred_forall, Set.ofPred_and]
34  refine isClosed_iInter fun l => IsClosed.inter (isClosed_iInter fun s => isClosed_iInter fun t => ?_)
35    (isClosed_iInter fun t => isClosed_iInter fun ht => ?_)
36  · by_cases hp : |(t : ℝ) - s| ≤ δ l
37    · have := isClosed_le ((hev t).dist (hev s)) (continuous_const (y := 1 / ((l : ℝ) + 1)))
38      simpa [hp] using this
39    · simp [hp]
40  · exact (hKt l t).preimage (hev t)

42theorem ptc_K_isCompact [CompleteSpace E] (δ : ℕ → ℝ) (hδ : ∀ l, 0 < δ l)
43    (F : ℕ → Finset (Set.Icc (0 : ℝ) T))
44    (hcov : ∀ l, ∀ t : Set.Icc (0 : ℝ) T, ∃ t' ∈ F l, |(t : ℝ) - t'| ≤ δ l)
45    (Kt : ℕ → Set.Icc (0 : ℝ) T → Set E) (hKt : ∀ l t, IsCompact (Kt l t)) :
46    IsCompact (ptc_K T δ F Kt) := by
47  set K := ptc_K T δ F Kt with hK
48  have hclosed : IsClosed K := ptc_K_isClosed T δ F Kt fun l t => (hKt l t).isClosed
49  -- range set
50  set S : Set E := {y | ∃ z ∈ K, ∃ t : Set.Icc (0 : ℝ) T, y = z t} with hS
51  have hSbdd : TotallyBounded S := by
52    rw [Metric.totallyBounded_iff]
53    intro ε hε
54    obtain ⟨l, hl⟩ := exists_nat_one_div_lt (half_pos hε)
55    have hQ : IsCompact (⋃ t ∈ F l, Kt l t) := (F l).isCompact_biUnion fun t _ => hKt l t
56    obtain ⟨c, hcQ, hcfin, hcov'⟩ := hQ.finite_cover_balls (half_pos hε)
57    refine ⟨c, hcfin, ?_⟩
58    rintro y ⟨z, hz, t, rfl⟩
59    obtain ⟨t', ht', hdt⟩ := hcov l t
60    have h1 : dist (z t) (z t') ≤ 1 / ((l : ℝ) + 1) := (hz l).1 t' t hdt
61    have h2 : z t' ∈ ⋃ t ∈ F l, Kt l t := Set.mem_biUnion ht' ((hz l).2 t' ht')
62    obtain ⟨y', hy', hy'b⟩ := Set.mem_iUnion₂.1 (hcov' h2)
63    refine Set.mem_iUnion₂.2 ⟨y', hy', ?_⟩
64    rw [Metric.mem_ball] at hy'b ⊢
65    calc dist (z t) y' ≤ dist (z t) (z t') + dist (z t') y' := dist_triangle _ _ _
66      _ < ε / 2 + ε / 2 := add_lt_add_of_le_of_lt (h1.trans hl.le) hy'b
67      _ = ε := add_halves ε
68  have hScpt : IsCompact (closure S) :=
69    (isCompact_iff_totallyBounded_isComplete.2 ⟨hSbdd.closure, isClosed_closure.isComplete⟩)
70  -- Arzelà-Ascoli through `C(I,E) ≃ I →ᵇ E`
71  let e := ContinuousMap.equivBoundedOfCompact (Set.Icc (0 : ℝ) T) E
72  have he : Topology.IsEmbedding e :=
73    (ContinuousMap.isUniformEmbedding_equivBoundedOfCompact (Set.Icc (0 : ℝ) T) E).isEmbedding
74  have hecl : Topology.IsClosedEmbedding e :=
75    ⟨he, by simp [Set.range_eq_univ.2 e.surjective]⟩
76  have hA : IsCompact (closure (e '' K)) := by
77    refine BoundedContinuousFunction.arzela_ascoli (closure S) hScpt _ ?_ ?_
78    · rintro f x ⟨z, hz, rfl⟩
79      exact subset_closure ⟨z, hz, x, rfl⟩
80    · intro x0
81      rw [Metric.equicontinuousAt_iff]
82      intro ε hε
83      obtain ⟨l, hl⟩ := exists_nat_one_div_lt hε
84      refine ⟨δ l, hδ l, fun x hx f => ?_⟩
85      obtain ⟨f, z, hz, rfl⟩ := f
86      have hx' : |(x : ℝ) - x0| ≤ δ l := by
87        rw [← Real.dist_eq]; exact hx.le
88      have := (hz l).1 x0 x hx'
89      rw [dist_comm]
90      exact lt_of_le_of_lt this hl
91  rw [he.isCompact_iff]
92  rw [(hecl.isClosedMap K hclosed).closure_eq] at hA
93  exact hA

97/-- Finite `δ`-nets in `[0,T]` made of points of a dense subset `D`. -/
98theorem ptc_cover (T : ℝ) (D : Set ℝ) (hDT : D ⊆ Set.Icc (0 : ℝ) T)
99    (hD : Set.Icc (0 : ℝ) T ⊆ closure D) (δ : ℝ) (hδ : 0 < δ) :
100    ∃ F : Finset (Set.Icc (0 : ℝ) T), (∀ t ∈ F, (t : ℝ) ∈ D) ∧
101      ∀ t : Set.Icc (0 : ℝ) T, ∃ t' ∈ F, |(t : ℝ) - t'| ≤ δ := by
102  classical
103  obtain ⟨F0, hF0⟩ := (isCompact_univ (X := Set.Icc (0 : ℝ) T)).elim_finite_subcover
104    (ι := {d : Set.Icc (0 : ℝ) T // (d : ℝ) ∈ D})
105    (fun d => Metric.ball (d : Set.Icc (0 : ℝ) T) δ) (fun _ => Metric.isOpen_ball) (by
106      intro x _
107      obtain ⟨b, hb, hxb⟩ := Metric.mem_closure_iff.1 (hD x.2) δ hδ
108      refine Set.mem_iUnion.2 ⟨⟨⟨b, hDT hb⟩, hb⟩, ?_⟩
109      rw [Metric.mem_ball]
110      simpa [Subtype.dist_eq] using hxb)
111  refine ⟨F0.image Subtype.val, ?_, ?_⟩
112  · intro t ht
113    obtain ⟨d, _, rfl⟩ := Finset.mem_image.1 ht
114    exact d.2
115  · intro t
116    obtain ⟨d, hd, hdt⟩ := Set.mem_iUnion₂.1 (hF0 (Set.mem_univ t))
117    refine ⟨d.1, Finset.mem_image.2 ⟨d, hd, rfl⟩, ?_⟩
118    rw [Metric.mem_ball, Subtype.dist_eq, Real.dist_eq] at hdt
119    exact hdt.le

122/-- Lemma S.A.21, statement copied verbatim from `HRS.Whole.AppendixALemmas.path_tightness_criterion`. -/
123theorem path_tightness_criterion_proof {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E]
124    [CompleteSpace E] [TopologicalSpace.SeparableSpace E] [MeasurableSpace E] [BorelSpace E]
125    (T : ℝ) (hT : 0 < T) {Ω : ℕ → Type*} [∀ N, MeasurableSpace (Ω N)] (P : ∀ N, Measure (Ω N))
126    [∀ N, IsProbabilityMeasure (P N)] (Z : ∀ N, ℝ → Ω N → E)
127    (hZmeas : ∀ N : ℕ, 1 ≤ N → ∀ t, Measurable (Z N t))
128    (hcont : ∀ N : ℕ, 1 ≤ N → ∀ᵐ ζ ∂(P N), Continuous fun t : Set.Icc (0 : ℝ) T => Z N t ζ)
129    (hpath : ∀ N : ℕ, 1 ≤ N → Measurable fun ζ => contPathOf T (fun t => Z N t ζ))
130    (D : Set ℝ) (hDT : D ⊆ Set.Icc (0 : ℝ) T) (hD : Set.Icc (0 : ℝ) T ⊆ closure D)
131    (htight : ∀ t ∈ D, IsTightMeasureSet {μ | ∃ N : ℕ, 1 ≤ N ∧ μ = (P N).map (Z N t)})
132    (hmod : ∀ ε : ℝ, 0 < ε → Tendsto (fun δ : ℝ =>
133      ⨆ N : ℕ, ⨆ (_ : 1 ≤ N), P N {ζ | ENNReal.ofReal ε < omegaE (fun t => Z N t ζ) T δ})
134      (𝓝[>] 0) (𝓝 0)) :
135    IsTightMeasureSet {μ | ∃ N : ℕ, 1 ≤ N ∧
136      μ = (P N).map (fun ζ => contPathOf T (fun t => Z N t ζ))} := by
137  classical
138  rw [isTightMeasureSet_iff_exists_isCompact_measure_compl_le]
139  intro ε hε
140  obtain ⟨ε', hε'pos, hsum⟩ := ENNReal.exists_pos_sum_of_countable
141    (ENNReal.half_pos hε.ne').ne' ℕ
142  have hε'ne : ∀ l, (ε' l : ℝ≥0∞) ≠ 0 := fun l => by exact_mod_cast (hε'pos l).ne'
143  -- step 1: scales `δ l`
144  have hδ : ∀ l : ℕ, ∃ δ : ℝ, 0 < δ ∧ ∀ N : ℕ, 1 ≤ N →
145      P N {ζ | ENNReal.ofReal (1 / ((l : ℝ) + 1)) < omegaE (fun t => Z N t ζ) T δ} ≤ ε' l := by
146    intro l
147    have h := hmod (1 / ((l : ℝ) + 1)) (by positivity)
148    have hlt := h.eventually (Iio_mem_nhds (pos_iff_ne_zero.2 (hε'ne l)))
149    obtain ⟨u, hu, hsub⟩ := mem_nhdsGT_iff_exists_Ioo_subset.1 hlt
150    have hu' : (0 : ℝ) < u := hu
151    refine ⟨u / 2, half_pos hu', fun N hN => ?_⟩
152    have h2 := hsub ⟨half_pos hu', half_lt_self hu'⟩
153    exact (le_iSup₂ (f := fun (N : ℕ) (_ : 1 ≤ N) =>
154      P N {ζ | ENNReal.ofReal (1 / ((l : ℝ) + 1)) < omegaE (fun t => Z N t ζ) T (u / 2)})
155      N hN).trans (le_of_lt h2)
156  choose δ hδpos hδbd using hδ
157  -- step 2: nets
158  have hF : ∀ l : ℕ, ∃ F : Finset (Set.Icc (0 : ℝ) T), (∀ t ∈ F, (t : ℝ) ∈ D) ∧
159      ∀ t : Set.Icc (0 : ℝ) T, ∃ t' ∈ F, |(t : ℝ) - t'| ≤ δ l :=
160    fun l => ptc_cover T D hDT hD (δ l) (hδpos l)
161  choose F hFD hFcov using hF
162  -- step 3: compacts in `E`
163  have hKt : ∀ (l : ℕ) (t : Set.Icc (0 : ℝ) T), ∃ K : Set E, IsCompact K ∧
164      ((t : ℝ) ∈ D → ∀ N : ℕ, 1 ≤ N →
165        P N {ζ | Z N t ζ ∉ K} ≤ (ε' l : ℝ≥0∞) / ((F l).card : ℝ≥0∞)) := by
166    intro l t
167    by_cases ht : (t : ℝ) ∈ D
168    · obtain ⟨K, hK, hKb⟩ := (isTightMeasureSet_iff_exists_isCompact_measure_compl_le.1
169        (htight t ht)) ((ε' l : ℝ≥0∞) / ((F l).card : ℝ≥0∞))
170        (ENNReal.div_pos (hε'ne l) (ENNReal.natCast_ne_top _))
171      refine ⟨K, hK, fun _ N hN => ?_⟩
172      have h := hKb _ ⟨N, hN, rfl⟩
173      rwa [Measure.map_apply (hZmeas N hN t) hK.isClosed.measurableSet.compl] at h
174    · exact ⟨∅, isCompact_empty, fun h => absurd h ht⟩
175  choose Kt hKtc hKtb using hKt
176  refine ⟨ptc_K T δ F Kt, ptc_K_isCompact T δ hδpos F hFcov Kt hKtc, ?_⟩
177  rintro μ ⟨N, hN, rfl⟩
178  have hKcl : IsClosed (ptc_K T δ F Kt) := ptc_K_isClosed T δ F Kt fun l t => (hKtc l t).isClosed
179  rw [Measure.map_apply (hpath N hN) hKcl.measurableSet.compl]
180  -- step 6: union bound
181  have hsub : (fun ζ => contPathOf T (fun t => Z N t ζ)) ⁻¹' (ptc_K T δ F Kt)ᶜ ⊆
182      {ζ | ¬ Continuous fun t : Set.Icc (0 : ℝ) T => Z N t ζ} ∪
183        ⋃ l : ℕ, ({ζ | ENNReal.ofReal (1 / ((l : ℝ) + 1)) < omegaE (fun t => Z N t ζ) T (δ l)} ∪
184          ⋃ t ∈ F l, {ζ | Z N t ζ ∉ Kt l t}) := by
185    intro ζ hζ
186    by_contra hnot
187    simp only [Set.mem_union, Set.mem_iUnion, Set.mem_ofPred_eq, not_or, not_exists, not_lt,
188      not_not] at hnot
189    obtain ⟨hc, hl⟩ := hnot
190    apply hζ
191    have hpe : ∀ t : Set.Icc (0 : ℝ) T, contPathOf T (fun t => Z N t ζ) t = Z N t ζ := by
192      intro t
193      simp [contPathOf, hc]
194    intro l
195    simp only [hpe]
196    refine ⟨fun s t hst => ?_, fun t ht => hl l |>.2 t ht⟩
197    have hle : ‖Z N t ζ - Z N s ζ‖ₑ ≤ omegaE (fun t => Z N t ζ) T (δ l) :=
198      le_iSup₂_of_le (s : ℝ) s.2 (le_iSup₂_of_le (t : ℝ) t.2 (le_iSup_of_le hst le_rfl))
199    have h1 := hle.trans (hl l).1
200    have hr : (0 : ℝ) ≤ 1 / ((l : ℝ) + 1) := by positivity
201    rw [← ofReal_norm] at h1
202    rw [dist_eq_norm]
203    exact (ENNReal.ofReal_le_ofReal_iff hr).1 h1
204  have hnull : P N {ζ | ¬ Continuous fun t : Set.Icc (0 : ℝ) T => Z N t ζ} = 0 :=
205    ae_iff.1 (hcont N hN)
206  have hcard : ∀ l : ℕ, ∑ t ∈ F l, (ε' l : ℝ≥0∞) / ((F l).card : ℝ≥0∞) ≤ ε' l := by
207    intro l
208    rw [Finset.sum_const, nsmul_eq_mul]
209    exact ENNReal.mul_div_le
210  calc P N ((fun ζ => contPathOf T (fun t => Z N t ζ)) ⁻¹' (ptc_K T δ F Kt)ᶜ)
211      ≤ P N ({ζ | ¬ Continuous fun t : Set.Icc (0 : ℝ) T => Z N t ζ} ∪
212        ⋃ l : ℕ, ({ζ | ENNReal.ofReal (1 / ((l : ℝ) + 1)) < omegaE (fun t => Z N t ζ) T (δ l)} ∪
213          ⋃ t ∈ F l, {ζ | Z N t ζ ∉ Kt l t})) := measure_mono hsub
214    _ ≤ 0 + ∑' l : ℕ, ((ε' l : ℝ≥0∞) + ε' l) := by
215        refine (measure_union_le _ _).trans (add_le_add hnull.le ?_)
216        refine (measure_iUnion_le _).trans (ENNReal.tsum_le_tsum fun l => ?_)
217        refine (measure_union_le _ _).trans (add_le_add (hδbd l N hN) ?_)
218        refine (measure_biUnion_finset_le _ _).trans ?_
219        refine (Finset.sum_le_sum fun t ht => hKtb l t (hFD l t ht) N hN).trans (hcard l)
220    _ = ∑' l : ℕ, (ε' l : ℝ≥0∞) + ∑' l : ℕ, (ε' l : ℝ≥0∞) := by
221        rw [zero_add, ENNReal.tsum_add]
222    _ ≤ ε / 2 + ε / 2 := add_le_add hsum.le hsum.le
223    _ = ε := ENNReal.add_halves ε
