36theorem uw_ball_le {m : ℕ} {χ : Euc d → ℝ} (hχ : IsUlocCutoff χ) {u : Euc d → ℝ}
37    (hu : ContDiff ℝ m u) {n : ℕ} (hn : n ≤ m) (i : Fin n → Fin d) (w : Euc d) :
38    ∫⁻ x in Metric.ball w 1, ENNReal.ofReal ((partialDeriv i u x) ^ 2) ≤ ulocNormE m χ u ^ 2 := by
39  set f : Euc d → ℝ := fun x => χ (x - w) * u x with hf
40  have hχm : ContDiff ℝ m χ := hχ.1.of_le (by exact_mod_cast le_top)
41  have hfd : ContDiff ℝ m f := (hχm.comp (contDiff_id.sub contDiff_const)).mul hu
42  have hχc : HasCompactSupport (fun x => χ (x - w)) :=
43    hχ.2.1.comp_homeomorph (Homeomorph.subRight w)
44  have hfc : HasCompactSupport f := hχc.mul_right
45  calc ∫⁻ x in Metric.ball w 1, ENNReal.ofReal ((partialDeriv i u x) ^ 2)
46      = ∫⁻ x in Metric.ball w 1, ENNReal.ofReal ((partialDeriv i f x) ^ 2) := by
47        refine setLIntegral_congr_fun Metric.isOpen_ball.measurableSet fun x hx => ?_
48        rw [hf, uw_pd_eq hχ u i w x hx]

64theorem uw_cell_le {m : ℕ} {χ : Euc d → ℝ} (hχ : IsUlocCutoff χ) {u : Euc d → ℝ}
65    (hu : ContDiff ℝ m u) {n : ℕ} (hn : n ≤ m) (i : Fin n → Fin d) (Y : Finset (Euc d))
66    (hY : ∀ p : Euc d, ‖p‖ ≤ Real.sqrt d → ∃ y ∈ Y, ‖p - y‖ < 1) (z : Fin d → ℤ) :
67    ∫⁻ x in uw_cell z, ENNReal.ofReal ((partialDeriv i u x) ^ 2) ≤
68      (Y.card : ℝ≥0∞) * ulocNormE m χ u ^ 2 := by
69  have hsub : uw_cell z ⊆ ⋃ y : Y, Metric.ball (latticePt z + (y : Euc d)) 1 := by
70    intro x hx
71    have h1 : ‖x - latticePt z‖ ≤ Real.sqrt d :=
72      Real.le_sqrt_of_sq_le (uw_cell_norm hx)
73    obtain ⟨y, hy, hxy⟩ := hY _ h1
74    refine Set.mem_iUnion.2 ⟨⟨y, hy⟩, ?_⟩
75    rw [Metric.mem_ball, dist_eq_norm]
76    have : x - (latticePt z + y) = x - latticePt z - y := by abel
77    rw [this]; exact hxy
78  calc ∫⁻ x in uw_cell z, ENNReal.ofReal ((partialDeriv i u x) ^ 2)
79      ≤ ∫⁻ x in ⋃ y : Y, Metric.ball (latticePt z + (y : Euc d)) 1,
80          ENNReal.ofReal ((partialDeriv i u x) ^ 2) := lintegral_mono_set hsub
81    _ ≤ ∑' y : Y, ∫⁻ x in Metric.ball (latticePt z + (y : Euc d)) 1,
82          ENNReal.ofReal ((partialDeriv i u x) ^ 2) := lintegral_iUnion_le _ _
83    _ ≤ ∑' _y : Y, ulocNormE m χ u ^ 2 :=
84        ENNReal.tsum_le_tsum fun y => uw_ball_le hχ hu hn i _
85    _ = (Y.card : ℝ≥0∞) * ulocNormE m χ u ^ 2 := by
86        rw [tsum_fintype, Finset.sum_const, Finset.card_univ, Fintype.card_coe, nsmul_eq_mul]

125theorem uw_part1 (m : ℕ) (χ : Euc d → ℝ) (hχ : IsUlocCutoff χ) :
126    ∃ C : ℝ, ∀ ϖ : Euc d → ℝ, (∀ x, 0 ≤ ϖ x) → ∀ u : Euc d → ℝ, ContDiff ℝ m u →
127      weightedHE m ϖ u ≤ ENNReal.ofReal C * (cellSupSum ϖ) ^ (1 / 2 : ℝ) * ulocNormE m χ u := by
128  obtain ⟨Y, hY⟩ := uw_net d
129  set M : ℕ := ∑ j ∈ Finset.range (m + 1), d ^ j with hM
130  refine ⟨Real.sqrt ((M : ℝ) * (Y.card : ℝ)), fun ϖ hϖ u hu => ?_⟩
131  set U := ulocNormE m χ u with hU
132  set S := cellSupSum ϖ with hS
133  set c : ℝ≥0∞ := S * ((Y.card : ℝ≥0∞) * U ^ 2) with hc
134  have hsum : ∑ j ∈ Finset.range (m + 1), ∑ i : Fin j → Fin d,
135      ∫⁻ x, ENNReal.ofReal (ϖ x * (partialDeriv i u x) ^ 2) ≤ (M : ℝ≥0∞) * c := by
136    calc _ ≤ ∑ j ∈ Finset.range (m + 1), ∑ _i : Fin j → Fin d, c := by
137          refine Finset.sum_le_sum fun j hj => Finset.sum_le_sum fun i _ => ?_
138          exact uw_tuple_le hχ hu (Nat.lt_succ_iff.1 (Finset.mem_range.1 hj)) i Y hY ϖ hϖ
139      _ = (M : ℝ≥0∞) * c := by
140          have hcount : ∀ j : ℕ, ∑ _i : Fin j → Fin d, c = (d ^ j : ℕ) • c := by
141            intro j
142            rw [Finset.sum_const, Finset.card_univ, Fintype.card_fun, Fintype.card_fin,
143              Fintype.card_fin]
144          simp_rw [hcount, nsmul_eq_mul]
145          rw [← Finset.sum_mul, hM]
146          push_cast
147          rfl
148  unfold weightedHE
149  calc (∑ j ∈ Finset.range (m + 1), ∑ i : Fin j → Fin d,
150          ∫⁻ x, ENNReal.ofReal (ϖ x * (partialDeriv i u x) ^ 2)) ^ (1 / 2 : ℝ)
151      ≤ ((M : ℝ≥0∞) * c) ^ (1 / 2 : ℝ) := ENNReal.rpow_le_rpow hsum (by norm_num)
152    _ = ENNReal.ofReal (Real.sqrt ((M : ℝ) * (Y.card : ℝ))) * S ^ (1 / 2 : ℝ) * U := by
153        have e1 : (M : ℝ≥0∞) * c = (ENNReal.ofReal ((M : ℝ) * (Y.card : ℝ)) * S) * U ^ 2 := by
154          rw [hc, ENNReal.ofReal_mul (Nat.cast_nonneg _), ENNReal.ofReal_natCast,
155            ENNReal.ofReal_natCast]
156          ring
157        rw [e1, ENNReal.mul_rpow_of_nonneg _ _ (by norm_num),
158          ENNReal.mul_rpow_of_nonneg _ _ (by norm_num),
159          ← ENNReal.rpow_natCast U 2, ← ENNReal.rpow_mul]
160        have e2 : ((2 : ℕ) : ℝ) * (1 / 2 : ℝ) = 1 := by norm_num
161        rw [e2, ENNReal.rpow_one,
162          ENNReal.ofReal_rpow_of_nonneg (mul_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _))
163            (by norm_num), Real.sqrt_eq_rpow]

184theorem uw_Sfine_lt_top :
185    (∑' z : Fin d → ℤ, ENNReal.ofReal (jbracket (cellPt d z) ^ (-((d : ℝ) + 3)))) < ⊤ := by
186  rcases Nat.eq_zero_or_pos d with hd | hd
187  · subst hd
188    haveI : Unique (Fin 0 → ℤ) := inferInstance
189    rw [tsum_fintype]
190    exact ENNReal.sum_lt_top.2 fun _ _ => ENNReal.ofReal_lt_top
191  · set k : ℝ := (⌈Real.sqrt d⌉ : ℝ) with hk
192    have hk1 : 1 ≤ k := by
193      have h1 : (1 : ℝ) ≤ Real.sqrt d := by
194        rw [show (1 : ℝ) = Real.sqrt 1 by simp]
195        exact Real.sqrt_le_sqrt (by exact_mod_cast hd)
196      have h2 : (1 : ℤ) ≤ ⌈Real.sqrt d⌉ := by
197        rw [Int.one_le_ceil_iff]; linarith
198      rw [hk]; exact_mod_cast h2
199    have hkpos : 0 < k := by linarith
200    have hp : 0 ≤ (d : ℝ) + 3 := by positivity
201    have hpt : ∀ z : Fin d → ℤ, ENNReal.ofReal (jbracket (cellPt d z) ^ (-((d : ℝ) + 3))) ≤
202        ENNReal.ofReal (k ^ ((d : ℝ) + 3)) *
203          ENNReal.ofReal (jbracket (latticePt z) ^ (-((d : ℝ) + 3))) := by
204      intro z
205      rw [← ENNReal.ofReal_mul (Real.rpow_nonneg hkpos.le _)]
206      apply ENNReal.ofReal_le_ofReal
207      have hc : cellPt d z = k⁻¹ • latticePt z := rfl
208      have hge := uw_jb_smul_ge (inv_nonneg.2 hkpos.le) (inv_le_one_of_one_le₀ hk1) (latticePt z)
209      have hjpos : 0 < jbracket (latticePt z) := Real.sqrt_pos.2 (by positivity)
210      have hcpos : 0 < k⁻¹ * jbracket (latticePt z) := mul_pos (inv_pos.2 hkpos) hjpos
211      rw [hc]
212      calc jbracket (k⁻¹ • latticePt z) ^ (-((d : ℝ) + 3))
213          ≤ (k⁻¹ * jbracket (latticePt z)) ^ (-((d : ℝ) + 3)) :=
214            Real.rpow_le_rpow_of_nonpos hcpos hge (by linarith)
215        _ = k ^ ((d : ℝ) + 3) * jbracket (latticePt z) ^ (-((d : ℝ) + 3)) := by
216            rw [Real.mul_rpow (inv_nonneg.2 hkpos.le) hjpos.le, Real.inv_rpow hkpos.le,
217              Real.rpow_neg hkpos.le, inv_inv]
218    calc (∑' z : Fin d → ℤ, ENNReal.ofReal (jbracket (cellPt d z) ^ (-((d : ℝ) + 3))))
219        ≤ ∑' z : Fin d → ℤ, ENNReal.ofReal (k ^ ((d : ℝ) + 3)) *
220            ENNReal.ofReal (jbracket (latticePt z) ^ (-((d : ℝ) + 3))) :=
221          ENNReal.tsum_le_tsum hpt
222      _ < ⊤ := by
223          rw [ENNReal.tsum_mul_left]
224          exact ENNReal.mul_lt_top ENNReal.ofReal_lt_top uw_S_lt_top
