25/-- Lemma 7.1 `M:O:lem:iid-edge` (lines 7729-7736), `C_b^6` form: for i.i.d. centered `X_i` in
26a separable real Hilbert space with finite fourth moment and `G` centered Gaussian with the same
27covariance, uniformly for `‖F‖_{C_b^6(H)} ≤ 1`,
28`|E[F(N^{-1/2} ∑ X_i)] - E[F(G)] - (6√N)⁻¹ E[D³F(G)[M₃]]| ≤ C/N`, with `C` depending only on
29the second, third and fourth moments.  Degenerate covariance allowed. -/
30theorem hilbert_edgeworth_C6 : ∃ C : ℝ → ℝ → ℝ → ℝ, ∀ {Ω : Type} [MeasurableSpace Ω]
31    (P : Measure Ω) [IsProbabilityMeasure P] (N : ℕ) (hN : 1 ≤ N)
32    (X₀ : Ω → H) (hmeas₀ : AEMeasurable X₀ P) (hcent : ∫ ω, X₀ ω ∂P = 0) (h4 : HasMoment P 4 X₀)
33    (X : Fin N → Ω → H) (hiid : IsIIDCopies P X) (hmeas : ∀ i, AEMeasurable (X i) P)
34    (hlaw : ∀ i, law P (X i) = law P X₀)
35    (G : Ω → H) (hG : IsCenteredGaussianRV P G fun x y => ProbabilityTheory.covarianceBilin (law P X₀) x y)
36    (F : H → ℝ) (hF : F ∈ CbUnitBall H 6),
37    |∫ ω, F ((Real.sqrt N)⁻¹ • ∑ i, X i ω) ∂P - ∫ ω, F (G ω) ∂P -
38        (1 / (6 * Real.sqrt N)) * thirdMomentPairing P F G X₀| ≤
39      C (momentNorm P 2 X₀) (momentNorm P 3 X₀) (momentNorm P 4 X₀) / N := by
40  refine ⟨EdgeAux.c6const, ?_⟩
41  intro Ω _ P _ N hN X₀ hmeas₀ hcent h4 X hiid hmeas hlaw G hG F hF
42  have h4' : Integrable (fun ω => ‖X₀ ω‖ ^ (4 : ℕ)) P := by
43    have h4'' : Integrable (fun ω => ‖X₀ ω‖ ^ (4 : ℝ)) P := h4
44    simpa using h4''
45  have := EdgeAux.c6_core (P := P) hN hmeas₀ hcent h4' hiid.1 hmeas hlaw hG hF
46  simpa [momentNorm] using this

49/-- Lemma 7.1, `C_b^4` form with `o(N^{-1/2})` remainder, uniform over families with a uniform
50fourth-moment bound and uniformly vanishing squared projection tails: for every such family
51(indexed by `ι`) and `ε > 0` there is `N₀` with
52`|E[F(N^{-1/2}∑X_i)] - E[F(G)] - (6√N)⁻¹ E[D³F(G)[M₃]]| ≤ ε N^{-1/2}` for `N ≥ N₀`,
53uniformly over the family and over `‖F‖_{C_b^4} ≤ 1`.  Tails are measured along a fixed
54Hilbert basis `e`.
55The copies `Xs i` are random vectors, hence a.e.-measurable; this is stated explicitly because
56`IsIIDCopies` does not include it and `law` of a non-measurable map is a junk Dirac mass
57(`docs/ISSUES.md` §E). -/
58theorem hilbert_edgeworth_C4 (e : HilbertBasis ℕ ℝ H) {ι : Type*}
59    {Ω : ι → Type} [∀ a, MeasurableSpace (Ω a)] (P : ∀ a, Measure (Ω a))
60    [∀ a, IsProbabilityMeasure (P a)] (X : ∀ a, Ω a → H) (hmeas : ∀ a, AEMeasurable (X a) (P a))
61    (hcent : ∀ a, ∫ ω, X a ω ∂(P a) = 0) (M₄ : ℝ)
62    (hint4 : ∀ a, Integrable (fun ω => ‖X a ω‖ ^ 4) (P a)) (h4 : ∀ a, ∫ ω, ‖X a ω‖ ^ 4 ∂(P a) ≤ M₄)
63    (htail : ∀ ε > 0, ∃ n₀ : ℕ, ∀ a, ∫ ω, ∑' n : {n : ℕ // n₀ ≤ n}, (inner ℝ (e n) (X a ω)) ^ 2 ∂(P a) ≤ ε)
64    (G : ∀ a, Ω a → H)
65    (hG : ∀ a, IsCenteredGaussianRV (P a) (G a) fun x y => ProbabilityTheory.covarianceBilin (law (P a) (X a)) x y) :
66    ∀ ε > 0, ∃ N₀ : ℕ, ∀ N ≥ N₀, ∀ a, ∀ (Xs : Fin N → Ω a → H), (∀ i, AEMeasurable (Xs i) (P a)) →
67      IsIIDCopies (P a) Xs → (∀ i, law (P a) (Xs i) = law (P a) (X a)) →
68      ∀ F : H → ℝ, F ∈ CbUnitBall H 4 →
69        |∫ ω, F ((Real.sqrt N)⁻¹ • ∑ i, Xs i ω) ∂(P a) - ∫ ω, F (G a ω) ∂(P a) -
70            (1 / (6 * Real.sqrt N)) * thirdMomentPairing (P a) F (G a) (X a)| ≤
71          ε / Real.sqrt N :=
72  EdgeAux.c4_main e P X hmeas hcent M₄ hint4 h4 htail G hG
