5# Lemma S.B.8(i) (`same_function_rd`) on `ℝ^d` (packet MS-SAMEFN-I)

7`same_function_rd_proof` restates `HRS.Whole.same_function_rd` (frozen in
8`HRS/Whole/AppendixBKernel.lean`, statement copied verbatim) and proves it in the style of the torus
9analogue `HRS/PDE/Toolbox/SameFunction.lean`.  Majorant of the symmetrized kernel:
10`|κ(x-y)| |a(y)-a(x)| ≤ K₀ (La 1_{|z|<1}|z|^{-s} + 2Λ 1_{|z|≥1}|z|^{-s-1})`, integrable because
11`s < d` and `s + 1 > d`; Schur's bound (`HRS.PDE.ToolboxRd.WeightAux.schur_bound`) gives absolute

14the symmetrized integral by oddness of `κ` and dominated convergence.

72/-- `t^{-r} ≤ 2^r (1+t)^{-r}` for `t ≥ 1`, `r ≥ 0`. -/
73theorem msf_rpow_tail {t r : ℝ} (ht : 1 ≤ t) (hr : 0 ≤ r) :
74    t ^ (-r) ≤ 2 ^ r * (1 + t) ^ (-r) := by
75  have ht0 : 0 < t := by linarith
76  have h1 : (2 * t) ^ (-r) ≤ (1 + t) ^ (-r) :=
77    Real.rpow_le_rpow_of_nonpos (by linarith) (by linarith) (by linarith)
78  rw [Real.mul_rpow (by norm_num : (0:ℝ) ≤ 2) ht0.le] at h1
79  have h2 : (2 : ℝ) ^ r * (2 : ℝ) ^ (-r) = 1 := by
80    rw [← Real.rpow_add (by norm_num : (0:ℝ) < 2)]; simp
81  calc t ^ (-r) = 2 ^ r * (2 ^ (-r) * t ^ (-r)) := by rw [← mul_assoc, h2, one_mul]

104/-- The truncation majorant: `|z|^{-s-1} ≤ ε⁻¹ 1_{|z|<1}|z|^{-s} + 1_{|z|≥1}|z|^{-s-1}` when `ε < |z|`. -/
105theorem msf_trunc_majorant {s ε : ℝ} (hε : 0 < ε) {z : Euc d} (hz : ε < ‖z‖) :
106    ‖z‖ ^ (-s - 1) ≤ ε⁻¹ * msf_I1 s z + msf_I2 s z := by
107  have ht : 0 < ‖z‖ := hε.trans hz
108  rcases lt_or_ge ‖z‖ 1 with h1 | h1
109  · rw [msf_I1_lt h1, msf_I2_lt h1, add_zero]
110    calc ‖z‖ ^ (-s - 1) = ‖z‖ ^ (-s) * ‖z‖⁻¹ := by
111          rw [Real.rpow_sub_one ht.ne', div_eq_mul_inv]
112      _ ≤ ‖z‖ ^ (-s) * ε⁻¹ :=
113          mul_le_mul_of_nonneg_left (inv_anti₀ hε hz.le) (Real.rpow_nonneg ht.le _)
114      _ = ε⁻¹ * ‖z‖ ^ (-s) := mul_comm _ _
115  · rw [msf_I1_ge h1, msf_I2_ge h1, mul_zero, zero_add]

131/-- Pointwise majorant of the symmetrized kernel. -/
132theorem msf_pair_bound {κ : Euc d → ℝ} {s K₀ La Λ : ℝ} (hK : 0 ≤ K₀) (hL : 0 ≤ La) (hΛ : 0 ≤ Λ)

135    (x y : Euc d) :

137  by_cases hxy : x = y
138  · have h0 : |κ (x - y)| * |a y - a x| = 0 := by
139      rw [hxy, sub_self, hκ0, abs_zero, zero_mul]
140    rw [h0]
141    exact mul_nonneg hK (add_nonneg (mul_nonneg hL (msf_I1_nonneg s _))
142      (mul_nonneg (mul_nonneg (by norm_num) hΛ) (msf_I2_nonneg s _)))
143  · have hz : x - y ≠ 0 := sub_ne_zero.2 hxy
144    have ht : 0 < ‖x - y‖ := norm_pos_iff.2 hz
145    have hΔ1 : |a y - a x| ≤ La * ‖x - y‖ := by
146      rw [abs_sub_comm]; exact ha x y
147    have hΔ2 : |a y - a x| ≤ 2 * Λ := by
148      have h1 : |a y - a x| ≤ |a y| + |a x| := by
149        rw [sub_eq_add_neg, ← abs_neg (a x)]; exact abs_add_le _ _
150      linarith [haΛ x, haΛ y]
151    rcases lt_or_ge ‖x - y‖ 1 with h1 | h1
152    · rw [msf_I1_lt h1, msf_I2_lt h1, mul_zero, add_zero]
153      have e : ‖x - y‖ ^ (-s - 1) * ‖x - y‖ = ‖x - y‖ ^ (-s) := by
154        rw [Real.rpow_sub_one ht.ne', div_mul_cancel₀ _ ht.ne']
155      calc |κ (x - y)| * |a y - a x|
156          ≤ (K₀ * ‖x - y‖ ^ (-s - 1)) * (La * ‖x - y‖) :=
157            mul_le_mul (hκ _ hz) hΔ1 (abs_nonneg _) (mul_nonneg hK (Real.rpow_nonneg (norm_nonneg _) _))
158        _ = K₀ * (La * ‖x - y‖ ^ (-s)) := by rw [← e]; ring
159    · rw [msf_I1_ge h1, msf_I2_ge h1, mul_zero, zero_add]
160      calc |κ (x - y)| * |a y - a x| ≤ (K₀ * ‖x - y‖ ^ (-s - 1)) * (2 * Λ) :=
161            mul_le_mul (hκ _ hz) hΔ2 (abs_nonneg _) (mul_nonneg hK (Real.rpow_nonneg (norm_nonneg _) _))
162        _ = K₀ * (2 * Λ * ‖x - y‖ ^ (-s - 1)) := by ring
