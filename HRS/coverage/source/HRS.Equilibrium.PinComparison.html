32/-- Two-sided comparison of a mixture. -/
33theorem mixture_le {ω a e : ι → ℝ≥0∞} {c ε : ℝ≥0∞} (hω : Measurable ω) (he : Measurable e)
34    (hω1 : ∫⁻ x, ω x ∂μ = 1) (hae : ∀ x, a x ≤ c + e x) (hea : ∀ x, c ≤ a x + e x)
35    (hε : ∫⁻ x, ω x * e x ∂μ ≤ ε) :
36    ∫⁻ x, ω x * a x ∂μ ≤ c + ε ∧ c ≤ ∫⁻ x, ω x * a x ∂μ + ε := by
37  have hc : ∫⁻ x, ω x * c ∂μ = c := by
38    rw [lintegral_mul_const _ hω, hω1, one_mul]
39  constructor
40  · calc ∫⁻ x, ω x * a x ∂μ ≤ ∫⁻ x, ω x * (c + e x) ∂μ := lintegral_mono fun x => mul_le_mul' le_rfl (hae x)
41      _ = ∫⁻ x, (ω x * c + ω x * e x) ∂μ := by simp_rw [mul_add]
42      _ = ∫⁻ x, ω x * c ∂μ + ∫⁻ x, ω x * e x ∂μ := lintegral_add_left (hω.mul_const c) _
43      _ ≤ c + ε := by rw [hc]; exact add_le_add le_rfl hε
44  · calc c = ∫⁻ x, ω x * c ∂μ := hc.symm
45      _ ≤ ∫⁻ x, ω x * (a x + e x) ∂μ := lintegral_mono fun x => mul_le_mul' le_rfl (hea x)
46      _ = ∫⁻ x, (ω x * a x + ω x * e x) ∂μ := by simp_rw [mul_add]
47      _ = ∫⁻ x, ω x * a x ∂μ + ∫⁻ x, ω x * e x ∂μ := lintegral_add_right _ (hω.mul he)
48      _ ≤ ∫⁻ x, ω x * a x ∂μ + ε := add_le_add le_rfl hε
49
50end Mixture

355theorem min_aux {t K C' C x : ℝ} (ht : t ≤ C' * x) (htK : t ≤ K) (hx : 0 ≤ x) (hCC' : C' ≤ C)
