341weight is at most `e^{K} e^{-tλ ∑_{a∈S} g(x-u_a)}`, and the normaliser of the conditional
342density of `u` given `v` is at least `e^{-K'}` by Jensen (`jensen_lintegral`,
343`KerData.jensen_block`; uses `λ n ≤ β`, `g ≥ -B`, `‖g*μ‖_∞ ≤ Cc`, `g ∈ L¹(μ)`).  Hence
344`W(u,v) ≤ C φ(u) ∫ W(u',v) du'` (`H_pole`), and Fubini over `Fin n ≃ S ⊔ Sᶜ`
345(`marginal_le_of_cond`, `pole_restr_le`) gives the marginal bounds.

351finite positive block (`A:eq:calibration`) and the bounded-density transfer

484def poleExpo (g : Torus d → ℝ) (μ : Measure (Torus d)) {n : ℕ} (lam β t : ℝ) (x : Torus d)
485    (Y : Config d n) : ℝ :=
486  (-lam * pairSum g Y + β * ∑ i, convMeasure g μ (Y i)) + (-(t * lam) * ∑ i, g (x - Y i))

488/-- single-site part of the exponent -/
489def site (g : Torus d → ℝ) (μ : Measure (Torus d)) (β t lam : ℝ) (x y : Torus d) : ℝ :=
490  β * convMeasure g μ y - (t * lam) * g (x - y)
491
492/-- the exponent of a block of coordinates `κ` (without the interaction with the rest) -/
493def blockExpo (g : Torus d → ℝ) (μ : Measure (Torus d)) {κ : Type*} [Fintype κ]
494    (lam β t : ℝ) (x : Torus d) (w : κ → Torus d) : ℝ :=
495  ∑ a, site g μ β t lam x (w a) -
496    (lam / 2) * (∑ a, ∑ b, g (w a - w b) - (Fintype.card κ : ℝ) * g 0)

498/-- cross interaction between two blocks -/
499def crossSum (g : Torus d → ℝ) {κ κ' : Type*} [Fintype κ] [Fintype κ']
500    (u : κ → Torus d) (v : κ' → Torus d) : ℝ := ∑ a, ∑ b, g (u a - v b)

508    intro i

510    simp

512  rw [Finset.sum_sub_distrib] at h

547        ∑ a : {a // p a}, ∑ b : {b // ¬ p b}, g (Y a - Y b) := by
