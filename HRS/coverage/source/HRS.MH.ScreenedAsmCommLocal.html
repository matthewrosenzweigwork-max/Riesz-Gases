28lemma acl_cont {La : ℝ} {ϕ : Euc d → ℝ} (hLa : 0 ≤ La) (hL : HRS.PDE.ToolboxRd.IsLipschitz La ϕ) :
29    Continuous ϕ := by
30  refine (LipschitzWith.of_dist_le_mul (K := La.toNNReal) (fun x y => ?_)).continuous
31  simpa [Real.dist_eq, dist_eq_norm, Real.coe_toNNReal _ hLa] using hL x y

33lemma acl_ind_mul_le {ϕ w : Euc d → ℝ} (hϕ : ∀ x, |ϕ x| ≤ 1) (hc : Continuous ϕ)
34    (hw : LocallyIntegrable w volume) {B : Set (Euc d)} (hB : MeasurableSet B) :
35    eLpNorm (B.indicator (fun x => ϕ x * w x)) 2 volume ≤ eLpNorm (B.indicator w) 2 volume := by
36  refine eLpNorm_mono ((hc.aestronglyMeasurable.mul hw.aestronglyMeasurable).indicator hB) fun x => ?_
37  by_cases hx : x ∈ B
38  · simp only [Set.indicator_of_mem hx, norm_mul, Real.norm_eq_abs]
39    calc |ϕ x| * |w x| ≤ 1 * |w x| := mul_le_mul_of_nonneg_right (hϕ x) (abs_nonneg _)
40      _ = |w x| := one_mul _
41  · simp [Set.indicator_of_notMem hx]

43lemma acl_alg {p y₁ : ℝ} (hp : 0 ≤ p) (_hy₁ : 0 ≤ y₁) (N : ℝ≥0∞) :
44    ENNReal.ofReal p * (ENNReal.ofReal y₁ * N) = ENNReal.ofReal (p * y₁) * N := by
45  rw [ENNReal.ofReal_mul hp, mul_assoc]

47lemma acl_alg2 {p y : ℝ} (hp : 0 ≤ p) (hy : 0 ≤ y) (N : ℝ≥0∞) :
48    ENNReal.ofReal p * N + ENNReal.ofReal p * (ENNReal.ofReal y * N) =
49      ENNReal.ofReal (p * (1 + y)) * N := by
50  rw [ENNReal.ofReal_mul hp, ENNReal.ofReal_add (by norm_num) hy, ENNReal.ofReal_one]

53theorem screened_comm_local_proof : type_of% @HRS.Whole.screened_comm_local := by
54  intro d hd s hs0 hsd m Cb T χ χh hχ
55  obtain ⟨Cl0, Cl, hloc⟩ := screened_local_proof hd hs0 hsd
56  obtain ⟨CF, hCF0, hF⟩ := flow_lip_sep_proof hd m Cb T χ χh hχ
57  obtain ⟨Lχ, hLχ0, hLχ1, hLχ2⟩ := cutoff_lipschitz_proof χ χh hχ
58  have hkg0 : 0 ≤ kgOrder d s := le_max_right _ _
59  have hkg1 : kgOrder d s < 1 := max_lt (by linarith) one_pos
60  obtain ⟨Cm, hCm0, hCm⟩ := mult_sobolev_proof hd hkg0 hkg1
61  have hd1 : (1 : ℝ) ≤ d := by exact_mod_cast hd
62  set X : ℝ := 1 + Cm * (2 * Lχ + 1) with hX
63  have hX1 : 1 ≤ X := by rw [hX]; nlinarith [mul_nonneg hCm0 (by linarith : (0:ℝ) ≤ 2 * Lχ + 1)]
64  set C : ℝ := max (max Cl 0 * X) CF with hC
65  have hCle1 : max Cl 0 * X ≤ C := le_max_left _ _
66  have hCle2 : CF ≤ C := le_max_right _ _
67  have hC0 : 0 ≤ C := le_trans hCF0 hCle2
68  refine ⟨Cl0, C, ?_⟩
69  intro b Ψ hb hΨ τ hτ z i ρ hρ c v hsupp hv
70  obtain ⟨-, hL2, -⟩ := hF b Ψ hb hΨ τ hτ
71  set L : ℝ := Real.exp (CF * τ) with hLdef
72  have hL1 : 1 ≤ L := Real.one_le_exp (mul_nonneg hCF0 hτ.1)
73  obtain ⟨hf0, hf1, -, -, -, -, hlipχ, hlipχh⟩ :=
74    advCutoff_facts_proof χ χh hχ Ψ τ L Lχ (by linarith) hL2 ⟨hLχ1, hLχ2⟩ z
75  have hLexp : L ≤ Real.exp (C * d * τ) := by
76    refine Real.exp_le_exp.2 ?_
77    have : CF ≤ C * d := le_trans hCle2 (le_mul_of_one_le_right hC0 hd1)
78    exact mul_le_mul_of_nonneg_right this hτ.1
79  -- the multipliers
80  set ϕ3 : Euc d → ℝ := advCutoff χ Ψ τ z with hϕ3
81  set ϕ1 : Euc d → ℝ := fun y => advCutoff χh Ψ τ z y - advCutoff χ Ψ τ z y with hϕ1
82  have hLχL : 0 ≤ Lχ * L := mul_nonneg hLχ0 (by linarith)
83  have hϕ3L : HRS.PDE.ToolboxRd.IsLipschitz (Lχ * L) ϕ3 := hlipχ
84  have hϕ3B : ∀ x, |ϕ3 x| ≤ 1 := fun x => by
85    have := hf0 x
86    rw [abs_le]; constructor <;> linarith [this.1, this.2]
87  have hϕ1L : HRS.PDE.ToolboxRd.IsLipschitz (2 * (Lχ * L)) ϕ1 := fun x y => by
88    have h1 := hlipχh x y
89    have h2 := hlipχ x y
90    have : ϕ1 x - ϕ1 y = (advCutoff χh Ψ τ z x - advCutoff χh Ψ τ z y) -
91        (advCutoff χ Ψ τ z x - advCutoff χ Ψ τ z y) := by simp only [hϕ1]; ring
92    rw [this]
93    calc _ ≤ |advCutoff χh Ψ τ z x - advCutoff χh Ψ τ z y| +
94          |advCutoff χ Ψ τ z x - advCutoff χ Ψ τ z y| := abs_sub _ _
95      _ ≤ 2 * (Lχ * L) * ‖x - y‖ := by nlinarith [norm_nonneg (x - y)]
96  have hϕ1B : ∀ x, |ϕ1 x| ≤ 1 := fun x => by
97    have a1 := hf0 x
98    have a2 := hf1 x
99    simp only [hϕ1]
100    rw [abs_le]; constructor <;> linarith [a1.1, a1.2, a2.1, a2.2]
101  have hc3 : Continuous ϕ3 := acl_cont hLχL hϕ3L
102  set N : ℝ≥0∞ := sobolevNormE (kgOrder d s) v with hNdef
103  have hN1 : sobolevNormE (kgOrder d s) (fun x => ϕ1 x * v x) ≤
104      ENNReal.ofReal (Cm * (2 * (Lχ * L) + 1)) * N := hCm _ _ ϕ1 v hϕ1L hϕ1B
105  have hN3 : sobolevNormE (kgOrder d s) (fun x => ϕ3 x * v x) ≤
106      ENNReal.ofReal (Cm * (Lχ * L + 1)) * N := hCm _ _ ϕ3 v hϕ3L hϕ3B
107  have hv1 : sobolevNormE (kgOrder d s) (fun x => ϕ1 x * v x) < ⊤ :=
108    lt_of_le_of_lt hN1 (ENNReal.mul_lt_top ENNReal.ofReal_lt_top hv)
109  have hv3 : sobolevNormE (kgOrder d s) (fun x => ϕ3 x * v x) < ⊤ :=
110    lt_of_le_of_lt hN3 (ENNReal.mul_lt_top ENNReal.ofReal_lt_top hv)
111  have hs1 : Function.support (fun x => ϕ1 x * v x) ⊆ Metric.closedBall c ρ :=
112    (Function.support_mul_subset_right _ _).trans hsupp
113  have hs3 : Function.support (fun x => ϕ3 x * v x) ⊆ Metric.closedBall c ρ :=
114    (Function.support_mul_subset_right _ _).trans hsupp
115  obtain ⟨w₁, hr1, hb1⟩ := hloc i ρ hρ c _ hs1 hv1
116  obtain ⟨w₂, hr2, hb2⟩ := hloc i ρ hρ c v hsupp hv
117  obtain ⟨w₃, hr3, hb3⟩ := hloc i ρ hρ c _ hs3 hv3
118  refine ⟨w₁, w₂, w₃, hr1, hr2, hr3, fun c' r' hr' => ?_⟩
119  have hρ0 : 0 ≤ ρ ^ Cl0 := (Real.rpow_pos_of_pos (by linarith) _).le
120  set p : ℝ := max Cl 0 * ρ ^ Cl0 with hp
121  have hp0 : 0 ≤ p := mul_nonneg (le_max_right _ _) hρ0
122  have hpl : ENNReal.ofReal (Cl * ρ ^ Cl0) ≤ ENNReal.ofReal p :=
123    ENNReal.ofReal_le_ofReal (mul_le_mul_of_nonneg_right (le_max_left _ _) hρ0)
124  have key : ∀ y : ℝ, 0 ≤ y → y ≤ X * L → p * y ≤ C * Real.exp (C * d * τ) * ρ ^ Cl0 := by
125    intro y hy0 hy
126    calc p * y ≤ p * (X * L) := mul_le_mul_of_nonneg_left hy hp0
127      _ = (max Cl 0 * X) * L * ρ ^ Cl0 := by rw [hp]; ring
128      _ ≤ C * Real.exp (C * d * τ) * ρ ^ Cl0 := by
129          gcongr
130  have hy1 : Cm * (2 * (Lχ * L) + 1) ≤ X * L := by
131    rw [hX]; nlinarith [mul_nonneg hCm0 hLχ0, mul_nonneg (mul_nonneg hCm0 hLχ0) (sub_nonneg.2 hL1),
132      mul_nonneg hCm0 (sub_nonneg.2 hL1)]
133  have hy3 : 1 + Cm * (Lχ * L + 1) ≤ X * L := by
134    rw [hX]; nlinarith [mul_nonneg hCm0 hLχ0, mul_nonneg (mul_nonneg hCm0 hLχ0) (sub_nonneg.2 hL1),
135      mul_nonneg hCm0 (sub_nonneg.2 hL1)]
136  have hy1' : 0 ≤ Cm * (2 * (Lχ * L) + 1) := by positivity
137  have hy3' : 0 ≤ Cm * (Lχ * L + 1) := by positivity
138  constructor
139  · calc eLpNorm ((Metric.closedBall c' r').indicator (fun x => advCutoff χ Ψ τ z x * w₁ x)) 2 volume
140        ≤ eLpNorm ((Metric.closedBall c' r').indicator w₁) 2 volume := acl_ind_mul_le hϕ3B hc3 (k5a_locInt i hr1) measurableSet_closedBall
141      _ ≤ ENNReal.ofReal (Cl * ρ ^ Cl0) * sobolevNormE (kgOrder d s) (fun x => ϕ1 x * v x) :=
142          hb1 c' r' hr'
143      _ ≤ ENNReal.ofReal p * (ENNReal.ofReal (Cm * (2 * (Lχ * L) + 1)) * N) :=
144          mul_le_mul' hpl hN1
145      _ = ENNReal.ofReal (p * (Cm * (2 * (Lχ * L) + 1))) * N := acl_alg hp0 hy1' N
146      _ ≤ ENNReal.ofReal (C * Real.exp (C * d * τ) * ρ ^ Cl0) * N := by
147          refine mul_le_mul' (ENNReal.ofReal_le_ofReal (key _ hy1' hy1)) le_rfl
148  · have hsplit : (Metric.closedBall c' r').indicator (fun x => advCutoff χ Ψ τ z x * w₂ x - w₃ x) =
149        fun x => (Metric.closedBall c' r').indicator (fun x => advCutoff χ Ψ τ z x * w₂ x) x -
150          (Metric.closedBall c' r').indicator w₃ x := by
151      funext x
152      by_cases hx : x ∈ Metric.closedBall c' r' <;> simp [Set.indicator, hx]
153    rw [hsplit]
154    calc eLpNorm (fun x => (Metric.closedBall c' r').indicator (fun x => advCutoff χ Ψ τ z x * w₂ x) x -
155            (Metric.closedBall c' r').indicator w₃ x) 2 volume
156        ≤ eLpNorm ((Metric.closedBall c' r').indicator (fun x => advCutoff χ Ψ τ z x * w₂ x)) 2 volume +
157          eLpNorm ((Metric.closedBall c' r').indicator w₃) 2 volume :=
158          eLpNorm_sub_le (by norm_num)
159      _ ≤ ENNReal.ofReal (Cl * ρ ^ Cl0) * N +
160          ENNReal.ofReal (Cl * ρ ^ Cl0) * sobolevNormE (kgOrder d s) (fun x => ϕ3 x * v x) :=
161          add_le_add ((acl_ind_mul_le hϕ3B hc3 (k5a_locInt i hr2) measurableSet_closedBall).trans (hb2 c' r' hr')) (hb3 c' r' hr')
162      _ ≤ ENNReal.ofReal p * N + ENNReal.ofReal p * (ENNReal.ofReal (Cm * (Lχ * L + 1)) * N) :=
163          add_le_add (mul_le_mul' hpl le_rfl) (mul_le_mul' hpl hN3)
164      _ = ENNReal.ofReal (p * (1 + Cm * (Lχ * L + 1))) * N := acl_alg2 hp0 hy3' N
165      _ ≤ ENNReal.ofReal (C * Real.exp (C * d * τ) * ρ ^ Cl0) * N := by
166          refine mul_le_mul' (ENNReal.ofReal_le_ofReal (key _ (by positivity) hy3)) le_rfl
