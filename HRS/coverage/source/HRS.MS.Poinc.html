5/-!
6# C-POI: the Gaussian Poincare inequality on `Euc d` and `euc_thermal_gaussian_bounds`
7
8Maurey-Pisier interpolation (`PoincAux1..3`) gives, for `γ = γ_{βa} dx`,
9`Var_γ f ≤ (π²/8) (βa)⁻¹ ∫ |∇f|² dγ` for `C¹` functions `f ∈ L²(γ)` with `∇f ∈ L²(γ)`
10(`cpoi_gauss_poincare`), i.e. `CbGaussPoincare d a β (π²/8)` of `ThmCons`.
11Combined with the merged transfer `cb_euc_thermal_gaussian_bounds_of_poincare` this proves
12`HRS.Whole.euc_thermal_gaussian_bounds` (`euc_thermal_gaussian_bounds_proof`).
13The constant in the Poincare inequality of the frozen statement is `C (1 + π²/8)/(c β a)`
14for the comparison constants of `cb_thermal_two_sided` (the frozen `C` is existential, so enlarging it
15is harmless).
16-/

27/-- **Gaussian Poincare inequality** (Maurey-Pisier constant `π²/8`). -/
28theorem cpoi_gauss_poincare {a β : ℝ} (ha : 0 < a) (hβ : 0 < β) {f : Euc d → ℝ}
29    (hf : ContDiff ℝ 1 f) (hL2 : MemLp f 2 (cpoi_gam d a β))
30    (hg : Integrable (fun x => ‖gradient f x‖ ^ 2) (cpoi_gam d a β)) :
31    variance f (cpoi_gam d a β) ≤
32      (Real.pi ^ 2 / 8) / (β * a) * ∫ x, ‖gradient f x‖ ^ 2 ∂(cpoi_gam d a β) := by
33  have hP := cpoi_gam_isProb (d := d) ha hβ
34  set γ := cpoi_gam d a β with hγ
35  set K := (β * a)⁻¹ * ∫ x, ‖gradient f x‖ ^ 2 ∂γ with hK
36  have hK0 : 0 ≤ K := by
37    rw [hK]; exact mul_nonneg (by positivity) (integral_nonneg fun x => by positivity)
38  have hfm : AEMeasurable f γ := hf.continuous.measurable.aemeasurable
39  set m := ∫ x, f x ∂γ with hm
40  set g : Euc d → ℝ := fun x => f x - m with hgdef
41  have hgL2 : MemLp g 2 γ := hL2.sub (memLp_const m)
42  have hg2 : Integrable (fun x => g x ^ 2) γ := hgL2.integrable_sq
43  have hg1 : Integrable g γ := hgL2.integrable one_le_two
44  have hg0 : ∫ x, g x ∂γ = 0 := by
45    simp only [hgdef]
46    rw [integral_sub (hL2.integrable one_le_two) (integrable_const m)]
47    simp [hm]
48  have hvar : variance f γ = ∫ x, g x ^ 2 ∂γ := by
49    rw [variance_eq_integral hfm]
50  -- the double integral
51  have hE : ∫ q, (f q.2 - f q.1) ^ 2 ∂(γ.prod γ) = 2 * ∫ x, g x ^ 2 ∂γ := by
52    have e : ∀ q : Euc d × Euc d, (f q.2 - f q.1) ^ 2 =
53        1 * g q.2 ^ 2 + (g q.1 ^ 2 * 1 - 2 * (g q.1 * g q.2)) := by
54      intro q; simp only [hgdef]; ring
55    simp_rw [e]
56    have i1 : Integrable (fun q : Euc d × Euc d => 1 * g q.2 ^ 2) (γ.prod γ) :=
57      Integrable.mul_prod (integrable_const (1 : ℝ)) hg2
58    have i2 : Integrable (fun q : Euc d × Euc d => g q.1 ^ 2 * 1) (γ.prod γ) :=
59      Integrable.mul_prod hg2 (integrable_const (1 : ℝ))
60    have i3 : Integrable (fun q : Euc d × Euc d => g q.1 * g q.2) (γ.prod γ) :=
61      Integrable.mul_prod hg1 hg1
62    have i4 : Integrable (fun q : Euc d × Euc d => 2 * (g q.1 * g q.2)) (γ.prod γ) := i3.const_mul 2
63    have i23 : Integrable (fun q : Euc d × Euc d => g q.1 ^ 2 * 1 - 2 * (g q.1 * g q.2)) (γ.prod γ) :=
64      i2.sub i4
65    have j1 : ∫ q : Euc d × Euc d, 1 * g q.2 ^ 2 ∂(γ.prod γ) = ∫ x, g x ^ 2 ∂γ := by
66      have := integral_prod_mul (μ := γ) (ν := γ) (fun _ => (1 : ℝ)) (fun x => g x ^ 2)
67      simpa using this
68    have j2 : ∫ q : Euc d × Euc d, g q.1 ^ 2 * 1 ∂(γ.prod γ) = ∫ x, g x ^ 2 ∂γ := by
69      have := integral_prod_mul (μ := γ) (ν := γ) (fun x => g x ^ 2) (fun _ => (1 : ℝ))
70      simpa using this
71    have j3 : ∫ q : Euc d × Euc d, g q.1 * g q.2 ∂(γ.prod γ) = 0 := by
72      have := integral_prod_mul (μ := γ) (ν := γ) g g
73      rw [this, hg0]; simp
74    rw [integral_add i1 i23, integral_sub i2 i4,
75      integral_const_mul (2 : ℝ) (fun q : Euc d × Euc d => g q.1 * g q.2), j1, j2, j3]
76    ring
77  -- the bound
78  have hbd := cpoi_lintegral_diff_le ha hβ hf hg
79  have hnn : (0 : ℝ) ≤ (Real.pi / 2) ^ 2 * K := by positivity
80  have hmeas : Measurable fun q : Euc d × Euc d => (f q.2 - f q.1) ^ 2 := by
81    have : Continuous fun q : Euc d × Euc d => (f q.2 - f q.1) ^ 2 := by
82      have := hf.continuous
83      fun_prop
84    exact this.measurable
85  have hle : ∫ q, (f q.2 - f q.1) ^ 2 ∂(γ.prod γ) ≤ (Real.pi / 2) ^ 2 * K := by
86    rw [integral_eq_lintegral_of_nonneg_ae (ae_of_all _ fun q => sq_nonneg _)
87      hmeas.aestronglyMeasurable]
88    have := ENNReal.toReal_mono ENNReal.ofReal_ne_top hbd
89    rwa [ENNReal.toReal_ofReal hnn] at this
90  rw [hE] at hle
91  have hv : variance f γ = ∫ x, g x ^ 2 ∂γ := hvar
92  rw [hK] at hle
93  rw [hv]
94  have : (Real.pi ^ 2 / 8) / (β * a) * ∫ x, ‖gradient f x‖ ^ 2 ∂γ =
95      (Real.pi / 2) ^ 2 * K / 2 := by
96    rw [hK]; field_simp; ring
97  rw [this]
98  linarith

100/-- The Gaussian Poincare inequality in the form `CbGaussPoincare` of `ThmCons`, with `κ = π²/8`. -/
101theorem cpoi_gaussPoincare {a β : ℝ} (ha : 0 < a) (hβ : 0 < β) :
102    HRS.MS.Thm.CbGaussPoincare d a β (Real.pi ^ 2 / 8) :=
103  fun _ hf hL2 hg => cpoi_gauss_poincare ha hβ hf hL2 hg
