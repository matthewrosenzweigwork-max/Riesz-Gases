6`energy_identity_of_feature_proof`: for a continuous `Ψ : 𝕋^d → H` into a real Hilbert space with
7`⟨Ψ(x),Ψ(y)⟩ = g(x-y)`, the modulated energy is `½ ‖N^{-1/2} ∑ (Ψ(x_i) - μ(Ψ))‖² - ½ g(0)` up to the
8factor `N`.  The proof is Hilbert-space algebra: expand the squared norm, use `⟨Ψ(x), μ(Ψ)⟩ = ∫ g(x-y) dμ`
9(`integral_inner`), then compare the double sums with the modulated energy.
10-/

27theorem fm3_norm_eq (g : Torus d → ℝ) (Ψ : Torus d → H)
28    (hg : ∀ x y : Torus d, inner ℝ (Ψ x) (Ψ y) = g (x - y)) (x : Torus d) :
29    ‖Ψ x‖ = Real.sqrt (g 0) := by
30  calc ‖Ψ x‖ = Real.sqrt (‖Ψ x‖ ^ 2) := (Real.sqrt_sq (norm_nonneg _)).symm
31    _ = Real.sqrt (g 0) := by rw [← real_inner_self_eq_norm_sq, hg x x, sub_self]

33theorem fm3_integrable (g : Torus d → ℝ) (Ψ : Torus d → H) (hΨ : Continuous Ψ)
34    (hg : ∀ x y : Torus d, inner ℝ (Ψ x) (Ψ y) = g (x - y)) (μ : Measure (Torus d))

39theorem fm3_inner_mean (g : Torus d → ℝ) (Ψ : Torus d → H) (hΨ : Continuous Ψ)
40    (hg : ∀ x y : Torus d, inner ℝ (Ψ x) (Ψ y) = g (x - y)) (μ : Measure (Torus d))
41    [IsProbabilityMeasure μ] (x : Torus d) :
42    inner ℝ (Ψ x) (∫ y, Ψ y ∂μ) = ∫ y, g (x - y) ∂μ := by

60theorem fm3_inner_centered (g : Torus d → ℝ) (Ψ : Torus d → H) (hΨ : Continuous Ψ)
61    (hg : ∀ x y : Torus d, inner ℝ (Ψ x) (Ψ y) = g (x - y)) (μ : Measure (Torus d))
62    [IsProbabilityMeasure μ] (x y : Torus d) :
63    inner ℝ (Ψ x - ∫ z, Ψ z ∂μ) (Ψ y - ∫ z, Ψ z ∂μ) =
64      g (x - y) - (∫ z, g (x - z) ∂μ) - (∫ z, g (y - z) ∂μ) + ∫ w, ∫ z, g (w - z) ∂μ ∂μ := by

66    fm3_inner_mean' g Ψ hΨ hg μ y, fm3_inner_mm g Ψ hΨ hg μ]

69theorem fm3_norm_sum_sq (g : Torus d → ℝ) (Ψ : Torus d → H) (hΨ : Continuous Ψ)
70    (hg : ∀ x y : Torus d, inner ℝ (Ψ x) (Ψ y) = g (x - y)) (μ : Measure (Torus d))
71    [IsProbabilityMeasure μ] {N : ℕ} (X : Config d N) :
72    ‖∑ i, (Ψ (X i) - ∫ z, Ψ z ∂μ)‖ ^ 2 =
73      ∑ i, ∑ j, (g (X i - X j) - (∫ z, g (X i - z) ∂μ) - (∫ z, g (X j - z) ∂μ)
74        + ∫ w, ∫ z, g (w - z) ∂μ ∂μ) := by
75  rw [← real_inner_self_eq_norm_sq, sum_inner]
76  refine Finset.sum_congr rfl fun i _ => ?_
77  rw [inner_sum]
78  refine Finset.sum_congr rfl fun j _ => ?_
79  exact fm3_inner_centered g Ψ hΨ hg μ (X i) (X j)

88theorem fm3_core (N : ℕ) (hN : 1 ≤ N) (k : Fin N → Fin N → ℝ) (G : Fin N → ℝ) (Γ g0 : ℝ)
89    (hk : ∀ i, k i i = g0) :
90    (N : ℝ) * ((1 / (2 * (N : ℝ) ^ 2)) * ∑ i, ∑ j ∈ Finset.univ.erase i, k i j
91        - (1 / (N : ℝ)) * ∑ i, G i + (1 / 2) * Γ) =
92      (1 / 2) * ((N : ℝ)⁻¹ * ∑ i, ∑ j, (k i j - G i - G j + Γ)) - (1 / 2) * g0 := by
93  have hN0 : (N : ℝ) ≠ 0 := Nat.cast_ne_zero.mpr (by omega)
94  have e1 : ∑ i, ∑ j, k i j = ∑ i, ∑ j ∈ Finset.univ.erase i, k i j + N * g0 := by
95    have h1 : ∀ i, ∑ j, k i j = g0 + ∑ j ∈ Finset.univ.erase i, k i j := fun i => by
96      rw [← hk i, Finset.add_sum_erase _ _ (Finset.mem_univ i)]
97    simp_rw [h1, Finset.sum_add_distrib, Finset.sum_const, Finset.card_univ, Fintype.card_fin,
98      nsmul_eq_mul]
99    ring
100  have e2 : ∑ i, ∑ j, (k i j - G i - G j + Γ) =
101      ∑ i, ∑ j, k i j - N * ∑ i, G i - N * ∑ j, G j + (N : ℝ) ^ 2 * Γ := by
102    simp only [Finset.sum_add_distrib, Finset.sum_sub_distrib, Finset.sum_const, Finset.card_univ,
103      Fintype.card_fin, nsmul_eq_mul, ← Finset.mul_sum]
104    ring
105  rw [e2, e1]
106  field_simp
107  ring

109theorem energy_identity_of_feature_proof : type_of% @HRS.energy_identity_of_feature := by
110  intro d H _ _ _ g Ψ hΨ hg μ _ N hN X
111  have hB : ‖(Real.sqrt N)⁻¹ • ∑ i, (Ψ (X i) - ∫ y, Ψ y ∂μ)‖ ^ 2 =
112      (N : ℝ)⁻¹ * ∑ i, ∑ j, (g (X i - X j) - (∫ z, g (X i - z) ∂μ) - (∫ z, g (X j - z) ∂μ)
113        + ∫ w, ∫ z, g (w - z) ∂μ ∂μ) := by
114    rw [fm3_scale, fm3_norm_sum_sq g Ψ hΨ hg μ X]
115  rw [hB]
116  exact fm3_core N hN (fun i j => g (X i - X j)) (fun i => ∫ z, g (X i - z) ∂μ)
117    (∫ w, ∫ z, g (w - z) ∂μ ∂μ) (g 0) (fun i => by simp)
