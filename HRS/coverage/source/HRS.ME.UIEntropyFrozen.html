4/-!
5# Proposition 5.10, last sentence: the entropy bound implies uniform integrability (frozen statement)
6
7Proof of `HRS.ui_of_entropy_bound` exactly as frozen, with no hypotheses on `d`, `s`, `V`, `β`.  For arbitrary `V` and `β`,
8`canonicalGibbs` is `c • μ` with `μ` the measure with density `exp (-β H)` and `c = (ofReal Z)⁻¹`, `Z` the Bochner integral
9of the density.  The density is positive, so `Z ≥ 0`, and there are two cases.
10
11* `Z > 0`: the density is integrable, `μ univ = ofReal Z`, and the Gibbs law is a probability measure.  Then
12  `HRS.ME.UIEntropy.ui_tail_of_relEntropy` applies.
13* `Z = 0`: `c = ⊤`.  From the entropy bound `ν ≪ Q`.  If `ν` had a Lebesgue decomposition with respect to `Q`, then
14  `ν = Q.withDensity (ν.rnDeriv Q)` would have total mass `⊤ * ∫⁻ rnDeriv dμ ∈ {0, ⊤}`, not `1`.  So there is no
15  Lebesgue decomposition, `rnDeriv = 0` by the junk-value convention, and the tail set is empty.
16-/

25/-- If `Q = ⊤ • μ` and `ν` is a probability measure with `ν ≪ Q`, then `ν.rnDeriv Q = 0`. -/
26lemma rnDeriv_top_smul {α : Type*} [MeasurableSpace α] (ν μ : Measure α) [IsProbabilityMeasure ν]
27    (hac : ν ≪ (⊤ : ℝ≥0∞) • μ) : ν.rnDeriv ((⊤ : ℝ≥0∞) • μ) = 0 := by
28  by_cases h : ν.HaveLebesgueDecomposition ((⊤ : ℝ≥0∞) • μ)
29  · exfalso
30    have e := Measure.withDensity_rnDeriv_eq ν ((⊤ : ℝ≥0∞) • μ) hac
31    have h1 := congrArg (fun m : Measure α => m Set.univ) e
32    simp only [measure_univ] at h1
33    rw [withDensity_apply _ MeasurableSet.univ, Measure.restrict_univ, lintegral_smul_measure] at h1
34    simp only [smul_eq_mul] at h1
35    by_cases h0 : ∫⁻ x, ν.rnDeriv ((⊤ : ℝ≥0∞) • μ) x ∂μ = 0
36    · rw [h0, mul_zero] at h1
37      exact zero_ne_one h1
38    · rw [ENNReal.top_mul h0] at h1
39      exact ENNReal.top_ne_one h1
40  · exact Measure.rnDeriv_of_not_haveLebesgueDecomposition h

42theorem ui_of_entropy_bound_proof {d : ℕ} : type_of% (@HRS.ui_of_entropy_bound d) := by
43  intro s β V ν hν H₀ hH
44  have hb : ∀ L : ℝ, 1 < L → uiTail d s β V ν L ≤ ENNReal.ofReal ((H₀ + Real.exp (-1)) / Real.log L) := by
45    intro L hL
46    unfold uiTail
47    refine iSup_le fun n => ?_
48    have := hν n

51    set Z := canonicalPartition (periodicRiesz d s) V N β with hZ
52    have hZ0 : 0 ≤ Z := integral_nonneg fun X => (Real.exp_pos _).le
53    rcases hZ0.eq_or_lt with hZ0 | hZpos
54    · -- junk case `Z = 0`
55      have hQ : canonicalGibbs (periodicRiesz d s) V N β =
56          (⊤ : ℝ≥0∞) • (volume : Measure (Config d N)).withDensity fun X => ENNReal.ofReal (f X) := by
57        unfold canonicalGibbs
58        rw [← hZ, ← hZ0, ENNReal.ofReal_zero, ENNReal.inv_zero]
59      have hkl := hH n
60      have hac : ν n ≪ canonicalGibbs (periodicRiesz d s) V N β := by
61        have : HRS.relEntropy (ν n) (canonicalGibbs (periodicRiesz d s) V N β) ≠ ⊤ :=
62          ne_top_of_le_ne_top ENNReal.ofReal_ne_top hkl
63        exact (InformationTheory.klDiv_ne_top_iff.mp this).1
64      rw [hQ] at hac ⊢
65      rw [rnDeriv_top_smul _ _ hac]
66      have : {X : Config d N | ENNReal.ofReal L < (0 : Config d N → ℝ≥0∞) X} = ∅ := by
67        ext X; simp
68      simp
69    · -- `Z > 0`: the Gibbs law is a probability measure
70      have hP : IsProbabilityMeasure (canonicalGibbs (periodicRiesz d s) V N β) := by
71        have hint : Integrable f := by
72          by_contra hni
73          have : Z = 0 := integral_undef hni
74          exact hZpos.ne' this
75        have hl := ofReal_integral_eq_lintegral_ofReal hint
76          (Filter.Eventually.of_forall fun X => (Real.exp_pos _).le)
77        refine ⟨?_⟩
78        unfold canonicalGibbs
79        rw [Measure.smul_apply, withDensity_apply _ MeasurableSet.univ, Measure.restrict_univ,
80          ← hl, smul_eq_mul]
81        exact ENNReal.inv_mul_cancel (by simpa using hZpos) ENNReal.ofReal_ne_top
82      exact (HRS.ME.UIEntropy.ui_tail_of_relEntropy (ν n) _ H₀ L hL (hH n)).2
83  refine ⟨hb, ?_⟩
84  have hlim : Tendsto (fun L : ℝ => ENNReal.ofReal ((H₀ + Real.exp (-1)) / Real.log L)) atTop (𝓝 0) := by
85    have h1 : Tendsto (fun L : ℝ => (H₀ + Real.exp (-1)) / Real.log L) atTop (𝓝 0) :=
86      tendsto_const_nhds.div_atTop Real.tendsto_log_atTop
87    have h2 := (ENNReal.continuous_ofReal.tendsto 0).comp h1
88    rw [ENNReal.ofReal_zero] at h2
89    exact h2
90  refine tendsto_of_tendsto_of_tendsto_of_le_of_le' tendsto_const_nhds hlim (Eventually.of_forall fun L => bot_le) ?_
91  filter_upwards [eventually_gt_atTop (1 : ℝ)] with L hL using hb L hL
