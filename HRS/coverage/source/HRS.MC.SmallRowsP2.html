25/-- `jp k ≤ 2π √(1 + |k|²)` (uses `4π² ≥ 1`). -/
26lemma p_sm2_jp_le_two_pi_sqrt {d : ℕ} (k : Lattice d) :
27    Lattice.jp k ≤ 2 * Real.pi * Real.sqrt (1 + Lattice.enorm k ^ 2) := by
28  unfold Lattice.jp
29  rw [Real.sqrt_le_iff]
30  have hsq : Real.sqrt (1 + Lattice.enorm k ^ 2) ^ 2 = 1 + Lattice.enorm k ^ 2 :=
31    Real.sq_sqrt (by positivity)
32  have key : (2 * Real.pi * Real.sqrt (1 + Lattice.enorm k ^ 2)) ^ 2 =
33      4 * Real.pi ^ 2 * (1 + Lattice.enorm k ^ 2) := by
34    rw [mul_pow, mul_pow, hsq]; ring
35  refine ⟨by positivity, ?_⟩
36  rw [key]
37  nlinarith [Real.pi_gt_three, sq_nonneg (Lattice.enorm k), sq_nonneg Real.pi]

50/-- `jp k ≤ 2π (1 + |k|)`. -/
51lemma p_sm2_jp_le_linear {d : ℕ} (k : Lattice d) :
52    Lattice.jp k ≤ 2 * Real.pi * (1 + Lattice.enorm k) := by
53  have he : 0 ≤ Lattice.enorm k := norm_nonneg _

57/-- NEW STATEMENT (clause P6/U2, TeX M L10061-10062, "`(1+4π²|k|²)^{m/2} ≤ (2π⟨k⟩)^m` for `m ≥ 0`",
58with `⟨k⟩ = (1+|k|²)^{1/2}`): numerical inequality for `jp k = (1+4π²|k|²)^{1/2}`.  Proof route:
59`1 + 4π²|k|² ≤ 4π²(1+|k|²)` (as `π ≥ 1/2`), `Real.sqrt_le_sqrt`, `Real.rpow_le_rpow`. -/
60theorem jp_rpow_le_two_pi_bracket {d : ℕ} (k : Lattice d) {m : ℝ} (hm : 0 ≤ m) :
61    Lattice.jp k ^ m ≤ (2 * Real.pi * Real.sqrt (1 + Lattice.enorm k ^ 2)) ^ m :=
62  Real.rpow_le_rpow (p_sm2_jp_nonneg k) (p_sm2_jp_le_two_pi_sqrt k) hm

64/-- NEW STATEMENT (clause P6/U2, TeX M L10061-10062): the Sobolev norm of the real mode
65`√2 cos(2πk·x)` / `√2 sin(2πk·x)` (`Torus.realMode k`) is at most `(2π⟨k⟩)^m`, `m ≥ 0`.  Proof
66route: `HRS.MP.MartMod.mmm_sobolevNorm_realMode` (`sobolevNorm m (realMode k) = jp k ^ m`) and
67`jp_rpow_le_two_pi_bracket`. -/
68theorem sobolevNorm_realMode_le_bracket {d : ℕ} (k : NonzeroLattice d) {m : ℝ} (hm : 0 ≤ m) :
69    sobolevNorm m (Torus.realMode k : Torus d → ℝ) ≤
70      (2 * Real.pi * Real.sqrt (1 + Lattice.enorm k.1 ^ 2)) ^ m := by
71  rw [HRS.MP.MartMod.mmm_sobolevNorm_realMode m k]
72  exact jp_rpow_le_two_pi_bracket k.1 hm

74/-- NEW STATEMENT (clause P8/U3(a), TeX M L10068: `‖e_k‖_{𝒞^r} ≤ C_r (1+|k|)^r`, integer `r = j`;
75the fractional Hölder-Zygmund case is a cited identification and is not formalised): the `C^j_b`
76norm (`Torus.cbNorm`, a sum over orders, which is within a factor `j+1` of the printed max norm) of
77the real plane waves grows at most like `(1+|k|)^j`.  Proof route: `Torus.cbNorm_realMode_le` and
78`jp k ≤ 1 + 2π|k| ≤ 2π(1+|k|)`, `C := (j+1)√2 (2π)^j`. -/
79theorem cbNorm_realMode_le_bracket {d : ℕ} (j : ℕ) :
80    ∃ C : ℝ, ∀ k : NonzeroLattice d,
81      Torus.cbNorm j (Torus.realMode k : Torus d → ℝ) ≤ C * (1 + Lattice.enorm k.1) ^ j := by
82  refine ⟨((j : ℝ) + 1) * Real.sqrt 2 * (2 * Real.pi) ^ j, fun k => ?_⟩
83  have h1 := Torus.cbNorm_realMode_le_aux j k
84  have h2 : Lattice.jp k.1 ^ j ≤ (2 * Real.pi * (1 + Lattice.enorm k.1)) ^ j :=
85    pow_le_pow_left₀ (p_sm2_jp_nonneg k.1) (p_sm2_jp_le_linear k.1) j
86  calc Torus.cbNorm j (Torus.realMode k : Torus d → ℝ)
87      ≤ ((j + 1) * Real.sqrt 2) * Lattice.jp k.1 ^ j := h1
88    _ ≤ ((j + 1) * Real.sqrt 2) * (2 * Real.pi * (1 + Lattice.enorm k.1)) ^ j :=
89        mul_le_mul_of_nonneg_left h2 (by positivity)
90    _ = ((j : ℝ) + 1) * Real.sqrt 2 * (2 * Real.pi) ^ j * (1 + Lattice.enorm k.1) ^ j := by
91        rw [mul_pow]; ring
