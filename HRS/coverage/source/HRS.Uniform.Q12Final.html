23lemma q12_hmSum_nonneg (f g : Torus d → ℝ) : 0 ≤ hmSum f g := by

30/-- Peyre's inequality with constant `1/2`. -/
31theorem q12_peyre (d : ℕ) : PeyreW2 d := by
32  refine ⟨1 / 2, by norm_num, fun f g hf hg hfs hgs hfpos m hm hgm => ?_⟩
33  obtain ⟨u, hus, hlap, hE⟩ := q12_poisson hf hg hfs hgs
34  let D : PeyreData d := ⟨f, g, u, m, hm, hf, hg, hfs, hgs, hus, hfpos, hgm, hlap⟩
35  have h1 : torusWassersteinE 2 (withDensityReal g) (withDensityReal f) ≤
36      ENNReal.ofReal (Real.sqrt (π ^ 2 * D.energy / m)) := D.W2_data
37  have hen : D.energy = hmSum f g / (4 * π ^ 2) := by
38    rw [D.energy_eq]; exact hE
39  have hH := q12_hmSum_nonneg f g
40  have hval : Real.sqrt (π ^ 2 * D.energy / m) = 1 / 2 / Real.sqrt m * Real.sqrt (hmSum f g) := by
41    rw [hen]
42    have hπ : (0 : ℝ) < π := Real.pi_pos
43    have : π ^ 2 * (hmSum f g / (4 * π ^ 2)) / m = (1 / 2) ^ 2 * (hmSum f g / m) := by
44      field_simp; ring
45    rw [this, Real.sqrt_mul (by positivity), Real.sqrt_sq (by norm_num), Real.sqrt_div hH]
46    ring
47  rw [hval] at h1
48  exact (q12_W_symm 2 (withDensityReal f) (withDensityReal g)).trans h1

50/-- Lemma 3.23, statement of `HRS/Uniform/Comparison.lean` verbatim. -/
51theorem energy_W2_proof (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (Λ m₀ : ℝ) (hm₀ : 0 < m₀) :
52    ∃ C₀min : ℝ, ∀ C₀ : ℝ, C₀min ≤ C₀ → ∃ C : ℝ, 0 < C ∧
53      ∀ N : ℕ, 2 ≤ N → ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) → (∀ x, m₀ ≤ ρ x) →
54      ∀ X : Config d N, X ∉ collisionSet d N →
55        0 ≤ rieszEnergy d s X (withDensityReal ρ) + oN d s C₀ N ∧
56        torusWassersteinE 2 (empirical X) (withDensityReal ρ) ≤
57          ENNReal.ofReal (C * ((N : ℝ) ^ (-(1 : ℝ) / d) +
58            (rieszEnergy d s X (withDensityReal ρ) + oN d s C₀ N) ^ ((1 : ℝ) / max 2 ((d : ℝ) - s)))) :=
59  energy_W2_of_peyre (q12_peyre d) hd s hs0 hsd Λ m₀ hm₀
