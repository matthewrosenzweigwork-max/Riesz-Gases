25/-- Monotonicity of `HasLogMarginalDensityBounds` in the constant (it enters both the sup bound and the rate). -/
26theorem wl11_hlmdb_mono {d : ℕ} {π μ : Measure (Euc d)} {C C' β : ℝ} {N : ℕ}
27    (H : HRS.Whole.HasLogMarginalDensityBounds π μ C β N) (hC : C ≤ C') :
28    HRS.Whole.HasLogMarginalDensityBounds π μ C' β N := by
29  obtain ⟨φ, hφm, hφeq, hφb, hφ2, hφn⟩ := H
30  refine ⟨φ, hφm, hφeq, fun x => ⟨(hφb x).1, (hφb x).2.trans ?_⟩, hφ2, hφn.trans ?_⟩
31  · refine mul_le_mul_of_nonneg_right hC (Real.rpow_nonneg (by positivity) _)
32  · have : (0 : ℝ) ≤ N := Nat.cast_nonneg N
33    exact div_le_div_of_nonneg_right hC this
