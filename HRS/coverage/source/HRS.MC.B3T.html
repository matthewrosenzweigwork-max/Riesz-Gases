3/-!
4# Packet B3T: Proposition B.3 at intermediate times
5
6The distributional forward Kolmogorov identity of `HRS.selected_law_fke_FKE` (proved there only at
7the horizon `T`) is extended to every time `t ∈ [0, T)`:
8`∫ Φ t dP^t − ∫ Φ 0 dP^0 = ∫_0^t ∫ (generator + interaction) dP^r dr`.
9
10The generator `fkeGenerator D T Φ` contains the time derivative `∂_r Φ` (as
11`derivWithin (configLift Φ · Y) (Icc 0 T)`), exactly as in `selected_law_fke_FKE`.
12The proof re-runs that argument with the Itô evaluation time `t` in place of `T`:
13`ito_formula_additive` (horizon `T`) already concludes at every time `≤ T`, so only the terminal
14evaluation, the martingale time and the time-integration domain change.
15-/

35    ∫ x, Φ t x ∂(selectedLaw P X t) - ∫ x, Φ 0 x ∂(selectedLaw P X 0)

65      liftProcM D W' X τ ω = liftProcM D W' X 0 ω
66        + (∫ s in Set.Icc (0 : ℝ) τ, driftProc D X (Real.toNNReal s) ω)
67        + ∑ k, (W' τ ω k - W' 0 ω k) • liftedNoise D k := by

69    rw [liftProcM_zero ω]
70    simp only [liftProcM]
71    rw [hω τ hτ]

73    intro s

75      funext fun ω => liftProcM_zero ω

77    have hg : Measurable (fun x : Config d N => fun i => Torus.rep (x i)) :=
78      measurable_pi_iff.2 fun i => (measurable_torusRep d).comp (measurable_pi_apply i)
79    exact hg.comp (hF.measurable_init s)

81  have hito := ito_formula_additive (P := P) (ℱ := ℱ) hF.brownian (T := T) hT0 (configLift Φ) hΦc
82    (liftedNoise D) (liftProcM D W' X) (measurable_liftProcM hV hF) hX₀ (driftProc D X)
83    (isProgressive_driftProc hV hF) hI0 hZint

85  have hτT : ((Real.toNNReal t : ℝ≥0) : ℝ) ≤ T := by
86    rw [hτ]
87    exact htT'
88  -- pathwise identity at the Itô time `t`
89  have hcore : ∀ᵐ ω ∂P, Φ t (X (Real.toNNReal t) ω) - Φ 0 (X 0 ω)
90      = (∫ s in Set.Icc (0:ℝ) t, fkeSample D T Φ X s ω)
91        + localItoIntegral P ℱ W' (itoIntegrand T Set.univ (configLift Φ)
92            (ofNNRealTime (liftProcM D W' X)) (fun _ _ => liftedNoise D)) t ω := by
93    filter_upwards [hito, hlift'] with ω hωito hω
94    have h := hωito.2 (Real.toNNReal t) hτT
95    rw [hτ] at h
96    have hL : configLift Φ t (liftProcM D W' X (Real.toNNReal t) ω)
97        = Φ t (X (Real.toNNReal t) ω) := by
98      show Φ t (fun i => Torus.mk (liftProcM D W' X (Real.toNNReal t) ω i)) = _
99      rw [← hω (Real.toNNReal t) hτT]
100    have hcol0 : configLift Φ 0 (liftProcM D W' X 0 ω) = Φ 0 (X 0 ω) := by
101      have h0 := hω 0 (by simpa using hT0)
102      show Φ 0 (fun i => Torus.mk (liftProcM D W' X 0 ω i)) = Φ 0 (X 0 ω)
103      rw [h0]

105        (ofNNRealTime (liftProcM D W' X)) (ofNNRealTime (driftProc D X))
106        (fun _ _ => liftedNoise D) s ω = ∫ s in Set.Icc (0:ℝ) t, fkeSample D T Φ X s ω := by
107      refine setIntegral_congr_fun measurableSet_Icc fun s hs => ?_
108      have hsT : s ∈ Set.Icc (0:ℝ) T := ⟨hs.1, hs.2.trans htT'⟩
109      have hsτ : ((Real.toNNReal s : ℝ≥0) : ℝ) ≤ T := by
110        rw [Real.coe_toNNReal _ hs.1]
111        exact hsT.2
112      have hx := hω _ hsτ
113      rw [itoDrift_eq_fkeIto D T Φ (liftProcM D W' X) X ω s hx]
114      unfold fkeSample
115      rw [clampTI_of_mem hsT]
116      have hmk : ∀ i, Torus.mk (repC (X (Real.toNNReal s) ω) i)
117          = Torus.mk (liftProcM D W' X (Real.toNNReal s) ω i) := by
118        intro i
119        show Torus.mk (Torus.rep (X (Real.toNNReal s) ω i)) = _
120        rw [Torus.mk_rep]
121        exact congrFun hx i
122      exact (fkeIto_eq_of_mk D T Φ s hmk).symm
123    rw [hL, hcol0, hint2] at h
124    linarith
125  -- the reference integrand and the martingale term
126  have hΨprog := isProgressive_fkeRef hF.progressive hT0 D Φ hΦc
127  have hΨL2 : IsProgressiveL2 P ℱ (fun s ω k => fkeRef D T Φ X s ω k) :=
128    IsProgressiveL2.of_bounded P hΨprog (max C1 0 * ∑ j : Fin N × Fin d, ‖liftedNoise D j‖)
129      (fun s ω k => fkeRef_bound hT0 D Φ hC1 s ω k)
130  obtain ⟨N0, hN0m, hN0P, hN0⟩ := exists_null_of_ae (P := P)
131    (p := fun ω => ∀ τ : ℝ≥0, (τ : ℝ) ≤ T → X τ ω = fun i => Torus.mk (liftProcM D W' X τ ω i)) hlift'
132  have hMart : ∀ᵐ ω ∂P, localItoIntegral P ℱ W' (itoIntegrand T Set.univ (configLift Φ)
133          (ofNNRealTime (liftProcM D W' X)) (fun _ _ => liftedNoise D)) t ω
134        = itoIntegral P ℱ W' (fun s ω k => fkeRef D T Φ X s ω k) t ω := by
135    have hcongr := localItoIntegral_congr_null hF.brownian hΨL2.locL2 hN0m hN0P
136      (Ψ := itoIntegrand T Set.univ (configLift Φ) (ofNNRealTime (liftProcM D W' X))
137        (fun _ _ => liftedNoise D))
138      (fun s ω hω => by
139        funext k
140        exact itoIntegrand_eq_fkeRef D Φ W' X ω (hN0 ω hω) s k)
141      (fun s hs ω => by
142        have hn : s ∉ Set.Icc (0:ℝ) T := fun h => absurd h.1 (not_le.2 hs)
143        funext k
144        simp [itoIntegrand, hn])
145      (fun s hs ω => by
146        have hn : s ∉ Set.Icc (0:ℝ) T := fun h => absurd h.1 (not_le.2 hs)
147        funext k
148        simp [fkeRef, hn])
149    have hloc := localItoIntegral_eq_itoIntegral hF.brownian hΨL2
150    filter_upwards [hcongr, hloc] with ω h1 h2
151    rw [h1 t, h2 t]
152  have hMLp : MemLp (itoIntegral P ℱ W' (fun s ω k => fkeRef D T Φ X s ω k) t) 2 P :=
153    memLp_two_itoIntegral hF.brownian hΨL2 t
154  have hMint : ∫ ω, itoIntegral P ℱ W' (fun s ω k => fkeRef D T Φ X s ω k) t ω ∂P = 0 :=
155    integral_itoIntegral hF.brownian hΨL2 t
156  have hsum : ∀ᵐ ω ∂P, Φ t (X (Real.toNNReal t) ω) - Φ 0 (X 0 ω)
157      = (∫ s in Set.Icc (0:ℝ) t, fkeSample D T Φ X s ω)
158        + itoIntegral P ℱ W' (fun s ω k => fkeRef D T Φ X s ω k) t ω := by
159    filter_upwards [hcore, hMart] with ω h1 h2
160    rw [h2] at h1
161    exact h1
162  -- integrability
163  haveI : IsFiniteMeasure (volume.restrict (Set.Icc (0:ℝ) t)) :=
164    ⟨by rw [Measure.restrict_apply_univ]; exact measure_Icc_lt_top⟩
165  have hprodm : Measurable (fun p : ℝ × Ω => fkeSample D T Φ X p.1 p.2) :=
166    measurable_fkeIto_sample hF.progressive hT0 D hV Φ hΦc
167  have hprodint : Integrable (fun p : ℝ × Ω => fkeSample D T Φ X p.1 p.2)
168      ((volume.restrict (Set.Icc (0:ℝ) t)).prod P) :=
169    Integrable.of_bound hprodm.aestronglyMeasurable Cf
170      (ae_of_all _ fun p => hCf _ (clampTI_mem hT0 p.1) (repC (X (Real.toNNReal p.1) p.2)))

172    hprodint.integral_prod_right

174    have e : (fun x : Config d N => Φ 0 x) = fun x => configLift Φ 0 (repC x) := by
175      funext x
176      show Φ 0 x = Φ 0 (fun i => Torus.mk (repC x i))
177      rw [mk_repC]
178    have hcont0 : Continuous (configLift Φ 0) :=
179      hΦc.cont.comp_continuous (continuous_const.prodMk continuous_id)
180        (fun Y => ⟨⟨le_refl (0:ℝ), hT0⟩, Set.mem_univ Y⟩)
181    rw [e]

185      funext x
186      show Φ t x = Φ t (fun i => Torus.mk (repC x i))
187      rw [mk_repC]

191    rw [e]
192    exact hcont.measurable.comp measurable_repC

194    have hb : ∀ x : Config d N, ‖Φ 0 x‖ ≤ C0 := fun x => by

200    exact Integrable.of_bound ((hm0.comp (hC.adapted 0)).aestronglyMeasurable) C0
201      (ae_of_all _ fun ω => hb _)
202  have hΦtint : Integrable (fun ω => Φ t (X (Real.toNNReal t) ω)) P := by
203    have hb : ∀ x : Config d N, ‖Φ t x‖ ≤ C0 := fun x => by
204      have e : Φ t x = configLift Φ t (repC x) := by
205        show Φ t x = Φ t (fun i => Torus.mk (repC x i))
206        rw [mk_repC]
207      rw [e]
208      exact hC0 t ⟨ht0, htT'⟩ (repC x)
209    exact Integrable.of_bound ((hmt.comp (hC.adapted (Real.toNNReal t))).aestronglyMeasurable) C0
210      (ae_of_all _ fun ω => hb _)
211  have hMi : Integrable (fun ω => itoIntegral P ℱ W' (fun s ω k => fkeRef D T Φ X s ω k) t ω) P :=
212    hMLp.integrable one_le_two
213  -- Fubini and the identification with the selected law on `[0, t]`
214  have hIeq : ∫ ω, (∫ s in Set.Icc (0:ℝ) t, fkeSample D T Φ X s ω) ∂P
215      = ∫ r in Set.Icc (0:ℝ) t, ∫ x, (fkeGenerator D T Φ r x + fkeInteraction D Φ r x)
216          ∂(selectedLaw P X r) := by
217    have hswap := integral_integral_swap (μ := volume.restrict (Set.Icc (0:ℝ) t)) (ν := P)
218      (f := fun s ω => fkeSample D T Φ X s ω) hprodint
219    rw [← hswap]
220    refine setIntegral_congr_fun measurableSet_Icc fun r hr => ?_
221    have hrT : r ∈ Set.Icc (0:ℝ) T := ⟨hr.1, hr.2.trans htT'⟩
222    have hg : Measurable (fun x : Config d N => fkeIto D T Φ (clampTI T r) (repC x)) :=
223      (measurable_fkeIto_lift hT0 D hV Φ hΦc).comp (measurable_const.prodMk measurable_repC)
224    have e : ∀ x : Config d N, fkeGenerator D T Φ r x + fkeInteraction D Φ r x
225        = fkeIto D T Φ (clampTI T r) (repC x) := by
226      intro x
227      rw [clampTI_of_mem hrT]
228      exact fkeGI_eq_fkeIto D hsd T Φ r x
229    simp only [e]
230    exact (integral_selectedLaw_eq hC r hg).symm
231  have hLHSt : ∫ x, Φ t x ∂(selectedLaw P X t) = ∫ ω, Φ t (X (Real.toNNReal t) ω) ∂P :=
232    integral_selectedLaw_eq hC t hmt
233  have hLHS : ∫ x, Φ 0 x ∂(selectedLaw P X 0) = ∫ ω, Φ 0 (X 0 ω) ∂P := by
234    rw [integral_selectedLaw_eq hC 0 hm0, Real.toNNReal_zero]

236      = ∫ ω, ((∫ s in Set.Icc (0:ℝ) t, fkeSample D T Φ X s ω)

238    integral_congr_ae hsum
