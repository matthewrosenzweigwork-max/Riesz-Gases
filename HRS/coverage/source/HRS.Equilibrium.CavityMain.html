25theorem rate_facts (hcrit : s = (d : ℝ) / 2) {N : ℕ} (hN : 2 ≤ N) :

260theorem cavity_response_proof {d : ℕ} {s : ℝ} (hd : 2 ≤ d) (hcrit : s = (d : ℝ) / 2) (β : ℝ) (hβ : 0 < β)
261    (ρ : Torus d → ℝ) (hρ : SmoothPosDensity ρ) (c : ℝ)
262    (hc : ∀ z, 0 ≤ periodicRiesz d s z + c) :
263    ∃ C : ℝ, ∀ N : ℕ, 2 ≤ N →
264      (∀ x : Torus d,
265        Integrable (fun Y : Config d (N - 1) => ∑ i, gPlus d s c (x - Y i))
266            (critCavity d s c β ρ N) ∧
267          HasDerivWithinAt (insertionPressure d s c β ρ N x)
268            (-((β / N) * ∫ Y, ∑ i, gPlus d s c (x - Y i) ∂(critCavity d s c β ρ N)))
269            (Set.Ici 0) 0 ∧
270          0 ≤ insertionDefect d s c β ρ N x ∧
271          ENNReal.ofReal (insertionDefect d s c β ρ N x) =
272            ∫⁻ t in Set.Ioc (0 : ℝ) 1, ENNReal.ofReal ((1 - t) * (β / N) ^ 2) *
273              evariance (fun Y : Config d (N - 1) => ∑ i, gPlus d s c (x - Y i))
274                (critPole d s c β ρ N x t) ∧
275          insertionDefect d s c β ρ N x ≤ C * deltaN d s N) ∧
276      (firstMarginal (modulatedGibbs (periodicRiesz d s) N β (withDensityReal ρ)) ≪
277          withDensityReal ρ ∧
278        firstMarginal (critCavity d s c β ρ N) ≪ withDensityReal ρ ∧
279        eLpNorm (fun y =>
280            ((firstMarginal (modulatedGibbs (periodicRiesz d s) N β (withDensityReal ρ))).rnDeriv
281                (withDensityReal ρ) y).toReal -
282              ((firstMarginal (critCavity d s c β ρ N)).rnDeriv (withDensityReal ρ) y).toReal)
283            2 (withDensityReal ρ) ≤
284          ENNReal.ofReal (C * Real.sqrt (Real.log N) / N)) ∧
285      eLpNorm (fun y =>
286          (((firstMarginal (modulatedGibbs (periodicRiesz d s) N β (withDensityReal ρ))).rnDeriv
287              (withDensityReal ρ) y).toReal - 1) +
288            β * Kop (periodicRiesz d s) (withDensityReal ρ)
289              (fun y' => ((firstMarginal (modulatedGibbs (periodicRiesz d s) N β
290                (withDensityReal ρ))).rnDeriv (withDensityReal ρ) y').toReal - 1) y -
291            projMean (withDensityReal ρ) (insertionDefect d s c β ρ N) y)
292          2 (withDensityReal ρ) ≤
293        ENNReal.ofReal (C * Real.sqrt (Real.log N) / N) ∧
294      (∀ x : Torus d, ∀ t ∈ Set.Icc (0 : ℝ) 1,
295        firstMarginal (critPole d s c β ρ N x t) ≪ withDensityReal ρ ∧
296          ∀ᵐ y ∂(withDensityReal ρ),
297            |((firstMarginal (critPole d s c β ρ N x t)).rnDeriv (withDensityReal ρ) y).toReal -
298                Real.exp (-(t * (β / N)) * gPlus d s c (x - y))| ≤
299              C * Real.sqrt (deltaN d s N) * Real.exp (-(t * (β / N)) * gPlus d s c (x - y))) ∧
300      (∀ᵐ y ∂(withDensityReal ρ),
301        |((firstMarginal (modulatedGibbs (periodicRiesz d s) N β (withDensityReal ρ))).rnDeriv
302            (withDensityReal ρ) y).toReal - 1| ≤ C * Real.sqrt (deltaN d s N)) := by
303  obtain ⟨hs0, hsd⟩ := PoleCavityAux.hs_aux hd hcrit
304  obtain ⟨CA, hA⟩ := CavResp.partA hd hs0 hsd β hβ hρ c hc
305  obtain ⟨C2, hC20, h2⟩ := CavResp.item2 hd hcrit hβ hρ (c := c)
306  obtain ⟨C3, hC30, h3⟩ := CavResp.item3 hd hcrit hβ hρ hc
307  obtain ⟨C4, hC40, h4⟩ := CavResp.pole_prefactor hd hcrit hβ hρ hc
308  obtain ⟨C5, hC50, h5⟩ := CavResp.item5 hd hcrit hβ hρ hc
309  obtain ⟨Cm, hCm⟩ : ∃ Cm : ℝ, Cm = max (max (max CA C2) (max C3 C4)) C5 := ⟨_, rfl⟩
310  have hCA : CA ≤ Cm := hCm ▸ ((le_max_left CA C2).trans ((le_max_left _ _).trans (le_max_left _ _)))
311  have hC2 : C2 ≤ Cm := hCm ▸ ((le_max_right CA C2).trans ((le_max_left _ _).trans (le_max_left _ _)))
312  have hC3 : C3 ≤ Cm := hCm ▸ ((le_max_left C3 C4).trans ((le_max_right _ _).trans (le_max_left _ _)))
313  have hC4 : C4 ≤ Cm := hCm ▸ ((le_max_right C3 C4).trans ((le_max_right _ _).trans (le_max_left _ _)))
314  have hC5 : C5 ≤ Cm := hCm ▸ (le_max_right _ _)
315  refine ⟨Cm, fun N hN => ⟨?_, ?_, ?_, ?_, ?_⟩⟩
316  · intro x
317    obtain ⟨a1, a2, a3, a4, a5⟩ := hA N hN x
318    refine ⟨a1, a2, a3, a4, a5.trans ?_⟩
