23/-- The canonical representative of a torus point lies in the box `[0,1)^d`, hence has norm at
24most `√d`. -/
25theorem p_sm1_rep_norm_le {d : ℕ} (x : Torus d) : ‖Torus.rep x‖ ≤ Real.sqrt d := by
26  have hcoord : ∀ i : Fin d, ‖Torus.rep x i‖ ≤ 1 := by
27    intro i
28    have h := (AddCircle.equivIco (1 : ℝ) 0 (x i)).2
29    have h' : 0 ≤ Torus.rep x i ∧ Torus.rep x i < 1 := by
30      simpa [Torus.rep] using h
31    rw [Real.norm_eq_abs, abs_le]
32    constructor <;> linarith [h'.1, h'.2]
33  rw [EuclideanSpace.norm_eq]
34  apply Real.sqrt_le_sqrt
35  calc ∑ i : Fin d, ‖Torus.rep x i‖ ^ 2 ≤ ∑ _i : Fin d, (1 : ℝ) := by
36        apply Finset.sum_le_sum
37        intro i _
38        have h0 : 0 ≤ ‖Torus.rep x i‖ := norm_nonneg _
39        nlinarith [hcoord i]
40    _ = d := by simp

44theorem negSpectral_force_not_lipschitz {d : ℕ} (hd : 1 ≤ d) {s c : ℝ} (hs1 : -1 < s)
45    (hs0 : s < 0) (hc : 0 < c) :
46    (∃ M : ℝ, ∀ x : Torus d, |negSpectralKernel d s c x| ≤ M) ∧
47    ¬ ∃ K : ℝ≥0, LipschitzWith K
48        (fun y : Euc d => Torus.grad (negSpectralKernel d s c) (Torus.mk y)) := by
49  refine ⟨?_, ?_⟩
50  · obtain ⟨M, hM⟩ := (isCompact_univ (X := Torus d)).exists_bound_of_continuousOn
51      (negSpectralKernel_continuous d hs0 hc).continuousOn
52    exact ⟨M, fun x => by rw [← Real.norm_eq_abs]; exact hM x (Set.mem_univ x)⟩
53  · rintro ⟨K, hK⟩
54    obtain ⟨x, -, -, hx⟩ := HRS.MC.L5566.negSpectral_grad_unbounded_proof hd hs1 hs0 hc
55      (‖Torus.grad (negSpectralKernel d s c) (Torus.mk 0)‖ + (K : ℝ) * Real.sqrt d)
56    have hlip := hK.dist_le_mul (Torus.rep x) 0
57    simp only [Torus.mk_rep, dist_eq_norm, sub_zero] at hlip
58    have h1 : ‖Torus.grad (negSpectralKernel d s c) x - Torus.grad (negSpectralKernel d s c) (Torus.mk 0)‖
59        ≤ (K : ℝ) * Real.sqrt d :=
60      hlip.trans (mul_le_mul_of_nonneg_left (p_sm1_rep_norm_le x) (NNReal.coe_nonneg K))
61    have h2 : ‖Torus.grad (negSpectralKernel d s c) x‖
62        ≤ ‖Torus.grad (negSpectralKernel d s c) x - Torus.grad (negSpectralKernel d s c) (Torus.mk 0)‖
63          + ‖Torus.grad (negSpectralKernel d s c) (Torus.mk 0)‖ := by
64      calc ‖Torus.grad (negSpectralKernel d s c) x‖
65          = ‖(Torus.grad (negSpectralKernel d s c) x - Torus.grad (negSpectralKernel d s c) (Torus.mk 0))
66              + Torus.grad (negSpectralKernel d s c) (Torus.mk 0)‖ := by rw [sub_add_cancel]
67        _ ≤ _ := norm_add_le _ _
68    linarith
