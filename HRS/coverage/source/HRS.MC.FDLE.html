20/-- `w_λ = 0` on `‖z‖ ≥ λ`. -/
21lemma fdle_W_vanish {ψ : Euc d → ℝ} (hψ : IsFDLProfile ψ) {s b lam : ℝ} {z : Euc d}
22    (hz : lam ≤ ‖z‖) (hlam : 0 < lam) : fdlWE ψ s b lam z = 0 := by
23  unfold fdlWE
24  have h0 : ∫ t in Ioc 0 lam, t ^ ((d : ℝ) - s) * fdlPhiScale ψ t z / t
25      = ∫ _ in Ioc 0 lam, (0 : ℝ) :=
26    setIntegral_congr_fun measurableSet_Ioc (fun t ht => by
27      show t ^ ((d : ℝ) - s) * fdlPhiScale ψ t z / t = 0
28      rw [fdle_phiScale_eq_zero hψ ht.1 (le_trans ht.2 hz), mul_zero, zero_div])
29  rw [h0, integral_zero, mul_zero]

31/-- Lower-bound integrability of `t ↦ t^{-s}/t` on `(a, b]`, `a > 0`. -/
32lemma fdle_lower_integrable (s : ℝ) {a b : ℝ} (ha : 0 < a) :
33    IntegrableOn (fun t : ℝ => t ^ (-s) / t) (Ioc a b) := by
34  have hc : ContinuousOn (fun t : ℝ => t ^ (-s) / t) (Icc a b) :=
35    ContinuousOn.div
36      (continuousOn_id.rpow_const (fun t ht => Or.inl (ne_of_gt (lt_of_lt_of_le ha ht.1))))
37      continuousOn_id (fun t ht => ne_of_gt (lt_of_lt_of_le ha ht.1))
38  exact hc.integrableOn_Icc.mono_set Ioc_subset_Icc_self

40/-- Lemma D.3(i) with the repair at `s = 0` (premise `‖z‖ ≤ 1` in the third bullet). -/
41theorem fdle_part_i (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {ψ : Euc d → ℝ}
42    (hψ : IsFDLProfile ψ) {b : ℝ} (hb : IsFDLConst ψ s b) :
43    ∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
44      (∀ z : Euc d, z ≠ 0 → 0 ≤ fdlWE ψ s b lam z) ∧
45      (∀ z : Euc d, lam ≤ ‖z‖ →
46        fdlWE ψ s b lam z = 0 ∧ fdlTruncE ψ s b lam z = Whole.rieszKernel d s z) ∧
47      (∀ z : Euc d, z ≠ 0 → (s = 0 → ‖z‖ ≤ 1) →
48        fdlWE ψ s b lam z ≤ Whole.rieszKernel d s z + C * (if s = 0 then 1 else 0)) ∧
49      ∫⁻ z : Euc d, ENNReal.ofReal (fdlWE ψ s b lam z) ≤
50        ENNReal.ofReal (C * lam ^ ((d : ℝ) - s)) := by
51  have hb0 : 0 < b := hb.1
52  have hds : 0 < (d : ℝ) - s := by linarith
53  have hI0 : 0 ≤ ∫ x, fdlPhi ψ x := integral_nonneg (fdle_phi_nonneg hψ)
54  refine ⟨b * (∫ x, fdlPhi ψ x) / ((d : ℝ) - s),
55    div_nonneg (mul_nonneg hb0.le hI0) hds.le, fun lam hlam0 hlam1 => ?_⟩
56  have hC0 : 0 ≤ b * (∫ x, fdlPhi ψ x) / ((d : ℝ) - s) :=
57    div_nonneg (mul_nonneg hb0.le hI0) hds.le
58  refine ⟨fun z hz => ?_, fun z hz => ?_, fun z hz hs1 => ?_, ?_⟩
59  · -- nonnegativity
60    unfold fdlWE
61    exact mul_nonneg hb0.le
62      (setIntegral_nonneg measurableSet_Ioc (fun t ht => fdle_F_nonneg hψ s ht.1 z))
63  · -- vanishing beyond `λ` and the truncated kernel
64    have hz0 : z ≠ 0 := fun h => by rw [h, norm_zero] at hz; linarith
65    have hW := fdle_W_vanish (s := s) (b := b) hψ hz hlam0
66    refine ⟨hW, ?_⟩
67    unfold fdlTruncE
68    rw [if_neg hz0, hW, sub_zero]
69  · -- the repaired third bullet
70    have hz0 : z ≠ 0 := hz
71    rcases hs0.eq_or_lt with h | h
72    · subst h
73      have hr1 : ‖z‖ ≤ 1 := hs1 rfl
74      rw [if_pos rfl, mul_one, fdle_rk0]
75      by_cases hr : lam ≤ ‖z‖
76      · rw [fdle_W_vanish (s := 0) (b := b) hψ hr hlam0]
77        have := Real.log_nonpos (norm_nonneg z) hr1
78        linarith
79      · push_neg at hr
80        have hr0 : 0 < ‖z‖ := norm_pos_iff.mpr hz0
81        have hφ0 := fdle_b_phi0 hd hb
82        unfold fdlWE
83        have hsplit : ∫ t in Ioc 0 lam, t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t
84            = ∫ t in Ioc ‖z‖ lam, t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t := by
85          refine setIntegral_eq_of_subset_of_forall_sdiff_eq_zero measurableSet_Ioc
86            (Ioc_subset_Ioc_left hr0.le) (fun t ht => ?_)
87          have htz : t ≤ ‖z‖ := by
88            by_contra h'
89            exact ht.2 ⟨not_le.mp h', ht.1.2⟩
90          show t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t = 0
91          rw [fdle_phiScale_eq_zero hψ ht.1.1 htz, mul_zero, zero_div]
92        have hFint : IntegrableOn (fun t : ℝ => t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t)
93            (Ioc ‖z‖ lam) :=
94          (fdle_F_integrable hs0 hb hz0 hlam0).mono_set (Ioc_subset_Ioc_left hr0.le)
95        have hGint : IntegrableOn (fun t : ℝ => fdlPhi ψ 0 * t⁻¹) (Ioc ‖z‖ lam) := by
96          have hc : ContinuousOn (fun t : ℝ => fdlPhi ψ 0 * t⁻¹) (Icc ‖z‖ lam) :=
97            continuousOn_const.mul (continuousOn_inv₀.mono (fun t ht => by
98              exact ne_of_gt (lt_of_lt_of_le hr0 ht.1)))
99          exact hc.integrableOn_Icc.mono_set Ioc_subset_Icc_self
100        have hmono : ∫ t in Ioc ‖z‖ lam, t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t
101            ≤ ∫ t in Ioc ‖z‖ lam, fdlPhi ψ 0 * t⁻¹ := by
102          refine setIntegral_mono_on hFint hGint measurableSet_Ioc (fun t ht => ?_)
103          have ht0 : 0 < t := lt_of_le_of_lt hr0.le ht.1
104          rw [fdle_F0_eq ht0 z, ← div_eq_mul_inv]
105          exact div_le_div_of_nonneg_right (fdle_phi_le_phi0 hψ _) ht0.le
106        have hval : ∫ t in Ioc ‖z‖ lam, fdlPhi ψ 0 * t⁻¹ = fdlPhi ψ 0 * Real.log (lam / ‖z‖) := by
107          rw [integral_const_mul, ← intervalIntegral.integral_of_le hr.le,
108            integral_inv_of_pos hr0 hlam0]
109        have hbound : b * ∫ t in Ioc 0 lam, t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t
110            ≤ Real.log (lam / ‖z‖) := by
111          rw [hsplit]
112          have := mul_le_mul_of_nonneg_left (hmono.trans hval.le) hb0.le
113          calc b * ∫ t in Ioc ‖z‖ lam, t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t
114              ≤ b * (fdlPhi ψ 0 * Real.log (lam / ‖z‖)) := by
115                exact mul_le_mul_of_nonneg_left (hmono.trans hval.le) hb0.le
116            _ = (b * fdlPhi ψ 0) * Real.log (lam / ‖z‖) := by ring
117            _ = Real.log (lam / ‖z‖) := by rw [hφ0, one_mul]
118        have hlog : Real.log (lam / ‖z‖) = Real.log lam - Real.log ‖z‖ :=
119          Real.log_div hlam0.ne' hr0.ne'
120        have hlam_log := Real.log_nonpos hlam0.le hlam1.le
121        linarith
122    · have hs' : s ≠ 0 := ne_of_gt h
123      rw [if_neg hs', mul_zero, add_zero]
124      have hrk := (hb.2.1 h z hz0).2
125      rw [hrk]
126      refine mul_le_mul_of_nonneg_left ?_ hb0.le
127      refine setIntegral_mono_set (hb.2.1 h z hz0).1 ?_ ?_
128      · exact (ae_restrict_iff' measurableSet_Ioi).mpr
129          (Eventually.of_forall fun t ht => fdle_F_nonneg hψ s (show (0:ℝ) < t from ht) z)
130      · exact Eventually.of_forall fun t ht => Ioc_subset_Ioi_self ht
131  · -- the `L¹` bound
132    rw [fdle_lintegral_W hd hs0 hsd hψ hb hlam0]
133    apply le_of_eq
134    congr 1
135    ring

137/-- Lemma D.3(iv), the three items of `fdl_kernel_euclid` verbatim. -/
138theorem fdle_part_iv (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {ψ : Euc d → ℝ}
139    (hψ : IsFDLProfile ψ) {b : ℝ} (hb : IsFDLConst ψ s b) :
140    (0 < s → ∃ c : ℝ, 0 < c ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d, lam ≤ ‖z‖ →
141      c * ‖z‖ ^ (-s) ≤ fdlTruncE ψ s b lam z) ∧
142    (s = 0 → ∃ c C : ℝ, 0 < c ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d, lam ≤ ‖z‖ →
143      ‖z‖ ≤ 1 / 2 → c * Real.log (1 / ‖z‖) - C ≤ fdlTruncE ψ s b lam z) ∧
144    (∀ K : ℝ, fdlK ≤ K → ∃ c : ℝ, 0 < c ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ z : Euc d, z ≠ 0 →
145      ‖z‖ ≤ 2 * lam → c * (if s = 0 then 1 else lam ^ (-s)) ≤ fdlWE ψ s b (K * lam) z) := by
146  have hb0 : 0 < b := hb.1
147  refine ⟨fun h => ?_, fun h0 => ?_, fun K hK => ?_⟩
148  · -- (iv), first item, `s > 0`
149    refine ⟨1, one_pos, fun lam hlam0 hlam1 z hz => ?_⟩
150    have hz0 : z ≠ 0 := fun h' => by rw [h', norm_zero] at hz; linarith
151    have hT : fdlTruncE ψ s b lam z = Whole.rieszKernel d s z := by
152      unfold fdlTruncE
153      rw [if_neg hz0, fdle_W_vanish hψ hz hlam0, sub_zero]
154    rw [hT, fdle_rk_pos s h.ne' z, one_mul]
155  · -- (iv), second item, `s = 0`
156    subst h0
157    refine ⟨1, 0, one_pos, fun lam hlam0 hlam1 z hz hz2 => ?_⟩
158    have hz0 : z ≠ 0 := fun h' => by rw [h', norm_zero] at hz; linarith
159    have hT : fdlTruncE ψ 0 b lam z = Whole.rieszKernel d 0 z := by
160      unfold fdlTruncE
161      rw [if_neg hz0, fdle_W_vanish hψ hz hlam0, sub_zero]
162    rw [hT, fdle_rk0, one_mul, sub_zero, one_div, Real.log_inv]
163  · -- (iv), third item
164    have hK16 : (16 : ℝ) ≤ K := hK
165    obtain ⟨c0, hc0pos, hc0⟩ := fdle_phi_min hψ
166    rcases hs0.eq_or_lt with h | h
167    · subst h
168      refine ⟨b * c0 * Real.log (K / 8), mul_pos (mul_pos hb0 hc0pos)
169        (Real.log_pos ((one_lt_div (by norm_num)).mpr (by linarith))),
170        fun lam hlam0 hlam1 z hz0 hzl => ?_⟩
171      rw [if_pos rfl, mul_one]
172      have hKl : 0 < K * lam := mul_pos (by linarith) hlam0
173      have h8 : 0 < 8 * lam := by linarith
174      have hpos8 : (0 : ℝ) ≤ 8 * lam := h8.le
175      have hle8 : 8 * lam ≤ K * lam := mul_le_mul_of_nonneg_right (by linarith) hlam0.le
176      have hr0 : 0 < ‖z‖ := norm_pos_iff.mpr hz0
177      have hFint : IntegrableOn (fun t : ℝ => t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t)
178          (Ioc 0 (K * lam)) := fdle_F_integrable hs0 hb hz0 hKl
179      have h1 : ∫ t in Ioc (8 * lam) (K * lam), t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t
180          ≤ ∫ t in Ioc 0 (K * lam), t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t :=
181        setIntegral_mono_set hFint
182          ((ae_restrict_iff' measurableSet_Ioc).mpr
183            (Eventually.of_forall fun t ht => fdle_F_nonneg hψ 0 ht.1 z))
184          (Eventually.of_forall fun t ht => Ioc_subset_Ioc_left hpos8 ht)
185      have h2 : ∫ t in Ioc (8 * lam) (K * lam), c0 * (t ^ (-(0 : ℝ)) / t)
186          ≤ ∫ t in Ioc (8 * lam) (K * lam), t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t := by
187        refine setIntegral_mono_on ((fdle_lower_integrable 0 h8).const_mul c0)
188          (hFint.mono_set (Ioc_subset_Ioc_left hpos8)) measurableSet_Ioc (fun t ht => ?_)
189        have ht0 : 0 < t := lt_trans h8 ht.1
190        have hzt : ‖t⁻¹ • z‖ ≤ 1 / 4 := by
191          rw [norm_smul, Real.norm_of_nonneg (inv_nonneg.2 ht0.le), ← div_eq_inv_mul,
192            div_le_iff₀ ht0]
193          linarith [ht.1]
194        exact fdle_F_lower 0 hc0 ht0 hzt
195      have h3 : ∫ t in Ioc (8 * lam) (K * lam), c0 * (t ^ (-(0 : ℝ)) / t)
196          = c0 * Real.log (K / 8) := by
197        have hsimp : ∀ t : ℝ, c0 * (t ^ (-(0 : ℝ)) / t) = c0 * t⁻¹ := by
198          intro t
199          simp
200        simp_rw [hsimp]
201        rw [integral_const_mul, ← intervalIntegral.integral_of_le (by linarith : 8 * lam ≤ K * lam),
202          integral_inv_of_pos h8 hKl, mul_div_mul_right _ _ hlam0.ne']
203      have hfin : c0 * Real.log (K / 8) ≤
204          ∫ t in Ioc 0 (K * lam), t ^ ((d : ℝ) - 0) * fdlPhiScale ψ t z / t := by
205        rw [← h3]
206        exact h2.trans h1
207      unfold fdlWE
208      rw [mul_assoc]
209      exact mul_le_mul_of_nonneg_left hfin hb0.le
210    · -- `s > 0`
211      have hs' : s ≠ 0 := ne_of_gt h
212      have hK8 : (8 : ℝ) < K := by linarith
213      have hpow : K ^ (-s) < (8 : ℝ) ^ (-s) :=
214        Real.rpow_lt_rpow_of_neg (by norm_num) hK8 (by linarith)
215      refine ⟨b * c0 * ((8 : ℝ) ^ (-s) - K ^ (-s)) / s,
216        div_pos (mul_pos (mul_pos hb0 hc0pos) (by linarith)) h,
217        fun lam hlam0 hlam1 z hz0 hzl => ?_⟩
218      rw [if_neg hs']
219      have hKl : 0 < K * lam := mul_pos (by linarith) hlam0
220      have h8 : 0 < 8 * lam := by linarith
221      have hpos8 : (0 : ℝ) ≤ 8 * lam := h8.le
222      have hle8 : 8 * lam ≤ K * lam := mul_le_mul_of_nonneg_right (by linarith) hlam0.le
223      have hr0 : 0 < ‖z‖ := norm_pos_iff.mpr hz0
224      have hFint : IntegrableOn (fun t : ℝ => t ^ ((d : ℝ) - s) * fdlPhiScale ψ t z / t)
225          (Ioc 0 (K * lam)) := fdle_F_integrable hs0 hb hz0 hKl
226      have h1 : ∫ t in Ioc (8 * lam) (K * lam), t ^ ((d : ℝ) - s) * fdlPhiScale ψ t z / t
227          ≤ ∫ t in Ioc 0 (K * lam), t ^ ((d : ℝ) - s) * fdlPhiScale ψ t z / t :=
228        setIntegral_mono_set hFint
229          ((ae_restrict_iff' measurableSet_Ioc).mpr
230            (Eventually.of_forall fun t ht => fdle_F_nonneg hψ s ht.1 z))
231          (Eventually.of_forall fun t ht => Ioc_subset_Ioc_left hpos8 ht)
232      have h2 : ∫ t in Ioc (8 * lam) (K * lam), c0 * (t ^ (-s) / t)
233          ≤ ∫ t in Ioc (8 * lam) (K * lam), t ^ ((d : ℝ) - s) * fdlPhiScale ψ t z / t := by
234        refine setIntegral_mono_on ((fdle_lower_integrable s h8).const_mul c0)
235          (hFint.mono_set (Ioc_subset_Ioc_left hpos8)) measurableSet_Ioc (fun t ht => ?_)
236        have ht0 : 0 < t := lt_trans h8 ht.1
237        have hzt : ‖t⁻¹ • z‖ ≤ 1 / 4 := by
238          rw [norm_smul, Real.norm_of_nonneg (inv_nonneg.2 ht0.le), ← div_eq_inv_mul,
239            div_le_iff₀ ht0]
240          linarith [ht.1]
241        exact fdle_F_lower s hc0 ht0 hzt
242      have h3 : ∫ t in Ioc (8 * lam) (K * lam), c0 * (t ^ (-s) / t)
243          = c0 * (((8 * lam) ^ (-s) - (K * lam) ^ (-s)) / s) := by
244        have hcongr : ∫ t in Ioc (8 * lam) (K * lam), t ^ (-s) / t
245            = ∫ t in Ioc (8 * lam) (K * lam), t ^ (-s - 1) :=
246          setIntegral_congr_fun measurableSet_Ioc (fun t ht => by
247            exact (Real.rpow_sub_one (ne_of_gt (lt_trans h8 ht.1)) (-s)).symm)
248        rw [integral_const_mul, hcongr, ← intervalIntegral.integral_of_le (by linarith : 8 * lam ≤ K * lam),
249          integral_rpow (Or.inr ⟨fun h' => by linarith, ?_⟩)]
250        · have hr1 : -s - 1 + 1 = -s := by ring
251          rw [hr1, div_neg]
252          ring
253        · rw [uIcc_of_le (by linarith : 8 * lam ≤ K * lam)]
254          intro h'
255          linarith [h'.1]
256      have key : c0 * (((8 * lam) ^ (-s) - (K * lam) ^ (-s)) / s)
257          ≤ ∫ t in Ioc 0 (K * lam), t ^ ((d : ℝ) - s) * fdlPhiScale ψ t z / t := by
258        rw [← h3]
259        exact h2.trans h1
260      rw [Real.mul_rpow (by norm_num) hlam0.le, Real.mul_rpow (by linarith) hlam0.le] at key
261      have := mul_le_mul_of_nonneg_left key hb0.le
262      unfold fdlWE
263      calc b * c0 * ((8 : ℝ) ^ (-s) - K ^ (-s)) / s * lam ^ (-s)
264          = b * (c0 * (((8 : ℝ) ^ (-s) * lam ^ (-s) - K ^ (-s) * lam ^ (-s)) / s)) := by ring
