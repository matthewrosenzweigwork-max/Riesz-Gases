28theorem field_chaos_proof (r : ℝ) (hr : r = 2 * (d : ℝ) + 6) (T : ℝ) (hT : 0 < T)
29    {Ω : ℕ → Type} [∀ N, MeasurableSpace (Ω N)] (P : ∀ N, Measure (Ω N))
30    [∀ N, IsProbabilityMeasure (P N)] (X₀ : ∀ N, Ω N → Config d N)
31    (W : ∀ N, ℝ≥0 → Ω N → (Fin N → Euc d)) (X : ∀ N, ℝ≥0 → Ω N → Config d N)
32    (hreal : ∀ N, 1 ≤ N → PreparedRealization R.D (P N)
33      (modulatedGibbs (periodicRiesz d R.D.s) N R.β₀ (withDensityReal R.ρ₀)) (X₀ N) (W N) (X N))
34    {Ω' : Type} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
35    (η : ℝ → Ω' → NegSobolev d r)
36    (hη : IsGaussianLimitField R.D R.ρ (withDensityReal R.ρ₀) R.β₀ r P' η) :
37    ∃ B : ℝ → Ω' → NegSobolev d r, Measurable (Function.uncurry B) ∧
38      (∀ᵐ ω ∂P', MemLp (fun t => B t ω) 2 (volume.restrict (Set.Icc (0 : ℝ) T))) ∧
39      (∀ᵐ t ∂(volume.restrict (Set.Icc (0 : ℝ) T)),
40        IsWickQuadraticField R.D.s P' (fun ω => η t ω) (withDensityReal (R.ρ t)) (B t)) ∧
41      (∀ G : PathSpace d r T × L2Path d r T → ℝ, Continuous G → (∃ M, ∀ z, |G z| ≤ M) →
42        Tendsto (fun N : ℕ => ∫ ω, G (pairN r T 0 R.D.s R.ρ (P N) (X N) ω) ∂(P N)) atTop
43          (𝓝 (∫ ω, G (pairLim r T 0 η B ω) ∂P'))) ∧
44      (∀ q : ℝ, 1 ≤ q →
45        Tendsto (fun N : ℕ => wassersteinE q (law (P N) (pairN r T 0 R.D.s R.ρ (P N) (X N)))
46            (law P' (pairLim r T 0 η B))) atTop (𝓝 0)) := by

49    hBL2 hBW⟩
