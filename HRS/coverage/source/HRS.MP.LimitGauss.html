158theorem mlg_X_bound {s t T C₂ C₃ u : ℝ} (hs : 0 ≤ s) (hst : s ≤ t) (htT : t ≤ T) :

163  have hδT : δ ≤ T := by linarith
164  have hsT : s ≤ T := by linarith

166  have h1 : s * δ ≤ T * T := mul_le_mul hsT hδT hδ hT
167  have h2 : s * δ ^ 2 ≤ T ^ 2 * δ := by
