654theorem hasFDerivAt_periodicRiesz {s : ℝ} (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d) {y0 : Euc d}
655    (hy0 : Torus.mk y0 ≠ 0) :
656    HasFDerivAt (fun y : Euc d => periodicRiesz d s (Torus.mk y))
657      (∑ j, (periodicRieszGrad d s (Torus.mk y0)) j • (EuclideanSpace.proj j : Euc d →L[ℝ] ℝ)) y0 := by
658  obtain ⟨x0, hxm, hxb⟩ := exists_mk_box (Torus.mk y0)
659  obtain ⟨m, hm⟩ := exists_lattice_of_mk_eq hxm
660  have hx0n : Torus.mk x0 ≠ 0 := by rw [hxm]; exact hy0
661  have hx0z : x0 ≠ 0 := mk_ne_zero_ne_zero hx0n
662  have h := hasFDerivAt_periodicRiesz_box hd hs0 hsd hxb hx0n
663  have h2 : HasFDerivAt (fun y : Euc d => periodicRiesz d s (Torus.mk (y + Lattice.toEuc m)))
664      (∑ j, gradComp s j (Torus.mk x0) • (EuclideanSpace.proj j : Euc d →L[ℝ] ℝ)) y0 := by
665    have hadd : HasFDerivAt (fun y : Euc d => y + Lattice.toEuc m) (ContinuousLinearMap.id ℝ (Euc d)) y0 :=
666      (hasFDerivAt_id y0).add_const _
667    have hm' : y0 + Lattice.toEuc m = x0 := hm.symm
668    have h' : HasFDerivAt (fun y : Euc d => periodicRiesz d s (Torus.mk y))
669        (∑ j, gradComp s j (Torus.mk x0) • (EuclideanSpace.proj j : Euc d →L[ℝ] ℝ)) (y0 + Lattice.toEuc m) := by
670      rw [hm']; exact h
671    exact (h'.comp y0 hadd).congr_fderiv (ContinuousLinearMap.comp_id _)
672  have hfun : (fun y : Euc d => periodicRiesz d s (Torus.mk (y + Lattice.toEuc m))) =
673      fun y => periodicRiesz d s (Torus.mk y) := by
674    funext y
675    rw [mk_add, mk_toEuc, add_zero]
676  rw [hfun] at h2
677  have hgrad := periodicRieszGrad_eq hsd x0 hxb hx0z
678  rw [hxm] at hgrad hx0n
679  rw [hxm] at h2
680  convert h2 using 2
681  rw [hgrad]
