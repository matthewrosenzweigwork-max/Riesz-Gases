58    have hD1 : 1 ≤ D := by nlinarith
59    have hDi : ∀ i, 1 ≤ i → i ≤ j → ‖iteratedFDeriv ℝ i φ x‖ ≤ D ^ i := fun i hi hij =>
60      calc _ ≤ C * (1 + ‖x‖) ^ k0 := hD i hi (hij.trans hj) x
61        _ ≤ D := by rw [hDdef]; nlinarith

65      rw [hρeq]; exact iteratedFDeriv_const_smul_apply' (Real.contDiff_exp.comp hφj).contDiffAt

68      rw [hρj, norm_smul, norm_inv, Real.norm_of_nonneg hZ.le, hρ x]

74      rw [hDdef, mul_pow, ← pow_mul]

92    (Function.hasTemperateGrowth_norm_sq (Euc d)).norm_iteratedFDeriv_le_uniform n

95  have hfun : (fun x : Euc d => -β * (a * ‖x‖ ^ 2 / 2 + U x)) =
96      (fun x : Euc d => (-β * a / 2) • ‖x‖ ^ 2) + (fun x => (-β) • U x) := by
97    funext x; simp only [Pi.add_apply, smul_eq_mul]; ring
98  have hx1 : 1 ≤ (1 + ‖x‖) ^ k0 := one_le_pow₀ (by linarith [norm_nonneg x])

100    (hqd.of_le (by exact_mod_cast hin)).contDiffAt
101  have hUi : ContDiffAt ℝ i U x := (hU.of_le (by exact_mod_cast hin)).contDiffAt
102  rw [hfun, iteratedFDeriv_add_apply (hqi.const_smul _) (hUi.const_smul _),
103    iteratedFDeriv_const_smul_apply' hqi, iteratedFDeriv_const_smul_apply' hUi]
104  calc ‖(-β * a / 2) • iteratedFDeriv ℝ i (fun x : Euc d => ‖x‖ ^ 2) x +
105        (-β) • iteratedFDeriv ℝ i U x‖
106      ≤ ‖(-β * a / 2) • iteratedFDeriv ℝ i (fun x : Euc d => ‖x‖ ^ 2) x‖ +
107        ‖(-β) • iteratedFDeriv ℝ i U x‖ := norm_add_le _ _
108    _ ≤ |(-β * a / 2)| * (Cq * (1 + ‖x‖) ^ k0) + |(-β)| * M := by
109        rw [norm_smul, norm_smul, Real.norm_eq_abs, Real.norm_eq_abs]
110        gcongr
111        · exact hq i hin x
112        · exact hUb i hi hin x
113    _ ≤ _ := by
114        have h := mul_le_mul_of_nonneg_left hx1 (mul_nonneg (abs_nonneg (-β)) hM)
115        nlinarith [h]

342lemma lreg_l_bound (n : ℕ) : ∃ M : ℝ, ∀ j, 1 ≤ j → j ≤ n → ∀ x : Euc d,
343    ‖iteratedFDeriv ℝ j (fun x : Euc d => (1 / 2 : ℝ) * Real.log (1 + ‖x‖ ^ 2)) x‖ ≤ M := by
344  obtain ⟨K, hK1, hK⟩ := lreg_psi_bound (d := d) n
345  refine ⟨2 ^ n * K * 2, fun j hj hjn x => ?_⟩
346  obtain ⟨m, rfl⟩ : ∃ m, j = m + 1 := ⟨j - 1, by omega⟩
347  rw [← norm_iteratedFDeriv_fderiv]
348  have hfd : fderiv ℝ (fun x : Euc d => (1 / 2 : ℝ) * Real.log (1 + ‖x‖ ^ 2)) =
349      fun y => lreg_psi y • (innerSL ℝ y : Euc d →L[ℝ] ℝ) := funext lreg_l_fderiv
350  rw [hfd]
351  have hTd : ContDiff ℝ m (fun y : Euc d => (innerSL ℝ y : Euc d →L[ℝ] ℝ)) :=
352    (innerSL ℝ : Euc d →L[ℝ] Euc d →L[ℝ] ℝ).contDiff
353  have h1 := norm_iteratedFDeriv_smul_le (𝕜 := ℝ) (N := (m : ℕ∞ω)) (lreg_psi_contDiff (n := m)) hTd x
354    (n := m) le_rfl

357      ‖iteratedFDeriv ℝ (m - i) (fun y : Euc d => (innerSL ℝ y : Euc d →L[ℝ] ℝ)) x‖ ≤
358      (m.choose i : ℝ) * (K * (lreg_psi x * (1 + ‖x‖))) := by
359    intro i hi
360    have hi' : i ≤ m := Nat.lt_succ_iff.1 (Finset.mem_range.1 hi)
361    have hab : ‖iteratedFDeriv ℝ i lreg_psi x‖ *
362        ‖iteratedFDeriv ℝ (m - i) (fun y : Euc d => (innerSL ℝ y : Euc d →L[ℝ] ℝ)) x‖ ≤
363        K * (lreg_psi x * (1 + ‖x‖)) :=
364      calc _ ≤ (K * lreg_psi x) * (1 + ‖x‖) :=
365            mul_le_mul (hK i (by omega) x) (lreg_T_bound _ x) (norm_nonneg _)
366              (mul_nonneg (by linarith) (lreg_psi_pos x).le)
367        _ = _ := by ring
368    calc (m.choose i : ℝ) * ‖iteratedFDeriv ℝ i lreg_psi x‖ *
369          ‖iteratedFDeriv ℝ (m - i) (fun y : Euc d => (innerSL ℝ y : Euc d →L[ℝ] ℝ)) x‖
370        = (m.choose i : ℝ) * (‖iteratedFDeriv ℝ i lreg_psi x‖ *
371          ‖iteratedFDeriv ℝ (m - i) (fun y : Euc d => (innerSL ℝ y : Euc d →L[ℝ] ℝ)) x‖) := by ring
372      _ ≤ _ := mul_le_mul_of_nonneg_left hab (Nat.cast_nonneg _)
373  refine (Finset.sum_le_sum hbd).trans ?_
374  rw [← Finset.sum_mul]
375  have hsum : ∑ i ∈ Finset.range (m + 1), (m.choose i : ℝ) = 2 ^ m := by
376    exact_mod_cast Nat.sum_range_choose m
377  rw [hsum]
378  have hpm := lreg_psi_mul_le x
379  have h2m : (2 : ℝ) ^ m ≤ 2 ^ n := pow_le_pow_right₀ (by norm_num) (by omega)
380  calc 2 ^ m * (K * (lreg_psi x * (1 + ‖x‖))) ≤ 2 ^ m * (K * 2) := by gcongr
381    _ ≤ 2 ^ n * (K * 2) := by gcongr
382    _ = _ := by ring

394lemma lreg_rdec_of_le {a β : ℝ} (ha : 0 < a) (hβ : 0 < β) {ρ : Euc d → ℝ} {C : ℝ}
395    (h0 : ∀ x, 0 ≤ ρ x)
396    (hle : ∀ x, ρ x ≤ C * ((1 + ‖x‖ ^ 2) ^ (β / 2) * Real.exp (-β * a * ‖x‖ ^ 2 / 2))) :
397    lreg_RDec ρ := by
398  intro N
399  set K0 := Real.exp (β * (a / 2 - 1 - Real.log (a / 2)) / 2) with hK0
400  obtain ⟨C1, hC1, hC1b⟩ := cb_rpow_le_exp (p := (N : ℝ)) (b := β * a / 16) (by positivity)
401    (by positivity)

410      _ = C * t67w a β x := by rw [hw]
411      _ ≤ max C 0 * t67w a β x := mul_le_mul_of_nonneg_right (le_max_left _ _) hwpos.le
412  have h1 : (1 + r) ^ N ≤ C1 * Real.exp (β * a / 16 * (1 + r) ^ 2) := by
413    have := hC1b (1 + r) (by positivity)
414    rwa [Real.rpow_natCast] at this
415  have h2 : (1 + r) ^ 2 ≤ 2 + 2 * r ^ 2 := by nlinarith [sq_nonneg (1 - r)]
416  have h3 : Real.exp (β * a / 16 * (1 + r) ^ 2) ≤
417      Real.exp (2 * (β * a / 16)) * Real.exp (2 * (β * a / 16) * r ^ 2) := by
418    rw [← Real.exp_add]
419    apply Real.exp_le_exp.2

426        refine mul_le_mul ?_ ?_ (h0 x) (by positivity)

428        · exact hρ.trans (mul_le_mul_of_nonneg_left h4 (le_max_right _ _))

430          Real.exp ((2 * (β * a / 16) - β * a / 4) * r ^ 2) := by
431        have : Real.exp ((2 * (β * a / 16) - β * a / 4) * r ^ 2) =
432            Real.exp (2 * (β * a / 16) * r ^ 2) * Real.exp (-(β * a / 4) * r ^ 2) := by
433          rw [← Real.exp_add]; congr 1; ring
434        rw [this]; ring
