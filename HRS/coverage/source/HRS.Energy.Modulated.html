53/-- The modulated energy `F_N^h(X_N, μ)` of `H:eq:modulated-energy` (lines 437-443) for a
54symmetric pair kernel `h`. -/
55def modulatedEnergy (h : Torus d → Torus d → ℝ) (X : Config d N) (μ : Measure (Torus d)) : ℝ :=
56  (1 / (2 * (N : ℝ) ^ 2)) * ∑ i, ∑ j ∈ Finset.univ.erase i, h (X i) (X j)
57    - (1 / (N : ℝ)) * ∑ i, ∫ y, h (X i) y ∂μ
58    + (1 / 2) * ∫ x, ∫ y, h x y ∂μ ∂μ

61def modulatedEnergyKernel (g : Torus d → ℝ) (X : Config d N) (μ : Measure (Torus d)) : ℝ :=
62  modulatedEnergy (fun x y => g (x - y)) X μ

109/-- The canonical Gibbs ensemble `P_{N,β}` of `H:eq:canonical` (lines 907-912), with
110Hamiltonian `∑_i V(x_i) + N⁻¹ ∑_{i<j} g(x_i - x_j)`. -/
111def canonicalHamiltonian (g : Torus d → ℝ) (V : Torus d → ℝ) (X : Config d N) : ℝ :=

129/-- `o_N = C₀ N^{s/d - 1} + 1_{s=0} log N /(2dN)` (`H:eq:energy-completion`, line 551). -/
130def oN (d : ℕ) (s : ℝ) (C₀ : ℝ) (N : ℕ) : ℝ :=
131  C₀ * (N : ℝ) ^ (s / d - 1) + (if s = 0 then Real.log N / (2 * d * N) else 0)

133/-- `δ_N` (source lines 720-735): `N⁻¹` (`s < d/2`), `N⁻¹ log(eN)` (`s = d/2`),
134`N^{1-d/s}` (`s > d/2`). -/
135def deltaN (d : ℕ) (s : ℝ) (N : ℕ) : ℝ :=
136  if s < (d : ℝ) / 2 then (N : ℝ)⁻¹
137  else if s = (d : ℝ) / 2 then (N : ℝ)⁻¹ * Real.log (Real.exp 1 * N)
138  else (N : ℝ) ^ (1 - (d : ℝ) / s)
