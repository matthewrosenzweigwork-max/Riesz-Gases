5/-!
6# The linear approximation and the limiting field (main manuscript, Appendix A)
7
8Specification file SPEC-A.  Sources (`HRS_periodic_main_20261008T010000Z.tex`):
9* the prose `B:sec:main-surrogate` (L10260-10292): the linear approximation
10  `𝔖_N^t[φ⃗] = (A_N ⟨U_N^{0,t}φ_a, η°⟩ + G_a)_a` (`B:eq:surrogate-def`), the Gaussian covariance
11  `𝗏_N^t[φ_a,φ_b]` (`B:eq:surrogate-cov`), the field `η_{N,lin}^t = A_N V_N^{t,0} η° + G_N^t`
12  (`B:eq:surrogate-field-torus`), and the `N`-independent limit field obtained by replacing the
13  coefficient quadruple `(β_N, μ_{β_N}, A_N, 2σ_N²/(Nβ_N))` by `(β_∞, μ_{β_∞}, A, 2/β_*)`;
14* Proposition A.33 `prop:periodic-limit-field` (L11310-11318).
15
16## Representation decisions
17
18* The *laws* of the linear approximations are given as explicit measures: the law of
19  `A_N ⟨U φ, η°⟩ + G` with `G` Gaussian independent of `η°` is the additive convolution
20  (`Measure.conv`) of the law of `A_N ⟨U φ, η°⟩` with the centered Gaussian `N(0, 𝗏)`.  This makes the
21  comparison statements independent of the existence of an independent `G` on the given probability
22  space (the TeX "`G_N^t` is chosen independent of `𝓕_0^in`").
23* The `H^{-γ}`-valued Gaussian field `G_N^t` is any law `νG` that is a centered Gaussian measure on
24  `NegSobolev d γ` with covariance form `HRS.noiseCovariance` (`IsLinearFieldNoise`); it is unique
25  (Gaussian laws on a separable Hilbert space are determined by their covariance on a dense class of
26  tests), and its existence is Proposition A.33 (`periodic_limit_field`).  Statements are made for
27  every such `νG`.
28* The transported initial field `V_N^{t,0} η°` is `transportField`:
29  `⟨e_k, V^{t,0} F⟩ := ⟨U^{0,t} e_k, F⟩` on the real trigonometric system `e_k = realMode k`
30  (zero mode dropped, as for all of `NegSobolev`), with `U` the project's classical adjoint
31  propagator `HRS.propagator`.
32* Weak convergence of laws is `ConvergesInLaw` (the portmanteau form: convergence of the integrals of
33  bounded continuous functions).
34-/

51variable {Ω : Type*} [MeasurableSpace Ω] {N : ℕ} {ι : Type*} [Fintype ι] [DecidableEq ι]

53/-- The particle statistics `(⟨φ_a, σ Ξ_N^{t_a}⟩)_a`, `Ξ_N^t = μ_N^t - μ_{β_N}^t`
54(`μ_N^t` the empirical measure of `X_N^t`, `μ_{β_N}^t = ρ t dx`), a vector in `ℝ^ι`. -/
55def fluctStat (σ : ℝ) (X : ℝ≥0 → Ω → Config d N) (ρ : ℝ → Torus d → ℝ) (ts : ι → ℝ)
56    (φ : ι → Torus d → ℝ) (ω : Ω) : EuclideanSpace ℝ ι :=
57  WithLp.toLp 2 fun a =>
58    σ * (pair (empirical (X (Real.toNNReal (ts a)) ω)) (φ a) - pair (withDensityReal (ρ (ts a))) (φ a))
59
60/-- The law of the particle statistics. -/
61def fluctStatLaw (P : Measure Ω) (σ : ℝ) (X : ℝ≥0 → Ω → Config d N) (ρ : ℝ → Torus d → ℝ)
62    (ts : ι → ℝ) (φ : ι → Torus d → ℝ) : Measure (EuclideanSpace ℝ ι) :=
63  law P (fluctStat σ X ρ ts φ)
64
65end Statistics

67section Surrogate

69variable {ι : Type*} [Fintype ι] [DecidableEq ι]

71/-- The covariance matrix of the Gaussian blocks `(G_a)` of `B:eq:surrogate-def`
72(`B:eq:surrogate-cov`, TeX L10272-10276, with the cross-covariance of the multi-time recipe of
73`B:rem:multitime`): `2q ∫_0^{t_a ∧ t_b} ⟨∇U^{r,t_a}φ_a · ∇U^{r,t_b}φ_b, μ^r⟩ dr`,
74`q = σ²/(Nβ)` (`2q = 2σ²/(Nβ)`); the same expression as in `HRS.IsLinearFluctuationSolution`. -/
75def surrogateCov (D : DynData d) (ρ : ℝ → Torus d → ℝ) (q : ℝ) (ts : ι → ℝ)
76    (φ : ι → Torus d → ℝ) : Matrix ι ι ℝ :=
77  Matrix.of fun a b =>
78    2 * q * ∫ r in Set.Icc 0 (min (ts a) (ts b)), ∫ x,
79      inner ℝ (Torus.grad (propagator D ρ r (ts a) (φ a)) x)
80        (Torus.grad (propagator D ρ r (ts b) (φ b)) x) * ρ r x
81
82/-- The law of the linear approximation `𝔖_N^t[φ⃗] = (A ⟨U^{0,t_a}φ_a, η°⟩ + G_a)_a`
83(`B:eq:surrogate-def`, TeX L10266-10270), `G` centered Gaussian with covariance `surrogateCov`
84independent of `η°`: the convolution of the law of `(A η°(U^{0,t_a}φ_a))_a` under `P` with the
85centered Gaussian of covariance `surrogateCov D ρ q`.  `η` is the random functional `η°`,
86`D, ρ` the coefficients (`β_N, μ_{β_N}`), `A = A_N`, `q = q_N`. -/
87def surrogateStatLaw {Ω : Type*} [MeasurableSpace Ω] (D : DynData d) (ρ : ℝ → Torus d → ℝ)
88    (A q : ℝ) (ts : ι → ℝ) (φ : ι → Torus d → ℝ) (P : Measure Ω)
89    (η : Ω → (Torus d → ℝ) → ℝ) : Measure (EuclideanSpace ℝ ι) :=
90  law P (fun ω => (WithLp.toLp 2 fun a => A * η ω (propagator D ρ 0 (ts a) (φ a)) :
91      EuclideanSpace ℝ ι)) ∗ multivariateGaussian 0 (surrogateCov D ρ q ts φ)
92
93end Surrogate

95/-! ### Fields -/
96
97/-- The transported initial field `V^{t,0} F` (the transpose of the adjoint propagator `U^{0,t}`):
98`⟨e_k, V^{t,0} F⟩ = ⟨U^{0,t} e_k, F⟩` for the real trigonometric system `e_k = realMode k`, as an
99element of `E_γ = NegSobolev d γ`. -/
100def transportField (D : DynData d) (ρ : ℝ → Torus d → ℝ) (t γ : ℝ) (F : NegSobolev d γ) :
101    NegSobolev d γ :=
102  NegSobolev.ofSeq γ fun k => NegSobolev.eval F (propagator D ρ 0 t (Torus.realMode k))
103
104/-- `νG` is the law of the Gaussian field `G^t` of the linear approximation at time `t`: a centered
105Gaussian probability measure on `E_γ = H^{-γ}_0` with covariance
106`2q ∫_0^t ⟨∇U^{r,t}φ·∇U^{r,t}ψ, μ^r⟩ dr` (`HRS.noiseCovariance`, `B:eq:surrogate-cov`),
107`q = σ_N²/(Nβ_N)` (resp. `β_*⁻¹` for the limit). -/
108def IsLinearFieldNoise (D : DynData d) (ρ : ℝ → Torus d → ℝ) (q t : ℝ) (γ : ℝ)
109    (νG : Measure (NegSobolev d γ)) : Prop :=
110  IsProbabilityMeasure νG ∧ IsCenteredGaussianWithCov γ νG (noiseCovariance D ρ q t)
111
112/-- The law of the particle field `σ Ξ_N^t = σ(μ_N^t - μ_{β_N}^t)` in `E_γ`. -/
113def particleFieldLaw {Ω : Type*} [MeasurableSpace Ω] {N : ℕ} (P : Measure Ω) (σ : ℝ) (γ : ℝ)
114    (X : ℝ≥0 → Ω → Config d N) (ρ : ℝ → Torus d → ℝ) (t : ℝ) : Measure (NegSobolev d γ) :=
115  law P (fun ω => σ • NegSobolev.ofMeasureDiff γ (empirical (X (Real.toNNReal t) ω))
116    (withDensityReal (ρ t)))
117
118/-- The law of the linear approximation field `η_{N,lin}^t = A V^{t,0} η° + G^t`
119(`B:eq:surrogate-field-torus`, TeX L10279): the convolution of the law of `A V^{t,0} η°`
120(`η° = η` a `NegSobolev d γ`-valued random element on `(Ω, P)`) with the Gaussian law `νG` of `G^t`
121(independent of `η°`). -/
122def linearFieldLaw {Ω : Type*} [MeasurableSpace Ω] (P : Measure Ω) (D : DynData d)
123    (ρ : ℝ → Torus d → ℝ) (A t : ℝ) (γ : ℝ) (η : Ω → NegSobolev d γ)
124    (νG : Measure (NegSobolev d γ)) : Measure (NegSobolev d γ) :=
125  haveI := InitialDataAux.negSobolev_secondCountable d γ
126  law P (fun ω => A • transportField D ρ t γ (η ω)) ∗ νG
127
128/-- The law of the `N`-independent limit field `η^t` (TeX L10285-10290): the coefficients are replaced
129by `(β_∞, μ_{β_∞}, A, 2/β_*)` (`LimitCoeffs`), `η°` has the fixed law `νin` (`η° ∼ νin`, an
130`N`-independent initial law), and `νGlim` is the Gaussian law of the limiting block (the cylindrical
131Wiener process is independent of `η°`). -/
132def limitFieldLaw (S : MainData d) (L : LimitCoeffs d) (γ : ℝ) (νin : Measure (NegSobolev d γ))
133    (t : ℝ) (νGlim : Measure (NegSobolev d γ)) : Measure (NegSobolev d γ) :=
134  linearFieldLaw νin (L.dyn S) L.ρ L.A t γ id νGlim
135
136/-- Weak convergence of laws (portmanteau form): `∫ f dμ_N → ∫ f dν` for every bounded continuous
137real `f`.  For random elements this is Mathlib's `TendstoInDistribution`. -/
138def ConvergesInLaw {E : Type*} [TopologicalSpace E] [MeasurableSpace E] (μ : ℕ → Measure E)
139    (ν : Measure E) : Prop :=
140  ∀ f : E →ᵇ ℝ, Tendsto (fun N => ∫ x, f x ∂(μ N)) atTop (𝓝 (∫ x, f x ∂ν))

142/-! ### Proposition A.33 -/
143
144/-- A distribution-valued solution of the linearized equation `H:eq:linear-equation` in mild form,
145*without* a path-continuity requirement: the same as `HRS.IsLinearFluctuationSolution` minus its
146field `cont`.  Proposition A.33 (TeX L11310-11318) only asserts a "distribution-valued solution";
147path continuity is claimed only for the continuous version in `H^{-(γ+2)}` (Proposition A.11,
148L10386-10391, which uses `IsLinearFluctuationSolution`). -/
149structure IsLinearFluctuationProcess {Ω : Type*} [MeasurableSpace Ω] (P : Measure Ω)
150    (D : DynData d) (ρ : ℝ → Torus d → ℝ) (m q T : ℝ) (Y₀ : Ω → NegSobolev d m)
151    (Y : ℝ → Ω → NegSobolev d m) : Prop where
152  init : ∀ᵐ ω ∂P, Y 0 ω = Y₀ ω
153  meas : ∀ t, Measurable (Y t)
154  gaussian_part : ∀ (p : ℕ) (ts : Fin p → ℝ) (φ : Fin p → Torus d → ℝ),
155    (∀ a, ts a ∈ Set.Icc 0 T) → (∀ a, Torus.IsSmooth (φ a)) →
156    let G : Ω → EuclideanSpace ℝ (Fin p) := fun ω => WithLp.toLp 2 fun a =>
157      NegSobolev.eval (Y (ts a) ω) (φ a) - NegSobolev.eval (Y₀ ω) (propagator D ρ 0 (ts a) (φ a))
158    ProbabilityTheory.IndepFun G Y₀ P ∧
159    law P G = ProbabilityTheory.multivariateGaussian 0 (Matrix.of fun a b =>
160      2 * q * ∫ r in Set.Icc 0 (min (ts a) (ts b)), ∫ x,
161        inner ℝ (Torus.grad (propagator D ρ r (ts a) (φ a)) x)
162          (Torus.grad (propagator D ρ r (ts b) (φ b)) x) * ρ r x)
163
164/-- Proposition A.33 `M:prop:periodic-limit-field` (TeX L11310-11318).  Under the hypotheses of the
165periodic field comparison (the adjoint hypotheses of Proposition 3.9 at the level `γ = m_*+d+3`,
166`AdjointData`, with regime `Rg`) and a noise level `q ≥ 0` (**technical**: `q = σ_N²/(Nβ_N) ≥ 0`
167makes the covariance positive semidefinite):
168(1) for every `t ∈ [0,T]` the covariance `B:eq:surrogate-cov` defines a centered Gaussian field in
169`H^{-γ}(𝕋^d)` (`IsLinearFieldNoise`); the independence from the initial fluctuation field is part
170of the solution concept (`gaussian_part`);
171(2) for every initial law `ν₁` on `E_γ` there is a distribution-valued solution `Y` of the linearized
172equation (`IsLinearFluctuationProcess`: initial field plus stochastic convolution, mild form) with
173`Y⁰ ∼ ν₁`, and it is unique: any two such solutions have the same finite-dimensional laws on smooth
174tests, `Law (Y_{t_a}(φ_a))_a` (the TeX asserts uniqueness among distribution-valued solutions, via the
175adjoint identity and density of smooth tests, and no path continuity here).
176The same statement applied to the limiting coefficients `(β_∞, μ_{β_∞}, β_*⁻¹)` gives the limiting
177field (the proposition is stated for arbitrary admissible coefficients `(D, ρ, q)`). -/
178theorem periodic_limit_field (hd : 1 ≤ d) (Rg : PDE.Regime) (ν₀ MB Λ VB : ℝ)
179    (D : DynData d) (T : ℝ) (ρ : ℝ → Torus d → ℝ)
180    (hA : PDE.AdjointData D.s (gammaIdx d) Rg MB Λ VB D ν₀ T ρ) (q : ℝ) (hq : 0 ≤ q) :
181    (∀ t ∈ Set.Icc (0 : ℝ) T, ∃ νG : Measure (NegSobolev d (gammaIdx d : ℝ)),
182        IsLinearFieldNoise D ρ q t (gammaIdx d : ℝ) νG) ∧
183    ∀ ν₁ : Measure (NegSobolev d (gammaIdx d : ℝ)), IsProbabilityMeasure ν₁ →
184      (∃ (Ω' : Type) (_ : MeasurableSpace Ω') (P' : Measure Ω') (_ : IsProbabilityMeasure P')
185          (Y₀ : Ω' → NegSobolev d (gammaIdx d : ℝ)) (Y : ℝ → Ω' → NegSobolev d (gammaIdx d : ℝ)),
186          Measurable Y₀ ∧ law P' Y₀ = ν₁ ∧
187          IsLinearFluctuationProcess P' D ρ (gammaIdx d : ℝ) q T Y₀ Y) ∧
188      ∀ (Ω₁ Ω₂ : Type) (_ : MeasurableSpace Ω₁) (_ : MeasurableSpace Ω₂)
189          (P₁ : Measure Ω₁) (P₂ : Measure Ω₂) (_ : IsProbabilityMeasure P₁)
190          (_ : IsProbabilityMeasure P₂)
191          (Y₀₁ : Ω₁ → NegSobolev d (gammaIdx d : ℝ)) (Y₀₂ : Ω₂ → NegSobolev d (gammaIdx d : ℝ))
192          (Y₁ : ℝ → Ω₁ → NegSobolev d (gammaIdx d : ℝ))
193          (Y₂ : ℝ → Ω₂ → NegSobolev d (gammaIdx d : ℝ)),
194        law P₁ Y₀₁ = ν₁ → law P₂ Y₀₂ = ν₁ →
195        IsLinearFluctuationProcess P₁ D ρ (gammaIdx d : ℝ) q T Y₀₁ Y₁ →
196        IsLinearFluctuationProcess P₂ D ρ (gammaIdx d : ℝ) q T Y₀₂ Y₂ →
197        ∀ (p : ℕ) (ts : Fin p → ℝ) (φ : Fin p → Torus d → ℝ),
198          (∀ a, ts a ∈ Set.Icc (0 : ℝ) T) → (∀ a, Torus.IsSmooth (φ a)) →
199          law P₁ (fun ω => (WithLp.toLp 2 fun a => NegSobolev.eval (Y₁ (ts a) ω) (φ a) :
200              EuclideanSpace ℝ (Fin p))) =
201            law P₂ (fun ω => (WithLp.toLp 2 fun a => NegSobolev.eval (Y₂ (ts a) ω) (φ a) :
202              EuclideanSpace ℝ (Fin p))) := by
203  have hJ : PDE.mStar d ≤ gammaIdx d := by unfold gammaIdx; omega
204  have hJ2 : PDE.mStar d + d ≤ gammaIdx d := by unfold gammaIdx; omega
205  let S : HRS.A33.A33Setup d :=
206    { J := gammaIdx d, hJ := hJ, hJ2 := hJ2, hd := hd, R := Rg, MB := MB, Λ := Λ, VB := VB,
207      ν₀ := ν₀, D := D, T := T, ρ := ρ, hA := hA }
208  obtain ⟨e⟩ := HRS.A33.a33_exists_equiv hd
209  refine ⟨?_, fun ν₁ hν₁ => ?_⟩
210  · intro t ht
211    obtain ⟨νG, hp, hc, hcov⟩ := HRS.A33.a33_noise_exists S q hq ht
212    refine ⟨νG, ?_⟩
213    unfold IsLinearFieldNoise IsCenteredGaussianWithCov
214    exact ⟨hp, hc, hcov⟩
215  · haveI := hν₁
216    refine ⟨?_, ?_⟩
217    · obtain ⟨Ω', m, P', hP', Y₀, Y, hY₀m, hlaw, hY⟩ := HRS.A33.a33_exists_process S q hq ν₁
218      exact ⟨Ω', m, P', hP', Y₀, Y, hY₀m, hlaw, ⟨hY.init, hY.meas, hY.gauss⟩⟩
219    · intro Ω₁ Ω₂ _ _ P₁ P₂ _ _ Y₀₁ Y₀₂ Y₁ Y₂ hl₁ hl₂ h₁ h₂ p ts φ hts hφ
220      exact HRS.A33.a33_unique S q e P₁ P₂ ν₁ Y₀₁ Y₀₂ Y₁ Y₂ hl₁ hl₂
221        ⟨h₁.init, h₁.meas, h₁.gaussian_part⟩ ⟨h₂.init, h₂.meas, h₂.gaussian_part⟩ p ts φ hts hφ
