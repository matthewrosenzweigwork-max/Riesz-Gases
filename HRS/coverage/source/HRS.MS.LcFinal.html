13`local_decay_proof : type_of% @HRS.Whole.local_decay` (the frozen statement, with `hk : 2 ≤ k`).

15Proof.  The constant `C` is the one of `lc_engine` (quantified before `ν, c, B`).  Data given on
16`[0,T]` only are clamped to `c (min · T)`, `B (min · T)` (`lcHyp_clamp`, `lcDecay_clamp`), which agree
17with the original data on `[0,T]`.  On a probability space with a Brownian filtration
18(`exists_isBrownianFiltration_type`):
19* existence: `lc_backward` for the clamped data and `f ∈ C^2` with bounded derivatives of order `≤ 2`
20  (from `decayNormE k b f < ⊤`, `k ≥ 2`, `lcf_bdd_iteratedFDeriv`), transported to the original drift by
21  `lcf_solution_congr` (a local backward solution only sees the drift on `[0,t]`);

23* the `C^k` regularity and the bound `decayNormE k b (u r) ≤ ofReal C * decayNormE k b f` are those of
24  `lc_engine` for `lcSg`.
25No statement adjustment is made.
26-/

36/-- **S4.11 `local_decay`**: proof (tie in `LcFinalTie`). -/
37theorem local_decay_proof : type_of% @HRS.Whole.local_decay := by
38  intro d hd s hs Q k hk T νstar bexp Λc ΛB hb
39  obtain ⟨C, hC⟩ := lc_engine.{0} s hs Q k T νstar bexp Λc ΛB hb
40  refine ⟨C, fun ν c B hνc hν hc hB hcb hBs t ht f hf hfd => ?_⟩
41  have hT0 : 0 ≤ T := ht.1.trans ht.2
42  obtain ⟨Ω, mΩ, P, hP, ℱ, W, hW⟩ := exists_isBrownianFiltration_type (Fin d)
43  haveI := hP
44  set c' : ℝ → Euc d := fun v => c (min v T) with hc'
45  set B' : ℝ → Euc d → Euc d := fun v => B (min v T) with hB'
46  have hH : LcHyp Q k Λc ΛB c' B' :=
47    lcHyp_clamp hT0 hc hB hcb fun v hv =>
48      ⟨by exact_mod_cast (hBs v hv).1, fun j hj x => (hBs v hv).2.1 j hj x⟩
49  have hH2 : LcHyp Q 2 Λc ΛB c' B' :=
50    lcHyp_clamp hT0 hc hB hcb fun v hv =>
51      ⟨(hBs v hv).1.of_le (by exact_mod_cast (by omega : 2 + 1 ≤ k + 1)),
52        fun j hj x => (hBs v hv).2.1 j (by omega) x⟩
53  have hdec := lcDecay_clamp hT0 (fun v hv => (hBs v hv).2.2)
54  -- the data
55  have hfk : ContDiff ℝ (k : WithTop ℕ∞) f := by
56    have : max k 2 = k := Nat.max_eq_left hk
57    rw [this] at hf
58    exact hf
59  have hf2 : ContDiff ℝ 2 f := hfk.of_le (by exact_mod_cast hk)
60  obtain ⟨M0, hM0⟩ := lcf_bdd_iteratedFDeriv hb.le hfd (Nat.zero_le k)
61  obtain ⟨M1, hM1⟩ := lcf_bdd_iteratedFDeriv hb.le hfd (by omega : 1 ≤ k)
62  obtain ⟨M2, hM2⟩ := lcf_bdd_iteratedFDeriv hb.le hfd hk
63  have hM : ∀ i ≤ 2, ∀ y, ‖iteratedFDeriv ℝ i f y‖ ≤ max (max M0 M1) M2 := by
64    intro i hi y
65    interval_cases i
66    · exact (hM0 y).trans ((le_max_left _ _).trans (le_max_left _ _))
67    · exact (hM1 y).trans ((le_max_right _ _).trans (le_max_left _ _))
68    · exact (hM2 y).trans (le_max_right _ _)
69  have hν0 : ∀ r, 0 ≤ ν r := fun r => (hν r).1
70  have hνs : ∀ r, ν r ≤ νstar := fun r => (hν r).2
71  -- the drift agrees on `[0,t]`
72  have hdr : ∀ r ∈ Icc 0 t, ∀ x : Euc d,
73      -(Torus.matVec Q x) - c' r + B' r x = -(Torus.matVec Q x) - c r + B r x := by
74    intro r hr x
75    have : min r T = r := min_eq_left (hr.2.trans ht.2)
76    simp only [hc', hB', this]
77  have hsol : IsLocalBackwardSolution ν (fun r x => -(Torus.matVec Q x) - c r + B r x) t f
78      (fun r => lcSg P W ν Q c' B' r t f) :=
79    lcf_solution_congr hdr (lc_backward hW hνc hν0 hνs hH2 ht.1 hf2 hM)
80  refine ⟨⟨_, hsol⟩, fun u hu r hr => ?_⟩
81  have hBn : ∀ r ∈ Icc 0 t, ∀ x, ‖B r x‖ ≤ ΛB := by
82    intro r hr x
83    have := (hBs r ⟨hr.1, hr.2.trans ht.2⟩).2.1 0 (Nat.zero_le _) x
84    rwa [norm_iteratedFDeriv_zero] at this
85  have huq := lc_unique ht.1 (fun r _ => hν r) (fun r hr => hcb r ⟨hr.1, hr.2.trans ht.2⟩) hBn
86    hu hsol
87  have hueq : u r = lcSg P W ν Q c' B' r t f := funext fun x => huq r hr x
88  rw [hueq]
89  exact hC P ℱ W ν c' B' hW hνc hν hH hdec r t hr.1 hr.2 ht.2 f hfk hfd
