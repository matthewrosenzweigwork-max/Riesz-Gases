25      eLpNorm (stationaryRemainder d s β V ρβ N φ) 2 (canonicalGibbs (periodicRiesz d s) V N β) ≤

32      (fun x => (CoercAux.J_bounds (by omega : 1 ≤ d) hs0 hsd one_pos hρ.density hΛ0.le hΛ x).1)

51    rel513_measurable_comm hsd (rel513_grad_continuous hφ)
52  have hsym : ∀ x y, h x y = h y x := fun x y => rel513_symm_comm hsd _ x y
53  have hhh : ∀ x y, x ≠ y → |h x y| ≤ B * singEnvelope s (Torus.eucDist x y) :=
54    fun x y hxy => hCenv L (Torus.grad φ) hL0 (rel513_gradLip hφ) x y hxy
55  have hlin := hKlin N hN B hB0 h hhm hhh
56  have hquad := hKq N hN B hB0 h hhm hsym hhh
57  have hfm : Measurable (fun y => Quad.Hc h μ y - Quad.Hc h m y) :=
58    (Quad.measurable_Hc hhm μ).sub (Quad.measurable_Hc hhm m)
59  have hfun : stationaryRemainder d s β V ρβ N φ = fun X : Config d N =>
60      Real.sqrt N * ((∫ y, (Quad.Hc h μ y - Quad.Hc h m y) ∂(empirical X))
61          - ∫ y, (Quad.Hc h μ y - Quad.Hc h m y) ∂m)
62      - Real.sqrt N * (modulatedEnergy h X m - ∫ X', modulatedEnergy h X' m ∂Q) := by
63    funext X
64    simp only [stationaryRemainder, hP N]
65    rfl
66  rw [hfun, hP N]
67  have hmin := eLpNorm_sub_le (p := 2) (μ := Q)
68    (f := fun X : Config d N => Real.sqrt N * ((∫ y, (Quad.Hc h μ y - Quad.Hc h m y) ∂(empirical X))
69          - ∫ y, (Quad.Hc h μ y - Quad.Hc h m y) ∂m))
