4/-!
5# MS-COMM: the commutator kernel of a Lipschitz field is in the S3.3 and S3.5 kernel classes
6
7Proves `HRS.Whole.commutatorKernel_isBounded` (S3.3, `B = sΛ`, `0 < s`) and
8`HRS.Whole.commutatorKernel_log_isBounded` (S3.5, `B = Λ`, `s = 0`) as
9`commutatorKernel_isBounded_proof` and `commutatorKernel_log_isBounded_proof`.
10
11Mathematics: `h(x,y) = ⟨v(x)-v(y), ∇g(x-y)⟩`.
12* symmetry: `∇g` is odd off the origin (explicit formula `movtr_rieszKernelGrad_eq`), and
13  `h(x,x) = 0` because `v(x)-v(x) = 0`;
14* measurability: `v` is continuous, `∇g` is measurable (`movtr_rieszKernelGrad_measurable`);
15* bound: `|h| ≤ Λ‖z‖·‖∇g(z)‖` with `‖∇g(z)‖ = σ‖z‖^{-s-1}` (`movtr_rieszKernelGrad_norm`);
16  for `s > 0` this gives `|h| ≤ sΛ‖z‖^{-s}`; for `s = 0` it gives `|h| ≤ Λ`, and the
17  log kernel `K(x,y) = -log‖x-y‖ + ℓ(x) + ℓ(y)` is nonnegative because
18  `‖x-y‖² ≤ (1+‖x‖²)(1+‖y‖²)` (the difference is `(‖x‖‖y‖-1)² + 2(‖x‖‖y‖+⟨x,y⟩) ≥ 0`).
19-/

32/-- The force is odd off the origin. -/
33lemma comm_grad_odd {z : Euc d} (hz : z ≠ 0) :
34    rieszKernelGrad d s (-z) = -rieszKernelGrad d s z := by
35  rw [movtr_rieszKernelGrad_eq s (neg_ne_zero.mpr hz), movtr_rieszKernelGrad_eq s hz,
36    norm_neg, smul_neg]

38/-- Symmetry of the commutator kernel. -/
39lemma comm_symm (v : Euc d → Euc d) (x y : Euc d) :
40    commutatorKernel d s v x y = commutatorKernel d s v y x := by
41  by_cases hxy : x = y
42  · subst hxy; rfl
43  have hz : x - y ≠ 0 := sub_ne_zero.mpr hxy
44  unfold commutatorKernel
45  rw [show y - x = -(x - y) by abel, comm_grad_odd hz, inner_neg_right,
46    ← neg_sub (v x) (v y), inner_neg_left, neg_neg]

55/-- The Lipschitz bound `‖v x - v y‖ ≤ Λ‖x - y‖`. -/
56lemma comm_lip {Λ : NNReal} {v : Euc d → Euc d} (hv : LipschitzWith Λ v) (x y : Euc d) :
57    ‖v x - v y‖ ≤ (Λ : ℝ) * ‖x - y‖ := by
58  have := hv.dist_le_mul x y
59  rwa [dist_eq_norm, dist_eq_norm] at this

61/-- The bound for `s > 0`: `|h| ≤ sΛ(1 + |x-y|^{-s})`, off the diagonal. -/
62lemma comm_bound_pos (hs0 : 0 < s) (Λ : NNReal) (v : Euc d → Euc d)
63    (hv : LipschitzWith Λ v) (x y : Euc d) (hxy : x ≠ y) :
64    |commutatorKernel d s v x y| ≤ s * (Λ : ℝ) * (1 + rieszWeight s (x - y)) := by
65  have hz : x - y ≠ 0 := sub_ne_zero.mpr hxy
66  have hzpos : 0 < ‖x - y‖ := norm_pos_iff.mpr hz
67  have hs : s ≠ 0 := hs0.ne'
68  have hΛ : (0 : ℝ) ≤ (Λ : ℝ) := NNReal.coe_nonneg Λ
69  have hw : rieszWeight s (x - y) = ‖x - y‖ ^ (-s) := by simp [rieszWeight, hs]
70  have hG : ‖rieszKernelGrad d s (x - y)‖ = s * ‖x - y‖ ^ (-s - 1) := by
71    rw [movtr_rieszKernelGrad_norm s hs0.le hz]
72    simp [movtr_sigma, hs]
73  have hpow : ‖x - y‖ ^ (-s) = ‖x - y‖ ^ (-s - 1) * ‖x - y‖ := by
74    rw [← Real.rpow_add_one hzpos.ne']
75    congr 1
76    ring
77  have key : |commutatorKernel d s v x y| ≤ s * (Λ : ℝ) * ‖x - y‖ ^ (-s) := by
78    unfold commutatorKernel
79    calc |inner ℝ (v x - v y) (rieszKernelGrad d s (x - y))|
80        ≤ ‖v x - v y‖ * ‖rieszKernelGrad d s (x - y)‖ := abs_real_inner_le_norm _ _
81      _ ≤ ((Λ : ℝ) * ‖x - y‖) * (s * ‖x - y‖ ^ (-s - 1)) :=
82          mul_le_mul (comm_lip hv x y) hG.le (norm_nonneg _) (by positivity)
83      _ = s * (Λ : ℝ) * ‖x - y‖ ^ (-s) := by rw [hpow]; ring
84  rw [hw]
85  have hsΛ : 0 ≤ s * (Λ : ℝ) := mul_nonneg hs0.le hΛ
86  rw [mul_add, mul_one]
87  linarith [key]

89/-- Nonnegativity of the spherical log kernel `K(x,y) = -log|x-y| + ℓ(x) + ℓ(y)`. -/
90lemma comm_log_nonneg (x y : Euc d) : 0 ≤ sphericalLogKernel x y := by
91  by_cases hxy : x = y
92  · subst hxy
93    unfold sphericalLogKernel logWeight
94    have h1 : 0 ≤ Real.log (1 + ‖x‖ ^ 2) := Real.log_nonneg (by have := sq_nonneg ‖x‖; linarith)
95    simp only [sub_self, norm_zero, Real.log_zero, neg_zero, zero_add]
96    linarith
97  have hc : ‖x - y‖ ^ 2 ≤ (1 + ‖x‖ ^ 2) * (1 + ‖y‖ ^ 2) := by
98    have h1 := norm_sub_sq_real x y
99    have h2 : -(‖x‖ * ‖y‖) ≤ inner ℝ x y := by
100      have := abs_real_inner_le_norm x y
101      linarith [neg_abs_le (inner ℝ x y)]
102    nlinarith [sq_nonneg (‖x‖ * ‖y‖ - 1)]
103  have hpos1 : 0 < 1 + ‖x‖ ^ 2 := by positivity
104  have hpos2 : 0 < 1 + ‖y‖ ^ 2 := by positivity
105  have hzpos : 0 < ‖x - y‖ := norm_pos_iff.mpr (sub_ne_zero.mpr hxy)
106  have h1 : Real.log (‖x - y‖ ^ 2) ≤ Real.log ((1 + ‖x‖ ^ 2) * (1 + ‖y‖ ^ 2)) :=
107    Real.log_le_log (by positivity) hc
108  rw [Real.log_pow, Real.log_mul hpos1.ne' hpos2.ne'] at h1
109  unfold sphericalLogKernel logWeight
110  push_cast at h1
111  linarith

113/-- The bound for `s = 0`: `|h| ≤ Λ(1 + K)`, off the diagonal. -/
114lemma comm_bound_log (Λ : NNReal) (v : Euc d → Euc d)
115    (hv : LipschitzWith Λ v) (x y : Euc d) (hxy : x ≠ y) :
116    |commutatorKernel d 0 v x y| ≤ (Λ : ℝ) * (1 + sphericalLogKernel x y) := by
117  have hz : x - y ≠ 0 := sub_ne_zero.mpr hxy
118  have hzpos : 0 < ‖x - y‖ := norm_pos_iff.mpr hz
119  have hΛ : (0 : ℝ) ≤ (Λ : ℝ) := NNReal.coe_nonneg Λ
120  have hG : ‖rieszKernelGrad d 0 (x - y)‖ = ‖x - y‖⁻¹ := by
121    rw [movtr_rieszKernelGrad_norm 0 le_rfl hz]
122    norm_num [movtr_sigma, Real.rpow_neg_one]
123  have key : |commutatorKernel d 0 v x y| ≤ (Λ : ℝ) := by
124    unfold commutatorKernel
125    calc |inner ℝ (v x - v y) (rieszKernelGrad d 0 (x - y))|
126        ≤ ‖v x - v y‖ * ‖rieszKernelGrad d 0 (x - y)‖ := abs_real_inner_le_norm _ _
127      _ ≤ ((Λ : ℝ) * ‖x - y‖) * ‖x - y‖⁻¹ :=
128          mul_le_mul (comm_lip hv x y) hG.le (norm_nonneg _) (by positivity)
129      _ = (Λ : ℝ) := by rw [mul_assoc, mul_inv_cancel₀ hzpos.ne', mul_one]
130  have hK := comm_log_nonneg (d := d) x y
131  rw [mul_add, mul_one]
132  linarith [key, mul_nonneg hΛ hK]

134/-- Theorem S3.3 commutator class membership (proof). -/
135theorem commutatorKernel_isBounded_proof (hs0 : 0 < s) (Λ : ℝ≥0) (v : Euc d → Euc d)
136    (hv : LipschitzWith Λ v) : IsBoundedPairKernel s (s * Λ) (commutatorKernel d s v) :=
137  ⟨comm_meas v hv.continuous.measurable, comm_symm v,
138    fun x y hxy => comm_bound_pos hs0 Λ v hv x y hxy⟩
