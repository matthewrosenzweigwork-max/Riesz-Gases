43/-- The set of `a` where the weighted coefficient sequence is in `ℓ²` is measurable. -/
44theorem mia_memℓp_meas {α : Type*} [MeasurableSpace α] {q : ℝ} (c : NonzeroLattice d → α → ℝ)
45    (hc : ∀ k, Measurable (c k)) :
46    MeasurableSet {a : α | Memℓp (fun k : NonzeroLattice d => Lattice.jp k.1 ^ (-q) * c k a) 2} := by
47  have key : ∀ a : α, Memℓp (fun k : NonzeroLattice d => Lattice.jp k.1 ^ (-q) * c k a) 2 ↔
48      ∃ M : ℕ, ∀ u : Finset (NonzeroLattice d),
49        ∑ k ∈ u, (Lattice.jp k.1 ^ (-q) * c k a) ^ 2 ≤ (M : ℝ) := by
50    intro a
51    have hn : ∀ k : NonzeroLattice d,
52        ‖Lattice.jp k.1 ^ (-q) * c k a‖ ^ ((2 : ℝ≥0∞).toReal) =
53          (Lattice.jp k.1 ^ (-q) * c k a) ^ 2 := by
54      intro k
55      rw [ENNReal.toReal_ofNat, Real.rpow_two, Real.norm_eq_abs, sq_abs]
56    rw [memℓp_gen_iff (by norm_num)]
57    simp only [hn]
58    constructor
59    · intro hs
60      refine ⟨⌈∑' k, (Lattice.jp k.1 ^ (-q) * c k a) ^ 2⌉₊, fun u => ?_⟩
61      exact (Summable.sum_le_tsum u (fun k _ => sq_nonneg _) hs).trans (Nat.le_ceil _)
62    · rintro ⟨M, hM⟩
63      exact summable_of_sum_le (fun k => sq_nonneg _) hM
64  have hset : {a : α | Memℓp (fun k : NonzeroLattice d => Lattice.jp k.1 ^ (-q) * c k a) 2} =
65      ⋃ M : ℕ, ⋂ u : Finset (NonzeroLattice d),
66        {a : α | ∑ k ∈ u, (Lattice.jp k.1 ^ (-q) * c k a) ^ 2 ≤ (M : ℝ)} := by
67    ext a
68    simp only [Set.mem_setOf_eq, Set.mem_iUnion, Set.mem_iInter]
69    exact key a
70  rw [hset]
71  refine MeasurableSet.iUnion fun M => MeasurableSet.iInter fun u => ?_
72  exact measurableSet_le
73    (Finset.measurable_sum u fun k _ => (measurable_const.mul (hc k)).pow_const 2) measurable_const

164/-- Transfer of `ContVersionTight` across processes that agree on `[0,T]`. -/
165theorem mia_cvt_transfer {E : Type*} [NormedAddCommGroup E] [MeasurableSpace E] [BorelSpace E]
166    {T : ℝ} {Ω : ℕ → Type*} [∀ N, MeasurableSpace (Ω N)] (P : ∀ N, Measure (Ω N))
167    (Z Z' Zc : ∀ N, ℝ → Ω N → E)
168    (hZ : ∀ N t ζ, t ∈ Set.Icc (0 : ℝ) T → Z' N t ζ = Z N t ζ)
169    (h : ContVersionTight T P Z' Zc) : ContVersionTight T P Z Zc := by
170  unfold ContVersionTight at h ⊢
171  obtain ⟨c1, c2, c3, c4, c5⟩ := h
172  refine ⟨fun N hN t ht => ?_, c2, c3, c4, fun htight => c5 fun t ht => ?_⟩
173  · filter_upwards [c1 N hN t ht] with ζ hζ
174    exact hζ.trans (hZ N t ζ ht)
175  · have hset : {μ : Measure E | ∃ N : ℕ, 1 ≤ N ∧ μ = (P N).map (Z' N t)} =
176        {μ : Measure E | ∃ N : ℕ, 1 ≤ N ∧ μ = (P N).map (Z N t)} := by
177      have hf : ∀ N, Z' N t = Z N t := fun N => funext fun ζ => hZ N t ζ ht
178      simp only [hf]
179    rw [hset]
180    exact htight t ht
