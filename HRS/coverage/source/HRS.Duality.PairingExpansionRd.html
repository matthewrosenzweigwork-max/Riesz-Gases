28/-- The symmetrized kernel `½ (∇f(x) - ∇f(y)) · M ∇g(x - y)` of `B:eq:B-symmetrized` on `ℝ^d`, `0` on the diagonal. -/
29def pairKernelRdB (f : Euc d → ℝ) (M : Matrix (Fin d) (Fin d) ℝ) (s : ℝ) (x y : Euc d) : ℝ :=
30  if x = y then 0 else
31    (1 / 2 : ℝ) * inner ℝ (gradient f x - gradient f y)
32      (Torus.matVec M (Whole.rieszKernelGrad d s (x - y)))

34/-- `B_f(α, λ)` in symmetrized, off-diagonal form on `ℝ^d`. -/
35def pairBRd (f : Euc d → ℝ) (M : Matrix (Fin d) (Fin d) ℝ) (s : ℝ) (α ν : Measure (Euc d)) : ℝ :=
36  ∫ p, pairKernelRdB f M s p.1 p.2 ∂(α.prod ν)

38/-- `B_f(ν - μ, ν - μ)` expanded bilinearly (four terms). -/
39def pairBSignedRd (f : Euc d → ℝ) (M : Matrix (Fin d) (Fin d) ℝ) (s : ℝ) (ν μ : Measure (Euc d)) : ℝ :=
40  pairBRd f M s ν ν - pairBRd f M s ν μ - pairBRd f M s μ ν + pairBRd f M s μ μ

42/-- `⟨φ, μ_N - μ⟩ = N⁻¹ ∑ φ(x_i) - ∫ φ dμ`. -/
43def pairEmpRd (φ : Euc d → ℝ) (X : Whole.EucConfig d N) (μ : Measure (Euc d)) : ℝ :=

51/-- Lemma 3.1 `B:lem:pairing-expansion` (TeX L1518-1524) on `𝔻 = ℝ^d`: for `f ∈ C²_b(ℝ^d)`, a probability density `ρ`
52(`μ = ρ dx`) in `C^{0,α}_b(ℝ^d)` with `α > max(0, s+1-d)` (and `∫⟨x⟩ dμ < ∞` if `s = 0`), and pairwise distinct
53`x_1, …, x_N`, the four quantities `B_f(μ,μ)`, `B_f(μ_N,μ)`, `B_f(μ,μ_N)` (absolutely convergent integrals), the
54convolution `∇g * μ` and `Div g * (μ M^*∇f)` (cancellation convention integrals) are finite and
55`B_f(μ_N,μ_N) - B_f(μ,μ) = ⟨∇f·M(∇g*μ), ρ_N⟩ - ⟨Div g*(μ M^*∇f), ρ_N⟩ + B_f(ρ_N,ρ_N)`.
56Same shape and conclusions as `HRS.Duality.pairing_expansion`.  Reading: `C^{0,α}` is the Hölder space of exponent
57`α ∈ (0,1]` (`hα1`; the torus statement leaves `α ≤ 1` implicit), boundedness of `ρ` is `hρb`, `⟨x⟩ = √(1+|x|²)`,
58`μ_N = N⁻¹∑δ_{x_i}` with `N ≥ 1`. -/
