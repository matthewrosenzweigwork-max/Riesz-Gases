39    {A : Matrix (Fin d) (Fin d) ℝ} (hA : A.transpose = -A) (hM : M = -(lam • (1 : Matrix (Fin d) (Fin d) ℝ)) + A) :
40    Torus.SymNonpos M := by

45    rw [h1, hM]
46    have : (-(lam • (1 : Matrix (Fin d) (Fin d) ℝ)) + A + (-(lam • (1 : Matrix (Fin d) (Fin d) ℝ)) + -A)) =

48      rw [two_smul]; abel

91  have hw := fcau_sub hd D hs0 hsd ρ t hS h₁ h₂
92  have hz := fcau_forward hd D hs0 hsd ρ t hS hw
93  have hzb := fcau_ulBounded hχ hw
94  have key := HRS.MS.Star.uloc_uniqueness_star hd D.s hs0 hsd m hm Rg T Λ ΛV G₀ MB νbar ν₀ χ hχ
95    D.M D.V D.invβ ρ hM hV hν hρ hRg t ht (fun σ => (fun r x => f₁ r x - f₂ r x) (t - σ)) hzb hz
96  intro r hr

104  have := congrFun h0 x

282      show max 0 (min r (tn (ns i))) = r
283      rw [min_eq_left (htn _).1.le, max_eq_right hr0]
284    rw [this]
285  have hgrad := s55m_grad_tendsto (hG1.2.2 1 (by norm_num) x)
286  have hcomp : Tendsto (fun i => gradient (F' i r) x a) atTop (𝓝 (gradient (g r) x a)) :=
287    ((EuclideanSpace.proj a : Euc d →L[ℝ] ℝ).continuous.tendsto _).comp hgrad
288  have e : gradComp a (propagatorW C r t₀ (ψ t₀)) x = gradient (g r) x a := by
289    rw [hfg r ⟨hr0, ht₀.1.le⟩]; rfl
290  rw [e]
291  refine hcomp.congr fun i => ?_
292  rw [hrle i]; rfl

296theorem s55m_continuousOn_star (b : WholeBounds d) (hb : b.Valid) (S : WholeSetup b) (N : ℕ)
297    (hReg : (∃ (Rg : PDE.Regime) (ν₀ : ℝ), RdRegimeHolds Rg d b.s b.M (S.invβ N) ν₀) ∨
298      UlocStarRegime d b.s b.M)
299    (α : MultiIndex d) (a : Fin d) {r : ℝ} (hr0 : 0 ≤ r) (x : Euc d) :
300    ContinuousOn (fun t => gradComp a (propagatorW (S.coeff N) r t
301      (generatorW (S.coeff N) t (oscTestFun b.m α))) x) (Set.Ioc r b.T) := by
302  intro t₀ ht₀
303  rw [ContinuousWithinAt, tendsto_iff_seq_tendsto]
304  intro tn htn
305  obtain ⟨hlim, hev⟩ := tendsto_nhdsWithin_iff.1 htn
306  obtain ⟨n₀, hn₀⟩ := eventually_atTop.1 hev
307  set tn' : ℕ → ℝ := fun n => tn (n + n₀) with htn'
308  have hlim' : Tendsto tn' atTop (𝓝 t₀) := hlim.comp (tendsto_add_atTop_nat n₀)
309  have hmem : ∀ n, tn' n ∈ Set.Ioc r b.T := fun n => hn₀ _ (by omega)
310  rw [← tendsto_add_atTop_iff_nat n₀]
311  refine tendsto_of_subseq_tendsto fun ns₁ hns₁ => ?_
312  obtain ⟨ns, hns, h⟩ := s55m_seq_star b hb S N hReg α hr0 ht₀ (tn := fun i => tn' (ns₁ i)) (fun i => hmem _)
313    (hlim'.comp hns₁) a x
314  exact ⟨ns, h⟩
