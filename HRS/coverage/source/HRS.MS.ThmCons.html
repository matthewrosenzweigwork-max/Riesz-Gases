4/-!
5# C-B: proof of `HRS.Whole.euc_thermal_moments` (`0 < s < d`, `d ≥ 2`)
6
7Both clauses follow from `thermalDensity ≤ C γ` and Gaussian integrability (`ThmConsAux2`).
8
9Also `cb_euc_thermal_gaussian_bounds_of_poincare`: the conclusion of `euc_thermal_gaussian_bounds` (two-sided Gaussian
10comparison and the Poincare inequality for `μ_β`) from the Gaussian Poincare inequality `CbGaussPoincare d a β κ` for the
11reference measure `γ dx` (the separate Maurey-Pisier packet C-POI; it is not in Mathlib or the repo).  The transfer from
12`γ` to `μ_β` is done here (`cb_poincare_transfer`) with `c = e^{-β M B}` and `C' = C (1 + κ)`; enlarging the upper constant
13is harmless since the upper bound only gets weaker.
14
15Also `cb_admissible_of_smooth`: `isAdmissiblePerturbation_thermal` follows from `ContDiff ℝ ∞ thermalDensity`
16(the first conjunct of the frozen, sorried `euc_thermal_regularity`) together with the lower Gaussian bound of
17`ThmConsAux1`; the smoothness itself is NOT proved here (C-REG packets).
18-/

38/-- `isAdmissiblePerturbation_thermal` from the smoothness of `thermalDensity` (supplied by C-REG):
39`√ρ ψ` is a smooth compactly supported (`ρ ≥ c γ > 0`), hence Schwartz, hence in every `Σ^m`. -/
40theorem cb_admissible_of_smooth (hd : 1 ≤ d) {s a β : ℝ} (hs0 : 0 < s) (hsd : s < d) (ha : 0 < a)
41    (hβ : 0 < β) (hsm : ContDiff ℝ ∞ (thermalDensity d s a β)) (εstar : ℝ) (hε : 0 < εstar)
42    (ψ : Euc d → ℝ) (hψ : ContDiff ℝ ∞ ψ) (hc : HasCompactSupport ψ) (h0 : ψ ≠ 0)
43    (hmean : ∫ x, ψ x * thermalDensity d s a β x = 0) :
44    ∃ ε₀ > 0, ∀ ε : ℝ, |ε| ≤ ε₀ →
45      IsAdmissiblePerturbation (thermalDensity d s a β) εstar ε ψ := by
46  obtain ⟨-, c, C, hc0, -, hcC⟩ := cb_thermal_two_sided hd hs0 hsd ha hβ
47  have hγ : ∀ x : Euc d, 0 < gaussianComparison d a β x := fun x => by
48    unfold gaussianComparison
49    have : 0 < β * a / (2 * Real.pi) := by positivity
50    positivity
51  have hpos : ∀ x, thermalDensity d s a β x ≠ 0 := fun x =>
52    (lt_of_lt_of_le (mul_pos hc0 (hγ x)) (hcC x).1).ne'
53  have hsq : ContDiff ℝ ∞ fun x => Real.sqrt (thermalDensity d s a β x) := hsm.sqrt hpos
54  have hg : ContDiff ℝ ∞ fun x => Real.sqrt (thermalDensity d s a β x) * ψ x := hsq.mul hψ
55  have hgc : HasCompactSupport fun x => Real.sqrt (thermalDensity d s a β x) * ψ x :=
56    hc.mul_left
57  refine isAdmissiblePerturbation_of_small _ εstar hε ψ hψ hc h0 hmean ?_
58  exact memOsc_schwartz _ (hgc.toSchwartzMap hg)

60/-! ### Transfer of a Gaussian Poincare inequality to `μ_β` -/

62/-- The Gaussian Poincare inequality for `γ_{βa} dx` with constant `κ / (β a)` (hypothesis of the conditional result). -/
63def CbGaussPoincare (d : ℕ) (a β κ : ℝ) : Prop :=
64  ∀ f : Euc d → ℝ, ContDiff ℝ 1 f → MemLp f 2 (HRS.Whole.withDensityReal (gaussianComparison d a β)) →
65    Integrable (fun x => ‖gradient f x‖ ^ 2) (HRS.Whole.withDensityReal (gaussianComparison d a β)) →
66      ProbabilityTheory.variance f (HRS.Whole.withDensityReal (gaussianComparison d a β)) ≤
67        κ / (β * a) * ∫ x, ‖gradient f x‖ ^ 2 ∂(HRS.Whole.withDensityReal (gaussianComparison d a β))

69lemma cb_withDensityReal_isProb {ρ : Euc d → ℝ} (hρ : HRS.Whole.IsProbDensity ρ) :
70    IsProbabilityMeasure (HRS.Whole.withDensityReal ρ) := by
71  constructor
72  unfold HRS.Whole.withDensityReal
73  rw [withDensity_apply _ MeasurableSet.univ, Measure.restrict_univ]
74  have hi : Integrable ρ := Integrable.of_integral_ne_zero (by rw [hρ.integral_one]; norm_num)
75  rw [← ofReal_integral_eq_lintegral_ofReal hi (ae_of_all _ hρ.nonneg), hρ.integral_one]
76  simp

78lemma cb_withDensityReal_le_smul {ρ ρ' : Euc d → ℝ} {C : ℝ} (hρ' : Measurable ρ') (hC : 0 ≤ C)
79    (h : ∀ x, ρ x ≤ C * ρ' x) :
80    HRS.Whole.withDensityReal ρ ≤ ENNReal.ofReal C • HRS.Whole.withDensityReal ρ' := by
81  unfold HRS.Whole.withDensityReal
82  have hm : Measurable fun x => ENNReal.ofReal (ρ' x) := hρ'.ennreal_ofReal
83  rw [← withDensity_smul _ hm]
84  refine withDensity_mono (ae_of_all _ fun x => ?_)
85  simp only [Pi.smul_apply, smul_eq_mul]
86  rw [← ENNReal.ofReal_mul hC]
87  exact ENNReal.ofReal_le_ofReal (h x)

89/-- Transfer of a Poincare inequality: if `c γ ≤ ρ ≤ C γ` and `γ dx` satisfies `Var ≤ κ/(βa) ∫|∇f|²`, then
90`ρ dx` satisfies `Var ≤ C (1+κ) / (c β a) ∫|∇f|²`. -/
91theorem cb_poincare_transfer {a β κ c C : ℝ} (ha : 0 < a) (hβ : 0 < β) (hκ : 0 ≤ κ) (hc : 0 < c)
92    (hC : 0 < C) (hP : CbGaussPoincare d a β κ) {ρ : Euc d → ℝ} (hρ : HRS.Whole.IsProbDensity ρ)
93    (hb : ∀ x, c * gaussianComparison d a β x ≤ ρ x ∧ ρ x ≤ C * gaussianComparison d a β x)
94    {f : Euc d → ℝ} (hf : ContDiff ℝ 1 f) (hf2 : MemLp f 2 (HRS.Whole.withDensityReal ρ))
95    (hg : Integrable (fun x => ‖gradient f x‖ ^ 2) (HRS.Whole.withDensityReal ρ)) :
96    ProbabilityTheory.variance f (HRS.Whole.withDensityReal ρ) ≤
97      C * (1 + κ) / (c * β * a) * ∫ x, ‖gradient f x‖ ^ 2 ∂(HRS.Whole.withDensityReal ρ) := by
98  set G := HRS.Whole.withDensityReal (gaussianComparison d a β) with hG
99  set μ := HRS.Whole.withDensityReal ρ with hμ
100  haveI : IsProbabilityMeasure μ := cb_withDensityReal_isProb hρ
101  haveI : IsProbabilityMeasure G := cb_withDensityReal_isProb (gaussianComparison_isProbDensity ha hβ)
102  have hγm := gaussianComparison_measurable (d := d) a β
103  have hGμ : G ≤ ENNReal.ofReal c⁻¹ • μ := by
104    refine cb_withDensityReal_le_smul hρ.measurable (inv_nonneg.2 hc.le) fun x => ?_
105    have := (hb x).1
106    rw [← mul_le_mul_iff_of_pos_left hc, ← mul_assoc, mul_inv_cancel₀ hc.ne', one_mul]
107    exact this
108  have hμG : μ ≤ ENNReal.ofReal C • G :=
109    cb_withDensityReal_le_smul hγm hC.le fun x => (hb x).2
110  have hf2G : MemLp f 2 G := (hf2.smul_measure ENNReal.ofReal_ne_top).mono_measure hGμ
111  have hgG : Integrable (fun x => ‖gradient f x‖ ^ 2) G :=
112    (hg.smul_measure ENNReal.ofReal_ne_top).mono_measure hGμ
113  have hfm : AEMeasurable f μ := hf.continuous.measurable.aemeasurable
114  have hfmG : AEMeasurable f G := hf.continuous.measurable.aemeasurable
115  set m := ∫ x, f x ∂G with hm
116  have h1 : ProbabilityTheory.variance f μ ≤ ∫ x, (f x - m) ^ 2 ∂μ := by
117    have e := ProbabilityTheory.variance_sub_const (μ := μ) (X := f) hfm.aestronglyMeasurable m
118    have := ProbabilityTheory.variance_le_expectation_sq (μ := μ) (X := fun x => f x - m)
119      (hfm.sub_const m).aestronglyMeasurable
120    rw [e] at this
121    simpa using this
122  have hsqG : Integrable (fun x => (f x - m) ^ 2) G :=
123    (hf2G.sub (memLp_const m)).integrable_sq
124  have h2 : ∫ x, (f x - m) ^ 2 ∂μ ≤ C * ∫ x, (f x - m) ^ 2 ∂G := by
125    have hsq' : Integrable (fun x => (f x - m) ^ 2) (ENNReal.ofReal C • G) :=
126      hsqG.smul_measure ENNReal.ofReal_ne_top
127    calc ∫ x, (f x - m) ^ 2 ∂μ ≤ ∫ x, (f x - m) ^ 2 ∂(ENNReal.ofReal C • G) :=
128          integral_mono_measure hμG (ae_of_all _ fun x => sq_nonneg _) hsq'
129      _ = C * ∫ x, (f x - m) ^ 2 ∂G := by
130          rw [integral_smul_measure, ENNReal.toReal_ofReal hC.le, smul_eq_mul]
131  have h3 : ∫ x, (f x - m) ^ 2 ∂G = ProbabilityTheory.variance f G :=
132    (ProbabilityTheory.variance_eq_integral hfmG).symm
133  have h4 := hP f hf hf2G hgG
134  have h5 : ∫ x, ‖gradient f x‖ ^ 2 ∂G ≤ c⁻¹ * ∫ x, ‖gradient f x‖ ^ 2 ∂μ := by
135    have hg' : Integrable (fun x => ‖gradient f x‖ ^ 2) (ENNReal.ofReal c⁻¹ • μ) :=
136      hg.smul_measure ENNReal.ofReal_ne_top
137    calc ∫ x, ‖gradient f x‖ ^ 2 ∂G ≤ ∫ x, ‖gradient f x‖ ^ 2 ∂(ENNReal.ofReal c⁻¹ • μ) :=
138          integral_mono_measure hGμ (ae_of_all _ fun x => sq_nonneg _) hg'
139      _ = c⁻¹ * ∫ x, ‖gradient f x‖ ^ 2 ∂μ := by
140          rw [integral_smul_measure, ENNReal.toReal_ofReal (inv_nonneg.2 hc.le), smul_eq_mul]
141  have hI : 0 ≤ ∫ x, ‖gradient f x‖ ^ 2 ∂μ := integral_nonneg fun x => sq_nonneg _
142  have hκβ : 0 ≤ κ / (β * a) := by positivity
143  calc ProbabilityTheory.variance f μ ≤ C * ProbabilityTheory.variance f G := by
144        rw [← h3]; exact h1.trans h2
145    _ ≤ C * (κ / (β * a) * (c⁻¹ * ∫ x, ‖gradient f x‖ ^ 2 ∂μ)) :=
146        mul_le_mul_of_nonneg_left (h4.trans (mul_le_mul_of_nonneg_left h5 hκβ)) hC.le
147    _ = (C * κ) / (c * β * a) * ∫ x, ‖gradient f x‖ ^ 2 ∂μ := by field_simp
148    _ ≤ C * (1 + κ) / (c * β * a) * ∫ x, ‖gradient f x‖ ^ 2 ∂μ := by
149        apply mul_le_mul_of_nonneg_right _ hI
150        apply div_le_div_of_nonneg_right _ (by positivity)
151        nlinarith

153/-- **Conditional form of `euc_thermal_gaussian_bounds`**: the two-sided comparison and the Poincare inequality for
154`μ_β`, given the Gaussian Poincare inequality for `γ` (not proved here). -/
155theorem cb_euc_thermal_gaussian_bounds_of_poincare (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 < s) (hsd : s < d)
156    (a β : ℝ) (ha : 0 < a) (hβ : 0 < β) {κ : ℝ} (hκ : 0 ≤ κ) (hP : CbGaussPoincare d a β κ) :
157    ∃ c > 0, ∃ C : ℝ,
158      (∀ x, c * gaussianComparison d a β x ≤ thermalDensity d s a β x ∧
159        thermalDensity d s a β x ≤ C * gaussianComparison d a β x) ∧
160      ∀ f : Euc d → ℝ, ContDiff ℝ 1 f → MemLp f 2 (thermalMeasure d s a β) →
161        Integrable (fun x => ‖gradient f x‖ ^ 2) (thermalMeasure d s a β) →
162        ProbabilityTheory.variance f (thermalMeasure d s a β) ≤
163          C / (c * β * a) * ∫ x, ‖gradient f x‖ ^ 2 ∂(thermalMeasure d s a β) := by
164  have hd1 : 1 ≤ d := by omega
165  obtain ⟨hρ, c, C, hc, hC, hb⟩ := cb_thermal_two_sided hd1 hs0 hsd ha hβ
166  refine ⟨c, hc, C * (1 + κ), ?_, fun f hf hf2 hg => ?_⟩
167  · intro x
168    refine ⟨(hb x).1, (hb x).2.trans ?_⟩
169    exact mul_le_mul_of_nonneg_right (by nlinarith) (gaussianComparison_pos ha hβ x).le
170  · exact cb_poincare_transfer ha hβ hκ hc hC hP hρ hb hf hf2 hg
