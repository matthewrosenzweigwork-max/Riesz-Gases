27/-- `L²`-convergence to `0` on a probability space implies `L¹`-convergence to `0`. -/
28theorem q08_tendsto_lintegral_enorm {Ω : Type*} [MeasurableSpace Ω] (P : Measure Ω)
29    [IsProbabilityMeasure P] (f : ℕ → Ω → ℝ) (hf : ∀ L, Measurable (f L))
30    (h : Tendsto (fun L => ∫⁻ ω, ‖f L ω‖ₑ ^ 2 ∂P) atTop (𝓝 0)) :
31    Tendsto (fun L => ∫⁻ ω, ‖f L ω‖ₑ ∂P) atTop (𝓝 0) := by
32  have h2 : Tendsto (fun L => (∫⁻ ω, ‖f L ω‖ₑ ^ 2 ∂P) ^ (1 / (2 : ℝ))) atTop (𝓝 0) := by
33    have := ENNReal.continuous_rpow_const (y := 1 / (2 : ℝ)) |>.tendsto 0
34    have h0 : (0 : ℝ≥0∞) ^ (1 / (2 : ℝ)) = 0 := by
35      rw [ENNReal.zero_rpow_of_pos (by norm_num)]
36    rw [h0] at this
37    exact this.comp h
38  refine tendsto_of_tendsto_of_tendsto_of_le_of_le tendsto_const_nhds h2 (fun L => bot_le) (fun L => ?_)
39  have := eLpNorm_le_eLpNorm_of_exponent_le (μ := P) (p := 1) (q := 2) (f := f L) (by norm_num)
40  rw [eLpNorm_one_eq_lintegral_enorm] at this
41  · refine this.trans (le_of_eq ?_)
42    rw [eLpNorm_eq_lintegral_rpow_enorm_toReal (by norm_num) (by norm_num)]
43    simp
44    all_goals exact (hf L).aestronglyMeasurable
45  · exact (hf L).aestronglyMeasurable

54    (hlaw : ∀ n : ℕ, law P (fun ω => (WithLp.toLp 2 fun i : Fin n => G i ω : EuclideanSpace ℝ (Fin n))) =
55        multivariateGaussian 0 (Matrix.of fun i j : Fin n => C i j))

59          H i j * (G i ω * G j ω - C i j)‖ₑ ^ 2 ∂P) atTop (𝓝 0))

62        F ((1 / 2) * ∑ i : Fin L, ∑ j : Fin L, H i j * (g i * g j - C i j),
63          fun b => ∑ i : Fin L, a b i * g i)

68    H i j * (G i ω * G j ω - C i j) with hTQ
69  set TB : ℕ → Fin p → Ω → ℝ := fun L b ω => ∑ i ∈ Finset.range L, a b i * G i ω with hTB

71    simp only [hTQ]
72    refine Measurable.const_mul (Finset.measurable_sum _ fun i _ => Finset.measurable_sum _ fun j _ => ?_) _
73    exact measurable_const.mul (((hGm i).mul (hGm j)).sub measurable_const)

75    simp only [hTB]
76    exact Finset.measurable_sum _ fun i _ => measurable_const.mul (hGm i)

82    hQm.prodMk (measurable_pi_iff.2 fun b => hLm b)
83  -- Step 1: Gaussian integrals as integrals over `Ω`
84  have step1 : ∀ L : ℕ, ∫ g : EuclideanSpace ℝ (Fin L),
85        F ((1 / 2) * ∑ i : Fin L, ∑ j : Fin L, H i j * (g i * g j - C i j),
86          fun b => ∑ i : Fin L, a b i * g i)
87        ∂(multivariateGaussian 0 (Matrix.of fun i j : Fin L => C i j)) =
88      ∫ ω, F (TQ L ω, fun b => TB L b ω) ∂P := by
89    intro L
90    rw [← hlaw L]
91    have hGv : Measurable (fun ω => (WithLp.toLp 2 fun i : Fin L => G i ω : EuclideanSpace ℝ (Fin L))) := by
92      refine (WithLp.measurable_toLp 2 (Fin L → ℝ)).comp (measurable_pi_iff.2 fun i => hGm i)

96      simp only [hTQ, hTB, Finset.sum_range]
97    · refine (hFm.comp ?_).aestronglyMeasurable
98      refine Measurable.prodMk ?_ (measurable_pi_iff.2 fun b => ?_)
99      · refine Measurable.const_mul (Finset.measurable_sum _ fun i _ => Finset.measurable_sum _ fun j _ => ?_) _
100        exact measurable_const.mul (((measurable_pi_apply i |>.comp (PiLp.continuous_ofLp 2 _ |>.measurable)).mul
101          (measurable_pi_apply j |>.comp (PiLp.continuous_ofLp 2 _ |>.measurable))).sub measurable_const)
102      · exact Finset.measurable_sum _ fun i _ => measurable_const.mul
103          (measurable_pi_apply i |>.comp (PiLp.continuous_ofLp 2 _ |>.measurable))

106    Integrable.of_bound (hFm.comp hf).aestronglyMeasurable M
107      (Eventually.of_forall fun ω => by simpa [Real.norm_eq_abs] using hM _)

109  have hpt : ∀ L ω, ‖F (TQ L ω, fun b => TB L b ω) - F (Q ω, fun b => Lin b ω)‖ₑ ≤
110      ENNReal.ofReal K * (‖Q ω - TQ L ω‖ₑ + ∑ b, ‖Lin b ω - TB L b ω‖ₑ) := by
111    intro L ω
112    have h1 : dist (F (TQ L ω, fun b => TB L b ω)) (F (Q ω, fun b => Lin b ω)) ≤
113        K * dist (TQ L ω, fun b => TB L b ω) (Q ω, fun b => Lin b ω) := hF.dist_le_mul _ _
114    have h2 : dist (TQ L ω, fun b => TB L b ω) (Q ω, fun b => Lin b ω) ≤
115        |Q ω - TQ L ω| + ∑ b, |Lin b ω - TB L b ω| := by
116      rw [Prod.dist_eq]
117      refine max_le ?_ ?_
118      · rw [Real.dist_eq, abs_sub_comm]
119        have : 0 ≤ ∑ b, |Lin b ω - TB L b ω| := Finset.sum_nonneg fun _ _ => abs_nonneg _
120        linarith
121      · rw [dist_pi_le_iff (by positivity)]
122        intro b
123        rw [Real.dist_eq, abs_sub_comm]
124        have : |Lin b ω - TB L b ω| ≤ ∑ b, |Lin b ω - TB L b ω| :=
125          Finset.single_le_sum (f := fun b => |Lin b ω - TB L b ω|) (fun _ _ => abs_nonneg _)
126            (Finset.mem_univ b)
127        linarith [abs_nonneg (Q ω - TQ L ω)]
128    have h3 : |F (TQ L ω, fun b => TB L b ω) - F (Q ω, fun b => Lin b ω)| ≤
129        K * (|Q ω - TQ L ω| + ∑ b, |Lin b ω - TB L b ω|) := by
130      rw [← Real.dist_eq]; exact h1.trans (mul_le_mul_of_nonneg_left h2 K.2)
131    rw [Real.enorm_eq_ofReal_abs]
132    calc ENNReal.ofReal |F (TQ L ω, fun b => TB L b ω) - F (Q ω, fun b => Lin b ω)|
133        ≤ ENNReal.ofReal (K * (|Q ω - TQ L ω| + ∑ b, |Lin b ω - TB L b ω|)) :=
134          ENNReal.ofReal_le_ofReal h3
135      _ = ENNReal.ofReal K * (‖Q ω - TQ L ω‖ₑ + ∑ b, ‖Lin b ω - TB L b ω‖ₑ) := by
136          rw [ENNReal.ofReal_mul (NNReal.coe_nonneg K), ENNReal.ofReal_add (abs_nonneg _)
137            (Finset.sum_nonneg fun _ _ => abs_nonneg _), ENNReal.ofReal_sum_of_nonneg
138            (fun _ _ => abs_nonneg _)]
139          simp [Real.enorm_eq_ofReal_abs]
140  have hQ1 := q08_tendsto_lintegral_enorm P (fun L ω => Q ω - TQ L ω)
141    (fun L => hQm.sub (hTQm L)) hQ
142  have hL1 : ∀ b, Tendsto (fun L => ∫⁻ ω, ‖Lin b ω - TB L b ω‖ₑ ∂P) atTop (𝓝 0) := fun b =>
143    q08_tendsto_lintegral_enorm P (fun L ω => Lin b ω - TB L b ω) (fun L => (hLm b).sub (hTBm L b))
144      (hLin b)
145  have hbound : Tendsto (fun L => ENNReal.ofReal K * (∫⁻ ω, ‖Q ω - TQ L ω‖ₑ ∂P +
146      ∑ b, ∫⁻ ω, ‖Lin b ω - TB L b ω‖ₑ ∂P)) atTop (𝓝 0) := by
147    have := ENNReal.Tendsto.const_mul (a := ENNReal.ofReal K)
148      (hQ1.add (tendsto_finsetSum (Finset.univ : Finset (Fin p)) fun b _ => hL1 b)) (Or.inr ENNReal.ofReal_ne_top)
149    simpa using this
150  refine tendsto_integral_of_L1 _ (hint _ hT0).aestronglyMeasurable (Eventually.of_forall fun L => hint _ (hTm L)) ?_
151  refine tendsto_of_tendsto_of_tendsto_of_le_of_le tendsto_const_nhds hbound (fun L => bot_le)
152    (fun L => ?_)
153  calc ∫⁻ ω, ‖F (TQ L ω, fun b => TB L b ω) - F (Q ω, fun b => Lin b ω)‖ₑ ∂P

164/-- Convergence of integrals of bounded Lipschitz functions implies convergence of integrals of bounded
165continuous functions (Portmanteau, via Mathlib's `ProbabilityMeasure` topology). -/
166theorem q08_upgrade {E : Type*} [PseudoMetricSpace E] [MeasurableSpace E] [OpensMeasurableSpace E]
167    {X : ℕ → Type*} [∀ N, MeasurableSpace (X N)] (ν : ∀ N, Measure (X N))
168    (hν : ∀ N, 1 ≤ N → IsProbabilityMeasure (ν N)) (T : ∀ N, X N → E) (hT : ∀ N, Measurable (T N))
169    (lam : Measure E) [IsProbabilityMeasure lam]
170    (hlip : ∀ F : E → ℝ, (∃ M, ∀ z, |F z| ≤ M) → (∃ K, LipschitzWith K F) →
171      Tendsto (fun N => ∫ x, F (T N x) ∂(ν N)) atTop (𝓝 (∫ z, F z ∂lam)))
172    (F : E → ℝ) (hFc : Continuous F) (hFb : ∃ M, ∀ z, |F z| ≤ M) :
173    Tendsto (fun N => ∫ x, F (T N x) ∂(ν N)) atTop (𝓝 (∫ z, F z ∂lam)) := by
174  have hprob : ∀ n : ℕ, IsProbabilityMeasure ((ν (n + 1)).map (T (n + 1))) := fun n => by
175    have := hν (n + 1) (by omega)
176    exact (Measure.isProbabilityMeasure_map_iff (hT (n + 1)).aemeasurable).2 inferInstance
177  let μs : ℕ → ProbabilityMeasure E := fun n => ⟨(ν (n + 1)).map (T (n + 1)), hprob n⟩
178  let L0 : ProbabilityMeasure E := ⟨lam, inferInstance⟩
179  have hint : ∀ (n : ℕ) (f : E → ℝ), Continuous f → ∫ z, f z ∂(μs n : Measure E) =
180      ∫ x, f (T (n + 1) x) ∂(ν (n + 1)) := fun n f hf => by
181    show ∫ z, f z ∂((ν (n + 1)).map (T (n + 1))) = _
182    exact integral_map (hT (n + 1)).aemeasurable hf.aestronglyMeasurable
183  have hlim : Tendsto μs atTop (𝓝 L0) := by
184    rw [tendsto_iff_forall_lipschitz_integral_tendsto]
185    intro f hfb hfl
186    obtain ⟨K, hK⟩ := hfl
187    obtain ⟨C, hC⟩ := hfb
188    have hb : ∃ M, ∀ z, |f z| ≤ M := by
189      by_cases hE : Nonempty E
190      · obtain ⟨z0⟩ := hE
191        refine ⟨|f z0| + C, fun z => ?_⟩
192        have := hC z z0
193        rw [Real.dist_eq] at this
194        have h2 : |f z| ≤ |f z0| + |f z - f z0| := by
195          have := abs_sub_abs_le_abs_sub (f z) (f z0)
196          linarith
197        linarith
198      · exact ⟨0, fun z => (hE ⟨z⟩).elim⟩
199    simp_rw [hint _ _ hK.continuous]
200    exact (tendsto_add_atTop_iff_nat 1).2 (hlip f hb ⟨K, hK⟩)
201  rw [ProbabilityMeasure.tendsto_iff_forall_integral_tendsto] at hlim
202  obtain ⟨M, hM⟩ := hFb
203  let f' : BoundedContinuousFunction E ℝ := BoundedContinuousFunction.ofNormedAddCommGroup F hFc M
204    (fun z => by simpa [Real.norm_eq_abs] using hM z)
205  have := hlim f'
206  simp_rw [show ∀ n, ∫ z, f' z ∂(μs n : Measure E) = ∫ x, F (T (n + 1) x) ∂(ν (n + 1)) from fun n =>
207    hint n f' f'.continuous] at this
208  exact (tendsto_add_atTop_iff_nat 1).1 this

215/-- The pair `(B_N(h), ζ_N(φ))` of the conclusion of Lemma 7.5 (`B_N` exactly centred under `Q_{N,β}`, the
216linear statistics centred at the Gibbs first marginal `m_N`). -/
217def q08Stat (N : ℕ) (X : Config d N) : ℝ × (Fin D.p → ℝ) :=
218  ((N : ℝ) * (modulatedEnergy D.h X D.μ - ∫ Y, modulatedEnergy D.h Y D.μ ∂(D.Gibbs N)),
219    fun a => Real.sqrt N * ((1 / (N : ℝ)) * ∑ i, D.φ a (X i) - pair (firstMarginal (D.Gibbs N)) (D.φ a)))
220
221/-- The finite-dimensional Wick chaos functional `E F(½ ∑_{i,j<L} H_ij (G_iG_j - C_ij), (∑_{i<L} a_{ai} G_i)_a)`
222for `(G_i)_{i<L} ~ N(0, C|_{L×L})`. -/
223def q08FiniteChaos (L : ℕ) (F : ℝ × (Fin D.p → ℝ) → ℝ) : ℝ :=
224  ∫ g : EuclideanSpace ℝ (Fin L),
225    F ((1 / 2) * ∑ i : Fin L, ∑ j : Fin L, D.Hmat i j * (g i * g j - D.Cmat i j),
226      fun b => ∑ i : Fin L, D.coef b i * g i)
227    ∂(multivariateGaussian 0 (Matrix.of fun i j : Fin L => D.Cmat i j))
228
229/-- **Intermediate layer** (the finite-rank approximation theorem, the heart of Lemma 7.5; to be proved
230from `Spectral`, `Features` and the `Chaos` bound, see the route below): for bounded Lipschitz `F`, the law
231of `(B_N, ζ_N)` under `Q_{N,β}` is within `ε` of the finite-rank Wick chaos functional `q08FiniteChaos D L`
232for all large `L` and then all large `N`.
233
234Route: (1) `q08_bounded_family` gives bounded orthonormal centred `u_{<L}`, with finite-rank (projected)
235kernels `g^U, h^U` and test-coefficients within `ε` in `L²`; (2) by `q08_weight_Lq` and `q08_energy_L2`
236(Hölder, `w_N ∈ L^3`, `L²` closeness of the statistics under `μ^{⊗N}`) one may replace the statistics and the
237Gibbs weight by the finite-rank ones `½(SᵀH^U S - tr H^U)`, `S^N = N^{-1/2} ∑_k u(X_k)`, with
238`e^{-(β/2)(SᵀK^U S - tr K^U)}`, up to `O(ε)` uniformly in `N` (the diagonal terms `(1/N)∑_k u(X_k)ᵀ K u(X_k)`
239converge by the LLN, `u` bounded); the exact centring of `B_N` under `Q_N` passes to the limit because the `L²`
240errors are controlled under `Q_N` by Cauchy-Schwarz with `w_N ∈ L²`; (3) `q08_clt_quadratic` gives the
241`N → ∞` limit as an integral against `N(0, I_L)`, `q08_gauss_tilt` turns the tilt into `N(0, (I+βK^U)^{-1})`;
242(4) the resolvent identity of `q08_Cmat_spec` and `‖K - K^U‖_{HS} ≤ ε` give `‖(I+βK^U)^{-1} - C‖ ≲ βε`, and
243`q08_chaos_l2_bound` (basis change from `u` to `e`) gives closeness to `q08FiniteChaos D L` for `L` large. -/
244theorem q08_gibbs_approx (F : ℝ × (Fin D.p → ℝ) → ℝ) (hb : ∃ M, ∀ z, |F z| ≤ M)
245    (hl : ∃ K : ℝ≥0, LipschitzWith K F) (ε : ℝ) (hε : 0 < ε) :
246    ∃ L0 : ℕ, ∀ L ≥ L0, ∀ᶠ N in atTop,
247      |∫ X, F (q08Stat D N X) ∂(D.Gibbs N) - q08FiniteChaos D L F| ≤ ε :=
248  q08gl_final D F hb hl ε hε

270    (p : ℕ) (φ : Fin p → Torus d → ℝ) (hφ : ∀ a, Measurable (φ a) ∧ ∃ M, ∀ x, |φ a x| ≤ M) :
271    (∀ N : ℕ, 1 ≤ N → Integrable (fun X : Config d N => modulatedEnergy h X (withDensityReal ρ))

276        multivariateGaussian 0 (Matrix.of fun i j : Fin n =>
277          susceptibilityCov (periodicRiesz d s) (withDensityReal ρ) β (e i) (e j))) ∧
278      (∀ a, Tendsto (fun L => ∫⁻ ω, ‖Lin a ω - ∑ i ∈ Finset.range L,

281          (∫ x, ∫ y, h x y * e i x * e j y ∂(withDensityReal ρ) ∂(withDensityReal ρ)) *
