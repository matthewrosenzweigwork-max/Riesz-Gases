34`|E h(M_t) - E h(G)| ≤ ½ sup_x ‖∇²h(x)‖_op E ∫_0^t ‖Σ_σ - sb_σ‖_{S_1} dσ`. -/

136theorem multivariate_martingale_clt_dc2 (ℱ : Filtration ℝ mΩ) (t : ℝ) (ht : 0 ≤ t)
137    (M : ℝ → Ω → (ℝ^p)) (Sg : ℝ → Ω → Matrix (Fin p) (Fin p) ℝ)
138    (hM : IsVectorMartingaleWithCovariation P ℱ t M Sg)
139    (sb : ℝ → Matrix (Fin p) (Fin p) ℝ) (hsbmeas : Measurable sb)
140    (hsbint : IntegrableOn (fun σ => s1Norm (sb σ)) (Set.Icc 0 t))
141    (hsbpsd : ∀ σ ∈ Set.Icc 0 t, (sb σ).PosSemidef) :
142    smoothDist 2 (law P (M t)) (ProbabilityTheory.multivariateGaussian 0 (Matrix.of fun i j => ∫ σ in Set.Icc 0 t, sb σ i j)) ≤
143      (1 / 2) * ∫ ω, (∫ σ in Set.Icc 0 t, s1Norm (Sg σ ω - sb σ)) ∂P := by
144  classical
145  have hmt : AEMeasurable (M t) P := (hM.sq_int t ⟨ht, le_rfl⟩).aestronglyMeasurable.aemeasurable
146  have hJ0 : 0 ≤ ∫ ω, (∫ σ in Set.Icc 0 t, s1Norm (Sg σ ω - sb σ)) ∂P :=
147    integral_nonneg fun ω => setIntegral_nonneg measurableSet_Icc fun σ _ => traceNormMatrix_nonneg _
148  refine Real.iSup_le (fun Φ => ?_) (by positivity)
149  obtain ⟨hΦ, hbd, hnorm⟩ := Φ.2
150  have h2 := hbd 2 le_rfl
151  have hle : (⨆ x, ‖iteratedFDeriv ℝ 2 Φ.1 x‖) ≤ 1 := by
152    refine le_trans ?_ hnorm
153    unfold cbNorm
154    simp only [Finset.sum_range_succ, Finset.range_zero, Finset.sum_empty, zero_add]

157    linarith
158  have hmain := multivariate_martingale_clt P ℱ t ht M Sg hM sb hsbmeas hsbint hsbpsd Φ.1 hΦ h2
159  have hH0 : 0 ≤ ⨆ x, ‖iteratedFDeriv ℝ 2 Φ.1 x‖ := Real.iSup_nonneg fun _ => norm_nonneg _
160  have e : ∫ z, Φ.1 z ∂(law P (M t)) = ∫ ω, Φ.1 (M t ω) ∂P :=
