24/-- Abstract form of the bridge: for nonnegative terms the maximum over `Fin (k+1)` is at most the
25sum over `range (k+1)`, which is at most `(k+1)` times the maximum. -/
26theorem p_sm3_bridge_abstract (k : ℕ) (f : ℕ → ℝ) (hf : ∀ j, 0 ≤ f j) :
27    (⨆ j : Fin (k + 1), f j) ≤ ∑ j ∈ Finset.range (k + 1), f j ∧
28      ∑ j ∈ Finset.range (k + 1), f j ≤ ((k : ℝ) + 1) * ⨆ j : Fin (k + 1), f j := by
29  constructor
30  · apply ciSup_le
31    intro j
32    exact Finset.single_le_sum (f := f) (fun i _ => hf i) (Finset.mem_range.2 j.isLt)
33  · have hle : ∀ j ∈ Finset.range (k + 1), f j ≤ ⨆ i : Fin (k + 1), f i := by
34      intro j hj
35      exact le_ciSup (Set.finite_range fun i : Fin (k + 1) => f i).bddAbove ⟨j, Finset.mem_range.1 hj⟩
36    have h := Finset.sum_le_card_nsmul (Finset.range (k + 1)) f _ hle
37    rw [Finset.card_range, nsmul_eq_mul] at h
38    push_cast at h
39    exact h

41/-- Clause U4 (TeX M L10064-10065): the printed max norm `M` and the Lean sum norm
42`Torus.cbNorm k φ` satisfy `M ≤ cbNorm k φ ≤ (k+1) M`, with no boundedness hypothesis. -/
43theorem cbNorm_max_bridge {d : ℕ} (k : ℕ) (φ : Torus d → ℝ) :
44    (⨆ j : Fin (k + 1), ⨆ x : Euc d, ‖iteratedFDeriv ℝ (j : ℕ) (Torus.lift φ) x‖) ≤
45        Torus.cbNorm k φ ∧

47        ((k : ℝ) + 1) * ⨆ j : Fin (k + 1), ⨆ x : Euc d, ‖iteratedFDeriv ℝ (j : ℕ) (Torus.lift φ) x‖ := by

53/-- Clause C8 (TeX M L1267-1270): `|⟨M, u⟩| ≤ ‖u‖ ‖M‖` for `u ∈ projTensor B 3`. -/
54theorem projTensor_pairing_le {B : Type*} [NormedAddCommGroup B] [NormedSpace ℝ B]
55    (u : projTensor B 3) (M : B [×3]→L[ℝ] ℝ) :
56    |((u : TensorDual B 3)) M| ≤ ‖u‖ * ‖M‖ := by
57  have h := (u : TensorDual B 3).le_opNorm M
58  have hu : ‖(u : TensorDual B 3)‖ = ‖u‖ := rfl
59  rw [hu, Real.norm_eq_abs] at h

63theorem norm_elemTensor_eq {B : Type*} [NormedAddCommGroup B] [NormedSpace ℝ B]
64    (v : Fin 3 → B) :
65    ‖(elemTensor v : TensorDual B 3)‖ = ∏ i, ‖v i‖ := by
66  refine le_antisymm (HRS.MC.E213T.e213t_norm_elemTensor_le v) ?_
67  choose g hg1 hgv using fun i : Fin 3 => exists_dual_vector'' ℝ (v i)
68  have hex : ∃ M : B [×3]→L[ℝ] ℝ, M v = ∏ i, ‖v i‖ ∧ ‖M‖ ≤ 1 := by
69    refine ⟨(ContinuousMultilinearMap.mkPiAlgebraFin ℝ 3 ℝ).compContinuousLinearMap
70      (fun i => (g i : B →L[ℝ] ℝ)), ?_, ?_⟩
71    · simp [ContinuousMultilinearMap.compContinuousLinearMap_apply,
72        ContinuousMultilinearMap.mkPiAlgebraFin_apply, hgv, Fin.prod_univ_three, mul_assoc]
73    · have hP : ‖ContinuousMultilinearMap.mkPiAlgebraFin ℝ 3 ℝ‖ ≤ 1 := by simp
74      have hprod : ∏ i, ‖(g i : B →L[ℝ] ℝ)‖ ≤ 1 :=

76      refine le_trans (ContinuousMultilinearMap.norm_compContinuousLinearMap_le _ _) ?_

79  obtain ⟨M, hMv, hM1⟩ := hex
