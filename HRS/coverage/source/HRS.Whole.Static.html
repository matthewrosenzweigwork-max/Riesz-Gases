23## Representation decisions
24
25* Variances are Mathlib's `ProbabilityTheory.evariance` (`ℝ≥0∞`), so "`N Var ≤ C‖f‖²`"
26  includes square integrability of the statistic (no junk value).
27* "The probability measure `Q_N` is well defined" (S line 929; "the normalizer belongs to
28  `(0,∞)`", line 1534) is the explicit structure `GibbsWellDefined`: the background terms of
29  the energy are integrable, the Gibbs weight is integrable, and the normaliser is positive.
30* A Radon–Nikodym statement "`dπ_N/dμ ≤ C`, `‖dπ_N/dμ - 1‖_{L²(μ)} ≤ Cδ_N`" is rendered as the
31  existence of a measurable density `φ` with `π_N = φ μ` and the two bounds (`MemLp` included).
32* Kernel bounds `|h(x,y)| ≤ B(1+|x-y|^{-s})` are imposed **off the diagonal only**
33  (`IsBoundedPairKernel`); Lean's junk `‖0‖^(-s) = 0` would otherwise impose a wrong condition
34  on the diagonal, which the energies delete anyway.
35* Sums of suprema `‖∇χ‖_∞ + ‖D²χ‖_∞ ≤ C` are written as "for all points `x, y`,
36  `‖∇χ(x)‖ + ‖D²χ(y)‖ ≤ C`", which is literally equivalent and avoids the real `iSup` junk.
37* Centered Gaussians are `HRS.IsCenteredGaussian` (Mathlib's `IsGaussian`, zero mean); the
38  covariance of S3.7 is read off from `G` (`C_ij = E[G_i G_j]`), so degenerate `C` is allowed.

52/-- The pair-kernel modulated energy `𝖥_N^h(X_N,μ)` of `W:eq:euclidean-F` (S lines 483-485),
53with the original deleted particle pair diagonal. -/
54def pairEnergy (h : Euc d → Euc d → ℝ) (X : EucConfig d N) (μ : Measure (Euc d)) : ℝ :=
55  (1 / (2 * (N : ℝ) ^ 2)) * ∑ i, ∑ j ∈ Finset.univ.erase i, h (X i) (X j)
56    - (1 / (N : ℝ)) * ∑ i, ∫ y, h (X i) y ∂μ
57    + (1 / 2) * ∫ x, ∫ y, h x y ∂μ ∂μ
58
59/-- `K^h_{N,b}(μ) = ∫ e^{-bN 𝖥_N^h} dμ^{⊗N}` (`W:eq:euclidean-K`, S line 486). -/
60def pairNormalizer (h : Euc d → Euc d → ℝ) (N : ℕ) (b : ℝ) (μ : Measure (Euc d)) : ℝ :=
61  ∫ X : EucConfig d N, Real.exp (-b * N * pairEnergy h X μ) ∂(Measure.pi fun _ => μ)
62
63/-- `Q^h_{N,b} = (K^h_{N,b})⁻¹ e^{-bN 𝖥_N^h} μ^{⊗N}` (S line 488). -/
64def pairGibbs (h : Euc d → Euc d → ℝ) (N : ℕ) (b : ℝ) (μ : Measure (Euc d)) :
65    Measure (EucConfig d N) :=
66  (ENNReal.ofReal (pairNormalizer h N b μ))⁻¹ •
67    (Measure.pi fun _ => μ).withDensity
68      fun X => ENNReal.ofReal (Real.exp (-b * N * pairEnergy h X μ))

81/-- "The law is well defined": every background term of the energy is integrable, the Gibbs
82weight is integrable against `μ^{⊗N}`, and the normaliser is strictly positive (so the
83Bochner integrals in `pairEnergy`, `pairNormalizer` are the source's integrals and `pairGibbs`
84is a probability measure). -/
85structure GibbsWellDefined (h : Euc d → Euc d → ℝ) (N : ℕ) (b : ℝ) (μ : Measure (Euc d)) :
86    Prop where
87  conv_integrable : ∀ x, Integrable (fun y => h x y) μ
88  pair_integrable : Integrable (fun p : Euc d × Euc d => h p.1 p.2) (μ.prod μ)
89  weight_integrable :
90    Integrable (fun X : EucConfig d N => Real.exp (-b * N * pairEnergy h X μ))
91      (Measure.pi fun _ => μ)
92  normalizer_pos : 0 < pairNormalizer h N b μ
93
94/-- The weight `W_s(z) - 1`: `|z|^{-s}` for `s > 0`, `(-log|z|)_+` for `s = 0`
95(S lines 508-509, 528-529). -/
96def rieszWeight (s : ℝ) (z : Euc d) : ℝ :=
97  if s = 0 then max (-Real.log ‖z‖) 0 else ‖z‖ ^ (-s)
98
99/-- A real measurable symmetric kernel with `|h(x,y)| ≤ B(1 + W_s(x-y) - 1)` off the diagonal
100(S lines 507-509; S lines 937-938 "off the diagonal"). -/
101structure IsBoundedPairKernel (s B : ℝ) (h : Euc d → Euc d → ℝ) : Prop where
102  measurable : Measurable fun p : Euc d × Euc d => h p.1 p.2
103  symm : ∀ x y, h x y = h y x
104  bound : ∀ x y, x ≠ y → |h x y| ≤ B * (1 + rieszWeight s (x - y))

106/-- The physical commutator kernel `h_v(x,y) = (v(x) - v(y))·∇g(x-y)` (S lines 950, 1561). -/
107def commutatorKernel (d : ℕ) (s : ℝ) (v : Euc d → Euc d) (x y : Euc d) : ℝ :=
108  inner ℝ (v x - v y) (rieszKernelGrad d s (x - y))

144/-- Proposition S3.1 `W:prop:euclidean-pressure`, main bound `W:eq:euclidean-pressure-bound`
145(S lines 497-510): `d ≥ 2`, `0 ≤ s < d`, `μ = ρ dx` a probability measure with `‖ρ‖_∞ ≤ M`
146(and `∫ log(2+|x|) dμ ≤ L` when `s = 0`), `0 < b₋ < b₊`.  For every `B` there are `θ₀ > 0`
147and `C` such that `|log K^{g-θh}_{N,b}(μ)| ≤ C N δ_N` for `N ≥ 2`, `b ∈ [b₋,b₊]`, `|θ| ≤ θ₀`
148and every real measurable symmetric `h` with `|h(x,y)| ≤ B(1+|x-y|^{-s})` (`s > 0`), resp.
149`B(1+(-log|x-y|)_+)` (`s = 0`).  The constants depend only on the displayed parameters
150`(d, s, M, L, b₋, b₊, B)`, hence are quantified before `ρ` and `h`.  The normaliser being in
151`(0,∞)` is part of the claim (`GibbsWellDefined`). -/
152theorem euclidean_pressure (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d) (M L bm bp B : ℝ)
153    (hbm : 0 < bm) (hb : bm < bp) :
154    ∃ θ₀ > 0, ∃ C : ℝ, ∀ ρ : Euc d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ M) →
155      (s = 0 → HasLogMoment (withDensityReal ρ) L) →
156      ∀ h : Euc d → Euc d → ℝ, IsBoundedPairKernel s B h →
157      ∀ N, 2 ≤ N → ∀ b ∈ Set.Icc bm bp, ∀ θ : ℝ, |θ| ≤ θ₀ →
158        GibbsWellDefined (fun x y => rieszKernel d s (x - y) - θ * h x y) N b (withDensityReal ρ) ∧
159        |Real.log (pairNormalizer (fun x y => rieszKernel d s (x - y) - θ * h x y) N b
160          (withDensityReal ρ))| ≤ C * N * deltaN d s N := by
161  sorry

163/-- Proposition S3.1, the two consequences `W:eq:euclidean-pair-entropy` (S lines 510-518):
164`log E_{Q^g_{N,b}}[e^{cN|𝖥_N^h|}] ≤ C N δ_N` and, for every probability law `f` with finite
165relative entropy `H(f | Q^g_{N,b})`, `E_f[N|𝖥_N^h|] ≤ C{H(f|Q^g_{N,b}) + N δ_N}`.  The small
166exponent `c > 0` and `C` depend only on the displayed parameters. -/

168    (hbm : 0 < bm) (hb : bm < bp) :

172      ∀ N, 2 ≤ N → ∀ b ∈ Set.Icc bm bp,

175          Real.log (∫ X : EucConfig d N, Real.exp (c * N * |pairEnergy h X (withDensityReal ρ)|)
176            ∂(modulatedGibbs (rieszKernel d s) N b (withDensityReal ρ))) ≤ C * N * deltaN d s N) ∧
177        ∀ f : Measure (EucConfig d N), IsProbabilityMeasure f →
178          relEntropy f (modulatedGibbs (rieszKernel d s) N b (withDensityReal ρ)) < ⊤ →
179          Integrable (fun X : EucConfig d N => N * |pairEnergy h X (withDensityReal ρ)|) f ∧
180          ∫ X, N * |pairEnergy h X (withDensityReal ρ)| ∂f ≤
181            C * ((relEntropy f (modulatedGibbs (rieszKernel d s) N b (withDensityReal ρ))).toReal +
182              N * deltaN d s N) := by

185/-- Proposition S3.1, last sentence (S lines 518-519): "the same two assertions hold with

224/-- Proposition S3.3 `W:prop:euc-static` (S lines 928-952), under the standing hypotheses of
225S lines 903-926: `d ≥ 2`, `0 < s < d`, `β > 0`, `μ = ρ dx` a probability measure with
226`ρ ∈ L^∞`, no lower bound on `ρ`; `g = |x|^{-s}` with the original normalisation and
227`Q_N = e^{-βN F_N^g}/K_N μ^{⊗N}`.  Constants depend only on `d, s, β` and an upper bound `M`
228for `‖ρ‖_∞`, not on a support radius, and are uniform for `β` in compact subsets of
229`(0,∞)`: hence `C` is quantified before `ρ` and before `β ∈ [β₋,β₊]`.

231Conclusions: `Q_N` is well defined; with `π_N = Q_N^{(1)}`:
232`N Var_{Q_N}[μ_N(f)] ≤ C‖f‖²_{L²(μ)}`, `‖dπ_N/dμ - 1‖_{L²(μ)} ≤ Cδ_N`, `dπ_N/dμ ≤ C`

234`|h(x,y)| ≤ B(1+|x-y|^{-s})` off the diagonal and both `ν = μ`, `ν = π_N`,
235`N Var_{Q_N}[D_N(h;ν)] ≤ C B² δ_N` (`W:eq:euc-quadratic`). -/

237    (hβ : βm ≤ βp) :

239      ∀ N, 2 ≤ N →

241        (∀ f : Euc d → ℝ, MemLp f 2 (withDensityReal ρ) →

247        ∀ B : ℝ, ∀ h : Euc d → Euc d → ℝ, IsBoundedPairKernel s B h →
248          QuadraticVarianceBound h (modulatedGibbs (rieszKernel d s) N β (withDensityReal ρ))
249            (withDensityReal ρ) (C * B ^ 2 * deltaN d s N / N) ∧
250          QuadraticVarianceBound h (modulatedGibbs (rieszKernel d s) N β (withDensityReal ρ))
251            (firstMarginal (modulatedGibbs (rieszKernel d s) N β (withDensityReal ρ)))
252            (C * B ^ 2 * deltaN d s N / N) := by

435/-- The finite-coordinate Stein-type identity of S3.7 (S lines 1990-1994):
436`E[X_i f(X)] = E[R_i f(X)] + ∑_j E[T_ij ∂_j f(X)]` for every bounded smooth `f` with bounded

438def SteinIdentity {n : ℕ} (X R : Ω → EuclideanSpace ℝ (Fin n))
439    (T : Ω → Matrix (Fin n) (Fin n) ℝ) : Prop :=
440  ∀ f : EuclideanSpace ℝ (Fin n) → ℝ, ContDiff ℝ ∞ f → (∃ Cf, ∀ x, |f x| ≤ Cf) →
441    (∃ Cf, ∀ x, ‖fderiv ℝ f x‖ ≤ Cf) → ∀ i : Fin n,
442      ∫ w, X w i * f (X w) ∂P =
443        ∫ w, R w i * f (X w) ∂P +
444          ∑ j : Fin n, ∫ w, T w i j * fderiv ℝ f (X w) (EuclideanSpace.single j 1) ∂P

446/-- Lemma S3.7 `W:lem:r2-static-Stein`, finite-dimensional part (S lines 1986-2001):
447`X, R` integrable random vectors in `ℝⁿ`, `T` a random matrix with integrable trace norm
448satisfying the Stein-type identity; `G` a centered Gaussian of any positive semidefinite
449covariance `C` (`C_ij = E[G_i G_j]`, degenerate allowed); `F ∈ C²` with bounded first and
450second derivatives.  Then
451`|E F(X) - E F(G)| ≤ ‖DF‖_∞ E|R| + ½ ‖D²F‖_∞ E‖T - C‖₁` (`W:eq:r2-owning-static-Stein`). -/
