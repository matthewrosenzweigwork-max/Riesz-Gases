43/-! ### Contact-allowing solution concepts -/
44
45/-- A strong solution in the sense of `IsStrongSolution` **without** the collision-free
46clause: the same `lift`, `init`, `cont`, `adapted` fields.  This is the notion the
47manuscript uses for the Cépa–Lépingle flow (case (c) of Assumption A.1, L10136) in which
48contacts are allowed.  At a contact, `particleDrift` evaluates the junk value
49`periodicRieszGrad d s 0`; see `NullContactOccupation`. -/
50structure IsContactStrongSolution (P : Measure Ω) (D : DynData d) (T : ℝ)
51    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d))
52    (X : ℝ≥0 → Ω → Config d N) : Prop where
53  lift : ∃ Y : ℝ≥0 → Ω → (Fin N → Euc d),
54    (∀ᵐ ω ∂P, ∀ t : ℝ≥0, (t : ℝ) ≤ T →
55      (∀ i, IntegrableOn (fun s : ℝ => particleDrift D (projectPath Y (Real.toNNReal s) ω) i)
56        (Set.Icc 0 (t : ℝ))) ∧
57      ∀ i, Y t ω i = Y 0 ω i + (∫ s in Set.Icc (0 : ℝ) t, particleDrift D (projectPath Y (Real.toNNReal s) ω) i)
58        + Real.sqrt (2 * D.invβ) • (W t ω i - W 0 ω i)) ∧
59    X = projectPath Y
60  init : ∀ᵐ ω ∂P, X 0 ω = X₀ ω
61  cont : ∀ᵐ ω ∂P, ∀ i, Continuous fun t => X t ω i
62  adapted : ∀ t : ℝ≥0, Measurable (X t)
63
64theorem IsStrongSolution.toContact {P : Measure Ω} {D : DynData d} {T : ℝ}
65    {X₀ : Ω → Config d N} {W : ℝ≥0 → Ω → (Fin N → Euc d)} {X : ℝ≥0 → Ω → Config d N}
66    (h : IsStrongSolution P D T X₀ W X) : IsContactStrongSolution P D T X₀ W X :=
67  ⟨h.lift, h.init, h.cont, h.adapted⟩

69/-- The set of contact times has Lebesgue measure zero, a.s. (CL 1997, Thm 2.2:
70the ordered gap processes are Bessel of dimension δ > 1 and so spend zero time at 0). -/
71def NullContactOccupation (P : Measure Ω) (T : ℝ) (X : ℝ≥0 → Ω → Config d N) : Prop :=
72  ∀ᵐ ω ∂P, volume {s : ℝ | s ∈ Set.Icc 0 T ∧ X (Real.toNNReal s) ω ∈ collisionSet d N} = 0
73
74/-- Cyclic order of a lift `y : Fin N → Euc d` under a labelling `π`: along the lift the
75labelled points are nondecreasing and the total spread is at most one period
76(`B:eq:…` L11665: "ordered on the circle, with the order preserved").  Stated for every
77coordinate so the definition type-checks for general `d`; it is used only with `d = 1`. -/
78def CyclicallyOrdered (π : Equiv.Perm (Fin N)) (y : Fin N → Euc d) : Prop :=
79  ∀ j : Fin d, (∀ k k' : Fin N, k ≤ k' → y (π k) j ≤ y (π k') j) ∧
80    ∀ k k' : Fin N, y (π k') j ≤ y (π k) j + 1
81
82/-- The Cépa–Lépingle ordered flow (Prop B.2, L11662-11670): a contact-allowing strong
83solution whose lift is cyclically ordered for all times by a (random, measurable)
84labelling fixed at time 0, and whose contact set is Lebesgue-null a.s. -/
85structure IsCepaLepingleFlow (P : Measure Ω) (D : DynData d) (T : ℝ)
86    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d))
87    (X : ℝ≥0 → Ω → Config d N) : Prop where
88  toContact : IsContactStrongSolution P D T X₀ W X
89  ordered : ∃ π : Ω → Equiv.Perm (Fin N), (∀ σ, MeasurableSet {ω | π ω = σ}) ∧
90    ∀ᵐ ω ∂P, ∃ Y : ℝ≥0 → Fin N → Euc d,
91      (∀ t : ℝ≥0, (t : ℝ) ≤ T → X t ω = fun i => Torus.mk (Y t i)) ∧
92      Continuous Y ∧ ∀ t : ℝ≥0, (t : ℝ) ≤ T → CyclicallyOrdered (π ω) (Y t)
93  occupation : NullContactOccupation P T X

95/-! ### Assumption A.1 and the selected law -/
96
97/-- Assumption A.1 (L10127-10143).  Cases (a), (b) and (d) are all "the unique strong
98collision-free solution" and collapse into one disjunct; case (c) is the Cépa–Lépingle flow. -/
99structure SelectedFlow (P : Measure Ω) (D : DynData d) (T : ℝ)
100    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d))
101    (X : ℝ≥0 → Ω → Config d N) : Prop where
102  init_meas : Measurable X₀
103  init_distinct : ∀ᵐ ω ∂P, X₀ ω ∉ collisionSet d N
104  brownian : IsVectorBrownian P W
105  indep : IndepFun X₀ (fun ω => fun t => W t ω) P
106  flow :
107    (IsStrongSolution P D T X₀ W X ∧ PathwiseUnique P D T X₀ W) ∨
108    (d = 1 ∧ D.s = 0 ∧ 0 < D.invβ ∧ (∃ a : ℝ, 0 < a ∧ D.M = -(a • 1)) ∧
109      IsCepaLepingleFlow P D T X₀ W X)

111/-- `P_N^t := Law(X_N^t)` (`B:eq:selected-particle-law` L10141). -/
112def selectedLaw (P : Measure Ω) (X : ℝ≥0 → Ω → Config d N) (t : ℝ) : Measure (Config d N) :=
113  law P (X (Real.toNNReal t))
114
115/-- `f_N^t` exists iff `P_N^t ≪ volume`; it is then `rnDeriv`. -/
116def HasSelectedDensity (P : Measure Ω) (X : ℝ≥0 → Ω → Config d N) (t : ℝ) : Prop :=
117  selectedLaw P X t ≪ (volume : Measure (Config d N))
118
119/-- The density `f_N^t = dP_N^t/dX` (a junk measurable function unless `HasSelectedDensity`). -/
120def selectedDensity (P : Measure Ω) (X : ℝ≥0 → Ω → Config d N) (t : ℝ) : Config d N → ℝ≥0∞ :=
121  (selectedLaw P X t).rnDeriv volume

125/-- Regularity of a time-dependent test `f : ℝ → Torus d → ℝ` on `[0,t]`: the exact
126regularity the adjoint solution of Definition 3.5 enjoys (`IsClassicalAdjointSolution`
127fields `c2`, `cont_c2`, `cont_time_deriv`), with the time derivative `ft` given on
128`Ioo 0 t` and continuous on `Icc 0 t × Torus d`. -/
129structure IsC12Test (t : ℝ) (f ft : ℝ → Torus d → ℝ) : Prop where
130  c2 : ∀ r ∈ Set.Icc 0 t, Torus.IsCk 2 (f r)
131  cont_c2 : ∀ j ≤ 2, Continuous fun p : Set.Icc (0:ℝ) t × Euc d =>
132    iteratedFDeriv ℝ j (Torus.lift (f p.1)) p.2
133  time_deriv : ∀ x, ∀ r ∈ Set.Ioo 0 t, HasDerivAt (fun r => f r x) (ft r x) r
134  cont_time_deriv : Continuous fun p : Set.Icc (0:ℝ) t × Torus d => ft p.1 p.2
135
136theorem IsClassicalAdjointSolution.isC12Test {D : DynData d} {ρ : ℝ → Torus d → ℝ} {t : ℝ}
137    {φ : Torus d → ℝ} {f : ℝ → Torus d → ℝ} (h : IsClassicalAdjointSolution D ρ t φ f) :
138    IsC12Test t f (fun r x => -(backwardGenerator D (ρ r) (f r) x)) :=
139  ⟨h.c2, h.cont_c2, h.time_deriv, h.cont_time_deriv.neg⟩

141/-- The integrand of the empirical martingale `M_N^t[f]` (`B:eq:…` L1645):
142`N⁻¹ √(2/β) ∂_j f(r, x_i(r))` for the coordinate `k = (i,j)`, zero outside `[0,t]`. -/
143def empiricalNoiseIntegrand (D : DynData d) (t : ℝ) (f : ℝ → Torus d → ℝ)
144    (X : ℝ≥0 → Ω → Config d N) : ℝ → Ω → (Fin N × Fin d) → ℝ :=
145  fun s ω k => if s ∈ Set.Icc 0 t then
146    (Real.sqrt (2 * D.invβ) / N) * (Torus.grad (f s) (X (Real.toNNReal s) ω k.1)) k.2 else 0
147
148/-- The empirical martingale `M_N^τ[f] = ∫_0^τ ∑_{i,j} (empiricalNoiseIntegrand)_{(i,j)} dW'_{(i,j)}`. -/
149def empiricalMartingale (P : Measure Ω) (ℱ : Filtration ℝ mΩ) (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ)
150    (D : DynData d) (t : ℝ) (f : ℝ → Torus d → ℝ) (X : ℝ≥0 → Ω → Config d N) : ℝ → Ω → ℝ :=
151  itoIntegral P ℱ W' (empiricalNoiseIntegrand D t f X)
152
153/-- Label-off-diagonal empirical interaction (`B:eq:…` L1500, "the CL label-off-diagonal
154sum"): `(2N²)⁻¹ ∑_i ∑_{j ≠ i} ⟨∇ψ(x_i) − ∇ψ(x_j), M ∇g(x_i − x_j)⟩`.  At a contact
155`x_i = x_j` the summand contains the junk value `∇g(0)`. -/
156def empiricalInteraction (D : DynData d) (ψ : Torus d → ℝ) (x : Config d N) : ℝ :=
157  (1 / (2 * (N:ℝ)^2)) * ∑ i, ∑ j ∈ Finset.univ.erase i,
158    inner ℝ (Torus.grad ψ (x i) - Torus.grad ψ (x j))
159      (Torus.matVec D.M (periodicRieszGrad d D.s (x i - x j)))

161/-! ### The test class and generators of Proposition B.3 -/
162
163/-- `Φ` read on the covering space. -/
164def configLift (Φ : ℝ → Config d N → ℝ) (r : ℝ) (Y : Fin N → Euc d) : ℝ :=
165  Φ r (fun i => Torus.mk (Y i))
166
167/-- `C_c^{1,2}([0,T) × Ω_N)`: `C^{1,2}` on the covering space (the exact regularity Itô's
168formula needs), vanishing for `r ≥ T'` for some `T' < T`, and vanishing outside a compact
169`K ⊆ Ω_N`. -/
170structure IsFKETest (T : ℝ) (Φ : ℝ → Config d N → ℝ) : Prop where
171  c12 : IsC12On T Set.univ (configLift Φ)
172  time_support : ∃ T' < T, ∀ r, T' ≤ r → ∀ X, Φ r X = 0
173  space_support : ∃ K : Set (Config d N), IsCompact K ∧ Disjoint K (collisionSet d N) ∧
174    ∀ r X, X ∉ K → Φ r X = 0
175
176/-- The basis vector `e_k := Pi.single k.1 (EuclideanSpace.single k.2 1)` of the covering
177space `Fin N → Euc d`. -/
178def configBasis (d N : ℕ) (k : Fin N × Fin d) : Fin N → Euc d :=
179  Pi.single k.1 (EuclideanSpace.single k.2 1)
180
181/-- `∂_r Φ + β⁻¹ Δ_X Φ − ∑_i ∇V(x_i)·∇_{x_i} Φ` at `(r, X)`, computed through the lift
182at `Y = rep ∘ X`. -/
183def fkeGenerator (D : DynData d) (T : ℝ) (Φ : ℝ → Config d N → ℝ) (r : ℝ) (X : Config d N) : ℝ :=
184  let Y : Fin N → Euc d := fun i => Torus.rep (X i)
185  derivWithin (fun r => configLift Φ r Y) (Set.Icc 0 T) r
186  + D.invβ * ∑ k : Fin N × Fin d,
187      fderiv ℝ (fderiv ℝ (configLift Φ r)) Y (configBasis d N k) (configBasis d N k)
188  - ∑ i, fderiv ℝ (configLift Φ r) Y (Pi.single i (Torus.grad D.V (X i)))
189
190/-- `∑_i N⁻¹ ∑_{j ≠ i} ∇_{x_i}Φ · M∇g(x_i − x_j)`, written label-off-diagonal and
191antisymmetrised, `w_ij := M∇g(x_i − x_j)`. -/
192def fkeInteraction (D : DynData d) (Φ : ℝ → Config d N → ℝ) (r : ℝ) (X : Config d N) : ℝ :=
193  let Y : Fin N → Euc d := fun i => Torus.rep (X i)
194  (1 / (N : ℝ)) * ∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j),
195    (fderiv ℝ (configLift Φ r) Y
196        (Pi.single i (Torus.matVec D.M (periodicRieszGrad d D.s (X i - X j))))
197     - fderiv ℝ (configLift Φ r) Y
198        (Pi.single j (Torus.matVec D.M (periodicRieszGrad d D.s (X i - X j)))))
