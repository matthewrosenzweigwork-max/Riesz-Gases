30theorem screened_local_proof : type_of% @HRS.Whole.screened_local := by
31  intro d hd s hs0 hsd
32  by_cases hs : s < (d : ℝ) - 1
33  · obtain ⟨a, A, hA⟩ := screened_local_lt_proof hd hs0 hs
34    refine ⟨max a 0, max A 0, fun i ρ hρ c v hsupp hv => ?_⟩
35    obtain ⟨w, hw, hbd⟩ := hA i ρ hρ c v hsupp hv
36    refine ⟨w, hw, fun c' r' hr' => ?_⟩
37    have h1 : ρ ^ a ≤ ρ ^ (max a 0) := Real.rpow_le_rpow_of_exponent_le hρ (le_max_left _ _)
38    have h2 : 0 < ρ ^ a := Real.rpow_pos_of_pos (by linarith) a
39    have h3 : A ≤ max A 0 := le_max_left _ _
40    have h4 : 0 ≤ max A 0 := le_max_right _ _
41    have key : A * ρ ^ a ≤ max A 0 * ρ ^ (max a 0) :=
42      calc A * ρ ^ a ≤ max A 0 * ρ ^ a := mul_le_mul_of_nonneg_right h3 h2.le
43        _ ≤ max A 0 * ρ ^ (max a 0) := mul_le_mul_of_nonneg_left h1 h4
44    exact (hbd c' r' hr').trans (mul_le_mul_of_nonneg_right (ENNReal.ofReal_le_ofReal key) zero_le)
45  · have hs1 : (d : ℝ) - 1 ≤ s := not_lt.mp hs
46    have hkg : kgOrder d s = s + 1 - d := by
47      unfold kgOrder
48      exact max_eq_left (by linarith)
49    obtain ⟨c0, hc0, hid⟩ := kg_fourier_h_proof hd hs0 hsd
50    obtain ⟨C, hC⟩ := kg_rep_of_symbol_proof hd hsd hs1 c0 hc0 hid
51    refine ⟨0, max C 0, fun i ρ _ c v _ hv => ?_⟩
52    rw [hkg] at hv ⊢
53    obtain ⟨w, hw, _, hnorm⟩ := hC i v hv
54    refine ⟨w, hw, fun c' r' _ => ?_⟩
55    have hm : MeasurableSet (Metric.closedBall c' r') := measurableSet_closedBall
56    have key : C ≤ max C 0 * ρ ^ (0 : ℝ) := by
57      rw [Real.rpow_zero, mul_one]
58      exact le_max_left _ _
59    exact (eLpNorm_indicator_le w hm).trans (hnorm.trans (mul_le_mul_of_nonneg_right (ENNReal.ofReal_le_ofReal key) zero_le))
