6# GAP-4: extension of the Poincare inequality to `C^1` functions, assembly of `reflection_gap_poincare`
7
8Supplement foundation packet GAP-4 (`docs/S4_PLAN.md`, section 2.4, steps 6 and 7; row S4.9).  Namespace
9`HRS.MS.Gap`, helper prefix `gap234_`.
10
11* `gap_ext`: for `GapRef d ρ q B` (`μ = ρ dx`), `q > 0`, `B ∈ C_b^2`, there is `c > 0` such that every `f ∈ C^1`
12  with bounded gradient and `f ∈ L²(μ)` satisfies `c Var_μ f ≤ ∫ ‖∇f‖² dμ`.
13  Route: (A) for `h ∈ C^1` bounded with bounded gradient, the mollifications `h_n = η_n * h`
14  (`η_n = invBump n` normalised, `PoisInvAux1.invMoll`) lie in the bounded `C²` class `gap234_Sm`
15  (`‖D h_n‖ ≤ ‖Dh‖_∞`, `D² h_n` bounded because `D²(h * η) = h * D²η` for bounded `h`), so the smooth inequality
16  `gap_smooth_const` (applied to `h_n - ∫ h_n`) gives it for `h_n`; `h_n → h`, `∇h_n → ∇h` pointwise with uniform
17  bounds, so dominated convergence passes to the limit.  (B) for general `f`, the truncations
18  `n sin(f/n)` are bounded `C^1` with `‖∇‖ ≤ ‖∇f‖_∞`, converge to `f` pointwise with `|n sin(f/n)| ≤ |f|`, and
19  dominated convergence passes to the limit again.
20* `reflection_gap_poincare_proof : type_of% @HRS.Whole.reflection_gap_poincare` (the frozen S4.9 statement).

22Statement adjustments relative to the plan sketch (documented, harmless): the truncation is `n sin(f/n)` (bounded,
23`1`-Lipschitz, `n sin(y/n) → y`) and the mollification is applied to the truncated function (so that the second
24derivative of the mollification is bounded by `‖h‖_∞ ∫‖D²η‖`); the variance is handled through the formula
25`Var f = ∫ f² - (∫ f)²` (`ProbabilityTheory.variance_eq_sub`).

80  obtain ⟨K, hK⟩ := hfd.bounded_above_of_compact_support (hκc.fderiv ℝ)
81  have hM0 : 0 ≤ M := (abs_nonneg _).trans (hM 0)
82  have hK0 : 0 ≤ K := (norm_nonneg _).trans (hK 0)
83  set S : Set (Euc d) := Metric.closedBall x₀ (R + 1) with hS
84  have hSm : MeasurableSet S := measurableSet_closedBall
85  have hbint : Integrable (fun t => S.indicator (fun _ => M * K) t) volume :=
86    (integrable_indicator_iff hSm).2 (integrableOn_const (measure_closedBall_lt_top.ne))

88      HasFDerivAt (fun x => h t • κ (x - t)) (h t • fderiv ℝ κ (x - t)) x := by
89    intro x t

91      (hasFDerivAt_id x).sub_const t

93    have h3 := h2.const_smul (h t)
94    convert h3 using 1 <;> first | rfl | simp
95  refine hasFDerivAt_integral_of_dominated_of_fderiv_le (𝕜 := ℝ) (μ := volume)
96    (s := Metric.ball x₀ 1)
97    (F := fun (x : Euc d) (t : Euc d) => h t • κ (x - t))
98    (F' := fun (x : Euc d) (t : Euc d) => h t • fderiv ℝ κ (x - t)) (x₀ := x₀)
99    (bound := fun t => S.indicator (fun _ => M * K) t) (Metric.ball_mem_nhds x₀ one_pos) ?_ ?_ ?_ ?_

102      (hh.smul (hκ0.comp (continuous_const.sub continuous_id))).aestronglyMeasurable

104      (by
105        have : HasCompactSupport (fun t : Euc d => κ (x₀ - t)) := by

107          simpa [Function.comp_def, sub_eq_add_neg] using this
108        exact this.smul_left)
109  · exact (hh.smul (hfd.comp (continuous_const.sub continuous_id))).aestronglyMeasurable
110  · refine ae_of_all _ fun t x hx => ?_
111    by_cases ht : t ∈ S
112    · rw [Set.indicator_of_mem ht, norm_smul, Real.norm_eq_abs]
113      exact mul_le_mul (hM t) (hK _) (norm_nonneg _) hM0
114    · rw [Set.indicator_of_notMem ht]
115      have hxt : x - t ∉ tsupport κ := by
116        intro hmem
117        have h1 := hR hmem
118        rw [mem_closedBall_zero_iff] at h1
119        apply ht
120        rw [hS, Metric.mem_closedBall, dist_eq_norm]
121        have hx' : ‖x - x₀‖ < 1 := by simpa [dist_eq_norm] using hx
122        calc ‖t - x₀‖ = ‖(x - x₀) - (x - t)‖ := by congr 1; abel
123          _ ≤ ‖x - x₀‖ + ‖x - t‖ := norm_sub_le _ _
124          _ ≤ R + 1 := by linarith
125      rw [fderiv_of_notMem_tsupport ℝ hxt]
126      simp
127  · exact ae_of_all _ fun t x _ => hdiff x t

200/-- The second derivative of a mollification of a bounded `C^1` function with bounded gradient is bounded

202theorem gap234_moll_second_bound {h : Euc d → ℝ} (hh : ContDiff ℝ 1 h) {M C : ℝ}
203    (hM : ∀ t, |h t| ≤ M) (hC : ∀ y, ‖fderiv ℝ h y‖ ≤ C) (n : ℕ) (x : Euc d) :

205  have hη := gap234_eta_smooth (d := d) n
206  have hη1 : ContDiff ℝ 1 (gap234_eta d n) := hη.of_le (by norm_num)
207  have hc : Continuous h := hh.continuous
208  have hC0 : 0 ≤ C := (norm_nonneg _).trans (hC 0)
209  have hfd : Continuous (fderiv ℝ (gap234_eta d n)) := hη1.continuous_fderiv one_ne_zero
210  have hcs : HasCompactSupport (fderiv ℝ (gap234_eta d n)) := (gap234_eta_compact n).fderiv ℝ
211  have hint : Integrable (fun s => ‖fderiv ℝ (gap234_eta d n) s‖) := gap234_int_norm_fderiv hη1 (gap234_eta_compact n)
212  have hI0 : 0 ≤ ∫ s, ‖fderiv ℝ (gap234_eta d n) s‖ := integral_nonneg fun s => norm_nonneg _
213  have s1 : ∀ x, HasFDerivAt (invMoll n h) (∫ t, h t • fderiv ℝ (gap234_eta d n) (x - t)) x := by
214    intro x
215    have := gap234_conv_hasFDerivAt_left (V := ℝ) hη1 (gap234_eta_compact n) hc hM x
216    have e : invMoll n h = fun x => ∫ t, h t • gap234_eta d n (x - t) :=
217      funext (gap234_invMoll_eq' n h)
218    rw [e]
219    exact this
220  have hD : ∀ x, fderiv ℝ (invMoll n h) x = ∫ s, h (x - s) • fderiv ℝ (gap234_eta d n) s := by
221    intro x
222    rw [(s1 x).fderiv]
223    have := integral_sub_left_eq_self (fun s => h (x - s) • fderiv ℝ (gap234_eta d n) s) volume x
224    simp only [sub_sub_cancel] at this
225    exact this
226  have hI : ∀ x, Integrable (fun s => h (x - s) • fderiv ℝ (gap234_eta d n) s) := fun x =>
227    ((hc.comp (continuous_const.sub continuous_id)).smul hfd).integrable_of_hasCompactSupport hcs.smul_left
228  have hlipf : ∀ a b : Euc d, |h a - h b| ≤ C * ‖a - b‖ := fun a b => by
229    have := (convex_univ (𝕜 := ℝ) (E := Euc d)).norm_image_sub_le_of_norm_fderiv_le
230      (f := h) (C := C) (fun z _ => (hh.differentiable one_ne_zero) z) (fun z _ => hC z)
231      (mem_univ b) (mem_univ a)
232    rwa [Real.norm_eq_abs] at this
233  have hlip : LipschitzWith (Real.toNNReal (C * ∫ s, ‖fderiv ℝ (gap234_eta d n) s‖))

235    refine LipschitzWith.of_dist_le_mul fun a b => ?_
236    rw [dist_eq_norm, dist_eq_norm, hD a, hD b, ← integral_sub (hI a) (hI b),
237      Real.coe_toNNReal _ (mul_nonneg hC0 hI0)]
238    have hb : Integrable (fun s => (C * ‖a - b‖) * ‖fderiv ℝ (gap234_eta d n) s‖) := hint.const_mul _
239    refine (norm_integral_le_of_norm_le hb (ae_of_all _ fun s => ?_)).trans (le_of_eq ?_)
240    · rw [← sub_smul, norm_smul, Real.norm_eq_abs]
241      refine mul_le_mul_of_nonneg_right ?_ (norm_nonneg _)
242      have := hlipf (a - s) (b - s)
243      simpa [sub_sub_sub_cancel_right] using this
244    · rw [integral_const_mul]; ring
245  exact (norm_fderiv_le_of_lipschitz ℝ hlip).trans (by rw [Real.coe_toNNReal _ (mul_nonneg hC0 hI0)])

292    simp only [Pi.add_apply]
293    nlinarith [sq_nonneg (D x - 1)]
294  have hmu : ∀ n, MemLp (u n) 2 μ := fun n =>
295    gap234_memLp_of_bound (hu n).continuous hD2 (hbd n)
296  have hmf : MemLp f 2 μ := gap234_memLp_of_bound hf.continuous hD2 hfD
297  have T1 : Tendsto (fun n => ∫ x, u n x ^ 2 ∂μ) atTop (𝓝 (∫ x, f x ^ 2 ∂μ)) := by
298    refine tendsto_integral_of_dominated_convergence (fun x => D x ^ 2)
299      (fun n => ((hu n).continuous.pow 2).aestronglyMeasurable) hD2 (fun n => ae_of_all _ fun x => ?_)
300      (ae_of_all _ fun x => (hlim x).pow 2)
301    rw [Real.norm_eq_abs, abs_pow, sq_abs]
302    exact sq_le_sq.2 ((hbd n x).trans (le_abs_self _))
303  have T2 : Tendsto (fun n => ∫ x, u n x ∂μ) atTop (𝓝 (∫ x, f x ∂μ)) := by
304    refine tendsto_integral_of_dominated_convergence D
305      (fun n => (hu n).continuous.aestronglyMeasurable) hD1 (fun n => ae_of_all _ fun x => ?_)
306      (ae_of_all _ hlim)

317      (𝓝 (c * ProbabilityTheory.variance f μ)) := by

326theorem gap234_smooth_var {q : ℝ} {B : Euc d → Euc d} {ρ : Euc d → ℝ} (hR : GapRef d ρ q B)
327    {c : ℝ}
328    (hc : ∀ {f : Euc d → ℝ} {M : ℝ}, gap234_Sm M f → ∫ x, f x ∂(HRS.Whole.withDensityReal ρ) = 0 →
329      c * ∫ x, f x ^ 2 ∂(HRS.Whole.withDensityReal ρ) ≤
330        ∫ x, ‖gradient f x‖ ^ 2 ∂(HRS.Whole.withDensityReal ρ))
331    {f : Euc d → ℝ} {M : ℝ} (hf : gap234_Sm M f) :
332    c * ProbabilityTheory.variance f (HRS.Whole.withDensityReal ρ) ≤
333      ∫ x, ‖gradient f x‖ ^ 2 ∂(HRS.Whole.withDensityReal ρ) := by
334  have : IsProbabilityMeasure (HRS.Whole.withDensityReal ρ) := gap1_isProb hR
335  set m : ℝ := ∫ x, f x ∂(HRS.Whole.withDensityReal ρ) with hm
336  have hfi : Integrable f (HRS.Whole.withDensityReal ρ) :=
337    (integrable_const M).mono' hf.1.continuous.aestronglyMeasurable
338      (ae_of_all _ fun x => gap234_sm_zero hf x)
339  have hfd : fderiv ℝ (fun x => f x - m) = fderiv ℝ f := by
340    funext x; exact fderiv_sub_const m
341  have hg : gap234_Sm (M + |m|) (fun x => f x - m) := by
342    have hM0 := gap234_sm_nonneg hf
343    refine gap234_sm_of_bounds (hf.1.sub contDiff_const) (fun y => ?_) (fun y => ?_) (fun y => ?_)
344    · rw [Real.norm_eq_abs]
345      have := gap234_sm_zero hf y
346      rw [Real.norm_eq_abs] at this
347      calc |f y - m| ≤ |f y| + |m| := abs_sub _ _
348        _ ≤ M + |m| := by linarith
349    · rw [hfd]; have := gap234_sm_one hf y; linarith [abs_nonneg m]
350    · rw [hfd]; have := gap234_sm_two hf y; linarith [abs_nonneg m]
351  have hcen : ∫ x, (f x - m) ∂(HRS.Whole.withDensityReal ρ) = 0 := by
352    rw [integral_sub hfi (integrable_const m)]
353    simp [hm]
354  have h1 := hc hg hcen
355  have hgrad : ∀ x, gradient (fun x => f x - m) x = gradient f x := fun x => by
356    unfold gradient; rw [hfd]
357  simp_rw [hgrad] at h1
358  have hv : ProbabilityTheory.variance f (HRS.Whole.withDensityReal ρ) =
359      ∫ x, (f x - m) ^ 2 ∂(HRS.Whole.withDensityReal ρ) :=
360    ProbabilityTheory.variance_eq_integral hf.1.continuous.aemeasurable
361  rw [hv]
362  exact h1

405theorem gap234_trunc_tendsto (a : ℝ) :
406    Tendsto (fun n : ℕ => ((n : ℝ) + 1) * Real.sin (a / ((n : ℝ) + 1))) atTop (𝓝 a) := by
407  have hdiff : ∀ᶠ n : ℕ in atTop, |((n : ℝ) + 1) * Real.sin (a / ((n : ℝ) + 1)) - a| ≤
408      |a| ^ 3 / ((n : ℝ) + 1) ^ 2 := by
409    obtain ⟨N, hN⟩ := exists_nat_gt |a|
410    filter_upwards [eventually_ge_atTop N] with n hn
411    have hpos : (0 : ℝ) < (n : ℝ) + 1 := by positivity
412    have hle : |a| ≤ (n : ℝ) + 1 := by
413      have : (N : ℝ) ≤ n := by exact_mod_cast hn
414      linarith
415    have hz : |a / ((n : ℝ) + 1)| ≤ 1 := by
416      rw [abs_div, abs_of_pos hpos, div_le_one hpos]; exact hle
417    have h1 := gap234_sin_sub _ hz
418    have e : ((n : ℝ) + 1) * Real.sin (a / ((n : ℝ) + 1)) - a =
419        ((n : ℝ) + 1) * (Real.sin (a / ((n : ℝ) + 1)) - a / ((n : ℝ) + 1)) := by
420      field_simp
421    rw [e, abs_mul, abs_of_pos hpos]
422    calc ((n : ℝ) + 1) * |Real.sin (a / ((n : ℝ) + 1)) - a / ((n : ℝ) + 1)|
423        ≤ ((n : ℝ) + 1) * |a / ((n : ℝ) + 1)| ^ 3 := mul_le_mul_of_nonneg_left h1 hpos.le
424      _ = |a| ^ 3 / ((n : ℝ) + 1) ^ 2 := by
425        rw [abs_div, abs_of_pos hpos]; field_simp
426  have hzero : Tendsto (fun n : ℕ => |a| ^ 3 / ((n : ℝ) + 1) ^ 2) atTop (𝓝 0) := by
427    refine tendsto_const_nhds.div_atTop ?_
428    exact (tendsto_pow_atTop (by norm_num : 2 ≠ 0)).comp (tendsto_natCast_atTop_atTop.atTop_add tendsto_const_nhds)
429  have : Tendsto (fun n : ℕ => ((n : ℝ) + 1) * Real.sin (a / ((n : ℝ) + 1)) - a) atTop (𝓝 0) :=
430    squeeze_zero_norm' (by simpa [Real.norm_eq_abs] using hdiff) hzero
431  simpa using this.add_const a

433theorem gap234_stageB {q : ℝ} {B : Euc d → Euc d} {ρ : Euc d → ℝ} (hR : GapRef d ρ q B) {c : ℝ}

435      c * ∫ x, f x ^ 2 ∂(HRS.Whole.withDensityReal ρ) ≤

437    {f : Euc d → ℝ} (hf : ContDiff ℝ 1 f) {Cf : ℝ} (hCf : ∀ x, ‖gradient f x‖ ≤ Cf)

442  set u : ℕ → Euc d → ℝ := fun n x => ((n : ℝ) + 1) * Real.sin (f x / ((n : ℝ) + 1)) with hu
443  have hpos : ∀ n : ℕ, (0 : ℝ) < (n : ℝ) + 1 := fun n => by positivity

445    contDiff_const.mul (Real.contDiff_sin.comp (hf.div_const _))
446  have hfd : ∀ n x, fderiv ℝ (u n) x = Real.cos (f x / ((n : ℝ) + 1)) • fderiv ℝ f x := by

455    rw [e2]

469          mul_le_mul_of_nonneg_left Real.abs_sin_le_abs (hpos n).le
470      _ = |f x| := by rw [abs_div, abs_of_pos (hpos n)]; field_simp

479          mul_le_mul_of_nonneg_right (Real.abs_cos_le_one _) (norm_nonneg _)

485      have := (Real.continuous_cos.tendsto 0).comp hz

487      exact this

489    simpa using hcos.smul_const (gradient f x)

491      simp only [hu]
