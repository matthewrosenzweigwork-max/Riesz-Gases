6Relocation of `HRS.convGradConv_coeff_of_holder`.
7
8Route (decided before writing code; est. 130 lines, far below the 1200-line limit).  The repo's
9proof of the `C¹` version (`KernelGradD1.coeff_convGradConv_lip_of_one_le`, `GradD1.lean`, and its
10`d ≥ 2` twin in `GradFourier.lean`) regularises the kernel by the heat kernel `heatRieszGrad ε`
11(uniformly `GoodKernel`, with continuous kernel, so the Fourier coefficient of the convolution is
12explicit: `KernelGrad.coeff_convGradConv_heat`) and passes to the limit `ε → 0` by dominated
13convergence; the only place where `C¹` (Lipschitz) is used is the *domination* of the integrand
14`kern G_ε h x` and the uniform bound of `convGradConv G_ε h`.  Replacing the Lipschitz
15domination `L C (|z|^{-s}+d) + B C (...)` by the Hölder domination of
16`Duality.kern_norm_holder`/`integrable_dom_holder`
17(`Λ C (|z|^{β-s-1} + |z|^β) + B C (...)`, valid for all uniformly good kernels at once) gives the
18Hölder version verbatim: no mollification of `ρ` is needed.
19-/

30/-- The Hölder dominating function. -/
31def chDom (s C Λ B β : ℝ) (d : ℕ) (z : Torus d) : ℝ :=
32  Λ * C * (‖z‖ ^ (β - s - 1) + ‖z‖ ^ β) + B * C * ((1 / 4 : ℝ) ^ (-s - 1) + 1)

34theorem ch_integrable_dom (hG : KernelGrad.GoodKernel d s C G) {Λ B β : ℝ} (hβ0 : 0 < β)
35    (hβ : s + 1 - d < β) : Integrable (chDom s C Λ B β d) :=
36  Duality.integrable_dom_holder hG hβ0 hβ

38theorem ch_kern_bound_ae (hG : KernelGrad.GoodKernel d s C G) {h : Torus d → ℝ} {B Λ β : ℝ}
39    (hB : ∀ x, |h x| ≤ B) (hB0 : 0 ≤ B) (hΛ : 0 ≤ Λ)
40    (hH : ∀ x z : Torus d, ‖z‖ ≤ 1 / 4 → |h (x - z) - h x| ≤ Λ * ‖z‖ ^ β) (x : Torus d) :
41    ∀ᵐ z : Torus d, ‖KernelGrad.kern G h x z‖ ≤ chDom s C Λ B β d z := by
42  filter_upwards [KernelGrad.torus_ae_ne' hG.pos] with z hz

45/-- Continuity of the cancellation convolution of a Hölder function. -/
46theorem ch_continuous_op (hG : KernelGrad.GoodKernel d s C G) {h : Torus d → ℝ} (hc : Continuous h)
47    {B Λ β : ℝ} (hB : ∀ x, |h x| ≤ B) (hB0 : 0 ≤ B) (hΛ : 0 ≤ Λ) (hβ0 : 0 < β)
48    (hβ : s + 1 - d < β)

50    Continuous (HRS.convGradConv G h) := by

74    {h : Torus d → ℝ} (hh : Continuous h) {B Λ β : ℝ} (hB : ∀ x, |h x| ≤ B) (hB0 : 0 ≤ B)
75    (hΛ : 0 ≤ Λ) (hβ0 : 0 < β) (hβ : s + 1 - d < β)
76    (hH : ∀ x z : Torus d, ‖z‖ ≤ 1 / 4 → |h (x - z) - h x| ≤ Λ * ‖z‖ ^ β) (x : Torus d) :
77    Tendsto (fun ε => HRS.convGradConv (heatRieszGrad d s ε) h x) (𝓝[>] 0)
78      (𝓝 (HRS.convGradConv (periodicRieszGrad d s) h x)) := by

83    exact (KernelGrad.measurable_kern (hGε ε hε) hh.measurable x).aestronglyMeasurable

93/-- Fourier coefficients of `∇g * h` (cancellation convention) for a Hölder function `h`
94(copy of `KernelGradD1.coeff_convGradConv_lip_of_one_le` with the Hölder domination). -/
95theorem ch_coeff_holder {s : ℝ} (hd : 1 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d) {h : Torus d → ℝ}
96    (hcont : Continuous h) {B Λ β : ℝ} (hB : ∀ x, |h x| ≤ B) (hB0 : 0 ≤ B) (hΛ : 0 ≤ Λ)
97    (hβ0 : 0 < β) (hβ : s + 1 - d < β)
98    (hH : ∀ x z : Torus d, ‖z‖ ≤ 1 / 4 → |h (x - z) - h x| ≤ Λ * ‖z‖ ^ β)
99    (k : Lattice d) (i : Fin d) :
100    ∫ x, Torus.mode (-k) x * ((HRS.convGradConv (periodicRieszGrad d s) h x i : ℝ) : ℂ) =
101      (∫ y, Torus.mode (-k) y * (h y : ℂ)) *
102        ((2 * Real.pi * Complex.I * ((k i : ℤ) : ℂ)) * ((rieszCoeff d s k : ℝ) : ℂ)) := by
103  obtain ⟨C, hC0, hGε⟩ := KernelGradD1.goodKernel_heat_of_one_le hd hs0 hsd
104  set l : Filter ℝ := 𝓝[>] (0 : ℝ) with hl
105  have hLHS : Tendsto (fun ε : ℝ => ∫ x, Torus.mode (-k) x *
106      ((HRS.convGradConv (heatRieszGrad d s ε) h x i : ℝ) : ℂ)) l
107      (𝓝 (∫ x, Torus.mode (-k) x * ((HRS.convGradConv (periodicRieszGrad d s) h x i : ℝ) : ℂ))) := by
108    refine tendsto_integral_filter_of_norm_le_const ?_ ?_ ?_
109    · filter_upwards [self_mem_nhdsWithin] with ε hε
110      have hc := ch_continuous_op (hGε ε hε) hcont hB hB0 hΛ hβ0 hβ hH
111      have : Continuous fun x : Torus d => Torus.mode (-k) x *
112          ((HRS.convGradConv (heatRieszGrad d s ε) h x i : ℝ) : ℂ) :=
113        (Torus.continuous_mode _).mul (Complex.continuous_ofReal.comp
114          ((EuclideanSpace.proj i : Euc d →L[ℝ] ℝ).continuous.comp hc))
115      exact this.aestronglyMeasurable
116    · refine ⟨∫ z, chDom s C Λ B β d z, ?_⟩
117      filter_upwards [self_mem_nhdsWithin] with ε hε
118      refine Eventually.of_forall fun x => ?_
119      rw [norm_mul, Torus.norm_mode, one_mul, Complex.norm_real, Real.norm_eq_abs]
120      refine le_trans ?_ (ch_bound_op (hGε ε hε) hB hB0 hΛ hβ0 hβ hH x)
121      have := PiLp.norm_apply_le (HRS.convGradConv (heatRieszGrad d s ε) h x) i
122      simpa using this
123    · refine Eventually.of_forall fun x => ?_
124      have := ch_tendsto_heat hsd hGε hcont hB hB0 hΛ hβ0 hβ hH x
125      have h2 := ((EuclideanSpace.proj i : Euc d →L[ℝ] ℝ).continuous.tendsto _).comp this
126      exact (tendsto_const_nhds.mul (Complex.continuous_ofReal.continuousAt.tendsto.comp h2))
127  set R : ℝ → ℂ := fun ε => (∫ y, Torus.mode (-k) y * (h y : ℂ)) *
128    ((2 * Real.pi * Complex.I * ((k i : ℤ) : ℂ)) *
129      ((rieszCoeff d s k * Real.exp (-ε * Lattice.enorm k ^ 2) : ℝ) : ℂ)) with hR
130  have hRc : Continuous R := by
131    simp only [hR]
132    fun_prop
133  have hRHS : Tendsto R l (𝓝 ((∫ y, Torus.mode (-k) y * (h y : ℂ)) *
134      ((2 * Real.pi * Complex.I * ((k i : ℤ) : ℂ)) * ((rieszCoeff d s k : ℝ) : ℂ)))) := by
135    have := (hRc.tendsto 0).mono_left (nhdsWithin_le_nhds (s := Set.Ioi (0 : ℝ)))
136    simpa [hR] using this
137  have hEq : (fun ε : ℝ => ∫ x, Torus.mode (-k) x *
138      ((HRS.convGradConv (heatRieszGrad d s ε) h x i : ℝ) : ℂ)) =ᶠ[l] R := by
139    filter_upwards [self_mem_nhdsWithin] with ε hε
140    exact KernelGrad.coeff_convGradConv_heat hsd hε (hGε ε hε) hcont hB k i
141  exact tendsto_nhds_unique (hLHS.congr' hEq) hRHS
