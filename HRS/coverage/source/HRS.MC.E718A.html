3/-!
4# E718A: Proposition 7.18, non-tightness and absence of a square-mean limit
5
6Proofs of the two statements `HRS.chaos_laws_not_tight` and `HRS.chaos_no_L2_limit` (stated in `HRS/MC/S7Chaos.lean`, with
7the statements below copied verbatim).  Both follow from the proved packets `S7E4.e4_sigma_top` (the variance of `Q_ε`
8tends to infinity) and `S7E5.s7_E5_not_tight_proof` (`P(|Q_ε| > R) → 1`).
9
10* Not tight: a tight family has, for the level `1/2`, a compact set `K ⊆ [-R,R]` with `P(Q_{ε_n} ∉ K) ≤ 1/2` for all `n`,
11  hence `P(|Q_{ε_n}| > R) ≤ 1/2`, against `P(|Q_{ε_n}| > R) → 1`.
12* No limit in square mean: if `E|Q_{ε_n} - Q₀|² → 0`, then `Q₀` is square integrable (as `Q_{ε_n}` is for one `n`, since
13  `E Q_{ε_n}² > 0` there), and `E Q_{ε_n}² ≤ 2 E|Q_{ε_n} - Q₀|² + 2 E Q₀²` stays bounded, against `E Q_ε² → ∞`.
14-/

27theorem chaos_laws_not_tight_proof (hd : 2 ≤ d) {s : ℝ} (hs : (d : ℝ) / 2 ≤ s) (hsd : s < d) {β : ℝ}
28    (hβ : 0 < β) (ρ : Torus d → ℝ) (hρ : SmoothPosDensity ρ) (φ : Torus d → ℝ)
29    (hφ : Torus.IsSmooth φ) (hnc : ∃ x y, φ x ≠ φ y) (e : ℕ → Torus d → ℝ)
30    (he : IsONBL2Zero (withDensityReal ρ) e) {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω)
31    [IsProbabilityMeasure P] (G : ℕ → Ω → ℝ) (hG : s7_GaussData d s β ρ e P G) (Q : ℝ → Ω → ℝ)
32    (hQ : ∀ ε, 0 < ε → AEMeasurable (Q ε) P ∧ s7_IsChaosLimit d s β ρ e P G (hCut d s φ ε) (Q ε))
33    (ε : ℕ → ℝ) (hε0 : ∀ n, 0 < ε n) (hε : Tendsto ε atTop (𝓝 0)) :
34    ¬ IsTightMeasureSet (Set.range fun n : ℕ => P.map (Q (ε n))) := by
35  intro htight
36  rw [isTightMeasureSet_iff_exists_isCompact_measure_compl_le] at htight
37  obtain ⟨K, hK, hKb⟩ := htight (1 / 2) (by norm_num)
38  obtain ⟨R, hR⟩ := hK.isBounded.exists_norm_le
39  have h5 := HRS.MC.Sec7.E5.s7_E5_not_tight_proof hd hs hsd hβ ρ hρ φ hφ hnc e he P G hG Q hQ R
40  have hεw : Tendsto ε atTop (𝓝[>] 0) :=
41    tendsto_nhdsWithin_iff.2 ⟨hε, Eventually.of_forall hε0⟩
42  have h6 := h5.comp hεw
43  have h7 : ∀ᶠ n in atTop, (1 / 2 : ℝ≥0∞) < P {ω | R < |Q (ε n) ω|} :=
44    h6.eventually (lt_mem_nhds (by
45      rw [ENNReal.div_lt_iff (Or.inl (by norm_num)) (Or.inl (by norm_num))]; norm_num))
46  obtain ⟨n, hn⟩ := h7.exists
47  have hmeas : AEMeasurable (Q (ε n)) P := (hQ (ε n) (hε0 n)).1
48  have h8 := hKb (P.map (Q (ε n))) ⟨n, rfl⟩
49  rw [Measure.map_apply_of_aemeasurable hmeas hK.isClosed.measurableSet.compl] at h8
50  have hsub : {ω | R < |Q (ε n) ω|} ⊆ Q (ε n) ⁻¹' Kᶜ := by
51    intro ω hω hmem
52    have := hR _ hmem
53    rw [Real.norm_eq_abs] at this
54    exact absurd hω (not_lt.2 this)
55  exact absurd (lt_of_lt_of_le hn ((measure_mono hsub).trans h8)) (lt_irrefl _)

60theorem enorm_sq_le (a b : ℝ) : ‖a‖ₑ ^ 2 ≤ 2 * ‖a - b‖ₑ ^ 2 + 2 * ‖b‖ₑ ^ 2 := by
61  rw [enorm_sq_eq, enorm_sq_eq, enorm_sq_eq]
62  have h : a ^ 2 ≤ 2 * (a - b) ^ 2 + 2 * b ^ 2 := by nlinarith [sq_nonneg (a - 2 * b)]
63  calc ENNReal.ofReal (a ^ 2) ≤ ENNReal.ofReal (2 * (a - b) ^ 2 + 2 * b ^ 2) := ENNReal.ofReal_le_ofReal h
64    _ = 2 * ENNReal.ofReal ((a - b) ^ 2) + 2 * ENNReal.ofReal (b ^ 2) := by
65        rw [ENNReal.ofReal_add (by positivity) (by positivity), ENNReal.ofReal_mul (by norm_num),
66          ENNReal.ofReal_mul (by norm_num)]
67        simp

75theorem chaos_no_L2_limit_proof (hd : 2 ≤ d) {s : ℝ} (hs : (d : ℝ) / 2 ≤ s) (hsd : s < d) {β : ℝ}
76    (hβ : 0 < β) (ρ : Torus d → ℝ) (hρ : SmoothPosDensity ρ) (φ : Torus d → ℝ)
77    (hφ : Torus.IsSmooth φ) (hnc : ∃ x y, φ x ≠ φ y) (e : ℕ → Torus d → ℝ)
78    (he : IsONBL2Zero (withDensityReal ρ) e) {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω)
79    [IsProbabilityMeasure P] (G : ℕ → Ω → ℝ) (hG : s7_GaussData d s β ρ e P G) (Q : ℝ → Ω → ℝ)
80    (hQ : ∀ ε, 0 < ε → AEMeasurable (Q ε) P ∧ s7_IsChaosLimit d s β ρ e P G (hCut d s φ ε) (Q ε))
81    (ε : ℕ → ℝ) (hε0 : ∀ n, 0 < ε n) (hε : Tendsto ε atTop (𝓝 0)) :
82    ¬ ∃ Q₀ : Ω → ℝ, AEMeasurable Q₀ P ∧
83      Tendsto (fun n : ℕ => ∫⁻ ω, ‖Q (ε n) ω - Q₀ ω‖ₑ ^ 2 ∂P) atTop (𝓝 0) := by
84  rintro ⟨Q₀, hQ₀, hlim⟩
85  have hεw : Tendsto ε atTop (𝓝[>] 0) :=
86    tendsto_nhdsWithin_iff.2 ⟨hε, Eventually.of_forall hε0⟩
87  have hσ := (HRS.MC.Sec7.E4.e4_sigma_top hd hs hsd hβ ρ hρ φ hφ hnc e he P G hG Q hQ).comp hεw
88  have hmeas : ∀ n, AEMeasurable (Q (ε n)) P := fun n => (hQ (ε n) (hε0 n)).1
89  have hd1 : ∀ᶠ n in atTop, ∫⁻ ω, ‖Q (ε n) ω - Q₀ ω‖ₑ ^ 2 ∂P ≤ 1 :=
90    hlim.eventually (ge_mem_nhds (by norm_num))
91  have hbig : ∀ᶠ n in atTop, (1 : ℝ) ≤ ∫ ω, Q (ε n) ω ^ 2 ∂P := hσ.eventually_ge_atTop 1
92  obtain ⟨n0, hn0a, hn0b⟩ := (hd1.and hbig).exists
93  have hfin : ∫⁻ ω, ‖Q (ε n0) ω‖ₑ ^ 2 ∂P ≠ ⊤ := by
94    intro htop
95    rw [integral_sq_eq_toReal P (hmeas n0), htop] at hn0b
96    simp at hn0b
97    linarith
98  -- `Q₀` is square integrable
99  have hmeasD : ∀ n, AEMeasurable (fun ω => ‖Q (ε n) ω - Q₀ ω‖ₑ ^ 2) P := fun n =>
100    (((hmeas n).sub hQ₀).enorm).pow_const 2
101  have hmeasQ : ∀ n, AEMeasurable (fun ω => ‖Q (ε n) ω‖ₑ ^ 2) P := fun n =>
102    ((hmeas n).enorm).pow_const 2
103  have hmeas0 : AEMeasurable (fun ω => ‖Q₀ ω‖ₑ ^ 2) P := hQ₀.enorm.pow_const 2
104  set L0 : ℝ≥0∞ := ∫⁻ ω, ‖Q₀ ω‖ₑ ^ 2 ∂P with hL0def
105  have hL0 : L0 ≤ 2 * 1 + 2 * ∫⁻ ω, ‖Q (ε n0) ω‖ₑ ^ 2 ∂P := by
106    calc L0 ≤ ∫⁻ ω, (2 * ‖Q (ε n0) ω - Q₀ ω‖ₑ ^ 2 + 2 * ‖Q (ε n0) ω‖ₑ ^ 2) ∂P := by
107          refine lintegral_mono fun ω => ?_
108          have := enorm_sq_le (Q₀ ω) (Q (ε n0) ω)
109          rwa [enorm_sub_rev] at this
110      _ = 2 * ∫⁻ ω, ‖Q (ε n0) ω - Q₀ ω‖ₑ ^ 2 ∂P + 2 * ∫⁻ ω, ‖Q (ε n0) ω‖ₑ ^ 2 ∂P := by
111          rw [lintegral_add_left' ((hmeasD n0).const_mul 2), lintegral_const_mul'' _ (hmeasD n0),
112            lintegral_const_mul'' _ (hmeasQ n0)]
113      _ ≤ _ := by gcongr
114  have hL0fin : L0 ≠ ⊤ :=
115    ne_top_of_le_ne_top (by simp [ENNReal.mul_eq_top, hfin]) hL0
116  -- the second moments of `Q (ε n)` stay bounded
117  set B : ℝ≥0∞ := 2 * 1 + 2 * L0 with hBdef
118  have hBfin : B ≠ ⊤ := by simp [hBdef, ENNReal.mul_eq_top, hL0fin]
119  have hbd : ∀ᶠ n in atTop, ∫ ω, Q (ε n) ω ^ 2 ∂P ≤ B.toReal := by
120    filter_upwards [hd1] with n hn
121    rw [integral_sq_eq_toReal P (hmeas n)]
122    refine ENNReal.toReal_mono hBfin ?_
123    calc ∫⁻ ω, ‖Q (ε n) ω‖ₑ ^ 2 ∂P
124        ≤ ∫⁻ ω, (2 * ‖Q (ε n) ω - Q₀ ω‖ₑ ^ 2 + 2 * ‖Q₀ ω‖ₑ ^ 2) ∂P :=
125          lintegral_mono fun ω => enorm_sq_le (Q (ε n) ω) (Q₀ ω)
126      _ = 2 * ∫⁻ ω, ‖Q (ε n) ω - Q₀ ω‖ₑ ^ 2 ∂P + 2 * L0 := by
127          rw [lintegral_add_left' ((hmeasD n).const_mul 2), lintegral_const_mul'' _ (hmeasD n),
128            lintegral_const_mul'' _ hmeas0]
129      _ ≤ 2 * 1 + 2 * L0 := by gcongr
130  obtain ⟨n1, hn1⟩ := (hbd.and (hσ.eventually_ge_atTop (B.toReal + 1))).exists
131  have h1 := hn1.1
132  have h2 := hn1.2
133  have h2' : B.toReal + 1 ≤ ∫ ω, Q (ε n1) ω ^ 2 ∂P := h2
134  linarith
