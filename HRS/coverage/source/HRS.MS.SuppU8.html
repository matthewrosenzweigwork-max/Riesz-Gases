5/-!
6# Worker U8: proofs of `HRS.Whole.fractional_decay` and `HRS.Whole.sharp_tail_bound`
7
8Route: for `α > 0` and `ℓ = ⌊α⌋ + 1` the multiplier `(2π|ξ|)^α` is a constant multiple of
9`∫ (2 - 2 cos (2π ⟪y, ξ⟫))^ℓ ‖y‖^{-d-α} dy`, i.e. `|D|^α` is the singular integral of the `ℓ`-th
10central difference.  For a function with decay norm `‖v‖_{k,b}` (`k > α`, `b > 0`), the near/far
11splitting of that integral gives `| |D|^α v (x) | ≲ ‖v‖_{k,b} ⟨x⟩^{-(b + α)}` pointwise (this uses only the
12orders `0` and `ℓ` of the derivatives), whence the `L^p` bound for `b + α > d/p`.  The identity
13`∫ v |D|^α φ = ∫ w φ` is the symmetry of the central difference (Fubini).
14-/

36    refine ⟨(max ((d : ℝ) / p - α) 0 + min bexp d) / 2, ?_, ?_, ?_, ?_⟩ <;>

44      have e5 := min_le_right bexp (d : ℝ)

61    rw [hN]; exact (ENNReal.ofReal_toReal hfin.ne).symm

63  have hvl : ContDiff ℝ ((⌊α⌋₊ + 1 : ℕ) : WithTop ℕ∞) v := hv.of_le (by exact_mod_cast hℓk)
64  have hconv := u8_conv hd1 k bexp hb.le v hfin
65  have hB0 : 0 ≤ N * u8_Cconv d k bexp := mul_nonneg hN0 (u8_Cconv_nonneg _ _ _)

67    intro z

81      Real.rpow_le_rpow_of_exponent_le (by linarith [norm_nonneg z]) (by linarith)

87  refine ⟨w, fun φ => hrep φ, hLp.1, ?_⟩

90      ≤ ENNReal.ofReal (C * (N * u8_Cconv d k bexp)) * Eh := by rw [hEh]; exact hLp.2

96    _ = ENNReal.ofReal (C * u8_Cconv d k bexp * Eh.toReal) * ENNReal.ofReal N :=
97        ENNReal.ofReal_mul (mul_nonneg (mul_nonneg hC0 (u8_Cconv_nonneg _ _ _)) ENNReal.toReal_nonneg)
