27theorem mpi_gen_bound (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (m : ℕ) (hm : 1 ≤ m)
28    (R : Regime) (MB Λ VB νbar : ℝ) :
29    ∃ K : ℝ, 0 ≤ K ∧ ∀ (D : DynData d) (ν₀ T : ℝ) (μ : ℝ → Torus d → ℝ)
30      (_ : AdjointData s m R MB Λ VB D ν₀ T μ), D.invβ ≤ νbar →
31      ∀ r ∈ Set.Icc (0 : ℝ) T, ∀ F : Coeff d, MemHs (m : ℝ) F →
32        MemHs ((m : ℝ) - 2) (forwardGen D (μ r) F) ∧
33        hsNormSq ((m : ℝ) - 2) (forwardGen D (μ r) F) ≤ K * hsNormSq (m : ℝ) F := by
34  obtain ⟨K1, hK1, hKb⟩ := TRB.firstOrder_bound hd s hs0 hsd m hm MB Λ VB
35  refine ⟨2 * (max νbar 0) ^ 2 + 2 * K1, by positivity, ?_⟩
36  intro D ν₀ T μ hA hν r hr F hF
37  obtain ⟨hP1, hP2⟩ := hKb D hA.hs hA.hM hA.hV (μ r) (hA.hμ.prob r hr) (hA.hμ.smooth r hr)
38    (hA.hμ.bound r hr) F hF
39  have hle : (m : ℝ) - 2 ≤ (m : ℝ) - 2 + TRB.eps d s := by
40    have := TRB.eps_pos hsd; linarith
41  have hP3 := hsNormSq_mono hle hP1
42  have hF2 : MemHs (((m : ℝ) - 2) + 2) F := by
43    have : ((m : ℝ) - 2) + 2 = m := by ring
44    rw [this]; exact hF
45  obtain ⟨hL1, hL2⟩ := StabAux.laplace_bound hF2
46  have e : ((m : ℝ) - 2) + 2 = m := by ring
47  rw [e] at hL2
48  obtain ⟨hQ1, hQ2⟩ := StabAux.hsNormSq_const_mul ((m : ℝ) - 2) ((D.invβ : ℝ) : ℂ) _ hL1
49  have hν2 : D.invβ ^ 2 ≤ (max νbar 0) ^ 2 := by
50    have h0 : 0 ≤ D.invβ := D.hβ
51    exact pow_le_pow_left₀ h0 (hν.trans (le_max_left _ _)) 2
52  have hQ3 : hsNormSq ((m : ℝ) - 2) (fun k => ((D.invβ : ℝ) : ℂ) * laplaceCoeff F k) ≤
53      (max νbar 0) ^ 2 * hsNormSq (m : ℝ) F := by
54    rw [hQ2, Complex.norm_real, Real.norm_eq_abs, sq_abs]
55    exact mul_le_mul hν2 hL2 (hsNormSq_nonneg _ _) (sq_nonneg _)
56  have hG : forwardGen D (μ r) F = (fun k => ((D.invβ : ℝ) : ℂ) * laplaceCoeff F k) +
57      (fun k => transportCoeff D (μ r) F k - nonlocalCoeff D (μ r) F k) := by
58    funext k
59    simp only [forwardGen, Pi.add_apply]
60    ring
61  have hP1' : MemHs ((m : ℝ) - 2) (fun k => transportCoeff D (μ r) F k - nonlocalCoeff D (μ r) F k) :=
62    hP1.mono hle
63  obtain ⟨hs1, hs2⟩ := ET.memHs_add_sq hQ1 hP1'
64  rw [hG]
65  refine ⟨hs1, hs2.trans ?_⟩
66  have := add_le_add (mul_le_mul_of_nonneg_left hQ3 (by norm_num : (0 : ℝ) ≤ 2))
67    (mul_le_mul_of_nonneg_left (hP3.trans hP2) (by norm_num : (0 : ℝ) ≤ 2))
68  refine this.trans (le_of_eq ?_)
69  ring

74theorem mpi_coeff_increment (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (m : ℕ) (hm : 1 ≤ m)
75    (R : Regime) (MB Λ VB νbar : ℝ) :
76    ∃ K : ℝ, 0 ≤ K ∧ ∀ (D : DynData d) (ν₀ T : ℝ) (μ : ℝ → Torus d → ℝ)
77      (_ : AdjointData s m R MB Λ VB D ν₀ T μ), D.invβ ≤ νbar →
78      ∀ t ∈ Set.Ioc (0 : ℝ) T, ∀ φc : Coeff d, MemHs (m : ℝ) φc →
79      ∀ B : ℝ, 0 ≤ B → (∀ σ ∈ Set.Icc (0 : ℝ) t, PDE.hsNorm (m : ℝ) (propagatorCoeff D μ (m : ℝ) t φc σ) ≤ B) →
80      ∀ τ ∈ Set.Icc (0 : ℝ) t, ∀ τ' ∈ Set.Icc (0 : ℝ) t,
81        MemHs ((m : ℝ) - 2) (propagatorCoeff D μ (m : ℝ) t φc τ' - propagatorCoeff D μ (m : ℝ) t φc τ) ∧
82        hsNormSq ((m : ℝ) - 2) (propagatorCoeff D μ (m : ℝ) t φc τ' - propagatorCoeff D μ (m : ℝ) t φc τ) ≤
83          10 * K * B ^ 2 * (τ' - τ) ^ 2 := by
84  obtain ⟨K, hK0, hKg⟩ := mpi_gen_bound hd s hs0 hsd m hm R MB Λ VB νbar
85  refine ⟨K, hK0, ?_⟩
86  intro D ν₀ T μ hA hν t ht φc hφ B hB0 hBd τ hτ τ' hτ'
87  obtain ⟨C, -, hspec⟩ := propagatorCoeff_spec hd s hs0 hsd m hm R MB Λ VB
88  obtain ⟨hsol, hw, -⟩ := hspec D ν₀ T μ hA t ht φc hφ
89  set F : ℝ → Coeff d := propagatorCoeff D μ (m : ℝ) t φc with hFdef
90  set G : ℝ → Coeff d := fun σ => forwardGen D (μ (t - σ)) (F σ) with hGdef
91  have hFB : ∀ σ ∈ Set.Icc (0 : ℝ) t, hsNormSq (m : ℝ) (F σ) ≤ B ^ 2 := by
92    intro σ hσ
93    have h1 := hBd σ hσ
94    unfold PDE.hsNorm at h1
95    exact (Real.sqrt_le_iff.mp h1).2
96  have hGb : ∀ σ ∈ Set.Icc (0 : ℝ) t, MemHs ((m : ℝ) - 2) (G σ) ∧
97      hsNormSq ((m : ℝ) - 2) (G σ) ≤ K * B ^ 2 := by
98    intro σ hσ
99    obtain ⟨h1, h2⟩ := hKg D ν₀ T μ hA hν (t - σ) (ET.tsub_mem_Icc ht hσ) (F σ) (hw.1 σ hσ)
100    exact ⟨h1, h2.trans (mul_le_mul_of_nonneg_left (hFB σ hσ) hK0)⟩
101  by_cases hne : τ' = τ
102  · subst hne
103    simp only [sub_self, ne_eq, OfNat.ofNat_ne_zero, not_false_eq_true, zero_pow,
104      mul_zero]
105    refine ⟨by simp [MemHs], ?_⟩
106    simp [hsNormSq]
107  · have hG : ContinuousInHs ((m : ℝ) - 2) G t :=
108      (adjoint_torus_time_regularity hd s hs0 hsd m hm R MB Λ VB D ν₀ T μ hA t ht φc hφ).1
109    have hint : ∀ k, ∀ u ∈ Set.Icc (0 : ℝ) t, IntervalIntegrable (fun σ => G σ k) volume 0 u := by
110      intro k u hu
111      rw [intervalIntegrable_iff_integrableOn_Icc_of_le hu.1]
112      exact (hsol.integrable k).mono_set (Set.Icc_subset_Icc_right hu.2)
113    have hFG : ∀ k, ∀ τ ∈ Set.Icc (0 : ℝ) t, ∀ τ' ∈ Set.Icc (0 : ℝ) t,
114        F τ' k - F τ k = ∫ σ in τ..τ', G σ k := by
115      intro k τ hτ τ' hτ'
116      rw [hsol.eq k τ hτ, hsol.eq k τ' hτ']
117      rw [add_sub_add_left_eq_sub, intervalIntegral.integral_interval_sub_left (hint k τ' hτ')
118        (hint k τ hτ)]
119    set η : ℝ := 2 * Real.sqrt K * B with hη
120    have hη0 : 0 ≤ η := by positivity
121    have hη2 : η ^ 2 = 4 * (K * B ^ 2) := by
122      rw [hη]; rw [mul_pow, mul_pow, Real.sq_sqrt hK0]; ring
123    have hsup : ∀ σ ∈ Set.uIcc τ τ', hsNormSq ((m : ℝ) - 2) (G σ - G τ) ≤ η ^ 2 := by
124      intro σ hσ
125      have hσ' : σ ∈ Set.Icc (0 : ℝ) t := Set.uIcc_subset_Icc hτ hτ' hσ
126      have h1 := hsNormSq_sub_le (hGb σ hσ').1 (hGb τ hτ).1
127      rw [hη2]
128      nlinarith [(hGb σ hσ').2, (hGb τ hτ).2]
129    obtain ⟨hq1, hq2⟩ := TRB.quotient_sq_bound hG hFG hτ hτ' hne η hη0 hsup
130    set h : ℂ := ((τ' - τ : ℝ) : ℂ) with hh
131    have hh0 : h ≠ 0 := by rw [hh]; exact_mod_cast sub_ne_zero.mpr hne
132    set q : Coeff d := fun k => (F τ' k - F τ k) / h - G τ k with hq
133    obtain ⟨hs1, hs2⟩ := ET.memHs_add_sq hq1 (hGb τ hτ).1
134    have heq : F τ' - F τ = fun k => h * ((q + G τ) k) := by
135      funext k
136      simp only [Pi.sub_apply, Pi.add_apply, hq]
137      field_simp
138      ring
139    rw [heq]
140    obtain ⟨hc1, hc2⟩ := StabAux.hsNormSq_const_mul ((m : ℝ) - 2) h (q + G τ) hs1
141    refine ⟨hc1, ?_⟩
142    rw [hc2, hh, Complex.norm_real, Real.norm_eq_abs, sq_abs]
143    have : hsNormSq ((m : ℝ) - 2) (q + G τ) ≤ 10 * K * B ^ 2 := by
144      have := (hGb τ hτ).2
145      nlinarith [hq2, hη2]
146    calc (τ' - τ) ^ 2 * hsNormSq ((m : ℝ) - 2) (q + G τ) ≤ (τ' - τ) ^ 2 * (10 * K * B ^ 2) :=
147          mul_le_mul_of_nonneg_left this (sq_nonneg _)
148      _ = _ := by ring

159theorem mpi_forward_sub (hd : 1 ≤ d) {s : ℝ} {m : ℕ} {R : Regime} {MB Λ VB : ℝ} {D : DynData d}

166    ⟨by linarith [hσ.2], by linarith [hσ.1, ht.2]⟩

168      forwardGen D (μ (t - σ)) (F₁ σ) - forwardGen D (μ (t - σ)) (F₂ σ) := fun σ hσ =>

170      (fun j => ((hKm _ (hrT σ hσ)).2 j).memHs0) (h₁.h1 σ hσ) (h₂.h1 σ hσ)
171  refine ⟨fun σ hσ => memHs_sub (h₁.h1 σ hσ) (h₂.h1 σ hσ), fun k => ?_, fun k τ hτ => ?_⟩
172  · refine ((h₁.integrable k).sub (h₂.integrable k)).congr_fun (fun σ hσ => ?_) measurableSet_Icc
173    simp only [hsub σ hσ, Pi.sub_apply]
174  · have hi : ∀ (F : ℝ → Coeff d), (∀ k, IntegrableOn (fun σ => forwardGen D (μ (t - σ)) (F σ) k)
175        (Set.Icc 0 t)) → IntervalIntegrable (fun σ => forwardGen D (μ (t - σ)) (F σ) k)
176          volume 0 τ := fun F hF => by
177      rw [intervalIntegrable_iff_integrableOn_Icc_of_le hτ.1]
178      exact (hF k).mono_set (Set.Icc_subset_Icc_right hτ.2)
179    show F₁ τ k - F₂ τ k = (φc - φc') k + ∫ σ in (0:ℝ)..τ,
180      forwardGen D (μ (t - σ)) (F₁ σ - F₂ σ) k
181    have hc : ∫ σ in (0:ℝ)..τ, forwardGen D (μ (t - σ)) (F₁ σ - F₂ σ) k =
182        ∫ σ in (0:ℝ)..τ, (forwardGen D (μ (t - σ)) (F₁ σ) k -
183          forwardGen D (μ (t - σ)) (F₂ σ) k) := by
184      refine intervalIntegral.integral_congr fun σ hσ => ?_
185      rw [Set.uIcc_of_le hτ.1] at hσ
186      simp only [hsub σ ⟨hσ.1, hσ.2.trans hτ.2⟩, Pi.sub_apply]
187    rw [hc, intervalIntegral.integral_sub (hi F₁ h₁.integrable) (hi F₂ h₂.integrable),
188      h₁.eq k τ hτ, h₂.eq k τ hτ]
189    simp only [Pi.sub_apply]
190    ring

291    refine ⟨h1, ?_⟩
