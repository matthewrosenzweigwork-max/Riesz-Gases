26theorem um_trans_bound (m : ℕ) (χ : Euc d → ℝ) (hχ : IsUlocCutoff χ) (k : Euc d) (z : Euc d) (ψ : ℝ) :
27    HRS.Whole.sobolevNormE (m : ℝ) (fun x => χ (x - z) * Real.cos (2 * Real.pi * inner ℝ k x + ψ)) ≤
28      ENNReal.ofReal ((HRS.Whole.sobolevNormE (m : ℝ) χ).toReal *
29        (Real.sqrt 2 * (1 + 2 * Real.pi)) ^ m * (1 + ‖k‖) ^ m) := by
30  have hfin := um_chi_finite hχ.1 hχ.2.1 (s := (m : ℝ)) (Nat.cast_nonneg m)
31  set N := (HRS.Whole.sobolevNormE (m : ℝ) χ).toReal with hN
32  have e : (fun x => χ (x - z) * Real.cos (2 * Real.pi * inner ℝ k x + ψ)) =
33      fun x => (fun y => χ y * Real.cos (2 * Real.pi * inner ℝ k y +
34        (2 * Real.pi * inner ℝ k z + ψ))) (x - z) := by
35    ext x
36    simp only [inner_sub_right]
37    congr 2
38    ring
39  rw [e]
40  refine (HRS.MS.SobMul.sobolevNormE_translate (m : ℝ) (fun y => χ y * Real.cos (2 * Real.pi * inner ℝ k y +
41    (2 * Real.pi * inner ℝ k z + ψ))) z).le.trans ?_
42  calc _ ≤ ENNReal.ofReal (Real.sqrt (2 ^ (m : ℝ) * wt k ^ (m : ℝ))) *
43        HRS.Whole.sobolevNormE (m : ℝ) χ := um_mod_bound' hχ.1 hχ.2.1 (Nat.cast_nonneg m) k _
44    _ = ENNReal.ofReal (Real.sqrt (2 ^ (m : ℝ) * wt k ^ (m : ℝ)) * N) := by
45        rw [ENNReal.ofReal_mul (Real.sqrt_nonneg _), hN, ENNReal.ofReal_toReal hfin]
46    _ ≤ _ := by
47        apply ENNReal.ofReal_le_ofReal
48        have h0 : 0 ≤ N := ENNReal.toReal_nonneg
49        calc Real.sqrt (2 ^ (m : ℝ) * wt k ^ (m : ℝ)) * N
50            ≤ ((Real.sqrt 2 * (1 + 2 * Real.pi)) ^ m * (1 + ‖k‖) ^ m) * N :=
51              mul_le_mul_of_nonneg_right (um_sqrt_le m k) h0
52          _ = _ := by ring

54/-- Lemma S.A.31(ii) -/
55theorem uloc_embedding_modes_proof (m : ℕ) (χ : Euc d → ℝ) (hχ : IsUlocCutoff χ) :
56    ∃ C : ℝ, ∀ k : Euc d,
57      ulocNormE m χ (fun x => Real.cos (2 * Real.pi * inner ℝ k x)) ≤
58        ENNReal.ofReal (C * (1 + ‖k‖) ^ m) ∧
59      ulocNormE m χ (fun x => Real.sin (2 * Real.pi * inner ℝ k x)) ≤
60        ENNReal.ofReal (C * (1 + ‖k‖) ^ m) := by
61  refine ⟨(HRS.Whole.sobolevNormE (m : ℝ) χ).toReal * (Real.sqrt 2 * (1 + 2 * Real.pi)) ^ m, fun k => ?_⟩
62  constructor
63  · unfold ulocNormE
64    refine iSup_le fun z => ?_
65    have := um_trans_bound m χ hχ k z 0
66    simpa [mul_assoc] using this
67  · unfold ulocNormE
68    refine iSup_le fun z => ?_
69    have := um_trans_bound m χ hχ k z (-(Real.pi / 2))
70    have e : (fun x => χ (x - z) * Real.cos (2 * Real.pi * inner ℝ k x + -(Real.pi / 2))) =
71        fun x => χ (x - z) * Real.sin (2 * Real.pi * inner ℝ k x) := by
72      ext x
73      rw [← sub_eq_add_neg, Real.cos_sub_pi_div_two]
74    rw [e] at this
75    simpa [mul_assoc] using this
