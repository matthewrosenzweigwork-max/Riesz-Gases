8Assembly of the two bounds `gff_core_Wp_trace` (P5a) and `gff_core_Wp_transport` (P5b).  If the energy term `En` is infinite, both right
9sides are infinite, because `σ = sigmaScale > 0` (`gfwp_sigN_pos`) and the constant is positive; otherwise each bound is one of P5a, P5b,

22theorem general_fluctuation_Wp_proof : type_of% @HRS.Uniform.general_fluctuation_Wp := by
23  intro d b p hp
24  obtain ⟨C₀a, hA⟩ := gff_core_Wp_trace_proof b p hp
25  obtain ⟨C₀b, hB⟩ := gff_core_Wp_transport_proof b p hp
26  refine ⟨max C₀a C₀b, fun C₀ hC₀ => ?_⟩
27  obtain ⟨Ca, hCa, hA'⟩ := hA C₀ (le_trans (le_max_left _ _) hC₀)
28  obtain ⟨Cb, hCb, hB'⟩ := hB C₀ (le_trans (le_max_right _ _) hC₀)
29  refine ⟨max Ca Cb, lt_max_of_lt_left hCa, ?_⟩
30  intro S hS hMF hPos N hN Ω _ P _ X₀ W hW X hX hF η₀ hη hind Y hY
31  have hσ : 0 < sigmaScale N (S.κ N) (S.invβ N) := gfwp_sigN_pos_proof S N (by omega)
32  have hC : (0 : ℝ) < max Ca Cb := lt_max_of_lt_left hCa
33  have h1 := hA' S hS hMF hPos N hN P X₀ W hW X hX hF η₀ hη hind Y hY
34  have h2 := hB' S hS hMF hPos N hN P X₀ W hW X hX hF η₀ hη hind Y hY
35  intro σ A q Ξ YN ε En Cv Wt
36  by_cases hEn : En = ⊤
37  · have hσ' : ENNReal.ofReal σ ≠ 0 := (ENNReal.ofReal_pos.2 hσ).ne'
38    have hC' : ENNReal.ofReal (max Ca Cb) ≠ 0 := (ENNReal.ofReal_pos.2 hC).ne'
39    have hsum1 : ENNReal.ofReal A * ε + ENNReal.ofReal σ * En + Cv = ⊤ := by
40      rw [hEn, ENNReal.mul_top hσ']; simp
41    have hsum2 : ENNReal.ofReal A * ε + ENNReal.ofReal σ * En +
42        ENNReal.ofReal (Real.sqrt q) * Wt = ⊤ := by
43      rw [hEn, ENNReal.mul_top hσ']; simp
44    rw [hsum1, hsum2, ENNReal.mul_top hC']
45    exact ⟨le_top, le_top⟩
46  · have e1 := h1 hEn
47    have e2 := h2 hEn
48    refine ⟨e1.trans ?_, e2.trans ?_⟩
49    · exact mul_le_mul' (ENNReal.ofReal_le_ofReal (le_max_left _ _)) le_rfl
50    · exact mul_le_mul' (ENNReal.ofReal_le_ofReal (le_max_right _ _)) le_rfl
