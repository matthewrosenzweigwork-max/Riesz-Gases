19/-- Evaluation of a path on a subset `D` of the time interval. -/
20def p_up4_rEval (m T : ℝ) (D : Set (Set.Icc (0 : ℝ) T)) :
21    PathSpace d m T → (D → NegSobolev d m) := fun w q => w.at' q.1

23theorem p_up4_rEval_continuous (m T : ℝ) (D : Set (Set.Icc (0 : ℝ) T)) :
24    Continuous (p_up4_rEval (d := d) m T D) := by
25  refine continuous_pi fun q => ?_
26  exact (continuous_eval_const (F := C(Set.Icc (0 : ℝ) T, NegSobolev d m)) q.1 :)

28theorem p_up4_rEval_injective (m T : ℝ) (D : Set (Set.Icc (0 : ℝ) T)) (hD : Dense D) :
29    Function.Injective (p_up4_rEval (d := d) m T D) := by
30  intro w w' h
31  have key : (fun t => w.at' t) = fun t => w'.at' t :=
32    Continuous.ext_on hD (f := fun t => w.at' t) (g := fun t => w'.at' t)
33      (w : C(Set.Icc (0 : ℝ) T, NegSobolev d m)).continuous
34      (w' : C(Set.Icc (0 : ℝ) T, NegSobolev d m)).continuous
35      (fun q hq => congrFun h ⟨q, hq⟩)
36  exact ContinuousMap.ext (congrFun key)

38theorem p_up4_embedding (m T : ℝ) (D : Set (Set.Icc (0 : ℝ) T)) [Countable D] (hD : Dense D) :

50theorem ret_window_event {N : ℕ} (hN : 1 ≤ N) {m : ℝ} (hm : (d : ℝ) < 2 * m) (T u : ℝ) (hu : 0 ≤ u)
51    (μ : Measure (Torus d)) [IsProbabilityMeasure μ]
52    (A : Set (PathSpace d m T)) (hA : MeasurableSet A) :
53    ∃ B : Set (Config d N × ({t : ℝ≥0 // (t : ℝ) ≤ u + T} → Config d N)), MeasurableSet B ∧
54      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω)
55        (X₀ : Ω → Config d N) (X : ℝ≥0 → Ω → Config d N),
56        (∀ t, Measurable (X t)) → (∀ᵐ ω ∂P, ∀ i, Continuous fun t => X t ω i) →
57        (windowGibbs m T u μ X ⁻¹' A) =ᵐ[P]
58          ((fun ω => (X₀ ω, fun t : {t : ℝ≥0 // (t : ℝ) ≤ u + T} => X t.1 ω)) ⁻¹' B) := by
59  obtain ⟨D, hDc, hDd⟩ := TopologicalSpace.exists_countable_dense (Set.Icc (0 : ℝ) T)
60  haveI : Countable D := hDc.to_subtype
61  have hemb := p_up4_embedding (d := d) m T D hDd
62  let g : Config d N × ({t : ℝ≥0 // (t : ℝ) ≤ u + T} → Config d N) → (D → NegSobolev d m) :=
63    fun z q => etaE m (z.2 (p_up4_idx T u hu q.1)) μ
64  have hg : Measurable g := by
65    refine measurable_pi_iff.2 fun q => ?_
66    have hc : Continuous fun x : Config d N => etaE m x μ :=
67      (HRS.MD.EQLawPath.eqlaw_continuous_ofMeasureDiff_empirical hm μ hN).const_smul (Real.sqrt N)
68    exact hc.measurable.comp ((measurable_pi_apply _).comp measurable_snd)
69  refine ⟨g ⁻¹' (p_up4_rEval m T D '' A), hg (hemb.measurableSet_image.2 hA), ?_⟩
70  intro Ω _ P X₀ X _ hcont
71  have hc : Continuous fun x : Config d N => etaE m x μ :=
72    (HRS.MD.EQLawPath.eqlaw_continuous_ofMeasureDiff_empirical hm μ hN).const_smul (Real.sqrt N)
73  filter_upwards [hcont] with ω hω
74  have hf : Continuous fun t : Set.Icc (0 : ℝ) T => etaE m (X (Real.toNNReal (u + t)) ω) μ := by
75    refine hc.comp ?_
76    refine continuous_pi fun i => ?_
77    exact (hω i).comp (continuous_real_toNNReal.comp (continuous_const.add continuous_subtype_val))
78  have hw : windowGibbs m T u μ X ω =
79      ⟨fun t : Set.Icc (0 : ℝ) T => etaE m (X (Real.toNNReal (u + t)) ω) μ, hf⟩ := by
80    unfold windowGibbs
81    exact dif_pos hf
82  have hrw : p_up4_rEval m T D (windowGibbs m T u μ X ω) =
83      g (X₀ ω, fun t : {t : ℝ≥0 // (t : ℝ) ≤ u + T} => X t.1 ω) := by
84    rw [hw]; rfl
85  show (windowGibbs m T u μ X ω ∈ A) = ((X₀ ω, fun t : {t : ℝ≥0 // (t : ℝ) ≤ u + T} => X t.1 ω) ∈
86    g ⁻¹' (p_up4_rEval m T D '' A))
87  apply propext
88  rw [Set.mem_preimage, ← hrw]
89  exact (p_up4_rEval_injective m T D hDd).mem_set_image.symm
