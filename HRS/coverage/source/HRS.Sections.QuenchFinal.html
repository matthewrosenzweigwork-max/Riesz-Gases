22theorem quench_modes_proof (hd : 2 ≤ d) {s β β₀ : ℝ} (hβ : 0 < β) (hβ₀ : 0 < β₀) (hne : β₀ ≠ β)
23    (D : DynData d) (hD : IsGradientData D s β (fun _ => 0)) (m : ℝ)
24    {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
25    (Z : ℝ → Ω → NegSobolev d m) (hZ : IsGaussianLimitField D (fun _ _ => 1) volume β₀ m P Z) :
26    (∀ (k : PosHalf d) (b : Bool), ∃ B : ℝ≥0 → Ω → ℝ, IsBrownianReal B P ∧
27        IndepFun (fun ω => NegSobolev.eval (Z 0 ω) (modeFn k.1 b)) (fun ω => fun t => B t ω) P ∧
28        ∀ᵐ ω ∂P, ∀ t : ℝ≥0, NegSobolev.eval (Z t ω) (modeFn k.1 b) =
29          NegSobolev.eval (Z 0 ω) (modeFn k.1 b) -
30            lamMode d s β k.1 * (∫ u in Set.Icc (0 : ℝ) t, NegSobolev.eval (Z u ω) (modeFn k.1 b)) +
31            Real.sqrt (2 / β) * (2 * Real.pi * Lattice.enorm k.1) * B t ω) ∧
32    (∀ (k : PosHalf d) (b : Bool),
33      ∫ ω, NegSobolev.eval (Z 0 ω) (modeFn k.1 b) ^ 2 ∂P = varMode d s β₀ k.1) ∧
34    iIndepFun (fun (kb : PosHalf d × Bool) ω => fun t : ℝ≥0 =>
35      NegSobolev.eval (Z t ω) (modeFn kb.1.1 kb.2)) P ∧
36    (∀ (k : PosHalf d) (b : Bool) (a t r : ℝ), 0 ≤ a → 0 ≤ t → 0 ≤ r →
37      ∫ ω, NegSobolev.eval (Z (a + t) ω) (modeFn k.1 b) * NegSobolev.eval (Z (a + r) ω) (modeFn k.1 b) ∂P =
38        varMode d s β k.1 * Real.exp (-lamMode d s β k.1 * |t - r|) +
39          deltaVar d s β β₀ k.1 * Real.exp (-2 * lamMode d s β k.1 * a) *
40            Real.exp (-lamMode d s β k.1 * (t + r))) :=

52    let Δ : ℝ → (PathSpace d m T [×2]→L[ℝ] ℝ) → ℝ := fun a M =>
53      ∫ ω, M ![procWin m T a Z ω, procWin m T a Z ω] ∂P -
54        ∫ ω, M ![procWin m T 0 Zs ω, procWin m T 0 Zs ω] ∂P'
55    (0 < lamStar d s β ∧ {k : PosHalf d | lamMode d s β k.1 = lamStar d s β}.Finite ∧
56      {k : PosHalf d | lamMode d s β k.1 = lamStar d s β}.Nonempty) ∧
57    (∀ a : ℝ, 0 ≤ a →
58      Summable (fun k : PosHalf d => |deltaVar d s β β₀ k.1| * Real.exp (-2 * lamMode d s β k.1 * a) *
59        (‖modePath m T (lamMode d s β k.1) (modeFn k.1 true)‖ ^ 2 +
60          ‖modePath m T (lamMode d s β k.1) (modeFn k.1 false)‖ ^ 2)) ∧
61      ∀ M : PathSpace d m T [×2]→L[ℝ] ℝ, Δ a M =
62        ∑' k : PosHalf d, deltaVar d s β β₀ k.1 * Real.exp (-2 * lamMode d s β k.1 * a) *
63          (M ![modePath m T (lamMode d s β k.1) (modeFn k.1 true),
64               modePath m T (lamMode d s β k.1) (modeFn k.1 true)] +
65            M ![modePath m T (lamMode d s β k.1) (modeFn k.1 false),
66               modePath m T (lamMode d s β k.1) (modeFn k.1 false)])) ∧
67    (∀ ε : ℝ, 0 < ε → ∃ a₀ : ℝ, ∀ a ≥ a₀, ∀ M : PathSpace d m T [×2]→L[ℝ] ℝ, ‖M‖ ≤ 1 →
68      |Real.exp (2 * lamStar d s β * a) * Δ a M - slowTensor d s β β₀ m T M| ≤ ε) ∧
69    (∃ M : PathSpace d m T [×2]→L[ℝ] ℝ, slowTensor d s β β₀ m T M ≠ 0) ∧
70    ∃ cT CT : ℝ, 0 < cT ∧ cT ≤ CT ∧ ∀ a : ℝ, 0 ≤ a →
71      (∀ M : PathSpace d m T [×2]→L[ℝ] ℝ, ‖M‖ ≤ 1 → |Δ a M| ≤ CT * Real.exp (-2 * lamStar d s β * a)) ∧
72      cT * Real.exp (-2 * lamStar d s β * a) ≤
73        ⨆ M : {M : PathSpace d m T [×2]→L[ℝ] ℝ // ‖M‖ ≤ 1}, |Δ a M.1|

82          (M ![modePath m T (lamMode d s β k.1) (modeFn k.1 true),
83               modePath m T (lamMode d s β k.1) (modeFn k.1 true)] +
84            M ![modePath m T (lamMode d s β k.1) (modeFn k.1 false),
85               modePath m T (lamMode d s β k.1) (modeFn k.1 false)]) :=

102  · intro ε hε
103    have ht := Qch.tendsto_tail (m := m) (T := T) hd hs0 hsd hβ hβ₀ hm hT
104    have h1 : ∀ᶠ a : ℝ in atTop, (∑' k, Qch.gk d s β β₀ m T a k) < ε := ht.eventually (gt_mem_nhds hε)
105    obtain ⟨a₀, ha₀⟩ := eventually_atTop.1 (h1.and (eventually_ge_atTop (0 : ℝ)))

136      ∫ ω, M ![procWin m T a Z ω, procWin m T a Z ω] ∂P =
137        ∫ ω, M ![procWin m T 0 Zs ω, procWin m T 0 Zs ω] ∂P'
