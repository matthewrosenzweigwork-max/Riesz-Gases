88  obtain ⟨C, hC0, hCap⟩ := u1_approx (volume : Measure (Euc d)) hs0 hsd

94  haveI : Nontrivial (Euc d) := Module.nontrivial_of_finrank_pos (R := ℝ) (by rw [hfr]; omega)
95  obtain ⟨x0, y0, hxy0⟩ := exists_pair_ne (Euc d)
96  have hB0 : 0 ≤ B := by
97    have h1 := hh.bound x0 y0 hxy0
98    have h2 : 0 < 1 + rieszWeight s (x0 - y0) := by
99      rw [← hV]; linarith [u1Vs_ge_one (s := s) (norm_nonneg (x0 - y0))]
100    have := (abs_nonneg (h x0 y0)).trans h1
101    exact (mul_nonneg_iff_of_pos_right h2).mp this
102  rcases hB0.eq_or_lt with hB | hBpos
103  · -- `B = 0`: the kernel vanishes a.e.
104    subst hB
105    refine ⟨fun _ _ _ => 0, ?_⟩
106    refine ⟨fun j => ⟨0, fun _ _ => 0, fun _ _ => 0, ⟨fun _ _ => rfl, fun a => a.elim0,
107      fun a => a.elim0⟩, by funext x y; simp [r2FiniteRankKernel]⟩, fun j x y hxy => by simp, ?_⟩

124        exact abs_nonpos_iff.mp h1

127    exact tendsto_const_nhds
128  · obtain ⟨rs, hrs1, hrs2, hrs3⟩ := u1_rseq hd hs0 hs
