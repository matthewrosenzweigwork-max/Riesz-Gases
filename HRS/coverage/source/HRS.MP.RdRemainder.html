339/-- The per-`N` form of the remainder bound. -/
340theorem mrr_perN (hS : RdStanding S R MB Λ ΛV G₀ m20 (Γ + 5) (Γ + 5) C₀ 𝔈) (hΛ : 1 ≤ Λ)
341    {K' : ℝ} {e : ℕ → ℝ} (he : ∀ N, 0 ≤ e N)
342    (hsel : SelectedCommutatorInput S (Γ : ℝ) S.T K' C₀ Λ e)
343    {N : ℕ} (hN : 2 ≤ N)
344    (hcoll : ∀ t ∈ Set.Icc (0 : ℝ) S.T, ∀ᵐ ζ ∂(R N).P,
345      (R N).X (Real.toNNReal t) ζ ∉ eucCollisionSet d N)
346    (Rm Rc : ℝ → (R N).Ω → HNeg d ((Γ + 2 : ℕ) : ℝ))
347    (hRem : IsRemainderW S R N (Γ + 2) Rm) (hCont : HasContVersion S.T (R N).P Rm Rc) :
348    ∫⁻ ζ, ⨆ t ∈ Set.Icc (0 : ℝ) S.T, ‖Rc t ζ‖ₑ ∂(R N).P ≤
349      ENNReal.ofReal (max K' 0 / 2 * S.T * S.sigma N * (𝔈 N + e N)) := by
350  classical
351  have hN1 : 1 ≤ N := by omega
352  have hT : 0 ≤ S.T := S.hT.le
353  obtain ⟨ℱ, W', hF⟩ := hS.filtered N
354  have hXm := mrr_prog_meas ℱ (R N).X hF.progressive hT
355  have hinjAE := mrr_coll_ae (R N).P hT (R N).X hXm (fun t ht => hcoll t ht)
356  set Gc : ℝ → (R N).Ω → ℝ := fun r ζ =>
357    rieszModulatedEnergy d S.s ((R N).X (Real.toNNReal (mrrClamp S.T r)) ζ)
358      (HRS.Whole.withDensityReal (S.ρ N (mrrClamp S.T r))) + oNΛ d S.s C₀ Λ N + e N with hGc
359  have hGm : Measurable fun p : ℝ × (R N).Ω => Gc p.1 p.2 :=
360    ((mrr_energy_meas hS hN1).comp (measurable_fst.prodMk hXm)).add measurable_const
361  have h𝔈 : 0 ≤ 𝔈 N := (hS.energy.pos N hN1).le
362  have hmom : ∀ r ∈ Set.Icc (0 : ℝ) S.T,
363      ∫⁻ ζ, ENNReal.ofReal (Gc r ζ) ∂(R N).P ≤ ENNReal.ofReal (𝔈 N + e N) := by
364    intro r hr
365    have hb := hS.energy.bound N hN1 r hr
366    calc ∫⁻ ζ, ENNReal.ofReal (Gc r ζ) ∂(R N).P
367        ≤ ∫⁻ ζ, (ENNReal.ofReal (correctedEnergyRd S C₀ Λ N ((R N).X (Real.toNNReal r) ζ) r) +
368            ENNReal.ofReal (e N)) ∂(R N).P := by
369          refine lintegral_mono fun ζ => ?_
370          simp only [hGc, mrrClamp_of_mem hr]
371          exact ENNReal.ofReal_add_le
372      _ = ∫⁻ ζ, ENNReal.ofReal (correctedEnergyRd S C₀ Λ N ((R N).X (Real.toNNReal r) ζ) r) ∂(R N).P +
373            ENNReal.ofReal (e N) := by
374          rw [lintegral_add_right _ measurable_const, lintegral_const, measure_univ, mul_one]
375      _ ≤ ENNReal.ofReal (𝔈 N) + ENNReal.ofReal (e N) := add_le_add hb le_rfl
376      _ = ENNReal.ofReal (𝔈 N + e N) := (ENNReal.ofReal_add h𝔈 (he N)).symm
377  have hZ := mrr_time_moment (R N).P Gc hGm _ hmom
378  set Z : (R N).Ω → ℝ≥0∞ := fun ζ => ∫⁻ r in Set.Icc (0 : ℝ) S.T, ENNReal.ofReal (Gc r ζ) with hZdef
379  set c : ℝ := S.sigma N / 2 * max K' 0 with hc
380  have hσ := mrr_sigma_nonneg S N
381  have hc0 : 0 ≤ c := mul_nonneg (by positivity) (le_max_right _ _)
382  obtain ⟨φ, hφt, hφn, hnormF⟩ := HRS.MP.HNegInfra.hneg_norming d (((Γ + 2 : ℕ) : ℝ))
383  have hΓle : (Γ : ℝ) ≤ ((Γ + 2 : ℕ) : ℝ) := by exact_mod_cast Nat.le_add_right Γ 2
384  -- the pathwise bound for one time and one test
385  have hpair : ∀ t ∈ Set.Icc (0 : ℝ) S.T, ∀ n : ℕ, ∀ᵐ ζ ∂(R N).P,
386      ENNReal.ofReal |HNeg.pairing (φ n) (Rc t ζ)| ≤ ENNReal.ofReal c * Z ζ := by
387    intro t ht n
388    filter_upwards [hCont.1 t ht, hRem.2 t ht (φ n) (hφt n), hinjAE] with ζ hRc hpairζ hinj
389    rw [hRc, hpairζ]
390    have hnn' : ∀ r ∈ Set.Icc (0 : ℝ) S.T, Function.Injective ((R N).X (Real.toNNReal r) ζ) →
391        0 ≤ rieszModulatedEnergy d S.s ((R N).X (Real.toNNReal r) ζ)
392          (HRS.Whole.withDensityReal (S.ρ N r)) + oNΛ d S.s C₀ Λ N + e N := by
393      intro r hr hinjr
394      have hreg := (hS.mean_field.regular N hN1).1 r hr
395      have hrho := (hS.mean_field.solution N hN1).density r hr
396      have hle : ∀ x, S.ρ N r x ≤ Λ := fun x =>
397        ((le_abs_self _).trans (mrr_abs_le_cbNorm hreg.1 x)).trans
398          (le_trans (le_add_of_nonneg_right ENNReal.toReal_nonneg) hreg.2.2)
399      have hcomp := hS.energy.completion.2 Λ hΛ N hN (S.ρ N r) hrho hle
400        (fun _ => Q09A.log_integrable_withDensityReal hrho
401          ((hS.mean_field.solution N hN1).second_moment r hr))
402        ((R N).X (Real.toNNReal r) ζ) (fun ⟨i, j, hij, heq⟩ => hij (hinjr heq))
403      exact add_nonneg hcomp (he N)
404    have hb := mrr_path_bound S hsel hN (fun r => (R N).X (Real.toNNReal r) ζ) hnn' hinj ht hΓle
405      (hφt n)
406    have hZeq : ∫⁻ r in Set.Icc (0 : ℝ) S.T, ENNReal.ofReal
407        (rieszModulatedEnergy d S.s ((R N).X (Real.toNNReal r) ζ) (HRS.Whole.withDensityReal (S.ρ N r)) +
408          oNΛ d S.s C₀ Λ N + e N) = Z ζ := by
409      refine setLIntegral_congr_fun measurableSet_Icc fun r hr => ?_
410      simp only [hGc, mrrClamp_of_mem hr]
411    rw [hZeq] at hb
412    rw [abs_mul, abs_of_nonneg (by positivity : (0 : ℝ) ≤ S.sigma N / 2),
413      ENNReal.ofReal_mul (by positivity : (0 : ℝ) ≤ S.sigma N / 2)]
414    have h1 : ENNReal.ofReal (max K' 0 * HRS.Whole.sobolevNorm (((Γ + 2 : ℕ) : ℝ)) (φ n)) ≤
415        ENNReal.ofReal (max K' 0) :=
416      ENNReal.ofReal_le_ofReal (mul_le_of_le_one_right (le_max_right _ _) (hφn n))
417    calc ENNReal.ofReal (S.sigma N / 2) * ENNReal.ofReal |∫ r in (0 : ℝ)..t,
418          commN d S.s (fun x => Torus.matVec S.M.transpose
419            (gradient (propagatorW (S.coeff N) r t (φ n)) x))
420            ((R N).X (Real.toNNReal r) ζ) (HRS.Whole.withDensityReal (S.ρ N r))|
421        ≤ ENNReal.ofReal (S.sigma N / 2) * (ENNReal.ofReal (max K' 0) * Z ζ) := by
422          gcongr
423          exact hb.trans (by gcongr)
424      _ = ENNReal.ofReal c * Z ζ := by
425          rw [← mul_assoc, ← ENNReal.ofReal_mul (by positivity : (0 : ℝ) ≤ S.sigma N / 2)]
426  -- the supremum over all times
427  obtain ⟨D, hDc, hDd⟩ := TopologicalSpace.exists_countable_dense (Set.Icc (0 : ℝ) S.T)
428  haveI : Countable D := hDc.to_subtype
429  have hall : ∀ᵐ ζ ∂(R N).P, ∀ (t : D) (n : ℕ),
430      ENNReal.ofReal |HNeg.pairing (φ n) (Rc t.1.1 ζ)| ≤ ENNReal.ofReal c * Z ζ :=
431    ae_all_iff.2 fun t => ae_all_iff.2 fun n => hpair t.1.1 t.1.2 n
432  have hsup : ∀ᵐ ζ ∂(R N).P, ⨆ t ∈ Set.Icc (0 : ℝ) S.T, ‖Rc t ζ‖ₑ ≤ ENNReal.ofReal c * Z ζ := by
433    filter_upwards [hall, hCont.2] with ζ hζ hcont
434    have hD : ∀ t : Set.Icc (0 : ℝ) S.T, t ∈ D → ‖Rc t.1 ζ‖ₑ ≤ ENNReal.ofReal c * Z ζ :=
435      fun t ht => mrr_enorm_le (hnormF (Rc t.1 ζ)) _ (fun n => hζ ⟨t, ht⟩ n)
436    have := mrr_closure_enorm D hDd (fun t => Rc t ζ) hcont _ hD
437    exact iSup₂_le this
438  calc ∫⁻ ζ, ⨆ t ∈ Set.Icc (0 : ℝ) S.T, ‖Rc t ζ‖ₑ ∂(R N).P
439      ≤ ∫⁻ ζ, ENNReal.ofReal c * Z ζ ∂(R N).P := lintegral_mono_ae hsup
440    _ = ENNReal.ofReal c * ∫⁻ ζ, Z ζ ∂(R N).P := lintegral_const_mul' _ _ ENNReal.ofReal_ne_top
441    _ ≤ ENNReal.ofReal c * (ENNReal.ofReal S.T * ENNReal.ofReal (𝔈 N + e N)) := by gcongr
442    _ = ENNReal.ofReal (max K' 0 / 2 * S.T * S.sigma N * (𝔈 N + e N)) := by
443        have h2 : 0 ≤ 𝔈 N + e N := add_nonneg h𝔈 (he N)
444        rw [← ENNReal.ofReal_mul hT, ← ENNReal.ofReal_mul hc0]
445        congr 1
446        rw [hc]; ring

448end PerN
