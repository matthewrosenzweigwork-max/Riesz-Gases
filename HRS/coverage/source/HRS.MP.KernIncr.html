112/-- the norm of `F ∈ NegSobolev` is at most `B` once the trigonometric tests satisfy
113`|F(φ)| ≤ B ‖φ‖_{H^m}`. -/
114theorem mpk_norm_le_of_eval {m : ℝ} (F : NegSobolev d m) (B : ℝ) (hB : 0 ≤ B)
115    (h : ∀ (S : Finset (NonzeroLattice d)) (b : NonzeroLattice d → ℝ),
116      |NegSobolev.eval F (FieldSingAux.trigPoly S b)| ≤
117        B * sobolevNorm m (FieldSingAux.trigPoly S b)) :
118    ‖F‖ ≤ B := by
119  classical
120  obtain ⟨hs, hn⟩ := NegSobolev.coord_sq F
121  have hS : ∀ S : Finset (NonzeroLattice d), ∑ k ∈ S, NegSobolev.coord F k ^ 2 ≤ B ^ 2 := by
122    intro S
123    have h1 : ∀ k, NegSobolev.fourierCoeff F k * mpkWt m (NegSobolev.coord F) k =
124        NegSobolev.coord F k ^ 2 := by
125      intro k
126      unfold NegSobolev.fourierCoeff
127      calc Lattice.jp k.1 ^ m * NegSobolev.coord F k * mpkWt m (NegSobolev.coord F) k
128          = NegSobolev.coord F k * (Lattice.jp k.1 ^ m * mpkWt m (NegSobolev.coord F) k) := by ring
129        _ = _ := by rw [mpkWt_mul]; ring
130    have hev := h S (mpkWt m (NegSobolev.coord F))
131    rw [mpk_eval_trigPoly, mpk_sobolevNorm_trigPoly] at hev
132    simp only [h1, mpkWt_sq] at hev
133    set Q := ∑ k ∈ S, NegSobolev.coord F k ^ 2 with hQ
134    have hQ0 : 0 ≤ Q := Finset.sum_nonneg fun k _ => sq_nonneg _
135    rw [abs_of_nonneg hQ0] at hev
136    have hq0 : 0 ≤ Real.sqrt Q := Real.sqrt_nonneg _
137    have hqq : Real.sqrt Q * Real.sqrt Q = Q := Real.mul_self_sqrt hQ0
138    by_cases hz : Real.sqrt Q = 0
139    · have hQz : Q = 0 := by
140        have := hqq
141        rw [hz] at this
142        linarith
143      rw [hQz]; positivity
144    · have hpos : 0 < Real.sqrt Q := lt_of_le_of_ne hq0 (Ne.symm hz)
145      have : Real.sqrt Q ≤ B := by
146        by_contra hc
147        have hc := not_le.mp hc
148        nlinarith
149      rw [← hqq]
150      nlinarith
151  have hle : ∑' k, NegSobolev.coord F k ^ 2 ≤ B ^ 2 := hs.tsum_le_of_sum_le hS
152  rw [← hn] at hle
153  exact (pow_le_pow_iff_left₀ (norm_nonneg _) hB two_ne_zero).mp hle

155/-- Riesz representation: a linear functional on smooth mean-zero tests, bounded by the `H^m` norm, is
156the pairing with an element of `NegSobolev d m`. -/
157theorem mpk_exists_rep {m : ℝ} (L : (Torus d → ℝ) → ℝ) (B : ℝ) (hB : 0 ≤ B)
158    (hlin : ∀ φ ψ : Torus d → ℝ, MPKTest φ → MPKTest ψ → ∀ α β : ℝ,
159      L (fun x => α * φ x + β * ψ x) = α * L φ + β * L ψ)
160    (hbd : ∀ φ : Torus d → ℝ, MPKTest φ → |L φ| ≤ B * sobolevNorm m φ) :
161    ∃ F : NegSobolev d m, ∀ φ : Torus d → ℝ, MPKTest φ → NegSobolev.eval F φ = L φ := by
162  classical
163  set a : NonzeroLattice d → ℝ := fun k => L (Torus.realMode k) with ha
164  set c : NonzeroLattice d → ℝ := mpkWt m a with hc
165  set b : NonzeroLattice d → ℝ := mpkWt m c with hb
166  have hL0 : L (fun _ => 0) = 0 := by
167    have := hlin _ _ mpk_test_zero mpk_test_zero 0 0
168    simpa using this
169  have hLtrig : ∀ (S : Finset (NonzeroLattice d)) (β : NonzeroLattice d → ℝ),
170      L (FieldSingAux.trigPoly S β) = ∑ k ∈ S, β k * a k := by
171    intro S β
172    induction S using Finset.induction_on with
173    | empty =>
174      have : FieldSingAux.trigPoly (∅ : Finset (NonzeroLattice d)) β = fun _ => 0 := by
175        funext x; simp [FieldSingAux.trigPoly]
176      rw [this, hL0]; simp
177    | insert k S hk ih =>
178      have e : FieldSingAux.trigPoly (insert k S) β =
179          fun x => β k * Torus.realMode k x + 1 * FieldSingAux.trigPoly S β x := by
180        funext x; simp [FieldSingAux.trigPoly, Finset.sum_insert hk]
181      rw [e, hlin _ _ (mpk_test_realMode k) (mpk_test_trigPoly S β), ih, Finset.sum_insert hk]
182      simp [ha]
183  have hQ : ∀ S : Finset (NonzeroLattice d), ∑ k ∈ S, c k ^ 2 ≤ B ^ 2 := by
184    intro S
185    have h1 : ∀ k, b k * a k = c k ^ 2 := by
186      intro k
187      have := mpkWt_mul m c k
188      have h2 := mpkWt_mul m a k
189      have hj := Torus.jp_pos k.1
190      have e1 : c k = Lattice.jp k.1 ^ (-m) * a k := rfl
191      have e2 : b k = Lattice.jp k.1 ^ (-m) * c k := rfl
192      rw [e2, sq]
193      nth_rewrite 2 [e1]
194      ring
195    have hev := hbd _ (mpk_test_trigPoly S b)
196    rw [hLtrig, mpk_sobolevNorm_trigPoly] at hev
197    simp only [h1] at hev
198    simp only [hb, mpkWt_sq] at hev
199    set Q := ∑ k ∈ S, c k ^ 2 with hQdef
200    have hQ0 : 0 ≤ Q := Finset.sum_nonneg fun k _ => sq_nonneg _
201    rw [abs_of_nonneg hQ0] at hev
202    have hq0 : 0 ≤ Real.sqrt Q := Real.sqrt_nonneg _
203    have hqq : Real.sqrt Q * Real.sqrt Q = Q := Real.mul_self_sqrt hQ0
204    by_cases hz : Real.sqrt Q = 0
205    · have hQz : Q = 0 := by
206        have := hqq
207        rw [hz] at this
208        linarith
209      rw [hQz]; positivity
210    · have hpos : 0 < Real.sqrt Q := lt_of_le_of_ne hq0 (Ne.symm hz)
211      have : Real.sqrt Q ≤ B := by
212        by_contra hcon
213        have hcon := not_le.mp hcon
214        nlinarith
215      rw [← hqq]
216      nlinarith
217  have hsum : Summable fun k => c k ^ 2 :=
218    summable_of_sum_le (fun k => sq_nonneg _) hQ
219  have hmem : Memℓp c 2 := (Torus.memℓp_two_iff c).2 hsum
220  let F : NegSobolev d m := (⟨c, hmem⟩ : lp (fun _ : NonzeroLattice d => ℝ) 2)
221  have hcoord : ∀ k, NegSobolev.coord F k = c k := fun k => rfl
222  have hfc : ∀ k, NegSobolev.fourierCoeff F k = a k := by
223    intro k
224    unfold NegSobolev.fourierCoeff
225    rw [hcoord, hc, mpkWt_mul]
226  refine ⟨F, fun φ hφ => ?_⟩
227  -- the Fourier series of `φ`
228  have hS := (A33.a33_memSob m hφ.1).2
229  set w : NonzeroLattice d → ℝ := fun k => Lattice.jp k.1 ^ (2 * m) * Torus.funRealCoeff φ k ^ 2
230    with hw
231  have hcontφ : Continuous φ := mpk_cont_of_smooth hφ.1
232  have hcomb : ∀ T : Finset (NonzeroLattice d),
233      MPKTest (fun x => 1 * φ x + (-1) * FieldSingAux.trigPoly T (Torus.funRealCoeff φ) x) :=
234    fun T => mpk_test_comb hφ (mpk_test_trigPoly T _) 1 (-1)
235  have hdiff : ∀ T : Finset (NonzeroLattice d),
236      |L φ - ∑ k ∈ T, a k * Torus.funRealCoeff φ k| ≤
237        B * Real.sqrt (∑' k, (if k ∈ T then 0 else w k)) := by
238    intro T
239    have hl := hlin φ _ hφ (mpk_test_trigPoly T (Torus.funRealCoeff φ)) 1 (-1)
240    rw [hLtrig] at hl
241    have e1 : L φ - ∑ k ∈ T, a k * Torus.funRealCoeff φ k =
242        L (fun x => 1 * φ x + (-1) * FieldSingAux.trigPoly T (Torus.funRealCoeff φ) x) := by
243      rw [hl]
244      have : ∑ k ∈ T, a k * Torus.funRealCoeff φ k = ∑ k ∈ T, Torus.funRealCoeff φ k * a k :=
245        Finset.sum_congr rfl fun k _ => mul_comm _ _
246      rw [this]; ring
247    rw [e1]
248    refine (hbd _ (hcomb T)).trans ?_
249    refine mul_le_mul_of_nonneg_left ?_ hB
250    unfold sobolevNorm
251    rw [(hcomb T).2, zero_pow two_ne_zero, zero_add]
252    refine Real.sqrt_le_sqrt (le_of_eq (tsum_congr fun k => ?_))
253    rw [A33.a33_funRealCoeff_lin hcontφ (PDE.ClassicalAux.continuous_of_isCk
254      (PDE.ClassicalAux.isCk_of_le (by exact_mod_cast le_top) (FieldSingAux.isSmooth_trigPoly T _)))
255      1 (-1) k, FieldSingAux.funRealCoeff_trigPoly]
256    by_cases hk : k ∈ T
257    · simp [hk]
258    · simp [hk, hw]
259  have htail : Tendsto (fun T : Finset (NonzeroLattice d) => ∑' k, (if k ∈ T then 0 else w k))
260      atTop (𝓝 0) := by
261    have e : ∀ T : Finset (NonzeroLattice d),
262        ∑' k, (if k ∈ T then 0 else w k) = (∑' k, w k) - ∑ k ∈ T, w k := by
263      intro T
264      have hsF : Summable (fun k => if k ∈ T then w k else 0) :=
265        summable_of_ne_finset_zero (s := T) fun k hk => by simp [hk]
266      have : (fun k => if k ∈ T then (0 : ℝ) else w k) =
267          fun k => w k - (if k ∈ T then w k else 0) := by
268        funext k; split_ifs <;> simp
269      have hind : ∑' k, (if k ∈ T then w k else 0) = ∑ k ∈ T, w k := by
270        rw [tsum_eq_sum (s := T) (fun k hk => by simp [hk])]
271        exact Finset.sum_congr rfl fun k hk => by simp [hk]
272      rw [this, hS.tsum_sub hsF, hind]
273    simp_rw [e]
274    have h2 : Tendsto (fun T : Finset (NonzeroLattice d) => (∑' k, w k) - ∑ k ∈ T, w k) atTop
275        (𝓝 ((∑' k, w k) - ∑' k, w k)) := tendsto_const_nhds.sub hS.hasSum
276    simpa using h2
277  have hlim : HasSum (fun k => a k * Torus.funRealCoeff φ k) (L φ) := by
278    have h3 : Tendsto (fun T : Finset (NonzeroLattice d) =>
279        B * Real.sqrt (∑' k, (if k ∈ T then 0 else w k))) atTop (𝓝 0) := by
280      have := (htail.sqrt).const_mul B
281      simpa using this
282    have h4 : Tendsto (fun T : Finset (NonzeroLattice d) =>
283        ‖(∑ k ∈ T, a k * Torus.funRealCoeff φ k) - L φ‖) atTop (𝓝 0) := by
284      refine squeeze_zero (fun T => norm_nonneg _) (fun T => ?_) h3
285      rw [Real.norm_eq_abs, abs_sub_comm]
286      exact hdiff T
287    exact tendsto_iff_norm_sub_tendsto_zero.2 h4
288  unfold NegSobolev.eval
289  simp_rw [hfc]
290  exact hlim.tsum_eq

349theorem mpk_prop_facts (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (m : ℕ)
350    (hm : mStar d ≤ m) (R : Regime) (MB Λ VB ν₀ T : ℝ) :
351    ∃ K : ℝ, 1 ≤ K ∧ ∀ (D : DynData d) (μ : ℝ → Torus d → ℝ),
352      AdjointData s m R MB Λ VB D ν₀ T μ → ∀ r t : ℝ, 0 ≤ r → r ≤ t → t ≤ T →
353      ∀ φ : Torus d → ℝ, Torus.IsSmooth φ →

394theorem mpk_pde (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (Rg : Regime)
395    (ν₀ MB Λ T VB νbar : ℝ) :
396    ∃ C₂ C₃ : ℝ, 0 ≤ C₂ ∧ 0 ≤ C₃ ∧ ∀ (D : DynData d) (μ : ℝ → Torus d → ℝ),
397      AdjointData s (qT d) Rg MB Λ VB D ν₀ T μ → D.invβ ≤ νbar → ∀ (j : Fin d) (x : Torus d),
398      (∀ r t : ℝ, 0 ≤ r → r ≤ t → t ≤ T → ∀ φ : Torus d → ℝ, Torus.IsSmooth φ →
399        |Torus.grad (propagator D μ r t φ) x j| ≤ C₂ * sobolevNorm ((qT d : ℕ) : ℝ) φ) ∧
400      (∀ r s' t : ℝ, 0 ≤ r → r ≤ s' → s' ≤ t → t ≤ T → ∀ φ : Torus d → ℝ, Torus.IsSmooth φ →
401        |Torus.grad (propagator D μ r t φ) x j - Torus.grad (propagator D μ r s' φ) x j| ≤
402          C₃ * |t - s'| * sobolevNorm ((qT d : ℕ) : ℝ) φ) ∧
403      (∀ r t : ℝ, 0 ≤ r → r ≤ t → t ≤ T → ∀ φ ψ : Torus d → ℝ, ∀ α β : ℝ,
404        Torus.IsSmooth φ → Torus.IsSmooth ψ →
405        Torus.grad (propagator D μ r t (fun y => α * φ y + β * ψ y)) x j =
406          α * Torus.grad (propagator D μ r t φ) x j + β * Torus.grad (propagator D μ r t ψ) x j) := by
407  have hm : mStar d ≤ qT d := by unfold qT Main.gammaIdx; omega
408  obtain ⟨K, hK1, hKf⟩ := mpk_prop_facts hd s hs0 hsd (qT d) hm Rg MB Λ VB ν₀ T
409  have hq1 : (d : ℝ) / 2 + 1 < ((qT d : ℕ) : ℝ) := by
410    rw [mpk_qT_cast]
411    have := Nat.cast_nonneg (α := ℝ) ⌊(d : ℝ) / 2⌋₊
412    have := Nat.cast_nonneg (α := ℝ) d
413    linarith
414  have hq2 : (d : ℝ) / 2 + 1 < ((Main.gammaIdx d : ℕ) : ℝ) := by
415    have : ((Main.gammaIdx d : ℕ) : ℝ) = ((qT d : ℕ) : ℝ) - 2 := by
416      unfold qT; push_cast; ring
417    rw [this, mpk_qT_cast]
418    have := Nat.cast_nonneg (α := ℝ) ⌊(d : ℝ) / 2⌋₊
419    have := Nat.cast_nonneg (α := ℝ) d
420    linarith
421  obtain ⟨Cs1, hCs1, hgq⟩ := mpk_grad_sup_le (d := d) (qT d) hq1
422  obtain ⟨Cs2, hCs2, hgg⟩ := mpk_grad_sup_le (d := d) (Main.gammaIdx d) hq2
423  obtain ⟨C1, hC1⟩ := PropIncr.mpi_main hd s hs0 hsd (Main.gammaIdx d)
424    (by unfold Main.gammaIdx; omega) Rg MB Λ VB νbar ν₀ T
425  refine ⟨Cs1 * K, Cs2 * |C1|, by positivity, by positivity, ?_⟩
426  intro D μ hA hν j x
427  refine ⟨?_, ?_, ?_⟩
428  · intro r t hr hrt htT φ hφ
429    obtain ⟨hck, hmem, hn⟩ := hKf D μ hA r t hr hrt htT φ hφ
430    refine (mpk_abs_grad_le _ x j).trans ((hgq _ hmem (mpk_isCk_one hck) x).trans ?_)
431    rw [mul_assoc]
432    exact mul_le_mul_of_nonneg_left hn hCs1
433  · intro r s' t hr hrs hst htT φ hφ
434    obtain ⟨hckt, hmemt, -⟩ := hKf D μ hA r t hr (hrs.trans hst) htT φ hφ
435    obtain ⟨hcks, hmems, -⟩ := hKf D μ hA r s' hr hrs (hst.trans htT) φ hφ
436    have hφS : MemSobolev ((qT d : ℕ) : ℝ) φ := MFSemigroup.memSobolev_of_isSmooth hφ _
437    have hφc : Continuous φ := mpk_cont_of_smooth hφ
438    obtain ⟨hmemd, hnd⟩ := hC1 D μ hA hν r s' t hr hrs hst htT φ hφS hφc
439    have hfe : (fun y => propagator D μ r t φ y - propagator D μ r s' φ y) =
440        fun y => 1 * propagator D μ r t φ y + (-1) * propagator D μ r s' φ y := by
441      funext y; ring
442    have hck1 : Torus.IsCk 1 (fun y => propagator D μ r t φ y - propagator D μ r s' φ y) := by
443      rw [hfe]
444      exact mpk_isCk_comb (mpk_isCk_one hckt) (mpk_isCk_one hcks) 1 (-1)
445    have hg := mpk_grad_comb (mpk_isCk_one hckt) (mpk_isCk_one hcks) 1 (-1) x j
446    have hdiff : Torus.grad (propagator D μ r t φ) x j - Torus.grad (propagator D μ r s' φ) x j =
447        Torus.grad (fun y => propagator D μ r t φ y - propagator D μ r s' φ y) x j := by
448      rw [hfe, hg]; ring
449    rw [hdiff]
450    refine (mpk_abs_grad_le _ x j).trans ((hgg _ hmemd hck1 x).trans ?_)
451    have h0 : 0 ≤ |t - s'| * sobolevNorm ((qT d : ℕ) : ℝ) φ :=
452      mul_nonneg (abs_nonneg _) (Real.sqrt_nonneg _)
453    calc Cs2 * sobolevNorm ((Main.gammaIdx d : ℕ) : ℝ)
454          (fun y => propagator D μ r t φ y - propagator D μ r s' φ y)
455        ≤ Cs2 * (C1 * |t - s'| * sobolevNorm ((qT d : ℕ) : ℝ) φ) :=
456          mul_le_mul_of_nonneg_left hnd hCs2
457      _ ≤ Cs2 * (|C1| * (|t - s'| * sobolevNorm ((qT d : ℕ) : ℝ) φ)) := by
458          refine mul_le_mul_of_nonneg_left ?_ hCs2
459          rw [mul_assoc]
460          exact mul_le_mul_of_nonneg_right (le_abs_self _) h0
461      _ = Cs2 * |C1| * |t - s'| * sobolevNorm ((qT d : ℕ) : ℝ) φ := by ring
462  · intro r t hr hrt htT φ ψ α β hφ hψ
463    have hcomb : Torus.IsSmooth (fun y => α * φ y + β * ψ y) := mpk_isCk_comb hφ hψ α β
464    obtain ⟨hckφ, -, -⟩ := hKf D μ hA r t hr hrt htT φ hφ
465    obtain ⟨hckψ, -, -⟩ := hKf D μ hA r t hr hrt htT ψ hψ
466    have hlin : propagator D μ r t (fun y => α * φ y + β * ψ y) =
467        fun y => α * propagator D μ r t φ y + β * propagator D μ r t ψ y := by
468      by_cases ht0 : t = 0
469      · subst ht0
470        obtain rfl : r = 0 := le_antisymm hrt hr
471        have hz : ∀ χ : Torus d → ℝ, Torus.IsSmooth χ → propagator D μ 0 0 χ = χ := fun χ hχ =>
472          mpk_prop_zero hd hA (MFSemigroup.isCk_nat_of_isSmooth hχ 2)
473        rw [hz _ hcomb, hz _ hφ, hz _ hψ]
474      · have ht : t ∈ Set.Ioc (0 : ℝ) T := ⟨lt_of_le_of_ne (hr.trans hrt) (Ne.symm ht0), htT⟩
475        exact A33.prop_lin hd s hs0 hsd (qT d) hm Rg MB Λ VB D ν₀ T μ hA ht (A33.dom_smooth hφ)
476          (A33.dom_smooth hψ) α β ⟨hr, hrt⟩
477    rw [hlin]
478    exact mpk_grad_comb (mpk_isCk_one hckφ) (mpk_isCk_one hckψ) α β x j

483    (ν₀ MB Λ T VB νbar : ℝ) :
484    ∃ C : ℝ, ∀ {ι : Type} (D : ι → DynData d) (ρ : ι → ℝ → Torus d → ℝ),
485      (∀ i, PDE.AdjointData s (Main.gammaIdx d + 2) Rg MB Λ VB (D i) ν₀ T (ρ i)) →
486      (∀ i, (D i).invβ ≤ νbar) →
487      (∀ i : ι, ∀ (j : Fin d) (r t : ℝ), 0 ≤ r → r ≤ t → t ≤ T → ∀ x : Torus d,
488        ∃! k : NegSobolev d ((qT d : ℕ) : ℝ), ∀ φ : Torus d → ℝ, Torus.IsSmooth φ →
489          Torus.mean φ = 0 → NegSobolev.eval k φ = (Torus.grad (propagator (D i) (ρ i) r t φ) x) j) ∧
490      ∀ Kern : ι → Fin d → ℝ → ℝ → Torus d → NegSobolev d ((qT d : ℕ) : ℝ),

494        ∀ i : ι, ∀ (j : Fin d) (r s' t : ℝ), 0 ≤ r → r ≤ s' → s' ≤ t → t ≤ T → ∀ x : Torus d,
495          ‖Kern i j r t x - Kern i j r s' x‖ ≤ C * |t - s'| ∧ ‖Kern i j r t x‖ ≤ C := by
496  by_cases hcase : 1 ≤ d ∧ 0 ≤ s ∧ s < d

498    obtain ⟨C₂, C₃, hC₂, hC₃, hP⟩ := mpk_pde hd s hs0 hsd Rg ν₀ MB Λ T VB νbar

508      refine ⟨F, fun φ hφs hφm => hF φ ⟨hφs, hφm⟩, fun F' hF' => ?_⟩
509      have hn : ‖F' - F‖ ≤ 0 := by

513        simp

515      exact sub_eq_zero.1 this

518      have hrt : r ≤ t := hrs.trans hst
519      have hrs' : r ≤ s' := hrs
520      have hs'T : s' ≤ T := hst.trans htT
521      refine ⟨?_, ?_⟩

523          refine mpk_norm_le_of_eval _ _ (mul_nonneg hC₃ (abs_nonneg _)) fun S b => ?_
524          rw [mpk_eval_sub_trig, hK i j r t hr hrt htT x _ (FieldSingAux.isSmooth_trigPoly S b)
525            (FieldSingAux.mean_trigPoly S b), hK i j r s' hr hrs' hs'T x _

527          exact hinc r s' t hr hrs hst htT _ (FieldSingAux.isSmooth_trigPoly S b)
