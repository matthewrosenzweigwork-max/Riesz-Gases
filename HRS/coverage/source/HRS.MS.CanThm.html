28/-- `exp(-β V) = k⁻¹ γ` for the confinement `V = a|x|²/2` and the Gaussian comparison density `γ`. -/
29lemma can2_w_eq {a β : ℝ} (ha : 0 < a) (hβ : 0 < β) (x : Euc d) :
30    Real.exp (-β * confinement a x) =
31      ((β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2))⁻¹ * gaussianComparison d a β x := by
32  have hk : 0 < (β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2) := by positivity
33  rw [gaussianComparison_eq, ← mul_assoc, inv_mul_cancel₀ hk.ne', one_mul]
34  unfold confinement
35  congr 1
36  ring

49/-- The abstract setting holds for the Riesz kernel with `0 < s < d` and the confinement. -/
50lemma can2_setting_pos (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 < s) (hsd : s < d) {a β : ℝ} (ha : 0 < a)
51    (hβ : 0 < β) :
52    ∃ B : ℝ, Can2Setting (rieszKernel d s) (confinement a) β B := by
53  obtain ⟨B₀, hB₀⟩ := riesz_conv_gaussian_bound (d := d) hd hs0 hsd ha hβ
54  have hk : 0 < (β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2) := by positivity
55  refine ⟨max B₀ 0, ?_⟩
56  have hVm : Measurable (confinement a : Euc d → ℝ) := by
57    unfold confinement; fun_prop
58  refine ⟨hβ, can2_rieszKernel_measurable hs0.ne', fun x => ?_, hVm, ?_, le_max_right _ _, fun y => ?_⟩
59  · simp only [rieszKernel, hs0.ne', ite_false]
60    exact Real.rpow_nonneg (norm_nonneg _) _
61  · have h := (gaussianComparison_integrable (d := d) ha hβ).const_mul
62      ((β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2))⁻¹
63    refine h.congr (ae_of_all _ fun x => ?_)
64    simp only [can2_w_eq ha hβ]
65  · have e : ∀ x : Euc d, rieszKernel d s (x - y) * Real.exp (-β * confinement a x) =
66        ((β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2))⁻¹ *
67          (rieszKernel d s (y - x) * gaussianComparison d a β x) := fun x => by
68      rw [can2_w_eq ha hβ, ← neg_sub y x, can2_rieszKernel_neg]
69      ring
70    simp only [e]
71    refine ⟨((hB₀ y).1.const_mul _), ?_⟩
72    have hL := can2_L_eq (d := d) ha hβ
73    unfold can2_L at hL
74    rw [integral_const_mul, hL]
75    have h1 := (hB₀ y).2
76    have h2 : ∫ x, rieszKernel d s (y - x) * gaussianComparison d a β x ≤ max B₀ 0 := h1.trans (le_max_left _ _)
77    calc ((β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2))⁻¹ *
78          ∫ x, rieszKernel d s (y - x) * gaussianComparison d a β x
79        ≤ ((β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2))⁻¹ * max B₀ 0 :=
80          mul_le_mul_of_nonneg_left h2 (inv_nonneg.2 hk.le)
81      _ = _ := mul_comm _ _

89/-- Clause 2 of `euc_thermal_canonical`. -/
90theorem can2_clause2_pos (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 < s) (hsd : s < d) {a β : ℝ}
91    (ha : 0 < a) (hβ : 0 < β) :
92    ∃ C : ℝ≥0, ∀ N, 2 ≤ N →
93      HRS.Whole.firstMarginal (HRS.Whole.canonicalGibbs (rieszKernel d s) (confinement a) N β) ≤
94        C • thermalMeasure d s a β ∧
95      HRS.Whole.secondMarginal (HRS.Whole.canonicalGibbs (rieszKernel d s) (confinement a) N β) ≤
96        C • (thermalMeasure d s a β).prod (thermalMeasure d s a β) := by
97  have hd1 : 1 ≤ d := by omega
98  obtain ⟨B, hS⟩ := can2_setting_pos hd1 hs0 hsd ha hβ
99  obtain ⟨hprob, c, C', hc, hC', hbd⟩ := cb_thermal_two_sided hd1 hs0 hsd ha hβ
100  set K : ℝ := Real.exp (β * B) / c with hK
101  have hK0 : 0 ≤ K := (div_pos (Real.exp_pos _) hc).le
102  have hdom : ∀ x : Euc d, (can2_L (confinement a : Euc d → ℝ) β * Real.exp (-(β * B)))⁻¹ *
103      Real.exp (-β * confinement a x) ≤ K * thermalDensity d s a β x := by
104    intro x
105    have hk : 0 < (β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2) := by positivity
106    rw [can2_w_eq ha hβ, can2_L_eq ha hβ]
107    have hρ := (hbd x).1
108    have e : (((β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2))⁻¹ * Real.exp (-(β * B)))⁻¹ *
109        (((β * a / (2 * Real.pi)) ^ ((d : ℝ) / 2))⁻¹ * gaussianComparison d a β x) =
110        Real.exp (β * B) * gaussianComparison d a β x := by
111      rw [mul_inv, inv_inv, ← Real.exp_neg, neg_neg]
112      field_simp
113    rw [e, hK, div_mul_eq_mul_div, le_div_iff₀ hc]
114    calc Real.exp (β * B) * gaussianComparison d a β x * c
115        = Real.exp (β * B) * (c * gaussianComparison d a β x) := by ring
116      _ ≤ Real.exp (β * B) * thermalDensity d s a β x :=
117          mul_le_mul_of_nonneg_left hρ (Real.exp_pos _).le
118      _ = _ := by ring
119  refine ⟨(K + K * K).toNNReal, fun N hN => ?_⟩
120  obtain ⟨n, rfl⟩ : ∃ n, N = n + 2 := ⟨N - 2, by omega⟩
121  have hρm := hprob.measurable
122  refine ⟨?_, ?_⟩
123  · have h1 := can2_first_final hS (n + 1) hK0 hdom
124    have e : HRS.Whole.firstMarginal (HRS.Whole.canonicalGibbs (rieszKernel d s) (confinement a) (n + 2) β) =
125        (HRS.Whole.canonicalGibbs (rieszKernel d s) (confinement a) (n + 1 + 1) β).map
126          (fun X : Fin (n + 1 + 1) → Euc d => X 0) := by
127      unfold HRS.Whole.firstMarginal
128      rw [dif_pos (by omega)]
129      rfl
130    rw [e]
131    refine h1.trans ?_
132    exact can2_smul_mono (by
133      change ENNReal.ofReal K ≤ ENNReal.ofReal (K + K * K)
134      exact ENNReal.ofReal_le_ofReal (by nlinarith)) _
135  · have h2 := can2_second_final hS n hρm hK0 hdom
136    have e : HRS.Whole.secondMarginal (HRS.Whole.canonicalGibbs (rieszKernel d s) (confinement a) (n + 2) β) =
137        (HRS.Whole.canonicalGibbs (rieszKernel d s) (confinement a) (n + 2) β).map
138          (fun X : Fin (n + 2) → Euc d => (X 0, X 1)) := by
139      unfold HRS.Whole.secondMarginal
140      rw [dif_pos (by omega)]
141      rfl
142    rw [e]
143    refine h2.trans ?_
144    refine can2_smul_mono ?_ _
145    rw [← ENNReal.ofReal_mul hK0]
146    change ENNReal.ofReal (K * K) ≤ ENNReal.ofReal (K + K * K)
147    exact ENNReal.ofReal_le_ofReal (by nlinarith)
