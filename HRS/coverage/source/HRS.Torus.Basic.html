13## Representation decisions (see `docs/PLAN.md` §(b))
14
15* The torus `𝕋^d = ℝ^d/ℤ^d` is `HRS.Torus d := UnitAddTorus (Fin d)`, i.e.
16  `Fin d → UnitAddCircle`; its `volume` is the product Haar measure, which has
17  total mass one (the paper's "volume one").
18* A configuration `X_N = (x_1,…,x_N)` is `Fin N → Torus d`.
19* Smoothness, derivatives and `C_b^k` norms of a function `φ : Torus d → ℝ` are
20  defined through its periodic lift `φ ∘ Torus.mk : EuclideanSpace ℝ (Fin d) → ℝ`.
21  This makes `ContDiff`/`iteratedFDeriv` from Mathlib available (Mathlib has no
22  smooth structure on `UnitAddTorus`).  Derivative norms are multilinear operator
23  norms with respect to the Euclidean norm on `ℝ^d`, exactly as in the paper
24  ("derivative norms are multilinear operator norms", source line 512).
25* Fourier modes `e^{2πi k·x}` are Mathlib's `UnitAddTorus.mFourier k`.  The real
26  orthonormal basis `e_k` of `L²_0(𝕋^d)` used by the paper (source lines 8274-8275)
27  is `√2 cos(2π k·x)` for `k` in the lexicographically positive half-lattice and
28  `√2 sin(2π k·x)` for `k` in the negative half.
29* `⟨k⟩ = (1+4π²|k|²)^{1/2}` (source line 8129) is `Lattice.jp`.
30-/

64/-- The covering map `ℝ^d → 𝕋^d`. -/
65def mk (x : Euc d) : Torus d := fun i => ((x i : ℝ) : UnitAddCircle)

67/-- The canonical representative in `[0,1)^d` of a point of the torus. -/
68def rep (x : Torus d) : Euc d :=
69  WithLp.toLp 2 fun i => ((AddCircle.equivIco (1 : ℝ) 0 (x i) : Set.Ico (0 : ℝ) (0 + 1)) : ℝ)

71/-- The periodic lift `φ ∘ mk : ℝ^d → E` of a function on the torus. -/
72def lift {E : Type*} (φ : Torus d → E) : Euc d → E := φ ∘ mk
73
74/-- `C^k` regularity of a function on the torus.  Beware Mathlib's order conventions for
75`ContDiff`: `k = ((⊤ : ℕ∞) : WithTop ℕ∞)` (notation `∞` under `open scoped ContDiff`) is `C^∞`,
76whereas `k = (⊤ : WithTop ℕ∞)` (notation `ω`) is *real-analytic*.  Use `IsSmooth` for `C^∞`. -/
77def IsCk {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
78    (k : WithTop ℕ∞) (φ : Torus d → E) : Prop :=
79  ContDiff ℝ k (lift φ)
80
81/-- `C^∞` regularity of a function on the torus through the periodic lift
82(`ContDiff ℝ ∞ (lift φ)`, not analyticity). -/
83def IsSmooth {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] (φ : Torus d → E) : Prop :=
84  IsCk ((⊤ : ℕ∞) : WithTop ℕ∞) φ
85
86/-- The `C_b^k` norm `∑_{j ≤ k} sup_x ‖D^j φ(x)‖` (source line 510, SUM over orders),
87with multilinear operator norms of the Euclidean derivatives. -/
88def cbNorm {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
89    (k : ℕ) (φ : Torus d → E) : ℝ :=
90  ∑ j ∈ Finset.range (k + 1), ⨆ x : Euc d, ‖iteratedFDeriv ℝ j (lift φ) x‖
91
92/-- Sup norm `‖φ‖_∞`. -/
93def supNorm {E : Type*} [NormedAddCommGroup E] (φ : Torus d → E) : ℝ :=
94  ⨆ x : Torus d, ‖φ x‖
95
96/-- Hölder seminorm `[φ]_{C^{0,α}}` with respect to the torus distance. -/
97def holderSeminorm (α : ℝ) (φ : Torus d → ℝ) : ℝ :=
98  ⨆ p : {p : Torus d × Torus d // p.1 ≠ p.2}, |φ p.1.1 - φ p.1.2| / ‖p.1.1 - p.1.2‖ ^ α
99
100/-- Gradient `∇φ` of a scalar function on the torus, as a vector in `ℝ^d`. -/
101def grad (φ : Torus d → ℝ) (x : Torus d) : Euc d := gradient (lift φ) (rep x)
102
103/-- Laplacian `Δφ`. -/
104def laplacian (φ : Torus d → ℝ) (x : Torus d) : ℝ :=
105  Laplacian.laplacian (lift φ) (rep x)
106
107/-- Divergence `Div v` of a vector field. -/
108def div (v : Torus d → Euc d) (x : Torus d) : ℝ :=
109  ∑ i, fderiv ℝ (lift v) (rep x) (EuclideanSpace.single i 1) i
110
111/-- Partial derivative `∂_j φ`. -/
112def partialDeriv (j : Fin d) (φ : Torus d → ℝ) (x : Torus d) : ℝ :=
113  fderiv ℝ (lift φ) (rep x) (EuclideanSpace.single j 1)

115/-- Translation of a function on the torus: `(translate a φ) x = φ (x - a)`. -/
116def translate (a : Torus d) (φ : Torus d → ℝ) : Torus d → ℝ := fun x => φ (x - a)

118/-- Apply a constant real matrix `M` to a vector in `ℝ^d`. -/
119def matVec (M : Matrix (Fin d) (Fin d) ℝ) (v : Euc d) : Euc d := Matrix.toEuclideanLin M v

121/-- `Sym M = (M + Mᵀ)/2`. -/
122def symPart (M : Matrix (Fin d) (Fin d) ℝ) : Matrix (Fin d) (Fin d) ℝ := (1/2 : ℝ) • (M + M.transpose)

124/-- `Sym M ≤ 0` (negative semidefinite symmetric part). -/

164/-- The real orthonormal basis `e_k` of `L²_0(𝕋^d)`, `k ≠ 0` (source lines 8274-8275):
165`√2 cos(2π k·x)` on the positive half-lattice and `√2 sin(2π k·x)` on the negative half. -/
166def realMode (k : NonzeroLattice d) (x : Torus d) : ℝ :=
167  if Lattice.IsPosHalf k.1 then Real.sqrt 2 * cosMode k.1 x else Real.sqrt 2 * sinMode k.1 x

169/-- Complex Fourier coefficient `μ̂(k) = ∫ e^{-2πi k·x} dμ(x)` of a finite measure. -/
170def measureFourierCoeff (μ : Measure (Torus d)) (k : Lattice d) : ℂ :=
171  ∫ x, mode (-k) x ∂μ
172
173/-- Real Fourier coefficient `∫ e_k dμ` of a finite measure against the real basis. -/
174def measureRealCoeff (μ : Measure (Torus d)) (k : NonzeroLattice d) : ℝ :=
175  ∫ x, realMode k x ∂μ

177/-- Complex Fourier coefficient `φ̂(k)` of a function (Mathlib's `mFourierCoeff`). -/
178def funFourierCoeff (φ : Torus d → ℂ) (k : Lattice d) : ℂ := UnitAddTorus.mFourierCoeff φ k
179
180/-- Real Fourier coefficient `∫ e_k φ dx` of a real function. -/
181def funRealCoeff (φ : Torus d → ℝ) (k : NonzeroLattice d) : ℝ :=
182  ∫ x, realMode k x * φ x

184/-- Haar mean `∫ φ dx`. -/
185def mean (φ : Torus d → ℝ) : ℝ := ∫ x, φ x

189/-- The empirical measure `μ_N = N⁻¹ ∑_i δ_{x_i}` (source line 192). -/
190def empirical {d N : ℕ} (X : Config d N) : Measure (Torus d) :=

193/-- Pairing `μ(φ) = ∫ φ dμ`. -/
194def pair {d : ℕ} (μ : Measure (Torus d)) (φ : Torus d → ℝ) : ℝ := ∫ x, φ x ∂μ

196/-- The collision set `𝔇_N = {X : x_i = x_j for some i ≠ j}` (source line ~10115). -/
197def collisionSet (d N : ℕ) : Set (Config d N) := {X | ∃ i j, i ≠ j ∧ X i = X j}

199/-- `L²(μ)` inner product `⟨f, g⟩_{L²(μ)} = ∫ f g dμ`. -/
200def l2Inner {d : ℕ} (μ : Measure (Torus d)) (f g : Torus d → ℝ) : ℝ := ∫ x, f x * g x ∂μ

202/-- Projection modulo constants `P_μ f = f - μ(f)` (source line 936 region). -/
203def projMean {d : ℕ} (μ : Measure (Torus d)) (f : Torus d → ℝ) : Torus d → ℝ :=

818/-- The standard lattice `ℤ^d ⊂ ℝ^d` as a `ℤ`-submodule. -/
819abbrev eucLattice (d : ℕ) : Submodule ℤ (Euc d) :=
820  Submodule.span ℤ (Set.range (EuclideanSpace.basisFun (Fin d) ℝ).toBasis)

822theorem Lattice.toEuc_mem (k : Lattice d) : Lattice.toEuc k ∈ eucLattice d := by
823  have : Lattice.toEuc k = ∑ i, (k i) • (EuclideanSpace.basisFun (Fin d) ℝ).toBasis i := by
824    ext j
825    simp [Lattice.toEuc, EuclideanSpace.basisFun_apply, Pi.single_apply, Finset.sum_apply]
826  rw [this]
827  exact Submodule.sum_mem _ fun i _ => Submodule.smul_mem _ _ (Submodule.subset_span ⟨i, rfl⟩)

829theorem summable_enorm_rpow_aux (r : ℝ) (hr : r < -(d : ℝ)) :
830    Summable fun k : NonzeroLattice d => Lattice.enorm k.1 ^ r := by
831  have hrk : Module.finrank ℤ (eucLattice d) = d := by
832    rw [ZLattice.rank ℝ (eucLattice d)]
833    simp
834  have hs : Summable fun z : eucLattice d => ‖z‖ ^ r := ZLattice.summable_norm_rpow (eucLattice d) r (by rw [hrk]; exact hr)
835  let f : NonzeroLattice d → eucLattice d := fun k => ⟨Lattice.toEuc k.1, Lattice.toEuc_mem k.1⟩
836  have hf : Function.Injective f := by
837    intro k l h
838    apply Subtype.ext
839    funext i
840    have := congrArg (fun z : eucLattice d => (z : Euc d) i) h
841    simpa [f, Lattice.toEuc] using this
842  have hs2 : Summable ((fun z : eucLattice d => ‖z‖ ^ r) ∘ f) := hs.comp_injective hf
843  refine hs2.congr fun k => ?_
844  simp only [Function.comp]
845  rfl

847theorem summable_jp_rpow_aux (m : ℝ) (hm : (d : ℝ) < 2 * m) :
848    Summable fun k : NonzeroLattice d => Lattice.jp k.1 ^ (-(2 * m)) := by
849  have hs := summable_enorm_rpow_aux (d := d) (-(2 * m)) (by linarith)
850  refine Summable.of_nonneg_of_le (fun k => by unfold Lattice.jp; positivity)
851    (fun k => ?_) (hs.mul_left ((2 * Real.pi) ^ (-(2 * m))))
852  have he : 0 < Lattice.enorm k.1 := by
853    unfold Lattice.enorm
854    exact norm_pos_iff.mpr (fun h => k.2 (by
855      funext i
856      have := congrFun (congrArg WithLp.ofLp h) i
857      simpa [Lattice.toEuc] using this))
858  have h1 : 2 * Real.pi * Lattice.enorm k.1 ≤ Lattice.jp k.1 := by
859    unfold Lattice.jp
860    apply Real.le_sqrt_of_sq_le
861    nlinarith [sq_nonneg (Lattice.enorm k.1)]
862  have hm0 : 0 < 2 * m := by
863    have : (0:ℝ) ≤ d := Nat.cast_nonneg d
864    linarith
865  calc Lattice.jp k.1 ^ (-(2 * m)) ≤ (2 * Real.pi * Lattice.enorm k.1) ^ (-(2 * m)) :=
866        Real.rpow_le_rpow_of_nonpos (by positivity) h1 (by linarith)
867    _ = _ := Real.mul_rpow (by positivity) he.le

883theorem jp_le_enorm {d : ℕ} (k : NonzeroLattice d) :
884    Lattice.jp k.1 ≤ (1 + 2 * Real.pi) * Lattice.enorm k.1 := by
885  have he : 1 ≤ Lattice.enorm k.1 := by
886    obtain ⟨i, hi⟩ : ∃ i, k.1 i ≠ 0 := by
887      by_contra h
888      push Not at h
889      exact k.2 (funext h)
890    have h1 : (1 : ℝ) ≤ |(k.1 i : ℝ)| := by
891      have : (1 : ℤ) ≤ |k.1 i| := Int.one_le_abs hi
892      exact_mod_cast this
893    calc (1 : ℝ) ≤ |(k.1 i : ℝ)| := h1
894      _ = ‖(Lattice.toEuc k.1) i‖ := by simp [Lattice.toEuc]
895      _ ≤ ‖Lattice.toEuc k.1‖ := PiLp.norm_apply_le _ i
896  unfold Lattice.jp
897  apply Real.sqrt_le_iff.mpr
898  refine ⟨by positivity, ?_⟩
899  nlinarith [Real.pi_gt_three, mul_pos Real.pi_pos Real.pi_pos]
