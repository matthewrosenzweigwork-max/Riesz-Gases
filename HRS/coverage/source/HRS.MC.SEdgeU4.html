6/-!
7# Packet U4: assembly of `static_edge_char` (relocated proof of `static_edge_char_proof`).
8
9For `p = 1`, the real and imaginary parts of `E exp(i t W_N)` are `E cos(t W_N)` and `E sin(t W_N)`; by
10`static_edge` (B14) the scaled remainders `sed_Uerr` of `cos(t .)`, `sin(t .)` tend to `0` for each `t`,
11by U3 they are equi-Lipschitz on `[-T, T]`, hence uniformly small (U1, composed with the clamp onto
12`[-T, T]`); U0 identifies the Gaussian integrals.
13-/

78lemma u4_identity (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hs : s < (d : ℝ) / 2) (hβ : 0 ≤ β)
79    (hρ : SmoothPosDensity ρ) (hφ : Torus.IsSmooth φ) {N : ℕ} (hN : 1 ≤ N) (t : ℝ) :
80    ∫ X, Complex.exp (Complex.I * (t : ℂ) * ((sed_Y (sed_Q d s N β ρ) φ X : ℝ) : ℂ))
81        ∂(sed_Q d s N β ρ) -
82      Complex.exp (-((sed_Cmat d s β ρ (fun _ : Fin 1 => φ) 0 0 : ℝ) : ℂ) * (t : ℂ) ^ 2 / 2) *
83        (1 + (Complex.I * (t : ℂ)) ^ 3 * ((sed_kappa d s β ρ (fun _ : Fin 1 => φ) 0 0 0 : ℝ) : ℂ) /
84          (6 * (Real.sqrt N : ℂ))) =
85    (((sed_Uerr d s β ρ φ true N t : ℝ) : ℂ) + Complex.I * ((sed_Uerr d s β ρ φ false N t : ℝ) : ℂ)) /
86      (Real.sqrt N : ℂ) := by
87  have hQ : IsProbabilityMeasure (sed_Q d s N β ρ) :=
88    HRS.MC.SEdge.B4b.b4b_isProb hd hs0 hs hβ hρ N hN
89  have hφ1 : ∀ a : Fin 1, Torus.IsSmooth ((fun _ : Fin 1 => φ) a) := fun _ => hφ
90  have hCpsd := HRS.MC.SEdge.B11a.sed_B11a_cov_psd_proof hd hs0 hs hβ hρ hφ1
91  set Cm : Matrix (Fin 1) (Fin 1) ℝ := sed_Cmat d s β ρ (fun _ : Fin 1 => φ) with hCm
92  have hv0 : 0 ≤ Cm 0 0 := hCpsd.diag_nonneg
93  have hCmeq : Matrix.of (fun _ _ : Fin 1 => Cm 0 0) = Cm := by
94    ext i j; fin_cases i; fin_cases j; rfl
95  have hφc : Continuous φ := HRS.Q66.q66_cont_of_smooth hφ
96  obtain ⟨hYm, MY, hMY⟩ := HRS.MC.SEdge.B12.b12_Y_facts hN hφc.measurable
97    (HRS.Q66.q66_bound_of_smooth hφ) (sed_Q d s N β ρ)
98  have hic : Integrable (fun X => Real.cos (t * sed_Y (sed_Q d s N β ρ) φ X)) (sed_Q d s N β ρ) :=
99    Integrable.of_bound (C := 1)
100      (Real.continuous_cos.measurable.comp (measurable_const.mul hYm)).aestronglyMeasurable
101      (Filter.Eventually.of_forall fun X => by rw [Real.norm_eq_abs]; exact Real.abs_cos_le_one _)
102  have his : Integrable (fun X => Real.sin (t * sed_Y (sed_Q d s N β ρ) φ X)) (sed_Q d s N β ρ) :=
103    Integrable.of_bound (C := 1)
104      (Real.continuous_sin.measurable.comp (measurable_const.mul hYm)).aestronglyMeasurable
105      (Filter.Eventually.of_forall fun X => by rw [Real.norm_eq_abs]; exact Real.abs_sin_le_one _)
106  have hexp : ∀ X, Complex.exp (Complex.I * (t : ℂ) * ((sed_Y (sed_Q d s N β ρ) φ X : ℝ) : ℂ)) =
107      ((Real.cos (t * sed_Y (sed_Q d s N β ρ) φ X) : ℝ) : ℂ) +
108        Complex.I * ((Real.sin (t * sed_Y (sed_Q d s N β ρ) φ X) : ℝ) : ℂ) := by
109    intro X
110    rw [show Complex.I * (t : ℂ) * ((sed_Y (sed_Q d s N β ρ) φ X : ℝ) : ℂ) =
111      ((t * sed_Y (sed_Q d s N β ρ) φ X : ℝ) : ℂ) * Complex.I by push_cast; ring,
112      Complex.exp_mul_I, ← Complex.ofReal_cos, ← Complex.ofReal_sin]
113    ring
114  simp_rw [hexp]
115  rw [integral_add hic.ofReal (his.ofReal.const_mul _), integral_const_mul, integral_complex_ofReal,
116    integral_complex_ofReal]
117  obtain ⟨h1, h2⟩ := HRS.MC.SEdge.U0.sed_U0_gauss_char_proof (Cm 0 0) hv0 t
118  obtain ⟨h3, h4⟩ := u4_gauss_d3 (Cm 0 0) hv0 t
119  rw [hCmeq] at h1 h2 h3 h4
120  have e1 : (∫ z, sed_Ucf true t z ∂(multivariateGaussian 0 Cm)) = Real.exp (-(Cm 0 0 * t ^ 2) / 2) := by
121    simpa [sed_Ucf] using h1
122  have e2 : (∫ z, sed_Ucf false t z ∂(multivariateGaussian 0 Cm)) = 0 := by
123    simpa [sed_Ucf] using h2
124  unfold sed_Uerr
125  rw [e1, e2, h3, h4]
126  simp only [u4_ucf_cs, ↓reduceIte, Bool.false_eq_true]
127  have hsN : (Real.sqrt N : ℂ) ≠ 0 := by
128    have : 0 < Real.sqrt N := Real.sqrt_pos.2 (by exact_mod_cast hN)
129    exact_mod_cast this.ne'
130  have hg : Complex.exp (-((Cm 0 0 : ℝ) : ℂ) * (t : ℂ) ^ 2 / 2) =
131      ((Real.exp (-(Cm 0 0 * t ^ 2) / 2) : ℝ) : ℂ) := by
132    rw [Complex.ofReal_exp]; congr 1; push_cast; ring
133  rw [hg]
134  have hI3 : (Complex.I * (t : ℂ)) ^ 3 = -(Complex.I * (t : ℂ) ^ 3) := by
135    rw [mul_pow, pow_succ Complex.I 2, Complex.I_sq]; ring
136  rw [hI3]
137  generalize Real.exp (-(Cm 0 0 * t ^ 2) / 2) = g
138  generalize (∫ X, Real.cos (t * sed_Y (sed_Q d s N β ρ) φ X) ∂(sed_Q d s N β ρ)) = Ec
139  generalize (∫ X, Real.sin (t * sed_Y (sed_Q d s N β ρ) φ X) ∂(sed_Q d s N β ρ)) = Es
140  generalize sed_kappa d s β ρ (fun _ : Fin 1 => φ) 0 0 0 = κ
141  push_cast
142  field_simp
143  ring

150lemma u4_clamp_lip (T t t' : ℝ) :
151    |max (-T) (min t T) - max (-T) (min t' T)| ≤ |t - t'| := by
152  have h1 := abs_max_sub_max_le_max (-T) (min t T) (-T) (min t' T)
153  have h2 := abs_min_sub_min_le_max t T t' T
154  simp only [sub_self, abs_zero] at h1 h2
155  rw [max_eq_right (abs_nonneg (min t T - min t' T))] at h1
156  rw [max_eq_left (abs_nonneg (t - t'))] at h2
157  exact h1.trans h2

159lemma u4_clamp_mem {T : ℝ} (hT : 0 < T) (t : ℝ) : max (-T) (min t T) ∈ Set.Icc (-T) T :=
160  ⟨le_max_left _ _, max_le (by linarith) (min_le_right _ _)⟩

162lemma u4_clamp_self {T t : ℝ} (ht : t ∈ Set.Icc (-T) T) : max (-T) (min t T) = t := by
163  rw [min_eq_left ht.2, max_eq_right ht.1]

169lemma u4_uniform (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hs : s < (d : ℝ) / 2) (hβ : 0 ≤ β)
170    (hρ : SmoothPosDensity ρ) (hφ : Torus.IsSmooth φ) (σ : Bool) (T : ℝ) (hT : 0 < T) (ε : ℝ)
171    (hε : 0 < ε) :
172    ∃ N₀ : ℕ, ∀ N ≥ N₀, 1 ≤ N → ∀ t ∈ Set.Icc (-T) T, |sed_Uerr d s β ρ φ σ N t| ≤ ε := by
173  obtain ⟨L, hL⟩ := HRS.MC.SEdge.U3.sed_U3_equilip_proof hd hs0 hs hβ hρ hφ T hT σ
174  set a : ℕ → ℝ → ℝ := fun n t => sed_Uerr d s β ρ φ σ (n + 1) (max (-T) (min t T)) with ha
175  have hLa : ∀ (n : ℕ) (t t' : ℝ), ‖a n t - a n t'‖ ≤ max L 0 * |t - t'| := by
176    intro n t t'
177    rw [Real.norm_eq_abs]
178    refine (hL (n + 1) (by omega) _ (u4_clamp_mem hT t) _ (u4_clamp_mem hT t')).trans ?_
179    calc L * |max (-T) (min t T) - max (-T) (min t' T)|
180        ≤ max L 0 * |max (-T) (min t T) - max (-T) (min t' T)| :=
181          mul_le_mul_of_nonneg_right (le_max_left _ _) (abs_nonneg _)
182      _ ≤ max L 0 * |t - t'| :=
183          mul_le_mul_of_nonneg_left (u4_clamp_lip T t t') (le_max_right _ _)
184  have hlim : ∀ t ∈ Set.Icc (-T) T, Tendsto (fun n => a n t) atTop (𝓝 ((fun _ : ℝ => (0 : ℝ)) t)) := by
185    intro t ht
186    simp only [ha, u4_clamp_self ht]
187    exact (tendsto_add_atTop_iff_nat (f := fun N => sed_Uerr d s β ρ φ σ N t) 1).2
188      (u4_pointwise hd hs0 hs hβ hρ hφ σ t)
189  obtain ⟨N₀, hN₀⟩ := HRS.MC.SEdge.U1.sed_U1_uniform_of_equilip_proof hLa hlim ε hε
190  refine ⟨N₀ + 1, fun N hN hN1 t ht => ?_⟩
191  obtain ⟨n, rfl⟩ : ∃ n, N = n + 1 := ⟨N - 1, by omega⟩
192  have := hN₀ n (by omega) t ht
193  simpa [ha, u4_clamp_self ht] using this
194
195end Uniform

203theorem static_edge_char_assembly_proof (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hs : s < (d : ℝ) / 2) (β : ℝ)
204    (hβ : 0 ≤ β) (ρ : Torus d → ℝ) (hρ : SmoothPosDensity ρ) (φ : Torus d → ℝ)
205    (hφ : Torus.IsSmooth φ) :
206    let μ := withDensityReal ρ
207    let r : Torus d → ℝ := responseInverse (periodicRiesz d s) μ β φ
208    let v : ℝ := ∫ x, φ x * r x ∂μ
209    let κ : ℝ := ∫ x, r x ^ 3 ∂μ
210    0 < v →
211    ∀ T : ℝ, 0 < T → ∀ ε : ℝ, 0 < ε → ∃ N₀ : ℕ, ∀ N ≥ N₀, 1 ≤ N → ∀ t : ℝ, t ∈ Set.Icc (-T) T →
212      ‖∫ X, Complex.exp (Complex.I * (t : ℂ) *
213            ((Real.sqrt N * (pair (empirical X) φ -
214              ∫ Y, pair (empirical Y) φ ∂(modulatedGibbs (periodicRiesz d s) N β μ)) : ℝ) : ℂ))
215          ∂(modulatedGibbs (periodicRiesz d s) N β μ) -
216        Complex.exp (-(v : ℂ) * (t : ℂ) ^ 2 / 2) *
217          (1 + (Complex.I * (t : ℂ)) ^ 3 * (κ : ℂ) / (6 * (Real.sqrt N : ℂ)))‖ ≤ ε / Real.sqrt N := by
218  dsimp only
219  intro _ T hT ε hε
220  obtain ⟨N₁, hN₁⟩ := u4_uniform hd hs0 hs hβ hρ hφ true T hT (ε / 2) (by positivity)
221  obtain ⟨N₂, hN₂⟩ := u4_uniform hd hs0 hs hβ hρ hφ false T hT (ε / 2) (by positivity)
222  refine ⟨max N₁ N₂, fun N hN hN1 t ht => ?_⟩
223  have hκ : (∫ x, responseInverse (periodicRiesz d s) (withDensityReal ρ) β φ x ^ 3
224      ∂(withDensityReal ρ)) = sed_kappa d s β ρ (fun _ : Fin 1 => φ) 0 0 0 := by
225    unfold sed_kappa
226    congr 1; funext x; ring
227  rw [hκ]
228  have hid := u4_identity hd hs0 hs hβ hρ hφ hN1 t
229  have hsN : 0 < Real.sqrt N := Real.sqrt_pos.2 (by exact_mod_cast hN1)
230  have h1 := hN₁ N (le_trans (le_max_left _ _) hN) hN1 t ht
231  have h2 := hN₂ N (le_trans (le_max_right _ _) hN) hN1 t ht
232  calc _ = ‖(((sed_Uerr d s β ρ φ true N t : ℝ) : ℂ) +
233        Complex.I * ((sed_Uerr d s β ρ φ false N t : ℝ) : ℂ)) / (Real.sqrt N : ℂ)‖ :=
234        congrArg norm hid
235    _ ≤ (ε / 2 + ε / 2) / Real.sqrt N := by
236        rw [norm_div, Complex.norm_real, Real.norm_eq_abs, abs_of_pos hsN]
237        gcongr
238        refine (norm_add_le _ _).trans (add_le_add ?_ ?_)
239        · rw [Complex.norm_real, Real.norm_eq_abs]; exact h1
240        · rw [norm_mul, Complex.norm_I, one_mul, Complex.norm_real, Real.norm_eq_abs]; exact h2
241    _ = ε / Real.sqrt N := by ring_nf
242
243end Final
