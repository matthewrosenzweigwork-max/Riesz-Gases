24theorem path_initial_modulus_rd_limit_proof (s : ℝ) (fd : FieldDensityRd d s)
25    (ν₀ MB Λ ΛV G₀ νbar T Kadj Cw Cin : ℝ) (m₀ : ℕ) (hm₀' : m₀ ≤ mStar d) (χ : Euc d → ℝ)
26    (hχ : IsUlocCutoff χ) :
27    ∃ C : ℝ, ∀ (S : WholeMainData d), S.s = s → S.T = T → PDE.MatBound MB S.M →
28      VHypRd (fd.Gamma + 5) ΛV G₀ S.V → ∀ L : LimitCoeffsRd d,
29      LimitPathHyp S L fd ν₀ Λ Kadj Cw m₀ → L.invβ ≤ νbar → L.A ≤ 1 →
30      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
31        (η : Ω → (Euc d → ℝ) → ℝ),
32      (∀ (ψ ψ' : Euc d → ℝ) (c : ℝ), Y0Class m₀ χ ψ → Y0Class m₀ χ ψ' →
33        ∀ᵐ ζ ∂P, η ζ (c • ψ + ψ') = c * η ζ ψ + η ζ ψ') →
34      (∀ ψ : Euc d → ℝ, Y0Class m₀ χ ψ → ∫⁻ ζ, ENNReal.ofReal ((η ζ ψ) ^ 2) ∂P ≤
35        ENNReal.ofReal (Cin ^ 2 * (varpiNormE m₀ χ ψ).toReal ^ 2)) →
36      ∀ J : ℝ → Ω → HNeg d ((fd.Gamma + 2 : ℕ) : ℝ), (∀ t, Measurable (J t)) →
37        (∀ t ∈ Set.Icc (0 : ℝ) T, ∀ φ : Euc d → ℝ, IsSobolevTest ((fd.Gamma + 2 : ℕ) : ℝ) φ →
38          ∀ᵐ ζ ∂P, HNeg.pairing φ (J t ζ) = L.A * η ζ (propagatorW (L.coeff S) 0 t φ)) →
39        (∀ s' ∈ Set.Icc (0 : ℝ) T, ∀ t ∈ Set.Icc (0 : ℝ) T,
40          ∫⁻ ζ, ‖J t ζ - J s' ζ‖ₑ ^ 2 ∂P ≤ ENNReal.ofReal (C * |t - s'| ^ 2)) ∧
41        ∃ Zc, ContVersionTight (Ω := fun _ : ℕ => Ω) T (fun _ => P) (fun _ t ζ => J t ζ) Zc := by
42  classical
43  by_cases hex : ∃ (S0 : WholeMainData d) (L0 : LimitCoeffsRd d), S0.s = s ∧
44      fd.HoldsAt S0.s S0.M L0.invβ ν₀
45  · obtain ⟨S0, L0, hS0s, hH0⟩ := hex
46    have hΓ : (mStar d : ℝ) + d / 2 + 4 < ((fd.Gamma : ℝ) + 2) := by
47      rw [hS0s] at hH0
48      exact HRS.MP.RdProp.rpi_hGamma fd hH0
49    obtain ⟨Cgen, hCgen0, hgen⟩ := HRS.MP.RdProp.rpi_increment (s := s) S0.hd (hS0s ▸ S0.hs0)
50      (hS0s ▸ S0.hsd) fd.Gamma m₀ hm₀' hΓ MB Λ ΛV G₀ νbar
51    set C₁ : ℝ := max Cw 0 * Real.exp (max Cw 0 * T) * Cgen with hC₁
52    obtain ⟨C, hC⟩ := HRS.MP.RdInit.path_initial_modulus_rd_limit'_of_increment s fd T Kadj Cin C₁ m₀ χ hχ
53    refine ⟨C, fun S hs hT hMB hV L hL hLν hA Ω _ P _ η hlin hη J hJm hJrep => ?_⟩
54    have hreg : IsRegularMFRd s (fd.Gamma + 2 + 3) Λ T L.ρ := by
55      have h : IsRegularMFRd S.s (fd.Gamma + 2 + 3) Λ S.T L.ρ := hL.coeff.regular
56      rw [hs, hT] at h
57      exact h
58    obtain ⟨U, hU⟩ := hL.weighted
59    have hinc : IncrementBoundRd (L.coeff S) T (fd.Gamma + 2) m₀ C₁ :=
60      hgen (L.dyn S) L.ρ T Kadj Cw U hs hMB hV hreg L.hinvβ hLν (hT ▸ hL.adjoint) (hT ▸ hU)
61    exact hC S hT L (hT ▸ hL.adjoint) hinc hA P η hlin hη J hJm hJrep
62  · refine ⟨0, fun S hs hT hMB hV L hL hLν hA Ω _ P _ η hlin hη J hJm hJrep => ?_⟩
63    exfalso
64    apply hex
65    refine ⟨S, L, hs, ?_⟩
66    exact hL.regime
