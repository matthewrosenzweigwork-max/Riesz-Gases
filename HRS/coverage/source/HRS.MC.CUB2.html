5/-!
6# Cubic tensor uniqueness, part 2: agreement on evaluation functionals and the main proof
7
8Two tensors satisfying `SatisfiesCubicFormula` agree on every product of evaluation functionals
9`ℓ_{t,k}` (`t ∈ [0,T]`, `k` a nonzero frequency), provided the cross-covariance densities exist
10(`CrossCovDensitiesExist`); the finite-rank form of `projPoly` then gives `P_L^{⊗3} K₁ = P_L^{⊗3} K₂`
11and the strong convergence `P_L^{⊗3} → id` on `projTensor` gives `K₁ = K₂`.
12-/

26/-- Existence of a witness `(ν, Ω', P', Z₀, Z)` for the cubic formula. -/
27theorem witness_exists (S : PreparedData d) (β₀ m T a : ℝ) (hE : EdgeworthHyp S β₀ m)
28    (hT : 0 < T) (ha : 0 ≤ a) :
29    ∃ ν : Measure (NegSobolev d m), IsCenteredGaussianWithCov m ν (preparedCov d S.s β₀ S.μ0) ∧
30      ∃ (Ω' : Type) (_ : MeasurableSpace Ω') (P' : Measure Ω') (_ : IsProbabilityMeasure P')
31        (Z₀ : Ω' → NegSobolev d m) (Z : ℝ → Ω' → NegSobolev d m), IsPreparedLinear S m (a + T) ν P' Z₀ Z := by
32  have hd : (0 : ℝ) ≤ d := Nat.cast_nonneg d
33  have hm := hE.m_gt
34  obtain ⟨ν, hν, hνc⟩ := preparedGaussian_exists_proof S β₀ hE.β₀_nonneg m (by linarith)
35  refine ⟨ν, hνc, ?_⟩
36  have := hν
37  obtain ⟨Ω', hΩ, P', hP', Z₀, Z, hZ⟩ :=
38    preparedLinear_exists_proof S m (by linarith) (a + T) (by linarith) ν
39  exact ⟨Ω', hΩ, P', hP', Z₀, Z, hZ⟩

46/-- Two tensors with the cubic formula agree on products of evaluation functionals. -/
47theorem agree_lf (S : PreparedData d) (β₀ m T a : ℝ) (hE : EdgeworthHyp S β₀ m)
48    (hT : 0 < T) (ha : 0 ≤ a) (hχ : CrossCovDensitiesExist S β₀ m T a)
49    (K₁ K₂ : projTensor (PathSpace d m T) 3)
50    (h₁ : SatisfiesCubicFormula S β₀ m T a K₁) (h₂ : SatisfiesCubicFormula S β₀ m T a K₂)
51    (n : Fin 3 → Set.Icc (0 : ℝ) T) (k : Fin 3 → NonzeroLattice d) :
52    ((K₁ : projTensor (PathSpace d m T) 3) : TensorDual (PathSpace d m T) 3)
53        (prodFormN 3 (fun i => S7PIC_lf m T (n i) (k i))) =
54      ((K₂ : projTensor (PathSpace d m T) 3) : TensorDual (PathSpace d m T) 3)
55        (prodFormN 3 (fun i => S7PIC_lf m T (n i) (k i))) := by
56  obtain ⟨ν, hν, Ω', hΩ, P', hP', Z₀, Z, hZ⟩ := witness_exists S β₀ m T a hE hT ha
57  set t : Fin 3 → ℝ := fun i => a + (n i : ℝ) with ht_def
58  set φ : Fin 3 → Torus d → ℝ := fun i => Torus.realMode (k i) with hφ_def
59  have ht : ∀ i, t i ∈ Set.Icc a (a + T) := fun i =>
60    ⟨le_add_of_nonneg_right (n i).2.1, by linarith [(n i).2.2]⟩
61  have hφ : ∀ i, Torus.IsSmooth (φ i) := fun i => FieldSingAux.isSmooth_realMode (k i)
62  have hℓ : ∀ i w, S7PIC_lf m T (n i) (k i) w = NegSobolev.eval (pathEval w (t i - a)) (φ i) := by
63    intro i w
64    have : t i - a = (n i : ℝ) := by simp [ht_def]
65    rw [this, pathEval_eq_pathAt, S7PIC_lf_eq_eval]
66  obtain ⟨χ, hχs, hχc⟩ := hχ t φ ht hφ ν hν P' Z₀ Z hZ
67  have e₁ := h₁ t φ ht hφ (fun i => S7PIC_lf m T (n i) (k i)) hℓ ν hν P' Z₀ Z hZ χ hχs hχc
68  have e₂ := h₂ t φ ht hφ (fun i => S7PIC_lf m T (n i) (k i)) hℓ ν hν P' Z₀ Z hZ χ hχs hχc
69  exact e₁.trans e₂.symm

71theorem cubic_tensor_unique_proof (S : PreparedData d) (β₀ m T a : ℝ) (hE : EdgeworthHyp S β₀ m)
72    (hT : 0 < T) (ha : 0 ≤ a) (hχ : CrossCovDensitiesExist S β₀ m T a)
73    (K₁ K₂ : projTensor (PathSpace d m T) 3)
74    (h₁ : SatisfiesCubicFormula S β₀ m T a K₁) (h₂ : SatisfiesCubicFormula S β₀ m T a K₂) :
75    K₁ = K₂ := by
76  obtain ⟨Pj, hPj, hnorm, hconv, hten⟩ := projection_strong_proof (d := d) m T hT
77  refine Subtype.ext ?_
78  refine eq_of_proj Pj _ _ (hten 3 _ K₁.2) (hten 3 _ K₂.2) (fun L => ?_)
79  refine tensorPow_eq_of_agree 3 (Pj L)
80    (fun a : Fin (2 ^ L + 1) × ↥(S7PIC_box d L) => S7PIC_lf m T (S7PIC_node T hT.le L a.1) a.2)
81    (fun a => S7PIC_vpath m T L a.1 a.2) (fun w => ?_) _ _ (fun r => ?_)
82  · rw [hPj, S7PIC_projPoly_eq_sum]
83  · exact agree_lf S β₀ m T a hE hT ha hχ K₁ K₂ h₁ h₂ (fun i => S7PIC_node T hT.le L (r i).1)
84      (fun i => (r i).2)
