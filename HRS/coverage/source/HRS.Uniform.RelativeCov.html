42/-- The finite-dimensional core: the Bures-type bound along the optimal transport map. -/
43theorem core_alg (C D T G : Matrix n n ℝ) (hTs : Tᵀ = T) (hTu : IsUnit T.det)
44    (hD : D = T * T) (hG : G.PosSemidef) (hGG : G * G = T * C * T)
45    (c : ℝ) (hc : 0 < c) (d : n → ℝ) (hd : ∀ i, 0 < d i)
46    (hcmp : (D - c • Matrix.diagonal d).PosSemidef) :
47    C.trace + D.trace - 2 * G.trace ≤
48      2 * c⁻¹ * ∑ i, ∑ j, (C - D) i j ^ 2 / (d i + d j) := by
49  set Ti : Matrix n n ℝ := T⁻¹ with hTi
50  have hTiT : Ti * T = 1 := Matrix.nonsing_inv_mul T hTu
51  have hTTi : T * Ti = 1 := Matrix.mul_nonsing_inv T hTu
52  have hTis : Tiᵀ = Ti := by rw [hTi, Matrix.transpose_nonsing_inv, hTs]
53  set Tm : Matrix n n ℝ := Ti * G * Ti with hTm
54  have hGs : Gᵀ = G := by
55    have := hG.isHermitian
56    rwa [Matrix.IsHermitian, Matrix.conjTranspose_eq_transpose_of_trivial] at this
57  have hTms : Tmᵀ = Tm := by
58    rw [hTm, Matrix.transpose_mul, Matrix.transpose_mul, hTis, hGs, Matrix.mul_assoc]
59  have hTmpsd : Tm.PosSemidef := by
60    have := hG.conjTranspose_mul_mul_same Ti
61    rwa [Matrix.conjTranspose_eq_transpose_of_trivial, hTis] at this
62  set N : Matrix n n ℝ := Tm - 1 with hN
63  have hNs : Nᵀ = N := by rw [hN, Matrix.transpose_sub, hTms, Matrix.transpose_one]
64  have hCeq : C = Tm * D * Tm := by
65    have h1 : C = Ti * (T * C * T) * Ti := by
66      calc C = (Ti * T) * C * (T * Ti) := by rw [hTiT, hTTi]; simp
67        _ = _ := by simp only [Matrix.mul_assoc]
68    have h2 : Tm * D * Tm = Ti * (G * G) * Ti := by
69      calc Tm * D * Tm = Ti * G * (Ti * T) * (T * Ti) * G * Ti := by
70            rw [hTm, hD]; simp only [Matrix.mul_assoc]
71        _ = _ := by rw [hTiT, hTTi]; simp only [Matrix.mul_assoc, Matrix.mul_one, Matrix.one_mul]
72    rw [h2, hGG]; exact h1
73  have hTmN : Tm = N + 1 := by simp [hN]
74  have hDs : Dᵀ = D := by rw [hD, Matrix.transpose_mul, hTs]
75  have hΔ : C - D = D * N + N * D + N * D * N := by
76    rw [hCeq, hTmN]; noncomm_ring
77  -- the squared cost
78  have hb2 : C.trace + D.trace - 2 * G.trace = (D * N * N).trace := by
79    have e1 : (D * N * N).trace = (D * Tm * Tm).trace - 2 * (D * Tm).trace + D.trace := by
80      have : D * N * N = D * Tm * Tm - 2 • (D * Tm) + D := by
81        rw [hN]; noncomm_ring
82      rw [this, Matrix.trace_add, Matrix.trace_sub, Matrix.trace_smul]
83      simp
84    have e2 : (D * Tm * Tm).trace = C.trace := by
85      rw [Matrix.trace_mul_cycle D Tm Tm, ← hCeq]
86    have e3 : (D * Tm).trace = G.trace := by
87      have : D * Tm = T * G * Ti := by
88        calc D * Tm = T * (T * Ti) * G * Ti := by rw [hD, hTm]; simp only [Matrix.mul_assoc]
89          _ = _ := by rw [hTTi]; simp only [Matrix.mul_one, Matrix.mul_assoc]
90      rw [this, Matrix.trace_mul_cycle T G Ti, hTiT, Matrix.one_mul]
91    rw [e1, e2, e3]; ring
92  have hτ : (( C - D) * N).trace = 2 * (D * N * N).trace + (D * N * N * N).trace := by
93    rw [hΔ]
94    have : (D * N + N * D + N * D * N) * N = D * N * N + N * D * N + N * D * N * N := by
95      noncomm_ring
96    rw [this, Matrix.trace_add, Matrix.trace_add]
97    have a1 : (N * D * N).trace = (D * N * N).trace := (Matrix.trace_mul_cycle D N N).symm
98    have a2 : (N * D * N * N).trace = (D * N * N * N).trace := by
99      simp only [Matrix.mul_assoc]
100      rw [Matrix.trace_mul_comm N]
101      simp only [Matrix.mul_assoc]
102    rw [a1, a2]; ring
103  -- positivity of `tr (D N N Tm)`
104  have hP : (Nᵀ * Tm * N).PosSemidef := by
105    have := hTmpsd.conjTranspose_mul_mul_same N
106    rwa [Matrix.conjTranspose_eq_transpose_of_trivial] at this
107  have hnn : 0 ≤ (D * N * N).trace + (D * N * N * N).trace := by
108    have h1 : (D * N * N).trace + (D * N * N * N).trace = (D * (Nᵀ * Tm * N)).trace := by
109      rw [hNs, hTmN, ← Matrix.trace_add]
110      congr 1
111      noncomm_ring
112    rw [h1, Matrix.trace_mul_comm]
113    have := trace_psd_mul_gram hP T
114    rwa [hD, hTs] at *
115  -- diagonal weight comparison
116  have hcd : c * ∑ i, ∑ j, d i * N i j ^ 2 ≤ (D * N * N).trace := by
117    have h1 := trace_psd_mul_gram hcmp N
118    rw [hNs, Matrix.sub_mul, Matrix.trace_sub, Matrix.smul_mul, Matrix.trace_smul] at h1
119    have h2 : ∑ i, ∑ j, d i * N i j ^ 2 = (Matrix.diagonal d * (N * N)).trace := by
120      simp only [Matrix.trace, Matrix.diag_apply, Matrix.diagonal_mul]
121      refine Finset.sum_congr rfl fun i _ => ?_
122      rw [Matrix.mul_apply, Finset.mul_sum]
123      refine Finset.sum_congr rfl fun j _ => ?_
124      have := congrFun (congrFun hNs j) i
125      simp only [Matrix.transpose_apply] at this
126      rw [this]; ring
127    rw [h2]
128    simp only [smul_eq_mul] at h1
129    rw [Matrix.mul_assoc D N N]
130    linarith
131  -- termwise AM-GM
132  have hterm : ∀ i j, (C - D) i j * N i j - (c / 4) * ((d i + d j) * N i j ^ 2) ≤
133      c⁻¹ * ((C - D) i j ^ 2 / (d i + d j)) := by
134    intro i j
135    have hs : 0 < d i + d j := add_pos (hd i) (hd j)
136    have hcs : 0 < c * (d i + d j) := mul_pos hc hs
137    have key : c⁻¹ * ((C - D) i j ^ 2 / (d i + d j)) -
138        ((C - D) i j * N i j - (c / 4) * ((d i + d j) * N i j ^ 2)) =
139        ((C - D) i j - c * (d i + d j) * N i j / 2) ^ 2 / (c * (d i + d j)) := by
140      field_simp
141      ring
142    have : 0 ≤ ((C - D) i j - c * (d i + d j) * N i j / 2) ^ 2 / (c * (d i + d j)) := by positivity
143    linarith
144  have hamgm : ∑ i, ∑ j, (C - D) i j * N i j - (c / 4) * ∑ i, ∑ j, (d i + d j) * N i j ^ 2 ≤
145      c⁻¹ * ∑ i, ∑ j, (C - D) i j ^ 2 / (d i + d j) := by
146    have h := Finset.sum_le_sum fun i (_ : i ∈ (Finset.univ : Finset n)) =>
147      Finset.sum_le_sum fun j (_ : j ∈ (Finset.univ : Finset n)) => hterm i j
148    simp only [Finset.sum_sub_distrib, ← Finset.mul_sum] at h
149    exact h
150  have hsym : ∑ i, ∑ j, (d i + d j) * N i j ^ 2 = 2 * ∑ i, ∑ j, d i * N i j ^ 2 := by
151    have : ∑ i, ∑ j, (d i + d j) * N i j ^ 2 =
152        ∑ i, ∑ j, d i * N i j ^ 2 + ∑ i, ∑ j, d j * N i j ^ 2 := by
153      simp only [add_mul, Finset.sum_add_distrib]
154    rw [this, Finset.sum_comm (f := fun i j => d j * N i j ^ 2)]
155    have h2 : ∑ j, ∑ i, d j * N i j ^ 2 = ∑ i, ∑ j, d i * N i j ^ 2 := by
156      refine Finset.sum_congr rfl fun j _ => Finset.sum_congr rfl fun i _ => ?_
157      have := congrFun (congrFun hNs j) i
158      simp only [Matrix.transpose_apply] at this
159      rw [this]
160    rw [h2]; ring
161  have hfro : ∑ i, ∑ j, (C - D) i j * N i j = ((C - D) * N).trace := fro_eq_trace _ _ hNs
162  rw [hfro, hsym, hτ] at hamgm
163  nlinarith [hamgm, hcd, hnn, hb2]

687    (c : ℝ) (hc : 0 < c) (hcD : ∀ x, c * D₀ x x ≤ D x x)
688    (hC : IsTraceClassCov C) (hD : IsTraceClassCov D) (q : ℝ)
689    (hq : ∑ i ∈ F, ∑ j ∈ F, (C - D) (e i) (e j) ^ 2 / (dd i + dd j) ≤ q) :
690    (1 / 2) * (∑ i ∈ F, C (e i) (e i) + ∑ i ∈ F, D (e i) (e i)) - c⁻¹ * q -
691        epsF e (formSqrt C) (formSqrt D) F ≤
692      traceNormForm ((innerSL ℝ).comp ((formSqrt D).comp (formSqrt C))) := by
693  classical
694  set S := formSqrt C with hSdef
695  set T := formSqrt D with hTdef
696  have hS : IsSqrtOfForm S C := formSqrt_spec hC
697  have hT : IsSqrtOfForm T D := formSqrt_spec hD
698  have hSs := summable_sqrt_diag e hS hC
699  have hTs := summable_sqrt_diag e hT hD
700  set Cm := fmat e F C with hCmdef
701  set Dm := fmat e F D with hDmdef
702  have hCm : Cm.PosSemidef := fmat_posSemidef e F hC
703  have hcmp := compare_psd e F hD dd hD₀e c hcD
704  have hDm : Dm.PosDef := by
705    have h1 : (c • Matrix.diagonal (fun i : F => dd i)).PosDef := by
706      refine Matrix.PosDef.smul ?_ hc
707      exact Matrix.PosDef.diagonal (fun i => hdd i)
708    have := h1.add_posSemidef hcmp
709    rwa [add_sub_cancel] at this
710  have hDm0 : Dm.PosSemidef := hDm.posSemidef
711  obtain ⟨Sx, hSxp, hSxSx⟩ := exists_psd_sqrt Cm hCm
712  obtain ⟨Tx, hTxp, hTxTx⟩ := exists_psd_sqrt Dm hDm0
713  have hTxs : Txᵀ = Tx := psd_symm hTxp
714  have hSxs : Sxᵀ = Sx := psd_symm hSxp
715  have hTxu : IsUnit Tx.det := by
716    have h1 : Tx.det * Tx.det = Dm.det := by rw [← Matrix.det_mul, hTxTx]
717    have h2 : Dm.det ≠ 0 := (Matrix.PosDef.det_pos hDm).ne'
718    have : IsUnit (Tx.det * Tx.det) := by rw [h1]; exact (Matrix.isUnit_iff_isUnit_det Dm).1 hDm.isUnit
719    exact (IsUnit.mul_iff.1 this).1
720  have hCTC : (Tx * Cm * Tx).PosSemidef := psd_congr_psd hCm Tx hTxs
721  obtain ⟨Gm, hGp, hGG⟩ := exists_psd_sqrt _ hCTC
722  have hcore := core_alg Cm Dm Tx Gm hTxs hTxu hTxTx.symm hGp hGG c hc (fun i : F => dd i)
723    (fun i => hdd i) hcmp
724  have hK : (Sx * Tx)ᵀ * (Sx * Tx) = Gm * Gm := by
725    rw [hGG, Matrix.transpose_mul, hTxs, hSxs, ← hSxSx]
726    simp only [Matrix.mul_assoc]
727  obtain ⟨m, u, v, hu, hv, hsum⟩ := svd_lower Gm (Sx * Tx) hGp hK
728  -- the finite-rank approximants
729  set SF : H →L[ℝ] H := mop e F Sx with hSFdef
730  set TF : H →L[ℝ] H := mop e F Tx with hTFdef
731  have hSF : IsSqrtOfForm SF (compF e F C) := by
732    refine ⟨mop_symm e F hSxs, mop_nonneg e F hSxp, fun x y => ?_⟩
733    rw [hSFdef, inner_mop, crd_mop, compF_apply, mulVec_dot_mulVec, hSxs, hSxSx]
734  have hTF : IsSqrtOfForm TF (compF e F D) := by
735    refine ⟨mop_symm e F hTxs, mop_nonneg e F hTxp, fun x y => ?_⟩
736    rw [hTFdef, inner_mop, crd_mop, compF_apply, mulVec_dot_mulVec, hTxs, hTxTx]
737  -- the orthonormal families
738  set xx : Fin m → H := fun a => vecOf e F (v a) with hxx
739  set yy : Fin m → H := fun a => vecOf e F (u a) with hyy
740  have hxo : Orthonormal ℝ xx := by
741    rw [orthonormal_iff_ite]
742    intro a b
743    rw [hxx]; simp only []
744    rw [inner_vecOf_vecOf, hv]
745  have hyo : Orthonormal ℝ yy := by
746    rw [orthonormal_iff_ite]
747    intro a b
748    rw [hyy]; simp only []
749    rw [inner_vecOf_vecOf, hu]
750  have hlow := sum_inner_le_traceNormForm e S T hT.symm hSs hTs xx yy hxo hyo
751  -- main term
752  have hmain : ∑ a, inner ℝ (TF (SF (xx a))) (yy a) = Gm.trace := by
753    rw [← hsum]
754    refine Finset.sum_congr rfl fun a _ => ?_
755    rw [hTFdef, hSFdef, hxx, hyy]
756    simp only [mop_apply, crd_vecOf, inner_vecOf_vecOf, Matrix.mulVec_mulVec]
757    rw [mulVec_dot_transpose, Matrix.transpose_mul, hTxs, hSxs]
758  -- error terms
759  have hdecomp : ∀ a, inner ℝ (T (S (xx a))) (yy a) = inner ℝ (TF (SF (xx a))) (yy a) +
760      inner ℝ (S (xx a)) ((T - TF) (yy a)) + inner ℝ ((S - SF) (xx a)) (TF (yy a)) := by
761    intro a
762    rw [hT.symm, hTF.symm]
763    simp only [_root_.sub_apply, inner_sub_left, inner_sub_right]
764    ring
765  obtain ⟨hHT1, hHT2⟩ := hs_sub_le e F hD hT hTF
766  obtain ⟨hHS1, hHS2⟩ := hs_sub_le e F hC hS hSF
767  obtain ⟨hHF1, hHF2⟩ := hs_compF_le e F hD hT hTF
768  have herr1 := sum_abs_inner_le e S (T - TF) hSs hHT1 xx yy hxo hyo
769  have herr2 := sum_abs_inner_le e (S - SF) TF hHS1 hHF1 xx yy hxo hyo
770  have hsum_lb : Gm.trace - (Real.sqrt (∑' j, ‖S (e j)‖ ^ 2) *
771        Real.sqrt (∑' j, ‖(T - TF) (e j)‖ ^ 2) +
772      Real.sqrt (∑' j, ‖(S - SF) (e j)‖ ^ 2) * Real.sqrt (∑' j, ‖TF (e j)‖ ^ 2)) ≤
773      ∑ a, inner ℝ (T (S (xx a))) (yy a) := by
774    rw [← hmain]
775    simp only [hdecomp, Finset.sum_add_distrib]
776    have h1 : -(∑ a, |inner ℝ (S (xx a)) ((T - TF) (yy a))|) ≤
777        ∑ a, inner ℝ (S (xx a)) ((T - TF) (yy a)) := by
778      rw [← Finset.sum_neg_distrib]
779      exact Finset.sum_le_sum fun a _ => neg_abs_le _
780    have h2 : -(∑ a, |inner ℝ ((S - SF) (xx a)) (TF (yy a))|) ≤
781        ∑ a, inner ℝ ((S - SF) (xx a)) (TF (yy a)) := by
782      rw [← Finset.sum_neg_distrib]
783      exact Finset.sum_le_sum fun a _ => neg_abs_le _
784    linarith
785  have htr : Cm.trace = ∑ i ∈ F, C (e i) (e i) := by
786    simp only [Matrix.trace, Matrix.diag, hCmdef, fmat, Matrix.of_apply]
787    exact Finset.sum_coe_sort F (fun i => C (e i) (e i))
788  have htrD : Dm.trace = ∑ i ∈ F, D (e i) (e i) := by
789    simp only [Matrix.trace, Matrix.diag, hDmdef, fmat, Matrix.of_apply]
790    exact Finset.sum_coe_sort F (fun i => D (e i) (e i))
791  have hQF : ∑ i : F, ∑ j : F, (Cm - Dm) i j ^ 2 / (dd i + dd j) =
792      ∑ i ∈ F, ∑ j ∈ F, (C - D) (e i) (e j) ^ 2 / (dd i + dd j) := by
793    rw [← Finset.sum_coe_sort F]
794    refine Finset.sum_congr rfl fun i _ => ?_
795    rw [← Finset.sum_coe_sort F]
796    refine Finset.sum_congr rfl fun j _ => ?_
797    simp [hCmdef, hDmdef, fmat]
798  have hG : (1 / 2) * (∑ i ∈ F, C (e i) (e i) + ∑ i ∈ F, D (e i) (e i)) - c⁻¹ * q ≤ Gm.trace := by
799    have h1 := hcore
800    rw [hQF, htr, htrD] at h1
801    have hc' : 0 ≤ c⁻¹ := by positivity
802    nlinarith [mul_le_mul_of_nonneg_left hq hc']
803  have he1 : Real.sqrt (∑' j, ‖S (e j)‖ ^ 2) * Real.sqrt (∑' j, ‖(T - TF) (e j)‖ ^ 2) ≤
804      Real.sqrt (∑' j, ‖S (e j)‖ ^ 2) * Real.sqrt (2 * Real.sqrt (∑' j, ‖T (e j)‖ ^ 2) *
805        Real.sqrt (tailS e T F)) :=
806    mul_le_mul_of_nonneg_left (Real.sqrt_le_sqrt hHT2) (Real.sqrt_nonneg _)
807  have he2 : Real.sqrt (∑' j, ‖(S - SF) (e j)‖ ^ 2) * Real.sqrt (∑' j, ‖TF (e j)‖ ^ 2) ≤
808      Real.sqrt (2 * Real.sqrt (∑' j, ‖S (e j)‖ ^ 2) * Real.sqrt (tailS e S F)) *
809        Real.sqrt (∑' j, ‖T (e j)‖ ^ 2) :=
810    mul_le_mul (Real.sqrt_le_sqrt hHS2) (Real.sqrt_le_sqrt hHF2) (Real.sqrt_nonneg _)
811      (Real.sqrt_nonneg _)
812  have : epsF e S T F = Real.sqrt (∑' j, ‖S (e j)‖ ^ 2) * Real.sqrt (2 * Real.sqrt (∑' j, ‖T (e j)‖ ^ 2) *
813        Real.sqrt (tailS e T F)) +
814      Real.sqrt (2 * Real.sqrt (∑' j, ‖S (e j)‖ ^ 2) * Real.sqrt (tailS e S F)) *
815        Real.sqrt (∑' j, ‖T (e j)‖ ^ 2) := rfl
816  rw [this] at *
817  linarith

869theorem bures2_le_real {C D : H →L[ℝ] H →L[ℝ] ℝ} (dd : ι → ℝ) (hdd : ∀ i, 0 < dd i)
870    (D₀ : H →L[ℝ] H →L[ℝ] ℝ) (hD₀e : ∀ i y, D₀ (e i) y = dd i * inner ℝ (e i : H) y)
871    (c : ℝ) (hc : 0 < c) (hcD : ∀ x, c * D₀ x x ≤ D x x)
872    (hC : IsTraceClassCov C) (hD : IsTraceClassCov D) (q : ℝ)
873    (hq : ∀ F : Finset ι, ∑ i ∈ F, ∑ j ∈ F, (C - D) (e i) (e j) ^ 2 / (dd i + dd j) ≤ q) :
874    Uniform.bures2 C D ≤ 2 * c⁻¹ * q := by
875  classical
876  set S := formSqrt C with hSdef
877  set T := formSqrt D with hTdef
878  have hS : IsSqrtOfForm S C := formSqrt_spec hC
879  have hT : IsSqrtOfForm T D := formSqrt_spec hD
880  have hSs := summable_sqrt_diag e hS hC
881  have hTs := summable_sqrt_diag e hT hD
882  obtain ⟨tC, hCt, -⟩ := hC.hasSum_basis e
883  obtain ⟨tD, hDt, -⟩ := hD.hasSum_basis e
884  have hnC : traceNormForm C = tC :=
885    le_antisymm ((traceNormForm_le_basis_trace hC e).trans hCt.tsum_eq.le)
886      (hCt.tsum_eq ▸ hC.tsum_basis_le e)
887  have hnD : traceNormForm D = tD :=
888    le_antisymm ((traceNormForm_le_basis_trace hD e).trans hDt.tsum_eq.le)
889      (hDt.tsum_eq ▸ hD.tsum_basis_le e)
890  have hbound : ∀ F : Finset ι, Uniform.bures2 C D ≤
891      (tC - ∑ i ∈ F, C (e i) (e i)) + (tD - ∑ i ∈ F, D (e i) (e i)) + 2 * c⁻¹ * q +
892        2 * epsF e S T F := by
893    intro F
894    have h := lower_bound_F e F dd hdd D₀ hD₀e c hc hcD hC hD q (hq F)
895    unfold Uniform.bures2
896    rw [hnC, hnD]
897    linarith
898  have hlim : Tendsto (fun F : Finset ι => (tC - ∑ i ∈ F, C (e i) (e i)) +
899      (tD - ∑ i ∈ F, D (e i) (e i)) + 2 * c⁻¹ * q + 2 * epsF e S T F) atTop
900      (𝓝 ((tC - tC) + (tD - tD) + 2 * c⁻¹ * q + 2 * 0)) := by
901    have a1 : Tendsto (fun F : Finset ι => ∑ i ∈ F, C (e i) (e i)) atTop (𝓝 tC) := hCt
902    have a2 : Tendsto (fun F : Finset ι => ∑ i ∈ F, D (e i) (e i)) atTop (𝓝 tD) := hDt
903    exact (((tendsto_const_nhds.sub a1).add (tendsto_const_nhds.sub a2)).add
904      tendsto_const_nhds).add ((tendsto_epsF e hSs hTs).const_mul 2)
905  have := ge_of_tendsto' hlim hbound
906  simpa using this

909    (D₀ : H →L[ℝ] H →L[ℝ] ℝ) (hD₀e : ∀ i y, D₀ (e i) y = dd i * inner ℝ (e i : H) y)

931        ENNReal.ofReal_le_ofReal_iff hq0] at h1
932      rw [Finset.sum_product] at h1
933      exact h1

935    calc ENNReal.ofReal (Uniform.bures2 C D) ≤ ENNReal.ofReal (2 * c⁻¹ * q) :=

937      _ = 2 * ENNReal.ofReal c⁻¹ * Uniform.relQuad e dd (C - D) := by
938          rw [hqe, ENNReal.ofReal_mul (by positivity), ENNReal.ofReal_mul (by norm_num)]
939          simp
