20private theorem fp81v01_final_weighted_integral_mono
21    (u H κ csmall cbig A B : ℝ)
22    (hB : 0 ≤ B) (hAB : A ≤ B) (hcc : csmall ≤ cbig) :
23    A * (∫ t in Icc 0 H,
24      Real.exp (-cbig * max (u - t) 0) * Real.exp (-κ * t)) ≤
25    B * (∫ t in Icc 0 H,
26      Real.exp (-csmall * max (u - t) 0) * Real.exp (-κ * t)) := by
27  let fbig : ℝ → ℝ := fun t =>
28    Real.exp (-cbig * max (u - t) 0) * Real.exp (-κ * t)
29  let fsmall : ℝ → ℝ := fun t =>
30    Real.exp (-csmall * max (u - t) 0) * Real.exp (-κ * t)
31  have hbigI : IntegrableOn fbig (Icc 0 H) := by
32    have hc : Continuous fbig := by
33      dsimp [fbig]
34      fun_prop
35    exact hc.continuousOn.integrableOn_Icc
36  have hsmallI : IntegrableOn fsmall (Icc 0 H) := by
37    have hc : Continuous fsmall := by
38      dsimp [fsmall]
39      fun_prop
40    exact hc.continuousOn.integrableOn_Icc
41  have hcmp : (∫ t in Icc 0 H, fbig t) ≤ (∫ t in Icc 0 H, fsmall t) := by
42    apply setIntegral_mono_on hbigI hsmallI measurableSet_Icc
43    intro t ht
44    have hage : 0 ≤ max (u - t) 0 := le_max_right _ _
45    have hcoef : csmall * max (u - t) 0 ≤ cbig * max (u - t) 0 :=
46      mul_le_mul_of_nonneg_right hcc hage
47    have hexp : Real.exp (-cbig * max (u - t) 0) ≤
48        Real.exp (-csmall * max (u - t) 0) := by
49      apply Real.exp_le_exp.mpr
50      nlinarith
51    dsimp [fbig, fsmall]
52    exact mul_le_mul_of_nonneg_right hexp (Real.exp_pos _).le
53  have hbig0 : 0 ≤ ∫ t in Icc 0 H, fbig t :=
54    setIntegral_nonneg measurableSet_Icc fun t _ =>
55      mul_nonneg (Real.exp_pos _).le (Real.exp_pos _).le
56  calc
57    A * (∫ t in Icc 0 H, fbig t) ≤ B * (∫ t in Icc 0 H, fbig t) :=
58      mul_le_mul_of_nonneg_right hAB hbig0
59    _ ≤ B * (∫ t in Icc 0 H, fsmall t) := mul_le_mul_of_nonneg_left hcmp hB
