21theorem triple_canonical_proof (hd : 2 ≤ d) (hcrit : s = (d : ℝ) / 2) (β : ℝ) (hβ : 0 < β)
22    (V : Torus d → ℝ) (hV : Torus.IsSmooth V) (ρ : Torus d → ℝ)

25      IsPairCorrector d s V β ρ f K₁ ψ →
26        IsSymmTriple (Gpsi d s ρ ψ) ∧ IsCanonicalTriple (withDensityReal ρ) (Gpsi d s ρ ψ) ∧
27        (∀ x y z, x ≠ y → x ≠ z → y ≠ z → |Gpsi d s ρ ψ x y z| ≤

29        IsSymmPair (Bpsi d s ρ ψ) ∧
