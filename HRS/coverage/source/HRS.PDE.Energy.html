23## Representation decisions (SPEC_NOTES §1–2)
24
25* Lemma 3.12 is an identity in `L²`, i.e. an identity of Fourier coefficients; it is stated for
26  the coefficient operators of `ForwardEq.lean` for `μ ∈ C²` and `F ∈ H²`.
27* Proposition 3.13 concerns `d/dτ ½‖F(τ)‖²_{H^m}` along the solution.  Its content is split into
28  (a) the **form estimate** `energy_form_estimate` for real trigonometric polynomials (the only
29  input the Fourier–Galerkin proof route needs; stated for every `m ∈ ℕ`, including `m = 0` which
30  Step 4 of the proof of Prop 3.9 uses, line 1974), and
31  (b) the **derivative statement** `energy_torus` for solutions regular enough for the derivative
32  to exist (`F ∈ C([0,t];H^{m+2})`), which is the literal display `B:eq:energy-torus`.
33  The `H^m` inner product of `B:eq:sobolev-def` is `∑_{|α|≤m} w_α ⟨∂^α·,∂^α·⟩` with multinomial
34  weights `w_α ≥ 1` (`hsNormSq_eq_sum_derivL2Sq`), so the unweighted dissipation sums of the
35  display are implied by the weighted ones.
36* The retained weighted dissipation `B_l(√μ ∂^αF, √μ ∂^αF)` is `quadForm (symbolK q_l)` of the
37  coefficients of `√μ ∂^αF`; its summability is part of the conclusion (no junk `tsum`).

49/-- The antisymmetric part `A = ½(M - Mᵀ)` (`M^* = S - A`, line 1828). -/
50def antisymPart (M : Matrix (Fin d) (Fin d) ℝ) : Matrix (Fin d) (Fin d) ℝ :=
51  (1 / 2 : ℝ) • (M - M.transpose)
52
53/-- A diagonalisation `S = -∑_l λ_l q_l q_lᵀ` of the symmetric part with `λ_l ≥ 0` and an
54orthonormal basis `(q_l)` (`B:eq:S-diag`, line 1828; exists iff `S ≤ 0`). -/
55structure IsSymDiag (M : Matrix (Fin d) (Fin d) ℝ) (lam : Fin d → ℝ) (q : Fin d → Euc d) : Prop where
56  nonneg : ∀ l, 0 ≤ lam l
57  orthonormal : Orthonormal ℝ q
58  diag : ∀ i j, Torus.symPart M i j = -∑ l, lam l * q l i * q l j
59
60/-- `D_q u = q·∇u` as a multiplier `2πi (q·k)`. -/
61def dirDerivCoeff (q : Euc d) (c : Coeff d) : Coeff d :=
62  fun k => (2 * Real.pi * Complex.I * (inner ℝ q (Lattice.toEuc k) : ℂ)) * c k
63
64/-- `K_q u = -g * D_q² u`: the multiplier `𝔪_q(k) = (2π q·k)² ĝ(k)` (`B:eq:Kl-def`). -/
65def KlCoeff (d : ℕ) (s : ℝ) (q : Euc d) (c : Coeff d) : Coeff d :=
66  fun k => ((symbolK d s q k : ℝ) : ℂ) * c k
67
68/-- `D_q g * u`: the multiplier `2πi (q·k) ĝ(k)`. -/
69def dirGradRieszCoeff (d : ℕ) (s : ℝ) (q : Euc d) (c : Coeff d) : Coeff d :=
70  dirDerivCoeff q (rieszMult d s c)

72/-- The right-hand side of `B:eq:G-structure` (line 1840) in coefficients:
73`∑_l λ_l (K_l(μF) + D_l g*(F D_lμ)) + ∑_{i,j} A_{ij} ∂_i g*(F ∂_jμ)`. -/
74def GStructureRHS (D : DynData d) (lam : Fin d → ℝ) (q : Fin d → Euc d) (μ : Torus d → ℝ)
75    (F : Coeff d) : Coeff d :=
76  fun k => (∑ l : Fin d, ((lam l : ℝ) : ℂ) *
77      (KlCoeff d D.s (q l) (mulCoeff (coeffOf μ) F) k +
78        dirGradRieszCoeff d D.s (q l) (mulCoeff F (dirDerivCoeff (q l) (coeffOf μ))) k)) +
79    ∑ i : Fin d, ∑ j : Fin d, ((antisymPart D.M i j : ℝ) : ℂ) *
80      gradRieszCoeff d D.s i (mulCoeff F (derivCoeff j (coeffOf μ))) k

82/-- Lemma 3.12 `B:lem:G-structure` (lines 1837–1843): for `μ ∈ C²(𝕋^d)`, `F ∈ H²(𝕋^d)` and a
83diagonalisation `S = -∑ λ_l q_l q_lᵀ`, as an identity in `L²` (= identity of all Fourier
84coefficients),
85`𝒢[F] = ∑_l λ_l (K_l(μF) + D_l g*(F D_lμ)) + ∑_{i,j} A_{ij} ∂_i g*(F ∂_jμ)`.
86(The lemma's standing frame is regime R2, `S ≤ 0`; the identity is algebraic and is stated for any
87diagonalisation.)  The antisymmetric part contributes only through operators of order `θ-1 < 1`
88applied to `F` times first derivatives of `μ` (line 1842). -/
89theorem G_structure (hd : 1 ≤ d) (D : DynData d) (lam : Fin d → ℝ) (q : Fin d → Euc d)
90    (hdiag : IsSymDiag D.M lam q) (μ : Torus d → ℝ) (hμ : Torus.IsCk 2 μ) (F : Coeff d)
91    (hF : MemHs 2 F) :
92    nonlocalCoeff D μ F = GStructureRHS D lam q μ F := by
93  have ha : MemHs 1 (coeffOf μ) :=
94    (memHs_one_coeffOf_cbNorm 2 (by norm_num) μ (by exact_mod_cast hμ)).1
95  have hF1 : MemHs 1 F := hF.mono (by norm_num)
96  have hM : ∀ i j, D.M j i = -(∑ l, lam l * q l i * q l j) - antisymPart D.M i j := by
97    intro i j
98    have := hdiag.diag i j
99    simp only [Torus.symPart, Matrix.smul_apply, Matrix.add_apply, Matrix.transpose_apply,
100      smul_eq_mul] at this
101    simp only [antisymPart, Matrix.smul_apply, Matrix.sub_apply, Matrix.transpose_apply,
102      smul_eq_mul]
103    linarith
104  have hA : ∀ i j, antisymPart D.M i j = -antisymPart D.M j i := by
105    intro i j
106    simp only [antisymPart, Matrix.smul_apply, Matrix.sub_apply, Matrix.transpose_apply,
107      smul_eq_mul]
108    ring
109  funext k
110  exact gstruct_mode D.s D.M (antisymPart D.M) lam q hM hA (coeffOf μ) F ha hF1 k

114/-- The retained weighted dissipation of `B:eq:energy-torus`:
115`∑_{|α|≤m} ∑_l λ_l B_l(√μ ∂^αF, √μ ∂^αF)` in Fourier form. -/
116def weightedDissipation (D : DynData d) (m : ℕ) (lam : Fin d → ℝ) (q : Fin d → Euc d)
117    (μ : Torus d → ℝ) (F : Coeff d) : ℝ :=
118  ∑ α ∈ multiIndices d m, ∑ l : Fin d, lam l *
119    quadForm (symbolK d D.s (q l)) (mulCoeff (coeffOf fun x => Real.sqrt (μ x)) (multiDerivCoeff α F))
120
121/-- Summability of every term of `weightedDissipation` (so that it is not a junk `tsum`). -/
122def WeightedDissipationFinite (D : DynData d) (m : ℕ) (q : Fin d → Euc d) (μ : Torus d → ℝ)
123    (F : Coeff d) : Prop :=
124  ∀ α ∈ multiIndices d m, ∀ l : Fin d,
125    Summable fun k => symbolK d D.s (q l) k *
126      ‖mulCoeff (coeffOf fun x => Real.sqrt (μ x)) (multiDerivCoeff α F) k‖ ^ 2
127
128/-- The right-hand side of `B:eq:energy-torus` (line 1868) evaluated at `F` with density `μ`:
129`Λ_m ‖F‖²_{H^m} - (ν/2) ∑_{|α|≤m} ‖∇∂^αF‖²_{L²} - ½ ∑_{|α|≤m} ∑_l λ_l B_l(√μ∂^αF, √μ∂^αF)`,
130the last sum present only in regime R2 (void in R2a since all `λ_l = 0`; not used in R1/R3). -/
131def energyRHS (R : Regime) (Λm : ℝ) (D : DynData d) (m : ℕ) (lam : Fin d → ℝ) (q : Fin d → Euc d)
132    (μ : Torus d → ℝ) (F : Coeff d) : ℝ :=
133  Λm * hsNormSq m F - (D.invβ / 2) * ∑ α ∈ multiIndices d m, gradDerivL2Sq α F -
134    (if R.IsR2 then (1 / 2) * weightedDissipation D m lam q μ F else 0)

139/-- `B:eq:transport-est` (line 1880): `|⟨∂^α(b·∇F), ∂^αF⟩| ≤ C ‖F‖²_{H^m}` for `|α| ≤ m`, with
140`C = C(d,m,‖b‖_{C^{m+1}})`. -/
141theorem transport_form_bound (hd : 1 ≤ d) (m : ℕ) (B : ℝ) :
142    ∃ C : ℝ, ∀ (b : Fin d → Torus d → ℝ), (∀ j, Torus.IsCk (m + 1) (b j)) →
143      (∀ j, Torus.cbNorm (m + 1) (b j) ≤ B) →
144      ∀ (α : Fin d → ℕ), α ∈ multiIndices d m →
145      ∀ (n : ℕ) (F : Coeff d), IsRealCoeff F → IsTrigPoly n F →
146        |(bilinForm (fun k => ∏ i, (2 * Real.pi * (k i : ℝ)) ^ (2 * α i))
147            (fun k => ∑ j, mulCoeff (coeffOf (b j)) (derivCoeff j F) k) F).re| ≤ C * hsNormSq m F := by
148  classical
149  set K : ℝ := max B 0 with hKdef
150  have hK0 : 0 ≤ K := le_max_right _ _
151  refine ⟨∑ α ∈ multiIndices d m,
152    ((d : ℝ) * (K / 2 + ∑ β ∈ (leibnizSet α).erase 0, (binomW α β : ℝ) * K)), ?_⟩
153  intro b hb hB α hα n F _ hF
154  have hαm : ∑ i, α i ≤ m := (Finset.mem_filter.mp hα).2
155  have h := transport_bound_fixed m K hK0 b
156    (fun j => by have := hb j; unfold Torus.IsCk at this; push_cast; exact this)
157    (fun j => (hB j).trans (le_max_left _ _)) α hαm F hF
158  refine h.trans (mul_le_mul_of_nonneg_right ?_ (hsNormSq_nonnegE _ _))
159  refine Finset.single_le_sum (f := fun α : Fin d → ℕ =>
160    ((d : ℝ) * (K / 2 + ∑ β ∈ (leibnizSet α).erase 0, (binomW α β : ℝ) * K)))
161    (fun α _ => ?_) hα
162  positivity

165/-- `B:eq:R1-est` (line 1885), regime R1 (`θ ≤ 0`): `‖∂^α 𝒢[F]‖_{L²} ≤ C ‖F‖_{H^m}`, hence
166`|⟨∂^α𝒢[F], ∂^αF⟩| ≤ C ‖F‖²_{H^m}`. -/
167theorem nonlocal_R1_bound (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hs : s ≤ (d : ℝ) - 2) (m : ℕ)
168    (MB Λ : ℝ) :
169    ∃ C : ℝ, ∀ (D : DynData d), D.s = s → MatBound MB D.M → ∀ (μ : Torus d → ℝ),
170      Torus.IsCk (m + 3) μ → Torus.cbNorm (m + 3) μ ≤ Λ →
171      ∀ (α : Fin d → ℕ), α ∈ multiIndices d m →
172      ∀ (n : ℕ) (F : Coeff d), IsRealCoeff F → IsTrigPoly n F →
173        derivL2Sq α (nonlocalCoeff D μ F) ≤ C * hsNormSq m F := by
174  classical
175  refine ⟨∑ α' ∈ multiIndices d m, r1Const d s MB Λ α', ?_⟩
176  intro D hDs hM μ hμ hΛ α hα n F _ hF
177  have hαm : ∑ i, α i ≤ m := (Finset.mem_filter.mp hα).2
178  have hμ' : ContDiff ℝ ((m + 3 : ℕ) : WithTop ℕ∞) (Torus.lift μ) := by
179    unfold Torus.IsCk at hμ; push_cast; exact hμ
180  have e : derivL2Sq α (nonlocalCoeff D μ F) =
181      ∑' k, ‖multiDerivCoeff α (nonlocalCoeff D μ F) k‖ ^ 2 := by
182    unfold derivL2Sq
183    refine tsum_congr fun k => ?_
184    rw [norm_sq_multiDerivCoeff]
185  rw [e]
186  refine le_trans (Real.tsum_le_of_sum_le (fun k => sq_nonneg _) fun S =>
187    nonlocal_R1_fixed s hs m MB Λ D hDs hM μ hμ' hΛ α hαm F hF S) ?_
188  refine mul_le_mul_of_nonneg_right ?_ (hsNormSq_nonnegE _ _)
189  exact Finset.single_le_sum (f := fun α' => r1Const d s MB Λ α') (fun α' _ => r1Const_nonneg s MB Λ α') hα

229/-- Regime R2c (Prop 3.14, proof lines 1946–1951), per-multi-index estimate with the weighted
230dissipation retained: for `d-2 < s ≤ d-1` (`0 < θ ≤ 1`), `S ≤ 0` and **no lower bound on `μ`**,
231`-⟨∂^α𝒢[F], w⟩ ≤ -½∑_l λ_l B_l(√μ w, √μ w) + C ‖F‖²_{H^m}`, `w = ∂^αF`.  The source proof expands
232the Leibniz rule before the rewriting of Lemma 3.12, bounds the `γ ≠ 0` terms by `C‖F‖²_{H^m}`
233(order `θ-1 ≤ 0`), and the leading parts by Lemma C.3(iii) (`-λ_l B_l(μw,w) = -λ_l B_l(√μw,√μw)
234- λ_l E` with `|E| ≤ C‖w‖²`) and Lemma C.4; it then drops the nonnegative `B_l` terms, which gives
235`energy_form_estimate_R2c`.  Keeping half of them gives this form, which is what the conclusion of
236Prop 3.13 (invoked by Prop 3.14) requires in regime R2. -/

241      ∀ (μ : Torus d → ℝ), IsProbDensity μ → Torus.IsCk (m + 3) μ → Torus.cbNorm (m + 3) μ ≤ Λ →
242      ∀ (α : Fin d → ℕ), α ∈ multiIndices d m →
243      ∀ (n : ℕ) (F : Coeff d), IsRealCoeff F → IsTrigPoly n F →
244        (∀ l, Summable fun k => symbolK d D.s (q l) k *
245          ‖mulCoeff (coeffOf fun x => Real.sqrt (μ x)) (multiDerivCoeff α F) k‖ ^ 2) ∧
246        -(bilinForm (fun k => ∏ i, (2 * Real.pi * (k i : ℝ)) ^ (2 * α i))
247            (nonlocalCoeff D μ F) F).re ≤
248          -(1 / 2) * ∑ l, lam l * quadForm (symbolK d D.s (q l))
249              (mulCoeff (coeffOf fun x => Real.sqrt (μ x)) (multiDerivCoeff α F)) +
250            C * hsNormSq m F := by
251  by_cases hs0 : 0 ≤ s
252  · have hsd : s < d := by linarith
253    have hK : R2Aux.R2KernelFacts d s := by
254      by_cases hd2 : 2 ≤ d
255      · exact R2Aux.r2KernelFacts_of_two_le hd2 hs0 hsd
256      · exact R2Aux.r2KernelFacts_one (by omega) hs0 hsd
257    obtain ⟨C, hC⟩ := R2Aux.r2c_main hd s hs0 hK hs1 hs2 m MB Λ

259    exact hC D hDs hM lam q hdiag.nonneg hdiag.orthonormal hdiag.diag μ hμ.nonneg hμ3 hΛ α hα n F
260      hF hFt
261  · -- no `DynData` with `D.s = s < 0` exists (`D.hs0`)
262    refine ⟨0, fun D hDs => ?_⟩
263    exact absurd (hDs ▸ D.hs0) hs0

265/-- `B:eq:R3-est` (line 1935), regime R3: for `ν₀ > 0`,
266`|⟨∂^α𝒢[F], w⟩| ≤ ν₀/(4(m+1)^d) ‖F‖²_{H^{m+1}} + C ν₀^{-θ/(2-θ)} ‖F‖²_{H^m}`, `C` independent of

268theorem nonlocal_R3_bound (hd : 1 ≤ d) (s : ℝ) (hs1 : (d : ℝ) - 2 < s) (hsd : s < d) (m : ℕ)
269    (MB Λ : ℝ) :
270    ∃ C : ℝ, ∀ (D : DynData d), D.s = s → MatBound MB D.M → ∀ (μ : Torus d → ℝ),
271      Torus.IsCk (m + 3) μ → Torus.cbNorm (m + 3) μ ≤ Λ → ∀ (ν₀ : ℝ), 0 < ν₀ →
272      ∀ (α : Fin d → ℕ), α ∈ multiIndices d m →
273      ∀ (n : ℕ) (F : Coeff d), IsRealCoeff F → IsTrigPoly n F →
274        |(bilinForm (fun k => ∏ i, (2 * Real.pi * (k i : ℝ)) ^ (2 * α i))
275            (nonlocalCoeff D μ F) F).re| ≤
276          ν₀ / (4 * ((m : ℝ) + 1) ^ d) * hsNormSq (m + 1) F +
277            C * ν₀ ^ (-(thetaOrder d s / (2 - thetaOrder d s))) * hsNormSq m F := by
278  classical
279  have hθ0 : 0 < s + 2 - d := by linarith
280  have hθ2 : s + 2 - d < 2 := by linarith
281  have hθ : thetaOrder d s = s + 2 - d := rfl
282  set lam : ℝ := (s + 2 - d) / 2 with hlam
283  have hl0 : 0 < lam := by rw [hlam]; linarith
284  have hl1 : lam < 1 := by rw [hlam]; linarith
285  set C₃ : ℝ := (∑ α' ∈ multiIndices d m, r3Const d s MB Λ α') + 1 with hC₃def
286  have hC₃ : 0 < C₃ := by
287    have : 0 ≤ ∑ α' ∈ multiIndices d m, r3Const d s MB Λ α' :=
288      Finset.sum_nonneg fun α' _ => r3Const_nonneg s MB Λ α'
289    rw [hC₃def]; linarith
