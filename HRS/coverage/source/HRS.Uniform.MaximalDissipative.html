4/-!
5# Lemma 3.17, dissipative alternative (packet KOTELENEZ)
6
7The stochastic convolution `Z_t = ∫₀ᵗ V(t,r) dM_r` for a quasi-contractive evolution (second alternative of
8Lemma 3.17) exists as an `L²`-limit of Riemann sums over arbitrary partitions, has a continuous version and satisfies
9`E sup_t ‖Z_t‖^p ≤ C(p,K,T) E (∫ Tr Σ)^{p/2}`.  The proof is Kotelenez's: the Riemann sums on a partition are, after a
10Sz.-Nagy-type dilation of the time-ordered products of the contractions `V(s_{l+1}, s_l)`, contractions of a martingale
11transform of `M` with deterministic step operators (`HRS/Uniform/KotDilation.lean`, `KotTransform.lean`), so the
12dimension-free Hilbert BDG inequality gives a bound uniform in the partition (`KotMaximal.lean`); differences of Riemann
13sums on a partition and a refinement are again of this form with operators tending to `0` strongly (`KotConv.lean`), which
14gives the Cauchy property, the continuous version (dyadic partitions, Borel–Cantelli), and the `L²`-convergence over all
15partitions.  The TeX's Galerkin difference argument (L2042-2045) is not used.
16-/

29theorem maximal_dissipative (T p K : ℝ) (hT : 0 ≤ T) (hp : 2 ≤ p) :
30    ∃ C : ℝ, 0 ≤ C ∧
31      ∀ (Ω : Type) [mΩ : MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
32        (ℱ : Filtration ℝ mΩ)
33        (H E : Type) [NormedAddCommGroup H] [InnerProductSpace ℝ H] [CompleteSpace H]
34        [MeasurableSpace H] [BorelSpace H] [SecondCountableTopology H]
35        [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]
36        [MeasurableSpace E] [BorelSpace E] [SecondCountableTopology E]
37        (S : EvolData H E) (hS : S.IsEvolution T ∧ S.Dissipative T K)
38        (M : ℝ → Ω → H) (Sg : ℝ → Ω → (H →L[ℝ] H →L[ℝ] ℝ))
39        (hM : IsHilbertMartingaleWithCovariation P ℱ T M Sg),
40        ∃ Z : ℝ → Ω → E,
41          (∀ t ∈ Set.Icc (0 : ℝ) T, IsStochConv P S.VE S.j M 0 t (Z t)) ∧
42          (∀ᵐ ω ∂P, ContinuousOn (fun t => Z t ω) (Set.Icc 0 T)) ∧
43          (∫⁻ ω, (⨆ t ∈ Set.Icc (0 : ℝ) T, ‖Z t ω‖ₑ) ^ p ∂P) ^ (1 / p) ≤
44            ENNReal.ofReal C *
45              (∫⁻ ω, (∫⁻ r in Set.Icc (0 : ℝ) T, ENNReal.ofReal (traceNormForm (Sg r ω))) ^ (p / 2) ∂P) ^
46                (1 / p) := by
47  have hp1 : 1 ≤ p := by linarith
48  have hp0 : 0 < p := by linarith
49  refine ⟨CD p K T ^ (1 / p), Real.rpow_nonneg (CD_nonneg _ _ _) _, ?_⟩
50  intro Ω mΩ P _ ℱ H E _ _ _ _ _ _ _ _ _ _ _ _ S hS M Sg hM
51  obtain ⟨hev, hD⟩ := hS
52  have hD0 := hD
53  obtain ⟨hj, hsurj, -, -, -⟩ := hD0
54  have hjE : ∀ x, ‖S.j x‖ₑ = ‖x‖ₑ := fun x => by
55    rw [← ofReal_norm_eq_enorm, ← ofReal_norm_eq_enorm, hj x]
56  have hcomp : IsCompleteFiltration P (augFiltration P ℱ) := augFiltration_isComplete ℱ
57  have hM' := isHMC_aug hM
58  have hV : EvolBound S.VH K T :=
59    { norm_le := fun r t hr hrt htT => by
60        refine ContinuousLinearMap.opNorm_le_bound _ (Real.exp_pos _).le fun x => ?_
61        exact EvolData.Dissipative.norm_VH_le S hD ⟨hr, hrt.trans htT⟩ ⟨hrt, htT⟩ x
62      comp := fun r t u hr hrt htu huT => by
63        have := hev.comp_H r t u hr hrt htu huT
64        exact this
65      self := fun t ht0 htT => hev.self_H t ⟨ht0, htT⟩ }
66  obtain ⟨m, Z', hm, hZc, hlim⟩ := exists_dyadic_limit hcomp hM' hV (EvolData.Dissipative.cont_VH hev hD) hT
67  refine ⟨fun t ω => S.j (Z' t ω), ?_, ?_, ?_⟩
68  · intro t ht ε hε
69    obtain ⟨δ, hδ, hconv⟩ := hconv_Z hcomp hM' hV (EvolData.Dissipative.cont_VH hev hD) hT hm hlim ht ε hε
70    refine ⟨δ, hδ, fun n r hr0 hrn hmono hmesh => ?_⟩
71    have hrs : ∀ ω, riemannSum S.VE S.j M t r ω =
72        S.j (∑ i : Fin n, S.VH t (r i.castSucc) (M (r i.succ) ω - M (r i.castSucc) ω)) := by
73      intro ω
74      unfold riemannSum
75      rw [map_sum]
76      refine Finset.sum_congr rfl fun i _ => ?_
77      have h0 : 0 ≤ r i.castSucc := hr0 ▸ hmono (Fin.zero_le _)
78      have h1 : r i.castSucc ≤ t := hrn ▸ hmono (Fin.le_last _)
79      have := congrArg (fun F : H →L[ℝ] E => F (M (r i.succ) ω - M (r i.castSucc) ω))
80        (hev.intertwine (r i.castSucc) t h0 h1 ht.2)
81      simpa using this
82    have := hconv n r hr0 hrn hmono hmesh
83    refine lt_of_le_of_lt (le_of_eq ?_) this
84    refine lintegral_congr fun ω => ?_
85    rw [hrs ω, ← map_sub, hjE]
86  · exact hZc.mono fun ω h => S.j.continuous.comp_continuousOn h
87  · have hb := sup_Z_bound hcomp hM' hV hT hZc hlim hp1
88    have e : ∀ ω, (⨆ t ∈ Set.Icc (0 : ℝ) T, ‖S.j (Z' t ω)‖ₑ) = ⨆ t ∈ Set.Icc (0 : ℝ) T, ‖Z' t ω‖ₑ := by
89      intro ω
90      refine iSup_congr fun t => iSup_congr fun _ => ?_
91      exact hjE _
92    simp_rw [e]
93    calc (∫⁻ ω, (⨆ t ∈ Set.Icc (0 : ℝ) T, ‖Z' t ω‖ₑ) ^ p ∂P) ^ (1 / p)
94        ≤ (ENNReal.ofReal (CD p K T) *
95            ∫⁻ ω, (∫⁻ r in Set.Icc (0 : ℝ) T, ENNReal.ofReal (traceNormForm (Sg r ω))) ^ (p / 2) ∂P) ^
96              (1 / p) := ENNReal.rpow_le_rpow hb (by positivity)
97      _ = _ := by
98        rw [ENNReal.mul_rpow_of_nonneg _ _ (by positivity),
99          ENNReal.ofReal_rpow_of_nonneg (CD_nonneg _ _ _) (by positivity)]

103theorem maximal_estimate_proof (T p K : ℝ) (hT : 0 ≤ T) (hp : 2 ≤ p) :
104    ∃ C : ℝ, 0 ≤ C ∧
105      (∀ (Ω : Type) [mΩ : MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
106        (ℱ : Filtration ℝ mΩ)
107        (H E : Type) [NormedAddCommGroup H] [InnerProductSpace ℝ H] [CompleteSpace H]
108        [MeasurableSpace H] [BorelSpace H] [SecondCountableTopology H]
109        [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]
110        [MeasurableSpace E] [BorelSpace E] [SecondCountableTopology E]
111        (S : EvolData H E) (hS : S.MaxHyp T K)
112        (M : ℝ → Ω → H) (Sg : ℝ → Ω → (H →L[ℝ] H →L[ℝ] ℝ))
113        (hM : IsHilbertMartingaleWithCovariation P ℱ T M Sg),
114        ∃ Z : ℝ → Ω → E,
115          (∀ t ∈ Set.Icc (0 : ℝ) T, IsStochConv P S.VE S.j M 0 t (Z t)) ∧
116          (∀ᵐ ω ∂P, ContinuousOn (fun t => Z t ω) (Set.Icc 0 T)) ∧
117          (∫⁻ ω, (⨆ t ∈ Set.Icc (0 : ℝ) T, ‖Z t ω‖ₑ) ^ p ∂P) ^ (1 / p) ≤
118            ENNReal.ofReal C *
119              (∫⁻ ω, (∫⁻ r in Set.Icc (0 : ℝ) T, ENNReal.ofReal (traceNormForm (Sg r ω))) ^ (p / 2) ∂P) ^ (1 / p)) ∧
120      (∀ (H E : Type) [NormedAddCommGroup H] [InnerProductSpace ℝ H]
121        [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]
122        (S : EvolData H E) (hS : S.MaxHyp T K) (b : ℝ → H),
123        ⨆ t ∈ Set.Icc (0 : ℝ) T, ‖∫ r in Set.Icc (0 : ℝ) t, S.VE t r (S.j (b r))‖ₑ ≤
124          ENNReal.ofReal C * ∫⁻ r in Set.Icc (0 : ℝ) T, ‖b r‖ₑ) := by
125  obtain ⟨C₁, hC₁0, hC₁, -⟩ := maximal_estimate_twoLevel T p K hT hp
126  obtain ⟨C₂, hC₂0, hC₂⟩ := maximal_dissipative T p K hT hp
127  refine ⟨max (max C₁ C₂) (detConst T K), le_max_of_le_left (le_max_of_le_left hC₁0), ?_, ?_⟩
128  · intro Ω mΩ P _ ℱ H E _ _ _ _ _ _ _ _ _ _ _ _ S hS M Sg hM
129    obtain ⟨hev, h | h⟩ := hS
130    · obtain ⟨Z, hZ1, hZ2, hZ3⟩ := hC₁ Ω P ℱ H E S ⟨hev, h⟩ M Sg hM
131      refine ⟨Z, hZ1, hZ2, hZ3.trans ?_⟩
132      exact mul_le_mul' (ENNReal.ofReal_le_ofReal (le_max_of_le_left (le_max_left _ _))) le_rfl
133    · obtain ⟨Z, hZ1, hZ2, hZ3⟩ := hC₂ Ω P ℱ H E S ⟨hev, h⟩ M Sg hM
134      refine ⟨Z, hZ1, hZ2, hZ3.trans ?_⟩
135      exact mul_le_mul' (ENNReal.ofReal_le_ofReal (le_max_of_le_left (le_max_right _ _))) le_rfl
136  · intro H E _ _ _ _ _ S hS b
137    exact (maximal_deterministic T K hT S hS b).trans
138      (mul_le_mul' (ENNReal.ofReal_le_ofReal (le_max_right _ _)) le_rfl)
