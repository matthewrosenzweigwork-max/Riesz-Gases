5# The feature map `O:eq:feature` and the bounded-diagonal identity `O:eq:bounded-diagonal`
6
7Statements only (every theorem ends in `sorry` except the two coordinate bounds that make `featureMap` well defined);
8relocated from `HRS/SA/Drafts/SAD2.lean` after the independent review recorded in `docs/ISSUES.md` and subsequent approval.
9Row `M:prose:L5521-5558` (TeX L5521-5538).
10
11The printed real feature map (one frequency of each pair `{k,-k}`, cos/sin coordinates with weight `√(2 ĝ(k))`) and the
12printed identity `N F_N^g(X_N,μ) = ½‖N^{-1/2} ∑ Z(x_i)‖² - ½ g(0)`, `Z = Ψ - μ(Ψ)`.  `HRS.StaticLimitAux.energy_identity`
13already proves the identity for the redundant full-lattice feature space; the declarations below state the printed form.
14Any other choice of one representative from each pair gives an isometric map.
15-/

28/-- The index set of the printed feature map `O:eq:feature`: one cos coordinate and one sin
29coordinate for every frequency `k` of the (lexicographically) positive half-lattice, i.e. one
30element of each pair `{k,-k}` of nonzero frequencies ("Choose one element from each pair"). -/
31abbrev HalfIdx (d : ℕ) : Type := {k : Lattice d // Lattice.IsPosHalf k} ⊕ {k : Lattice d // Lattice.IsPosHalf k}
32
33/-- Coordinate functions of the printed feature map
34`Ψ(x) = (√(2 ĝ(k)) cos(2π k·x), √(2 ĝ(k)) sin(2π k·x))_k`, `ĝ = a`. -/
35def featCoord (a : Lattice d → ℝ) : HalfIdx d → Torus d → ℝ
36  | Sum.inl k, x => Real.sqrt (2 * a k.1) * Torus.cosMode k.1 x
37  | Sum.inr k, x => Real.sqrt (2 * a k.1) * Torus.sinMode k.1 x
38
39/-- The dominating summable weights `2 ĝ(k)`. -/
40def featWeight (a : Lattice d → ℝ) : HalfIdx d → ℝ
41  | Sum.inl k => 2 * a k.1
42  | Sum.inr k => 2 * a k.1
43
44theorem summable_featWeight {a : Lattice d → ℝ} (has : Summable a) :
45    Summable (featWeight a) := by
46  have h : Summable fun k : {k : Lattice d // Lattice.IsPosHalf k} => 2 * a k.1 :=
47    (has.comp_injective Subtype.val_injective).mul_left 2
48  exact Summable.sum (f := featWeight a) h h
49
50theorem featCoord_sq_le {a : Lattice d → ℝ} (ha0 : ∀ k, 0 ≤ a k) (i : HalfIdx d) (x : Torus d) :
51    featCoord a i x ^ 2 ≤ featWeight a i := by
52  rcases i with k | k
53  · simp only [featCoord, featWeight, mul_pow]
54    rw [Real.sq_sqrt (by have := ha0 k.1; linarith)]
55    have h := Torus.norm_cosMode_le k.1 x
56    rw [Real.norm_eq_abs] at h
57    have : Torus.cosMode k.1 x ^ 2 ≤ 1 := by
58      have := sq_le_one_iff_abs_le_one _ |>.mpr h
59      exact this
60    have h2 := ha0 k.1
61    nlinarith
62  · simp only [featCoord, featWeight, mul_pow]
63    rw [Real.sq_sqrt (by have := ha0 k.1; linarith)]
64    have h := Torus.norm_sinMode_le k.1 x
65    rw [Real.norm_eq_abs] at h
66    have : Torus.sinMode k.1 x ^ 2 ≤ 1 := by
67      have := sq_le_one_iff_abs_le_one _ |>.mpr h
68      exact this
69    have h2 := ha0 k.1
70    nlinarith

72/-- The real separable Hilbert space `ℓ²` of the feature coordinates. -/
73abbrev FeatHilbert (d : ℕ) : Type := lp (fun _ : HalfIdx d => ℝ) 2
74
75/-- The feature map `Ψ : 𝕋^d → ℓ²` of `O:eq:feature`, for a nonnegative summable symbol
76`ĝ = a` (`a ≥ 0`, `∑ a < ∞` are exactly what makes the coordinates square-summable). -/
77def featureMap {a : Lattice d → ℝ} (ha0 : ∀ k, 0 ≤ a k) (has : Summable a) (x : Torus d) :
78    FeatHilbert d :=
79  StaticLimitAux.mkLp (fun i => featCoord a i x) (featWeight a) (summable_featWeight has)
80    (fun i => featCoord_sq_le ha0 i x)

82/-- A spectral symbol in the sense of `O:eq:negative-symbol`: nonnegative, summable, even
83(`ĝ(-k) = ĝ(k)`, so that `g` is real and even) and with vanishing zero mode `ĝ(0) = 0`. -/
84structure IsSpectralSymbol (a : Lattice d → ℝ) : Prop where
85  nonneg : ∀ k, 0 ≤ a k
86  summable : Summable a
87  even : ∀ k, a (-k) = a k
88  zero : a 0 = 0

90/-- The negative-exponent symbol `O:eq:negative-symbol` is a spectral symbol (`s < 0`, `c > 0`). -/
91theorem isSpectralSymbol_negSpectralCoeff (d : ℕ) {s c : ℝ} (hs : s < 0) (hc : 0 < c) :
92    IsSpectralSymbol (negSpectralCoeff d s c) := by
93  exact ⟨fun k => negSpectralCoeff_nonneg d s hc.le k,
94    summable_negSpectralCoeff d hs c,
95    negSpectralCoeff_neg d s c,
96    by simp [negSpectralCoeff]⟩

127/-- **`O:eq:bounded-diagonal`, abstract form** (M L5534-5538).  Let `Ψ : 𝕋^d → H` be a continuous
128map into a real Hilbert space with `⟨Ψ(x),Ψ(y)⟩ = g(x-y)` for all `x,y`.  Then for every
129probability measure `μ`, every `N ≥ 1` and every configuration `X_N`,
130`N F_N^g(X_N,μ) = ½‖N^{-1/2} ∑_i (Ψ(x_i) - μ(Ψ))‖² - ½ g(0)`.
131The deleted diagonal in `F_N^g` is the original particle diagonal (`modulatedEnergy` sums over
132`j ≠ i`).  `N ≥ 1` is necessary: at `N = 0` the left side is `0` and the right side is
133`-½ g(0)`. -/
134theorem energy_identity_of_feature {H : Type*} [NormedAddCommGroup H] [InnerProductSpace ℝ H]
135    [CompleteSpace H] (g : Torus d → ℝ) (Ψ : Torus d → H) (hΨ : Continuous Ψ)
136    (hg : ∀ x y : Torus d, inner ℝ (Ψ x) (Ψ y) = g (x - y))
137    (μ : Measure (Torus d)) [IsProbabilityMeasure μ] {N : ℕ} (hN : 1 ≤ N) (X : Config d N) :
138    (N : ℝ) * modulatedEnergyKernel g X μ =
139      (1 / 2) * ‖(Real.sqrt N)⁻¹ • ∑ i, (Ψ (X i) - ∫ y, Ψ y ∂μ)‖ ^ 2 - (1 / 2) * g 0 := by
140  sorry

142/-- **`O:eq:bounded-diagonal`, concrete form** (M L5534-5538): the printed identity for the
143printed feature map `Ψ` of `O:eq:feature`, with `g = specKernel ĝ` and `Z = Ψ - μ(Ψ)`. -/
144theorem energy_identity_featureMap {a : Lattice d → ℝ} (hsym : IsSpectralSymbol a)
145    (μ : Measure (Torus d)) [IsProbabilityMeasure μ] {N : ℕ} (hN : 1 ≤ N) (X : Config d N) :
146    (N : ℝ) * modulatedEnergyKernel (StaticLimitAux.specKernel a) X μ =
147      (1 / 2) * ‖(Real.sqrt N)⁻¹ •
148          ∑ i, (featureMap hsym.nonneg hsym.summable (X i) -
149                  ∫ y, featureMap hsym.nonneg hsym.summable y ∂μ)‖ ^ 2
150        - (1 / 2) * StaticLimitAux.specKernel a 0 := by
151  sorry

155theorem energy_identity_negSpectral {s c : ℝ} (hs : s < 0) (hc : 0 < c)
