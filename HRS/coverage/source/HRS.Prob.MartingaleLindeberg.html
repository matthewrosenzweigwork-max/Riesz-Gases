6Lemma A.21 `M:B:lem:lindeberg` (source lines 10545-10551; statements
7`HRS.multivariate_martingale_clt`, `HRS.multivariate_martingale_clt_dc2` in
8`HRS/Prob/CLT.lean`) is proved in the source (lines 10553-10598) with the backward heat
9function `u(σ, x) = E h(x + G_{σ,t})`, `G_{σ,t} ∼ N(0, ∫_σ^t sb)`, and Itô's formula.  Mathlib has
10no stochastic calculus, so the formalisation follows a **discrete** route along the uniform
11partitions `τ_k = k t / n` and never uses Itô's formula:
12
131. `gaussSmooth h Q x = ∫ h(x + y) dN(0, Q)(y)` (the source's `u(σ, ·)` with
14   `Q = ∫_σ^t sb`); it keeps the Hessian bound and the Hessian Lipschitz constant, satisfies the
15   semigroup identity, and `gaussSmooth h 0 = h`.
162. Telescoping `E h(M_t) - E h(G) = ∑_k E[u_{k+1}(M_{τ_{k+1}}) - u_k(M_{τ_k})]`, with
17   `u_k(x) = ∫ u_{k+1}(x + y) dN(0, Q_k)`, `Q_k = ∫_{τ_k}^{τ_{k+1}} sb`.
183. Second-order Taylor expansion of `u_{k+1}` at `M_{τ_k}` in the increment `ΔM_k` and in the
19   Gaussian variable `y`: first-order terms vanish (martingale property; mean zero), the
20   second-order terms give `½ E⟨∇²u_{k+1}(M_{τ_k}), ∫_{τ_k}^{τ_{k+1}} (Σ - sb)⟩` (covariation
21   martingale), bounded by trace duality by `½ H E∫_{τ_k}^{τ_{k+1}} ‖Σ - sb‖_{S_1}`.
224. Remainders `min(H|v|², L|v|³)`; their sums tend to zero along the uniform partitions
23   (discrete Lindeberg condition, from path continuity and uniform integrability of the
24   discrete quadratic variations after localisation).
255. A Gaussian smoothing of `h` (step 1 with `Q = ε² I`) reduces general `h ∈ C²` with bounded
26   Hessian to `h` with a Lipschitz Hessian, at uniform cost `½ H ε² p`.

700lemma trace_dual (H Q : Matrix (Fin p) (Fin p) ℝ) (β : ℝ) (hβ0 : 0 ≤ β)
701    (hβ : ∀ u v : Fin p → ℝ, |u ⬝ᵥ (H.mulVec v)| ≤ β * Real.sqrt (u ⬝ᵥ u) * Real.sqrt (v ⬝ᵥ v)) :
702    |∑ a, ∑ b, H a b * Q a b| ≤ β * traceNormMatrix Q := by
703  classical
704  have hdiag : ∀ a, |H a a| ≤ β := by
705    intro a
706    have := hβ (Pi.single a 1) (Pi.single a 1)
707    simpa [dotProduct, Matrix.mulVec, Pi.single_apply] using this
708  refine le_of_forall_pos_le_add fun δ hδ => ?_
709  -- choose ε small with Q + ε • 1 invertible
710  have hne : Q.charpoly ≠ 0 := (Matrix.charpoly_monic Q).ne_zero
711  have hfin : Set.Finite {ε : ℝ | Q.charpoly.IsRoot (-ε)} :=
712    (Polynomial.finite_setOfPred_isRoot hne).preimage (neg_injective.injOn)
713  set ε0 : ℝ := δ / (2 * (β * p + 1) + 1) with hε0
714  have hε0pos : 0 < ε0 := by positivity
715  obtain ⟨ε, hε, hεbad⟩ : ∃ ε ∈ Set.Ioo 0 ε0, ε ∉ {ε : ℝ | Q.charpoly.IsRoot (-ε)} := by

717    push Not at hcon
718    exact (Set.Ioo_infinite hε0pos).not_finite (hfin.subset fun ε hε => hcon ε hε)
719  have hdet : IsUnit (Q + ε • (1 : Matrix (Fin p) (Fin p) ℝ)).det := by
720    have h1 : Q.charpoly.eval (-ε) ≠ 0 := hεbad
721    rw [Matrix.eval_charpoly] at h1
722    have h2 : (Matrix.scalar (Fin p) (-ε) - Q) = -(Q + ε • (1 : Matrix (Fin p) (Fin p) ℝ)) := by
723      ext i j
724      by_cases hij : i = j

726      · simp [Matrix.scalar_apply, hij]
727    rw [h2, Matrix.det_neg] at h1
728    refine isUnit_iff_ne_zero.2 fun h0 => h1 ?_

730  have key := trace_dual_of_isUnit H (Q + ε • 1) β hβ0 hβ hdet
731  have hε1 : 0 ≤ ε := hε.1.le
732  have e1 : ∑ a, ∑ b, H a b * (Q + ε • (1 : Matrix (Fin p) (Fin p) ℝ)) a b =
733      ∑ a, ∑ b, H a b * Q a b + ε * ∑ a, H a a := by
734    simp only [Matrix.add_apply, Matrix.smul_apply, Matrix.one_apply, mul_add, Finset.sum_add_distrib]
735    congr 1
736    rw [Finset.mul_sum]
737    refine Finset.sum_congr rfl fun a _ => ?_
738    simp [mul_comm]
739  have e2 : |∑ a, H a a| ≤ β * p := by
740    calc |∑ a, H a a| ≤ ∑ a, |H a a| := Finset.abs_sum_le_sum_abs _ _
741      _ ≤ ∑ _a : Fin p, β := Finset.sum_le_sum fun a _ => hdiag a
742      _ = β * p := by simp [mul_comm]
743  have e3 := traceNormMatrix_add_le Q (ε • (1 : Matrix (Fin p) (Fin p) ℝ))
744  have e4 := traceNormMatrix_smul_one_le (p := p) ε hε1
745  rw [e1] at key
746  have e5 : |∑ a, ∑ b, H a b * Q a b| ≤ |∑ a, ∑ b, H a b * Q a b + ε * ∑ a, H a a| + ε * (β * p) := by
747    have := abs_sub (∑ a, ∑ b, H a b * Q a b + ε * ∑ a, H a a) (ε * ∑ a, H a a)
748    rw [add_sub_cancel_right] at this
749    calc _ ≤ _ := this
750      _ ≤ _ := by
751        gcongr
752        rw [abs_mul, abs_of_nonneg hε1]
753        exact mul_le_mul_of_nonneg_left e2 hε1
754  have hεle : ε ≤ ε0 := hε.2.le
755  have hp : (0 : ℝ) ≤ p := Nat.cast_nonneg p
756  have e6 : β * (traceNormMatrix Q + ε * p) + ε * (β * p) ≤ β * traceNormMatrix Q + δ := by
757    have : ε * (2 * (β * p)) ≤ ε0 * (2 * (β * p + 1) + 1) := by
758      have h0 : 0 ≤ 2 * (β * p) := by positivity
759      nlinarith [mul_nonneg hβ0 hp]
760    have h3 : ε0 * (2 * (β * p + 1) + 1) = δ := by
761      rw [hε0]; field_simp
762    nlinarith [mul_nonneg hβ0 hp]
763  calc _ ≤ _ := e5

765    _ ≤ β * (traceNormMatrix Q + ε * p) + ε * (β * p) := by
766        gcongr
767        exact e3.trans (add_le_add le_rfl e4)

956lemma cov_pair (ℱ : Filtration ℝ mΩ) (t : ℝ)
957    (M : ℝ → Ω → (ℝ^p)) (Sg : ℝ → Ω → Matrix (Fin p) (Fin p) ℝ)
958    (hM : IsVectorMartingaleWithCovariation P ℱ t M Sg) (τ τ' : ℝ) (hτ : 0 ≤ τ)
959    (hττ' : τ ≤ τ') (hτ' : τ' ≤ t) (z : Ω → ℝ) (hz : StronglyMeasurable[ℱ τ] z) (K : ℝ)
960    (hzK : ∀ ω, |z ω| ≤ K) (a b : Fin p) :
961    Integrable (fun ω => z ω * ((M τ' ω - M τ ω) a * (M τ' ω - M τ ω) b)) P ∧
962    Integrable (fun ω => z ω * (∫ σ in Set.Icc τ τ', Sg σ ω a b)) P ∧
963    ∫ ω, z ω * ((M τ' ω - M τ ω) a * (M τ' ω - M τ ω) b) ∂P =
964      ∫ ω, z ω * (∫ σ in Set.Icc τ τ', Sg σ ω a b) ∂P := by
965  have hm1 : MemLp (M τ') 2 P := hM.sq_int τ' ⟨hτ.trans hττ', hτ'⟩
966  have hm0 : MemLp (M τ) 2 P := hM.sq_int τ ⟨hτ, hττ'.trans hτ'⟩
967  have hz' : StronglyMeasurable z := hz.mono (ℱ.le τ)
968  have hzm : AEStronglyMeasurable z P := hz'.aestronglyMeasurable
969  have c1 := memLp_coord P hm1
970  have c0 := memLp_coord P hm0
971  set A : ℝ → Ω → ℝ := fun s ω => ∫ σ in Set.Icc 0 s, Sg σ ω a b with hA
972  set N : ℝ → Ω → ℝ := fun s ω => M s ω a * M s ω b - A s ω with hN
973  have iN : ∀ s, Integrable (N s) P := fun s => (hM.covariation a b).integrable s
974  have iA : ∀ s, MemLp (M s) 2 P → Integrable (A s) P := by
975    intro s hs
976    have hprod : Integrable (fun ω => M s ω a * M s ω b) P := (memLp_coord P hs a).integrable_mul (memLp_coord P hs b)
977    have : A s = fun ω => (M s ω a * M s ω b) - N s ω := by funext ω; simp [hN]
978    rw [this]; exact hprod.sub (iN s)
979  have iA1 := iA τ' hm1
980  have iA0 := iA τ hm0
981  -- the pathwise increment
982  have hD : (fun ω => ∫ σ in Set.Icc τ τ', Sg σ ω a b) =ᵐ[P] fun ω => A τ' ω - A τ ω := by
983    filter_upwards [hM.sigma_entry_int] with ω hω
984    exact (setInt_Icc_diff hτ hττ' hτ' (hω a b)).symm

986  set T1 : Ω → ℝ := fun ω => z ω * (N τ' ω - N τ ω) with hT1
987  set T2 : Ω → ℝ := fun ω => z ω * (A τ' ω - A τ ω) with hT2
988  set T3 : Ω → ℝ := fun ω => (z ω * M τ ω a) * (M τ' ω b - M τ ω b) with hT3
989  set T4 : Ω → ℝ := fun ω => (z ω * M τ ω b) * (M τ' ω a - M τ ω a) with hT4

991  have iT2 : Integrable T2 P := integrable_bdd_mul P hzm hzK (iA1.sub iA0)

993    intro c
994    refine MemLp.of_le_mul (c := K) (c0 c) (hzm.mul (c0 c).aestronglyMeasurable) (ae_of_all _ fun ω => ?_)
995    rw [norm_mul, Real.norm_eq_abs]
996    exact mul_le_mul_of_nonneg_right (hzK ω) (norm_nonneg _)
997  have iT3 : Integrable T3 P := (hW a).integrable_mul ((c1 b).sub (c0 b))
998  have iT4 : Integrable T4 P := (hW b).integrable_mul ((c1 a).sub (c0 a))
999  have e1 : ∫ ω, T1 ω ∂P = 0 := by
1000    have := integral_bdd_mul_martingale P ℱ (hM.covariation a b) hττ' hz hzK
1001    simp only [hT1, mul_sub]
1002    rw [integral_sub (integrable_bdd_mul P hzm hzK (iN τ')) (integrable_bdd_mul P hzm hzK (iN τ))]
1003    simp only [hN] at this ⊢
1004    rw [this, sub_self]
1005  have wm : ∀ c : Fin p, StronglyMeasurable[ℱ τ] (fun ω => z ω * M τ ω c) := fun c =>
1006    hz.mul ((hM.martingale c).1 τ)
1007  have e3 : ∫ ω, T3 ω ∂P = 0 := by
1008    have := integral_mul_martingale P ℱ (hM.martingale b) hττ' (wm a) (hW a) (c1 b)
1009    have i1 : Integrable (fun ω => (z ω * M τ ω a) * M τ' ω b) P := (hW a).integrable_mul (c1 b)
1010    have i0 : Integrable (fun ω => (z ω * M τ ω a) * M τ ω b) P := (hW a).integrable_mul (c0 b)
1011    simp only [hT3, mul_sub]
1012    rw [integral_sub i1 i0, this, sub_self]
1013  have e4 : ∫ ω, T4 ω ∂P = 0 := by
1014    have := integral_mul_martingale P ℱ (hM.martingale a) hττ' (wm b) (hW b) (c1 a)
1015    have i1 : Integrable (fun ω => (z ω * M τ ω b) * M τ' ω a) P := (hW b).integrable_mul (c1 a)
1016    have i0 : Integrable (fun ω => (z ω * M τ ω b) * M τ ω a) P := (hW b).integrable_mul (c0 a)
1017    simp only [hT4, mul_sub]
1018    rw [integral_sub i1 i0, this, sub_self]
1019  have hF : (fun ω => z ω * ((M τ' ω - M τ ω) a * (M τ' ω - M τ ω) b)) =
1020      fun ω => T1 ω + T2 ω - T3 ω - T4 ω := by
1021    funext ω
1022    simp only [hT1, hT2, hT3, hT4, hN, PiLp.sub_apply]

1025    hD.mono fun ω h => by simp only [hT2, h]

1029    have j2 : Integrable (fun ω => T1 ω + T2 ω - T3 ω) P := j1.sub iT3
1030    rw [hF, integral_sub j2 iT4, integral_sub j1 iT3, integral_add iT1 iT2, e1, e3, e4,
