33theorem poly_identity (q u : Fin d → ℝ) (c g : ℝ) :
34    (c * ∑ i, q i * u i) ^ 2 * g =
35      ∑ i, -(q i * ∑ j, q j) * ((c * u i) ^ 2 * g) +
36        ∑ i, ∑ j, q i * q j / 2 * ((c * (u i + u j)) ^ 2 * g) := by
37  have h1 : ∀ i, -(q i * ∑ j, q j) * ((c * u i) ^ 2 * g) =
38      ∑ j, (-(q i * q j) * ((c * u i) ^ 2 * g)) := by
39    intro i
40    rw [← Finset.sum_mul, Finset.sum_neg_distrib, ← Finset.mul_sum]
41  simp_rw [h1]
42  have h2 : (c * ∑ i, q i * u i) ^ 2 * g = ∑ i, ∑ j, c ^ 2 * g * (q i * q j * (u i * u j)) := by
43    rw [mul_pow, sq (∑ i, q i * u i), Finset.sum_mul_sum]
44    simp only [Finset.mul_sum, Finset.sum_mul]
45    exact Finset.sum_congr rfl fun i _ => Finset.sum_congr rfl fun j _ => by ring
46  rw [h2, ← Finset.sum_add_distrib]
47  have h3 : ∑ i, ∑ j, c ^ 2 * g * (q i * q j * (u j ^ 2 - u i ^ 2) / 2) = 0 := by
48    have : ∑ i, ∑ j, c ^ 2 * g * (q i * q j * (u j ^ 2 - u i ^ 2) / 2) =
49        -∑ i, ∑ j, c ^ 2 * g * (q i * q j * (u j ^ 2 - u i ^ 2) / 2) := by
50      conv_lhs => rw [Finset.sum_comm]
51      rw [← Finset.sum_neg_distrib]
52      refine Finset.sum_congr rfl fun i _ => ?_
53      rw [← Finset.sum_neg_distrib]
54      refine Finset.sum_congr rfl fun j _ => ?_
55      ring
56    linarith
57  have h4 : ∑ i, (∑ j, ((-(q i * q j) * ((c * u i) ^ 2 * g)) +
58        q i * q j / 2 * ((c * (u i + u j)) ^ 2 * g))) =
59      ∑ i, ∑ j, (c ^ 2 * g * (q i * q j * (u i * u j)) +
60        c ^ 2 * g * (q i * q j * (u j ^ 2 - u i ^ 2) / 2)) := by
61    refine Finset.sum_congr rfl fun i _ => Finset.sum_congr rfl fun j _ => by ring
62  have h5 : ∑ i, (∑ j, (-(q i * q j) * ((c * u i) ^ 2 * g)) +
63        ∑ j, q i * q j / 2 * ((c * (u i + u j)) ^ 2 * g)) =
64      ∑ i, (∑ j, ((-(q i * q j) * ((c * u i) ^ 2 * g)) +
65        q i * q j / 2 * ((c * (u i + u j)) ^ 2 * g))) := by
66    refine Finset.sum_congr rfl fun i _ => ?_
67    rw [Finset.sum_add_distrib]
68  rw [h5, h4]
69  simp only [Finset.sum_add_distrib]
70  rw [h3, add_zero]

81theorem abs_pcoef_le (hd : 1 ≤ d) (q : Euc d) (hq : ‖q‖ ≤ 1) (ι : PIdx d) :
82    |pcoef q ι| ≤ d := by
83  have hqi : ∀ i, |q i| ≤ 1 := fun i => by
84    have := PiLp.norm_apply_le q i
85    rw [Real.norm_eq_abs] at this

93        _ = d := by simp

122/-- Lemma C.3(iii) (weighted symmetrisation), uniformly over unit-ball directions `q`
123(polarisation of the finitely many symbols `𝔪_{e_i}`, `𝔪_{e_i+e_j}`). -/
124theorem weighted_uniform (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hs1 : (d : ℝ) - 2 < s) (hsd : s < d) :
125    ∃ C : ℝ, ∀ (q : Euc d), ‖q‖ ≤ 1 → ∀ (L La Lm Lw : ℝ) (a w : Torus d → ℝ),
126      (∀ x, 0 ≤ a x) → IsC11 L a → Torus.IsLipschitz La a →
127      Torus.IsLipschitz Lm (fun x => Real.sqrt (a x)) → Torus.IsLipschitz Lw w →
128      MemSobolev (thetaOrder d s / 2) w →
129      |symbolForm (symbolK d s q) (fun x => a x * w x) w -
130        symbolForm (symbolK d s q) (fun x => Real.sqrt (a x) * w x)
131          (fun x => Real.sqrt (a x) * w x)| ≤
132        C * (Lm ^ 2 + Torus.supNorm (fun x => Real.sqrt (a x)) ^ 2) * l2NormSq w := by
133  have hθ0 : 0 < thetaOrder d s := by unfold thetaOrder; linarith
134  have hθ2 : thetaOrder d s < 2 := by unfold thetaOrder; have : (s : ℝ) < d := hsd; linarith
135  have hd1 : 1 ≤ d := hd
136  have hH : ∀ ι : PIdx d, IsHomogSymbol (thetaOrder d s) (symbolK d s (pvec ι)) :=
137    fun ι => symbolK_isHomogSymbol hd s hs0 hs1 hsd _
138  have hK : ∀ ι : PIdx d, ∃ 𝔨 : Torus d → ℝ, IsLevyKernel (thetaOrder d s) (symbolK d s (pvec ι)) 𝔨 :=
139    fun ι => quadratic_toolbox_kernel hd _ hθ0 hθ2 _ (hH ι)
140  choose 𝔨 h𝔨 using hK
141  have hW : ∀ ι : PIdx d, ∃ C : ℝ, ∀ (L La Lm Lw : ℝ) (a w : Torus d → ℝ), (∀ x, 0 ≤ a x) →
142      IsC11 L a → Torus.IsLipschitz La a → Torus.IsLipschitz Lm (fun x => Real.sqrt (a x)) →
143      Torus.IsLipschitz Lw w →
144      |symbolForm (symbolK d s (pvec ι)) (fun x => a x * w x) w -
145        symbolForm (symbolK d s (pvec ι)) (fun x => Real.sqrt (a x) * w x)
146          (fun x => Real.sqrt (a x) * w x)| ≤
147        C * (Lm ^ 2 + Torus.supNorm (fun x => Real.sqrt (a x)) ^ 2) * l2NormSq w := by
148    intro ι
149    obtain ⟨C, hC⟩ := quadratic_toolbox_weighted hd _ hθ0 hθ2 _ (hH ι) (𝔨 ι) (h𝔨 ι)
150    refine ⟨C, fun L La Lm Lw a w ha hC11 hLa hLm hLw => ?_⟩
151    obtain ⟨-, -, h3⟩ := hC L La Lm Lw a w ha hC11 hLa hLm hLw
152    exact h3
153  choose CC hCC using hW
154  have hB : ∀ ι : PIdx d, ∃ C : ℝ, ∀ u v : Torus d → ℝ, MemSobolev (thetaOrder d s / 2) u →
155      MemSobolev (thetaOrder d s / 2) v →
156      Summable fun k => ((symbolK d s (pvec ι) k : ℝ) : ℂ) * (coeffOf u k * conj (coeffOf v k)) := by
157    intro ι
158    obtain ⟨C, hC⟩ := quadratic_toolbox_basic _ hθ0 hθ2 _ (hH ι)
159    exact ⟨C, fun u v hu hv => (hC u v hu hv).2.2.1⟩
160  choose CB hCB using hB
161  obtain ⟨CP, hCP⟩ := product_toolbox_frac (d := d) hd1 (thetaOrder d s / 2) (by linarith)
162    (by linarith)
163  refine ⟨d * ∑ ι, |CC ι|, fun q hq L La Lm Lw a w ha hC11 hLa hLm hLw hw => ?_⟩
164  have hu : MemSobolev (thetaOrder d s / 2) (fun x => a x * w x) := (hCP La a hLa w hw).1
165  have hv : MemSobolev (thetaOrder d s / 2) (fun x => Real.sqrt (a x) * w x) :=
166    (hCP Lm _ hLm w hw).1
167  rw [symbolForm_polarize s q _ _ (fun ι => hCB ι _ _ hu hw),
168    symbolForm_polarize s q _ _ (fun ι => hCB ι _ _ hv hv), ← Finset.sum_sub_distrib]
169  have hR : 0 ≤ (Lm ^ 2 + Torus.supNorm (fun x => Real.sqrt (a x)) ^ 2) * l2NormSq w := by
170    refine mul_nonneg (by positivity) ?_
171    unfold l2NormSq
172    exact integral_nonneg fun x => sq_nonneg _
173  calc |∑ ι, (pcoef q ι * symbolForm (symbolK d s (pvec ι)) (fun x => a x * w x) w -
174        pcoef q ι * symbolForm (symbolK d s (pvec ι)) (fun x => Real.sqrt (a x) * w x)
175          (fun x => Real.sqrt (a x) * w x))|
176      ≤ ∑ ι, |pcoef q ι * (symbolForm (symbolK d s (pvec ι)) (fun x => a x * w x) w -
177        symbolForm (symbolK d s (pvec ι)) (fun x => Real.sqrt (a x) * w x)
178          (fun x => Real.sqrt (a x) * w x))| := by
179        refine Finset.abs_sum_le_sum_abs _ _ |>.trans (le_of_eq ?_)
180        refine Finset.sum_congr rfl fun ι _ => ?_
181        rw [mul_sub]
182    _ ≤ ∑ ι, (d : ℝ) * (|CC ι| * ((Lm ^ 2 + Torus.supNorm (fun x => Real.sqrt (a x)) ^ 2) *
183          l2NormSq w)) := by
184        refine Finset.sum_le_sum fun ι _ => ?_
185        rw [abs_mul]
186        refine mul_le_mul (abs_pcoef_le hd1 q hq ι) ?_ (abs_nonneg _) (Nat.cast_nonneg d)
187        refine (hCC ι L La Lm Lw a w ha hC11 hLa hLm hLw).trans ?_
188        rw [mul_assoc]
189        exact mul_le_mul_of_nonneg_right (le_abs_self _) hR
190    _ = _ := by
191        rw [← Finset.mul_sum, ← Finset.sum_mul]
192        ring

214theorem weighted_bound (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hs1 : (d : ℝ) - 2 < s) (hsd : s < d) :

216      (∀ x, 0 ≤ a x) → (∀ x, a x ≤ A) → IsC11 L a → Torus.IsCk 1 w →

218      |symbolForm (symbolK d s q) (fun x => a x * w x) w -
219        symbolForm (symbolK d s q) (fun x => Real.sqrt (a x) * w x)
220          (fun x => Real.sqrt (a x) * w x)| ≤ C * (L / 2 + A) * l2NormSq w := by
221  obtain ⟨C, hC⟩ := weighted_uniform hd s hs0 hs1 hsd
222  refine ⟨|C|, abs_nonneg _, fun q hq L A a w hL ha hA hC11 hw1 hw => ?_⟩
223  have hLa := isLipschitz_of_cbNorm hC11.1
224  have hLm := sqrt_isLipschitz hL ha hC11
225  have hLw := isLipschitz_of_cbNorm hw1
226  refine (hC q hq _ _ _ _ a w ha hC11 hLa hLm hLw hw).trans ?_
