182/-- `u² e^{-λ u} ≤ min(u, K)²` for `K ≥ 2/λ`, `u ≥ 0`. -/
183theorem sq_mul_exp_le {lam K u : ℝ} (hlam : 0 < lam) (hK : 2 / lam ≤ K) (hu : 0 ≤ u) :
184    u ^ 2 * Real.exp (-lam * u) ≤ min u K ^ 2 := by
185  rcases le_total u K with h | h
186  · rw [min_eq_left h]
187    have : Real.exp (-lam * u) ≤ 1 := Real.exp_le_one_iff.2 (by nlinarith)
188    nlinarith [sq_nonneg u]
189  · rw [min_eq_right h]
190    have ht : (lam * u / 2) ≤ Real.exp (lam * u / 2) := by
191      have := Real.add_one_le_exp (lam * u / 2)
192      linarith
193    have h2 : u * Real.exp (-(lam * u / 2)) ≤ 2 / lam := by
194      have hpos : 0 < Real.exp (lam * u / 2) := Real.exp_pos _
195      rw [Real.exp_neg, ← div_eq_mul_inv, div_le_iff₀ hpos, div_mul_eq_mul_div, le_div_iff₀ hlam]
196      nlinarith
197    have h3 : u ^ 2 * Real.exp (-lam * u) = (u * Real.exp (-(lam * u / 2))) ^ 2 := by
198      rw [mul_pow, ← Real.exp_nat_mul]
199      congr 2
200      push_cast; ring
201    rw [h3]
202    have h4 : 0 ≤ u * Real.exp (-(lam * u / 2)) := mul_nonneg hu (Real.exp_pos _).le
203    have h5 : u * Real.exp (-(lam * u / 2)) ≤ K := h2.trans hK
204    nlinarith

206/-- **Retained-repulsion integral** (`A:eq:radial` as a distribution-function bound):
207for `K ≥ max(1, 2/λ)`, `∫ e(x-y)² e^{-λ e(x-y)} dμ(y) ≤ Λ c₁ gFun K`. -/

209    ∃ c₁ : ℝ, 0 < c₁ ∧ ∀ (ρ : Torus d → ℝ) (Λ : ℝ), (∀ y, ρ y ≤ Λ) → ∀ (lam K : ℝ), 0 < lam → 1 ≤ K →
210      2 / lam ≤ K → ∀ x : Torus d,
211      ∫⁻ y, ENNReal.ofReal (envT s (x - y) ^ 2 * Real.exp (-lam * envT s (x - y))) ∂(withDensityReal ρ) ≤
212        ENNReal.ofReal Λ * ENNReal.ofReal (c₁ * gFun d s K) := by
213  obtain ⟨c₁, hc₁, hc⟩ := lintegral_minsq_envT hd hs0 hsd
214  refine ⟨c₁, hc₁, fun ρ Λ hΛ lam K hlam hK1 hK x => ?_⟩
215  refine (KernelAux2.lintegral_conv_le ρ Λ hΛ
216    (fun z => ENNReal.ofReal (envT s z ^ 2 * Real.exp (-lam * envT s z))) x).trans ?_
217  gcongr
218  refine (lintegral_mono fun z => ENNReal.ofReal_le_ofReal
219    (sq_mul_exp_le hlam hK (zero_le_one.trans (one_le_envT z)))).trans (hc K hK1)

221end Repulsion

227/-- Schur test in `ℝ≥0∞` form for a translation kernel: if `∫ k(x-y) dμ(y) ≤ A` and
228`∫ k(x-y) dμ(x) ≤ A`, then `∫ (∫ k(x-y) f(y) dμ(y))² dμ(x) ≤ A² ∫ f² dμ`. -/
229theorem schur_sq {ν : Measure (Torus d)} [SFinite ν] {k f : Torus d → ℝ≥0∞} (hk : Measurable k)
230    (hf : Measurable f) {A : ℝ≥0∞}
231    (h1 : ∀ x, ∫⁻ y, k (x - y) ∂ν ≤ A) (h2 : ∀ y, ∫⁻ x, k (x - y) ∂ν ≤ A) :
232    ∫⁻ x, (∫⁻ y, k (x - y) * f y ∂ν) ^ 2 ∂ν ≤ A ^ 2 * ∫⁻ y, f y ^ 2 ∂ν := by
233  have hkm : Measurable fun p : Torus d × Torus d => k (p.1 - p.2) :=
234    hk.comp (measurable_fst.sub measurable_snd)
235  have hpt : ∀ x, (∫⁻ y, k (x - y) * f y ∂ν) ^ 2 ≤ A * ∫⁻ y, k (x - y) * f y ^ 2 ∂ν := by
236    intro x
237    have hφ : Measurable fun y => k (x - y) := hk.comp (measurable_const.sub measurable_id)
238    have := PoleCavityAux.weighted_cs ν (fun y => k (x - y)) f hφ hf
239    exact this.trans (mul_le_mul' (h1 x) le_rfl)
240  calc ∫⁻ x, (∫⁻ y, k (x - y) * f y ∂ν) ^ 2 ∂ν
241      ≤ ∫⁻ x, A * ∫⁻ y, k (x - y) * f y ^ 2 ∂ν ∂ν := lintegral_mono hpt
242    _ = A * ∫⁻ x, ∫⁻ y, k (x - y) * f y ^ 2 ∂ν ∂ν := by
243        rw [lintegral_const_mul]
244        exact Measurable.lintegral_prod_right' (f := fun p : Torus d × Torus d => k (p.1 - p.2) * f p.2 ^ 2)
245          (hkm.mul ((hf.comp measurable_snd).pow_const 2))
246    _ = A * ∫⁻ y, ∫⁻ x, k (x - y) * f y ^ 2 ∂ν ∂ν := by
247        congr 1
248        refine lintegral_lintegral_swap ?_
249        exact (hkm.mul ((hf.comp measurable_snd).pow_const 2)).aemeasurable
250    _ ≤ A * ∫⁻ y, A * f y ^ 2 ∂ν := by
251        gcongr with y
252        have hm : Measurable fun x : Torus d => k (x - y) := hk.comp (measurable_id.sub measurable_const)
253        rw [lintegral_mul_const _ hm]
254        exact mul_le_mul' (h2 y) le_rfl
255    _ = A ^ 2 * ∫⁻ y, f y ^ 2 ∂ν := by
256        rw [lintegral_const_mul _ (hf.pow_const 2), ← mul_assoc, sq]
257
258end Schur

390/-- The doubly centered kernel `h_c(x,y) = h(x,y) - H(x) - H(y) + h₀`. -/
391def cK (h : Torus d → Torus d → ℝ) (π : Measure (Torus d)) (x y : Torus d) : ℝ :=
392  h x y - Hc h π x - Hc h π y + h0c h π

402    + (1 - 2 * ((N : ℝ) - 1) / (N : ℝ) ^ 2) * h0c h π

440theorem sum_sum_erase_split {N : ℕ} (i : Fin N) (f : Fin N → Fin N → ℝ) :

667    {f : E → ℝ} (hf : Measurable f) (c : ℝ) :
668    ∫⁻ y, ‖∑ k, f (y k) - c‖ₑ ^ 2 ∂(pinnedLaw V (β / N) n a)
669      ≤ ENNReal.ofReal (2 * n) * ∫⁻ z, ‖f z‖ₑ ^ 2 *
670          ENNReal.ofReal (Real.exp (V.M' + β * V.γ + pinField V (β / N) a z)) ∂μ
671        + ENNReal.ofReal ((∑ k, ∫ z, f z ∂((pinnedLaw V (β / N) n a).map (fun y => y k)) - c) ^ 2) := by
672  have hlam : 0 ≤ β / N := div_nonneg hβ (Nat.cast_nonneg N)
673  have hPr := isProb_pinnedLaw V hlam n a
674  set P := pinnedLaw V (β / N) n a with hP
675  set A := ∫⁻ z, ‖f z‖ₑ ^ 2 * ENNReal.ofReal (Real.exp (V.M' + β * V.γ + pinField V (β / N) a z)) ∂μ with hA
676  have hn0 : (n : ℝ) ≠ 0 := by exact_mod_cast NeZero.ne n
677  by_cases hAtop : A = ⊤
678  · rw [hAtop, ENNReal.mul_top (by simpa using hn0)]
679    simp
680  have hS : Measurable fun y : Fin n → E => ∑ k, f (y k) :=
681    Finset.measurable_sum _ fun k _ => hf.comp (measurable_pi_apply k)
682  -- second moment of f against each marginal
683  have hmarg : ∀ k : Fin n, ∫⁻ z, ‖f z‖ₑ ^ 2 ∂(P.map (fun y => y k)) ≤ A := by
684    intro k
685    calc ∫⁻ z, ‖f z‖ₑ ^ 2 ∂(P.map (fun y => y k))
686        ≤ ∫⁻ z, ‖f z‖ₑ ^ 2 ∂(μ.withDensity fun z => ENNReal.ofReal
687            (Real.exp (V.M' + β * V.γ + pinField V (β / N) a z))) :=
688          lintegral_mono' (pinned_marginal_le_density V hβ hN0 hN a k) le_rfl
689      _ = A := by
690          rw [lintegral_withDensity_eq_lintegral_mul₀]
691          · simp only [Pi.mul_apply]
692            exact lintegral_congr fun z => mul_comm _ _
693          · exact (ENNReal.measurable_ofReal.comp (Real.measurable_exp.comp
694              (measurable_const.add (isField_pinField V hlam a).measurable))).aemeasurable
695          · exact (hf.enorm.pow_const 2).aemeasurable
696  have hfL2 : ∀ k : Fin n, MemLp f 2 (P.map (fun y => y k)) := by
697    intro k
698    refine (memLp_two_iff_integrable_sq hf.aestronglyMeasurable).2 ⟨hf.aestronglyMeasurable.pow 2, ?_⟩
699    simp only [HasFiniteIntegral, enorm_pow]
700    exact lt_of_le_of_lt (hmarg k) (lt_top_iff_ne_top.2 hAtop)
701  have hfi : ∀ k : Fin n, Integrable (fun y : Fin n → E => f (y k)) P := by
702    intro k
703    have := (hfL2 k).integrable (by norm_num)
704    exact (integrable_map_measure hf.aestronglyMeasurable (measurable_pi_apply k).aemeasurable).1 this
705  -- variance bound
706  have hvar : evariance (fun y : Fin n → E => ∑ k, f (y k)) P ≤ ENNReal.ofReal (2 * n) * A := by
707    have h1 := weak_linear_variance_symm (B := B) V hβ hN0 hN a hκ f hf
708    refine h1.trans ?_
709    have h2 : evariance f (P.map (fun x => x 0)) ≤ A := by
710      refine (evariance_le_lintegral_sub (P.map (fun x => x 0)) hf.aestronglyMeasurable 0).trans ?_
711      simpa using hmarg 0
712    have e : (2 : ℝ≥0∞) * n = ENNReal.ofReal (2 * n) := by
713      rw [ENNReal.ofReal_mul (by norm_num)]; simp
714    rw [e]
715    gcongr
716  have hvtop : evariance (fun y : Fin n → E => ∑ k, f (y k)) P < ⊤ :=
717    lt_of_le_of_lt hvar (ENNReal.mul_lt_top ENNReal.ofReal_lt_top (lt_top_iff_ne_top.2 hAtop))
718  have hSmem : MemLp (fun y : Fin n → E => ∑ k, f (y k)) 2 P :=
719    (evariance_lt_top_iff_memLp hS.aestronglyMeasurable).1 hvtop
720  have hSc : MemLp (fun y : Fin n → E => ∑ k, f (y k) - c) 2 P := hSmem.sub (memLp_const c)
721  rw [lintegral_enorm_sq_eq_ofReal hSc]
722  have hmean : ∫ y, ∑ k, f (y k) ∂P = ∑ k, ∫ z, f z ∂(P.map (fun y => y k)) := by
723    rw [integral_finsetSum _ fun k _ => hfi k]
724    refine Finset.sum_congr rfl fun k _ => ?_
725    rw [integral_map (measurable_pi_apply k).aemeasurable hf.aestronglyMeasurable]
726  have e1 := variance_eq_sub hSc
727  have e2 : variance (fun y : Fin n → E => ∑ k, f (y k) - c) P = variance (fun y : Fin n → E => ∑ k, f (y k)) P :=
728    variance_sub_const hS.aestronglyMeasurable c
729  have hint : ∫ y, (∑ k, f (y k) - c) ^ 2 ∂P
730      = variance (fun y : Fin n → E => ∑ k, f (y k)) P + (∫ y, ∑ k, f (y k) ∂P - c) ^ 2 := by

732    simp only [Pi.pow_apply] at this
733    rw [integral_sub (hSmem.integrable (by norm_num)) (integrable_const c)] at this
734    have hc : ∫ _y, c ∂P = c := by simp
735    rw [hc] at this
736    rw [e2] at this
737    linarith
738  rw [hint, hmean, ENNReal.ofReal_add (variance_nonneg _ _) (sq_nonneg _), ofReal_variance hSmem]
