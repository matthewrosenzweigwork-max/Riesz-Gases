6/-!
7# Worker U4: proof of `HRS.Whole.mov_scope_infinite_dim` (frozen in `HRS/Whole/Moving.lean`)
8
9Route (see the brief): `ρ₀ > 0` everywhere follows from the Gibbs formula and `∫ ρ₀ = 1`, so
10`√ρ₀` is smooth.  Bump functions `χ_k` with disjoint supports (centres on the first coordinate axis,
11distance at least `1`, radius `1/2`) give `m_k = ∫ χ_k ρ₀ > 0`; the `n` functions
12`ψ_i = m_{i+1}⁻¹ χ_{i+1} - m_0⁻¹ χ_0` are smooth with compact support, have `∫ ψ_i ρ₀ = 0`, and are
13linearly independent (evaluate at the centres).  Smallness: for `|c_i| ≤ δ` the `C_b^{j+1}` norm and the
14`Σ^m` norm of `√ρ₀ ∑ c_i ψ_i` are bounded by `∑ |c_i| (‖ψ_i‖_{C_b} + ‖√ρ₀ ψ_i‖_{Σ^m})`, which is at most
15`δ K`, and `δ = εstar / (K+1)` makes this at most `εstar`.
16-/

27/-- Centre of the `k`-th bump: `k` times a unit vector on the first coordinate axis. -/
28def u4Center (hd : 0 < d) (k : ℕ) : Euc d :=
29  (k : ℝ) • EuclideanSpace.single (⟨0, hd⟩ : Fin d) (1 : ℝ)

31/-- The `k`-th bump: radius `1/4` inside, `1/2` outside. -/
32def u4Bump (hd : 0 < d) (k : ℕ) : ContDiffBump (u4Center hd k) where
33  rIn := 1 / 4
34  rOut := 1 / 2
35  rIn_pos := by norm_num
36  rIn_lt_rOut := by norm_num

38lemma u4_dist_centers (hd : 0 < d) {k l : ℕ} (hkl : k ≠ l) :
39    1 ≤ dist (u4Center hd k) (u4Center hd l) := by
40  have hdist : dist (u4Center hd k) (u4Center hd l) = |(k : ℝ) - l| := by
41    simp only [dist_eq_norm, u4Center]
42    rw [← sub_smul, norm_smul, EuclideanSpace.norm_single]
43    simp
44  rw [hdist]
45  rcases lt_or_gt_of_ne hkl with h | h
46  · have h1 : k + 1 ≤ l := h
47    have h' : (k : ℝ) + 1 ≤ l := by exact_mod_cast h1
48    rw [abs_of_nonpos (by linarith)]
49    linarith
50  · have h1 : l + 1 ≤ k := h
51    have h' : (l : ℝ) + 1 ≤ k := by exact_mod_cast h1
52    rw [abs_of_nonneg (by linarith)]
53    linarith

55lemma u4_bump_center (hd : 0 < d) (k : ℕ) : u4Bump hd k (u4Center hd k) = 1 :=
56  (u4Bump hd k).one_of_mem_closedBall (Metric.mem_closedBall_self (u4Bump hd k).rIn_pos.le)

58lemma u4_bump_zero (hd : 0 < d) {k l : ℕ} (hkl : k ≠ l) : u4Bump hd k (u4Center hd l) = 0 := by
59  apply (u4Bump hd k).zero_of_le_dist
60  have h1 := u4_dist_centers hd hkl
61  have h2 : (u4Bump hd k).rOut = 1 / 2 := rfl
62  rw [dist_comm, h2]
63  linarith

74/-- `m_k = ∫ χ_k ρ₀`. -/
75def u4M (hd : 0 < d) (ρ : Euc d → ℝ) (k : ℕ) : ℝ := ∫ x, u4Bump hd k x * ρ x

77lemma u4_m_pos (hd : 0 < d) {ρ : Euc d → ℝ} (hρc : Continuous ρ) (hρp : ∀ x, 0 < ρ x) (k : ℕ) :
78    0 < u4M hd ρ k := by
79  unfold u4M
80  refine Continuous.integral_pos_of_hasCompactSupport_nonneg_nonzero (x := u4Center hd k) ?_ ?_ ?_ ?_
81  · exact (u4Bump hd k).continuous.mul hρc

83  · exact fun x => mul_nonneg (u4Bump hd k).nonneg (hρp x).le

85    rw [u4_bump_center, one_mul]
86    exact (hρp _).ne'

88/-- `ψ_i = m_{i+1}⁻¹ χ_{i+1} - m_0⁻¹ χ_0`. -/
89def u4Psi (hd : 0 < d) (ρ : Euc d → ℝ) (i : ℕ) (x : Euc d) : ℝ :=
90  (u4M hd ρ (i + 1))⁻¹ * u4Bump hd (i + 1) x + (-(u4M hd ρ 0)⁻¹) * u4Bump hd 0 x

92lemma u4Psi_cs (hd : 0 < d) (ρ : Euc d → ℝ) (i : ℕ) : U4CS (u4Psi hd ρ i) :=
93  u4_cs_add (u4_cs_const_mul _ (u4_bump_cs hd (i + 1))) (u4_cs_const_mul _ (u4_bump_cs hd 0))

95lemma u4Psi_mean (hd : 0 < d) {ρ : Euc d → ℝ} (hρc : Continuous ρ) (hρp : ∀ x, 0 < ρ x) (i : ℕ) :
96    ∫ x, u4Psi hd ρ i x * ρ x = 0 := by
97  have hfun : (fun x => u4Psi hd ρ i x * ρ x) = fun x =>
98      (u4M hd ρ (i + 1))⁻¹ * (u4Bump hd (i + 1) x * ρ x) +
99        (-(u4M hd ρ 0)⁻¹) * (u4Bump hd 0 x * ρ x) := by
100    funext x
101    unfold u4Psi
102    ring
103  rw [hfun, integral_add ((u4_int (u4_bump_cs hd (i + 1)) hρc).const_mul _)
104    ((u4_int (u4_bump_cs hd 0) hρc).const_mul _), integral_const_mul, integral_const_mul]
105  have h1 : ∫ x, u4Bump hd (i + 1) x * ρ x = u4M hd ρ (i + 1) := rfl
106  have h0 : ∫ x, u4Bump hd 0 x * ρ x = u4M hd ρ 0 := rfl
107  rw [h1, h0]
108  have hm1 : u4M hd ρ (i + 1) ≠ 0 := (u4_m_pos hd hρc hρp _).ne'
109  have hm0 : u4M hd ρ 0 ≠ 0 := (u4_m_pos hd hρc hρp _).ne'
110  linear_combination (inv_mul_cancel₀ hm1) - (inv_mul_cancel₀ hm0)

112lemma u4Psi_eval (hd : 0 < d) (ρ : Euc d → ℝ) (i j : ℕ) :
113    u4Psi hd ρ i (u4Center hd (j + 1)) = if i = j then (u4M hd ρ (j + 1))⁻¹ else 0 := by
114  unfold u4Psi
115  rw [u4_bump_zero hd (show (0 : ℕ) ≠ j + 1 by omega)]
116  by_cases hij : i = j
117  · subst hij
118    rw [u4_bump_center]
119    simp
120  · rw [u4_bump_zero hd (show i + 1 ≠ j + 1 by omega)]
121    simp [hij]

123lemma u4_lin_indep (hd : 0 < d) {ρ : Euc d → ℝ} (hρc : Continuous ρ) (hρp : ∀ x, 0 < ρ x)
124    (n : ℕ) : LinearIndependent ℝ (fun i : Fin n => u4Psi hd ρ (i : ℕ)) := by
125  rw [Fintype.linearIndependent_iff]
126  intro g hg i0
127  have hsum := congrFun hg (u4Center hd ((i0 : ℕ) + 1))
128  simp only [Finset.sum_apply, Pi.smul_apply, smul_eq_mul, Pi.zero_apply] at hsum
129  have hval : ∀ i : Fin n, g i * u4Psi hd ρ (i : ℕ) (u4Center hd ((i0 : ℕ) + 1)) =
130      if i = i0 then g i * (u4M hd ρ ((i0 : ℕ) + 1))⁻¹ else 0 := by
131    intro i
132    rw [u4Psi_eval]
133    by_cases h : i = i0
134    · subst h
135      simp
136    · have h' : (i : ℕ) ≠ (i0 : ℕ) := fun e => h (Fin.ext e)
137      simp [h, h']
138  rw [Finset.sum_congr rfl (fun i _ => hval i), Finset.sum_eq_single i0] at hsum
139  · rw [if_pos rfl] at hsum
140    exact (mul_eq_zero.1 hsum).resolve_right (inv_ne_zero (u4_m_pos hd hρc hρp _).ne')
141  · intro b _ hb
142    exact if_neg hb
143  · intro h
144    exact absurd (Finset.mem_univ i0) h

164/-- The smallness and the initial-class membership for the family `∑ c_i ψ_i`. -/
165lemma u4_small (j m : ℕ) {ρ : Euc d → ℝ} (hρc : Continuous ρ)
166    (hsq : ContDiff ℝ ∞ (fun x => Real.sqrt (ρ x))) (εstar : ℝ) (hε : 0 < εstar) {n : ℕ}
167    (ψ : Fin n → Euc d → ℝ) (hψ : ∀ i, U4CS (ψ i)) (hint : ∀ i, ∫ x, ψ i x * ρ x = 0) :
168    ∃ δ : ℝ, 0 < δ ∧ ∀ c : Fin n → ℝ, (∀ i, |c i| ≤ δ) →
169      MovInitialClass ρ j m εstar (fun x => ∑ i, c i * ψ i x) := by
170  have hP1_0 : cbNorm (j + 1) (fun _ : Euc d => (0 : ℝ)) ≤ 0 := (u4_cbNorm_zero _).le
171  have hP1_add : ∀ f g : Euc d → ℝ, U4CS f → U4CS g →
172      cbNorm (j + 1) (fun x => f x + g x) ≤ cbNorm (j + 1) f + cbNorm (j + 1) g :=
173    fun f g hf hg => u4_cbNorm_add_le (j + 1) hf hg
174  have hP1_mul : ∀ (a : ℝ) (f : Euc d → ℝ), U4CS f →
175      cbNorm (j + 1) (fun x => a * f x) = |a| * cbNorm (j + 1) f :=
176    fun a f hf => cbNorm_const_mul (j + 1) a f hf.1
177  have hP2_0 : oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * (0 : ℝ)) ≤ 0 := by
178    have e : (fun x => Real.sqrt (ρ x) * (0 : ℝ)) = fun _ : Euc d => (0 : ℝ) := by
179      funext x
180      ring
181    rw [e, u4_oscNorm_zero]
182  have hP2_add : ∀ f g : Euc d → ℝ, U4CS f → U4CS g →
183      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * (f x + g x)) ≤
184        oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * f x) +
185          oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * g x) := by
186    intro f g hf hg
187    have e : (fun x => Real.sqrt (ρ x) * (f x + g x)) =
188        fun x => Real.sqrt (ρ x) * f x + Real.sqrt (ρ x) * g x := by
189      funext x
190      ring
191    rw [e]
192    exact u4_oscNorm_add_le (m : ℝ) (u4_cs_mul hsq hf) (u4_cs_mul hsq hg)
193  have hP2_mul : ∀ (a : ℝ) (f : Euc d → ℝ), U4CS f →
194      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * (a * f x)) =
195        |a| * oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * f x) := by
196    intro a f _
197    have e : (fun x => Real.sqrt (ρ x) * (a * f x)) = fun x => a * (Real.sqrt (ρ x) * f x) := by
198      funext x
199      ring
200    rw [e, oscNorm_const_mul]
201  -- the constant `K = ∑ (‖ψ_i‖_{C_b} + ‖√ρ ψ_i‖_{Σ^m})`
202  obtain ⟨K, hK⟩ : ∃ K : ℝ, K = ∑ i, (cbNorm (j + 1) (ψ i) +
203      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x)) := ⟨_, rfl⟩
204  have hnn : ∀ i, 0 ≤ cbNorm (j + 1) (ψ i) +
205      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x) :=
206    fun i => add_nonneg (u4_cbNorm_nonneg _ _) (u4_oscNorm_nonneg _ _)
207  have hK0 : 0 ≤ K := by
208    rw [hK]
209    exact Finset.sum_nonneg fun i _ => hnn i
210  have hK1 : (0 : ℝ) < K + 1 := by linarith
211  have hδ : 0 < εstar / (K + 1) := div_pos hε hK1
212  refine ⟨εstar / (K + 1), hδ, fun c hc => ?_⟩
213  obtain ⟨hcs, hle1⟩ := u4_sum_bound (fun f => cbNorm (j + 1) f) hP1_0 hP1_add hP1_mul c ψ hψ
214    Finset.univ
215  obtain ⟨-, hle2⟩ := u4_sum_bound (fun f => oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * f x))
216    hP2_0 hP2_add hP2_mul c ψ hψ Finset.univ
217  dsimp only at hle1 hle2
218  have hsplit : ∑ i, |c i| * (cbNorm (j + 1) (ψ i) +
219      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x)) =
220      ∑ i, |c i| * cbNorm (j + 1) (ψ i) +
221        ∑ i, |c i| * oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x) := by
222    rw [← Finset.sum_add_distrib]
223    exact Finset.sum_congr rfl fun i _ => mul_add _ _ _
224  have hdelta : ∑ i, |c i| * (cbNorm (j + 1) (ψ i) +
225      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x)) ≤
226      ∑ i, εstar / (K + 1) * (cbNorm (j + 1) (ψ i) +
227        oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x)) :=
228    Finset.sum_le_sum fun i _ => mul_le_mul_of_nonneg_right (hc i) (hnn i)
229  have hsumK : ∑ i, εstar / (K + 1) * (cbNorm (j + 1) (ψ i) +
230      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x)) = εstar / (K + 1) * K := by
231    rw [hK, Finset.mul_sum]
232  have hfin : εstar / (K + 1) * K ≤ εstar := by
233    rw [div_mul_eq_mul_div, div_le_iff₀ hK1]
234    nlinarith
235  have hsmall : cbNorm (j + 1) (fun x => ∑ i, c i * ψ i x) +
236      oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ∑ i, c i * ψ i x) ≤ εstar := by
237    calc cbNorm (j + 1) (fun x => ∑ i, c i * ψ i x) +
238          oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ∑ i, c i * ψ i x)
239        ≤ ∑ i, |c i| * cbNorm (j + 1) (ψ i) +
240            ∑ i, |c i| * oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x) :=
241          add_le_add hle1 hle2
242      _ = ∑ i, |c i| * (cbNorm (j + 1) (ψ i) +
243            oscNorm (m : ℝ) (fun x => Real.sqrt (ρ x) * ψ i x)) := hsplit.symm
244      _ ≤ _ := hdelta
245      _ = εstar / (K + 1) * K := hsumK
246      _ ≤ εstar := hfin
247  have hcs' : U4CS (fun x => ∑ i, c i * ψ i x) := by
248    have := hcs
249    exact this
250  refine ⟨u4_memCb (j + 1) hcs', u4_memOsc (m : ℝ) (u4_cs_mul hsq hcs'), ?_, ?_⟩
251  · show ∫ x, (∑ i, c i * ψ i x) * ρ x = 0
252    have e : ∀ x, (∑ i, c i * ψ i x) * ρ x = ∑ i, c i * (ψ i x * ρ x) := by
253      intro x
254      rw [Finset.sum_mul]
255      exact Finset.sum_congr rfl fun i _ => mul_assoc _ _ _
256    simp_rw [e]
257    rw [integral_finsetSum]
258    · have hmean : ∑ i, ∫ x, c i * (ψ i x * ρ x) = ∑ i, c i * ∫ x, ψ i x * ρ x :=
259        Finset.sum_congr rfl fun i _ => integral_const_mul _ _
260      rw [hmean]
261      simp [hint]
262    · intro i _
263      exact (u4_int (hψ i) hρc).const_mul (c i)
264  · exact hsmall

266/-- The linearly independent family and the smallness statement (`n` bumps). -/
267lemma u4_main (hd : 0 < d) (j m : ℕ) {ρ : Euc d → ℝ} (hρc : Continuous ρ) (hρp : ∀ x, 0 < ρ x)
268    (hsq : ContDiff ℝ ∞ (fun x => Real.sqrt (ρ x))) (εstar : ℝ) (hε : 0 < εstar) (n : ℕ) :
269    ∃ ψ : Fin n → Euc d → ℝ, LinearIndependent ℝ ψ ∧
270      (∀ i, ContDiff ℝ ∞ (ψ i) ∧ HasCompactSupport (ψ i) ∧ ∫ x, ψ i x * ρ x = 0) ∧
271      ∃ δ : ℝ, 0 < δ ∧ ∀ c : Fin n → ℝ, (∀ i, |c i| ≤ δ) →
272        MovInitialClass ρ j m εstar (fun x => ∑ i, c i * ψ i x) := by
273  have hψcs : ∀ i : Fin n, U4CS (u4Psi hd ρ (i : ℕ)) := fun i => u4Psi_cs hd ρ (i : ℕ)
274  have hint : ∀ i : Fin n, ∫ x, u4Psi hd ρ (i : ℕ) x * ρ x = 0 :=
275    fun i => u4Psi_mean hd hρc hρp (i : ℕ)
276  obtain ⟨δ, hδ, hsmall⟩ := u4_small j m hρc hsq εstar hε (fun i : Fin n => u4Psi hd ρ (i : ℕ))
277    hψcs hint
278  exact ⟨fun i : Fin n => u4Psi hd ρ (i : ℕ), u4_lin_indep hd hρc hρp n,
279    fun i => ⟨(hψcs i).1, (hψcs i).2, hint i⟩, δ, hδ, hsmall⟩
