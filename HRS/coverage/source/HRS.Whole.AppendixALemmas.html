53/-- `‖ψ‖_{Ḣ^a(ℝ^d)}² = ∫ (2π|ξ|)^{2a}|ψ̂|² dξ` is finite (`ψ ∈ L²`, Plancherel transform). -/

55  MemLp ψ 2 volume ∧
56    Integrable fun ξ : Euc d => (2 * Real.pi * ‖ξ‖) ^ (2 * a) *
57      ‖HRS.PDE.ToolboxRd.fourierL2 ψ ξ‖ ^ 2

59/-- `‖ψ‖_{Ḣ^a(ℝ^d)} = (∫ (2π|ξ|)^{2a}|ψ̂|² dξ)^{1/2}` (meaningful under
60`HasFiniteHomSobolevRd`). -/
61def homSobolevRd (a : ℝ) (ψ : Euc d → ℝ) : ℝ :=
62  Real.sqrt (∫ ξ : Euc d, (2 * Real.pi * ‖ξ‖) ^ (2 * a) *

65/-- `B:prop:coercivity` on `ℝ^d` (M L2842-2857), the pair `(C₀, C)`: the lower bound
66`CompletionNonnegRd d s C₀` (`HRS.Whole.AppendixA`: `F_N + o_N ≥ 0`, `o_N = oNΛ`) and, for every compactly
67supported Lipschitz `ψ` (constant `L`) with finite `Ḣ^{(d-s)/2}` seminorm,
68`|∫ψ d(μ_N - μ)| ≤ C λ L + C ‖ψ‖_{Ḣ^{(d-s)/2}} (F_N + o_N(Λ))^{1/2}` with `λ = (NΛ)^{-1/d}`

70transform of `ψ` is the true one; this is the only class of tests used in `B:lem:test-cutoff`.)  Statement repair
71(docs/ISSUES.md, 2026-10-19): at `s = 0` the densities carry a finite logarithmic moment (as in `CompletionNonnegRd`). -/

77    ∀ (ψ : Euc d → ℝ) (L : ℝ), 0 ≤ L → HRS.PDE.ToolboxRd.IsLipschitz L ψ → HasCompactSupport ψ →

158theorem cnorm_unpack (χ : Euc d → ℝ) (h : Euc d → ℝ) (hh : cNormE χ h < ⊤) :

160      (∀ x, |h x| ≤ b) ∧ (∀ x y : Euc d, |h x - h y| ≤ l * ‖x - y‖) ∧
161      ∀ z : Euc d, HRS.PDE.ToolboxRd.MemSobolev ((mStar d - 1 : ℕ) : ℝ) (fun x => χ (x - z) * h x) ∧
162        HRS.PDE.ToolboxRd.sobolevNormSq ((mStar d - 1 : ℕ) : ℝ) (fun x => χ (x - z) * h x) ≤ Uv ^ 2 := by
163  classical
164  -- unpack the finiteness of `𝒩(h)`
165  unfold cNormE lipNormE at hh ⊢
166  set A : ℝ≥0∞ := ⨆ x : Euc d, ENNReal.ofReal |h x| with hA
167  set Q : ℝ≥0∞ := ⨆ p : {p : Euc d × Euc d // p.1 ≠ p.2},
168    ENNReal.ofReal (|h p.1.1 - h p.1.2| / ‖p.1.1 - p.1.2‖) with hQ
169  set Um : ℝ≥0∞ := ulocNormE (mStar d - 1) χ h with hUm
170  have hAQU : A + Q + Um < ⊤ := hh
171  have hA' : A < ⊤ := lt_of_le_of_lt (le_self_add.trans le_self_add) hAQU
172  have hQ' : Q < ⊤ := lt_of_le_of_lt (le_add_self.trans le_self_add) hAQU
173  have hU' : Um < ⊤ := lt_of_le_of_lt le_add_self hAQU
174  set b : ℝ := A.toReal with hbdef
175  set l : ℝ := Q.toReal with hldef
176  set Uv : ℝ := Um.toReal with hUvdef
177  have hn : (A + Q + Um).toReal = b + l + Uv := by
178    rw [ENNReal.toReal_add (ENNReal.add_ne_top.mpr ⟨hA'.ne, hQ'.ne⟩) hU'.ne,
179      ENNReal.toReal_add hA'.ne hQ'.ne]
180  have hb0 : 0 ≤ b := ENNReal.toReal_nonneg
181  have hl0 : 0 ≤ l := ENNReal.toReal_nonneg
182  have hU0 : 0 ≤ Uv := ENNReal.toReal_nonneg
183  have hbd : ∀ x, |h x| ≤ b := by
184    intro x
185    have : ENNReal.ofReal |h x| ≤ A := le_iSup (fun x : Euc d => ENNReal.ofReal |h x|) x
186    exact (ENNReal.ofReal_le_iff_le_toReal hA'.ne).mp this
187  have hlip : ∀ x y : Euc d, |h x - h y| ≤ l * ‖x - y‖ := by
188    intro x y
189    by_cases hxy : x = y
190    · subst hxy; simp
191    · have : ENNReal.ofReal (|h x - h y| / ‖x - y‖) ≤ Q :=
192        le_iSup (fun p : {p : Euc d × Euc d // p.1 ≠ p.2} =>
193          ENNReal.ofReal (|h p.1.1 - h p.1.2| / ‖p.1.1 - p.1.2‖)) ⟨(x, y), hxy⟩
194      have h2 := (ENNReal.ofReal_le_iff_le_toReal hQ'.ne).mp this
195      have hpos : 0 < ‖x - y‖ := norm_pos_iff.mpr (sub_ne_zero.mpr hxy)
196      rw [div_le_iff₀ hpos] at h2
197      exact h2
198  have hSob : ∀ z : Euc d, HRS.PDE.ToolboxRd.MemSobolev ((mStar d - 1 : ℕ) : ℝ)
199        (fun x => χ (x - z) * h x) ∧
200      HRS.PDE.ToolboxRd.sobolevNormSq ((mStar d - 1 : ℕ) : ℝ) (fun x => χ (x - z) * h x) ≤ Uv ^ 2 := by
201    intro z
202    have h1 : sobolevNormE ((mStar d - 1 : ℕ) : ℝ) (fun x => χ (x - z) * h x) ≤ Um :=
203      le_iSup (fun z : Euc d => sobolevNormE ((mStar d - 1 : ℕ) : ℝ) fun x => χ (x - z) * h x) z
204    have h2 : sobolevNormE ((mStar d - 1 : ℕ) : ℝ) (fun x => χ (x - z) * h x) ≠ ⊤ :=
205      ne_top_of_le_ne_top hU'.ne h1
206    unfold sobolevNormE at h1 h2
207    by_cases hM : HRS.PDE.ToolboxRd.MemSobolev ((mStar d - 1 : ℕ) : ℝ) (fun x => χ (x - z) * h x)
208    · rw [if_pos hM] at h1
209      refine ⟨hM, ?_⟩
210      have h3 := (ENNReal.ofReal_le_iff_le_toReal hU'.ne).mp h1
211      exact ((Real.sqrt_le_iff.mp h3).2)
212    · rw [if_neg hM] at h2
213      exact absurd rfl h2
214  refine ⟨b, l, Uv, hb0, hl0, hU0, ?_, hbd, hlip, hSob⟩
215  exact hn
