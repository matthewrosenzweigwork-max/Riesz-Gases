26/-- Measurability of the linear coordinate. -/
27theorem p_cp8_linCoord_meas {N : ℕ} (k : Lattice d) :
28    Measurable (fun X : Config d N => linCoord (volume : Measure (Torus d)) (PDE.cosTest k) X) := by
29  have hψ : Measurable (PDE.cosTest k : Torus d → ℝ) := (continuous_const.mul (Torus.continuous_cosMode k)).measurable
30  unfold linCoord
31  refine measurable_const.mul ((measurable_const.mul (Finset.measurable_sum _ fun i _ => ?_)).sub
32    measurable_const)
33  exact hψ.comp (measurable_pi_apply i)

35/-- Time-marginals of a selected flow are measurable (`d ≥ 2`). -/
36theorem p_cp8_adapted (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {β : ℝ} (hβ : 0 < β)
37    {N : ℕ} {T : ℝ} {Ω : Type} [MeasurableSpace Ω] {P : Measure Ω}
38    {X₀ : Ω → Config d N} {W : ℝ≥0 → Ω → (Fin N → Euc d)} {X : ℝ≥0 → Ω → Config d N}
39    (hsel : SelectedFlow P (PDE.mfData d s hs0 hsd (fun _ => (0 : ℝ)) β hβ) T X₀ W X) (τ : ℝ≥0) :
40    Measurable (X τ) := by
41  rcases hsel.flow with ⟨hs, -⟩ | ⟨hd1, -⟩
42  · exact hs.adapted τ
43  · exact absurd hd1 (by omega)

55/-- P8 -/
56theorem p8_assembly (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) (β : ℝ) (hβ : 0 < β)
57    (k : Lattice d) (hk : k ≠ 0) :
58    (∀ (t : ℝ), 0 ≤ t →
59      ∀ (Ω : ℕ → Type) [∀ N, MeasurableSpace (Ω N)] (P : ∀ N, Measure (Ω N))
60        [∀ N, IsProbabilityMeasure (P N)]
61        (X₀ : ∀ N, Ω N → Config d N) (W : ∀ N, ℝ≥0 → Ω N → (Fin N → Euc d))
62        (X : ∀ N, ℝ≥0 → Ω N → Config d N),
63        (∀ N : ℕ, 2 ≤ N → SelectedFlow (P N) (PDE.mfData d s hs0 hsd (fun _ => 0) β hβ) t
64          (X₀ N) (W N) (X N)) →
65        (∀ N : ℕ, 2 ≤ N → ∃ (ℱ : MeasureTheory.Filtration ℝ (inferInstance : MeasurableSpace (Ω N)))
66          (W' : ℝ≥0 → Ω N → (Fin N × Fin d) → ℝ), IsFilteredSolution (P N) ℱ W' (W N) (X N)) →
67        (∀ N : ℕ, 2 ≤ N → law (P N) (X₀ N) =
68          ProbabilityTheory.cond (canonicalGibbs (periodicRiesz d s) (fun _ => (0 : ℝ)) N β)
69            {Y | 0 < linCoord (volume : Measure (Torus d)) (PDE.cosTest k) Y}) →
70        ConvergesInLaw
71          (fun N : ℕ => law (P N) (fun ω =>
72            linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X N (Real.toNNReal t) ω)))
73          (halfNormalMixtureLaw d s β k t)) := by
74  classical
75  intro t ht Ω _ P _ X₀ W X hsel hfil hν f
76  have hV : Torus.IsSmooth (fun _ : Torus d => (0 : ℝ)) := HRS.d1_isSmooth_const 0
77  have hex : ∀ n : ℕ, ∃ (Ω : Type) (_ : MeasurableSpace Ω) (P : Measure Ω) (_ : IsProbabilityMeasure P)
78      (X₀ : Ω → Config d (n + 2)) (W : ℝ≥0 → Ω → (Fin (n + 2) → Euc d)) (X : ℝ≥0 → Ω → Config d (n + 2)),
79      IsStationaryGibbsFlow P s hs0 hsd (fun _ => (0 : ℝ)) β hβ t X₀ W X :=
80    fun n => HRS.MD.EQLawFlow.eq_flow_exists_proof hd hs0 hsd _ hV hβ (by omega) t ht
81  choose Ω₂ inst₂ P₂ hP₂ X₀₂ W₂ X₂ h₂ using hex
82  have hmeas : ∀ n τ, Measurable (X₂ n τ) := fun n τ => p_cp8_adapted hd hs0 hsd hβ (h₂ n).flow τ
83  have hjoint := condou_stationary_joint hd hs0 hsd hβ k hk t ht Ω₂ P₂ X₀₂ W₂ X₂ h₂
84  have hpm : ∀ n, Measurable (fun ω => (WithLp.toLp 2 ![
85        linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X₂ n 0 ω),
86        linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X₂ n (Real.toNNReal t) ω)] :
87        EuclideanSpace ℝ (Fin 2))) := fun n =>
88    (WithLp.measurable_toLp 2 (Fin 2 → ℝ)).comp (measurable_pi_iff.2 (Fin.forall_fin_two.2
89      ⟨(p_cp8_linCoord_meas k).comp (hmeas n 0), (p_cp8_linCoord_meas k).comp (hmeas n _)⟩))
90  have hνp : ∀ n, IsProbabilityMeasure (law (P₂ n) (fun ω => (WithLp.toLp 2 ![
91        linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X₂ n 0 ω),
92        linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X₂ n (Real.toNNReal t) ω)] :
93        EuclideanSpace ℝ (Fin 2)))) := fun n => inferInstance
94  -- P6 with `a = x ↦ x 0`, `g = 2 f (x 1)`
95  have hxm : Continuous (fun x : EuclideanSpace ℝ (Fin 2) => x 1) := by fun_prop
96  have hx0 : Continuous (fun x : EuclideanSpace ℝ (Fin 2) => x 0) := by fun_prop
97  let g : EuclideanSpace ℝ (Fin 2) →ᵇ ℝ := (2 : ℝ) • f.compContinuous ⟨fun x => x 1, hxm⟩
98  have hg : ∀ x, g x = 2 * f (x 1) := fun x => by simp [g]
99  have hlim := condou_tendsto_indicator2 _ (multivariateGaussian 0 (condouCov d s β k t))
100    (fun x : EuclideanSpace ℝ (Fin 2) => x 0) hx0 (condou_gauss_pair_nullset hd hs0 hsd hβ k t ht) hjoint g
101  have hg' : (fun x : EuclideanSpace ℝ (Fin 2) => Set.indicator {x : EuclideanSpace ℝ (Fin 2) | 0 < x 0} (fun x => g x) x) =
102      fun x => Set.indicator {x : EuclideanSpace ℝ (Fin 2) | 0 < x 0} (fun x => 2 * f (x 1)) x := by
103    funext x; simp only [hg]
104  simp only [hg'] at hlim
105  rw [condou_gauss_pair_integral hd hs0 hsd hβ k t ht f] at hlim
106  -- the sequence in question
107  have hseq : ∀ n : ℕ, ∫ x, f x ∂(law (P (n + 2)) (fun ω =>
108      linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X (n + 2) (Real.toNNReal t) ω))) =
109      ∫ x, Set.indicator {x : EuclideanSpace ℝ (Fin 2) | 0 < x 0} (fun x => 2 * f (x 1)) x
110        ∂(law (P₂ n) (fun ω => (WithLp.toLp 2 ![
111        linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X₂ n 0 ω),
112        linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X₂ n (Real.toNNReal t) ω)] :
113        EuclideanSpace ℝ (Fin 2)))) := by
114    intro n
115    have hN : 2 ≤ n + 2 := by omega
116    have hXm : Measurable (fun ω => linCoord (volume : Measure (Torus d)) (PDE.cosTest k)
117        (X (n + 2) (Real.toNNReal t) ω)) :=
118      (p_cp8_linCoord_meas k).comp (p_cp8_adapted hd hs0 hsd hβ (hsel (n + 2) hN) _)
119    have hprob := HRS.MH.CondOuP.conditioned_ou_prob_proof hd hs0 hsd β hβ k hk (n + 2) (by omega)
120    have hA : MeasurableSet {Y : Config d (n + 2) |
121        0 < linCoord (volume : Measure (Torus d)) (PDE.cosTest k) Y} :=
122      measurableSet_lt measurable_const (p_cp8_linCoord_meas k)
123    have hP1 := condou_weighted_integral hd hs0 hsd hβ hN t ht (P (n + 2)) (X₀ (n + 2)) (W (n + 2))
124      (X (n + 2)) (hsel _ hN) (hfil _ hN) (P₂ n) (X₀₂ n) (W₂ n) (X₂ n) (h₂ n)
125      {Y | 0 < linCoord (volume : Measure (Torus d)) (PDE.cosTest k) Y} hA hprob (hν _ hN)
126      (fun Y => linCoord (volume : Measure (Torus d)) (PDE.cosTest k) Y) (p_cp8_linCoord_meas k)
127      (Real.toNNReal t) (by rw [Real.coe_toNNReal _ ht]) f
128    have hind : Measurable (fun x : EuclideanSpace ℝ (Fin 2) =>
129        Set.indicator {x : EuclideanSpace ℝ (Fin 2) | 0 < x 0} (fun x => 2 * f (x 1)) x) :=
130      (measurable_const.mul (f.continuous.measurable.comp hxm.measurable)).indicator
131        (measurableSet_lt measurable_const hx0.measurable)
132    rw [law, integral_map hXm.aemeasurable f.continuous.aestronglyMeasurable, hP1, law,
133      integral_map (hpm n).aemeasurable hind.aestronglyMeasurable]
134    refine integral_congr_ae ?_
135    filter_upwards [p_cp8_init hd hs0 hsd hβ (h₂ n).flow] with ω hω
136    rw [← hω]
137    by_cases h : 0 < linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X₂ n 0 ω)
138    · simp [Set.indicator, h]
139    · simp [Set.indicator, h]
140  have hfin : Tendsto (fun N : ℕ => ∫ x, f x ∂(law (P N) (fun ω =>
141      linCoord (volume : Measure (Torus d)) (PDE.cosTest k) (X N (Real.toNNReal t) ω)))) atTop
142      (𝓝 (∫ y, f y ∂(halfNormalMixtureLaw d s β k t))) := by
143    rw [← tendsto_add_atTop_iff_nat 2]
144    simp only [hseq]
145    exact hlim
146  exact hfin
