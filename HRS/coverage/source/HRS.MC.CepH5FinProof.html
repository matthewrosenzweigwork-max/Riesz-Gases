50/-- A finite sum of terms `≠ ⊥` is `≠ ⊥` (glue; proved). -/
51theorem h5_sum_ne_bot {ι : Type*} (s : Finset ι) (f : ι → EReal) (h : ∀ i, f i ≠ ⊥) : ∑ i ∈ s, f i ≠ ⊥ := by
52  classical

55  | insert a s ha ih =>

59/-- `gibbsEntropy` is never `⊥` (proved): it is `↑(relEntropy) - ↑(log Z)` with `relEntropy ∈ [0,∞]` and `log Z` real. -/

61    gibbsEntropy D μ ≠ ⊥ := by

63  generalize relEntropy μ (gibbsRef D N) = x

68      rw [← EReal.coe_ennreal_toReal hx]
69    rw [this, ← EReal.coe_sub]
70    exact EReal.coe_ne_bot _

73/-- **B1** (finiteness of the components).  In a mixture `∑ p_i a_i + c` of `EReal` numbers `a_i ≠ ⊥` with weights
74`p_i ≥ 0` and a real `c`, finiteness of the total forces `a_i < ⊤` for every `i` with `p_i ≠ 0` (a term `p_i * ⊤ = ⊤`
75absorbs the finitely many other terms, none of which is `⊥`). -/
76theorem h5_components_lt_top {ι : Type*} [Fintype ι] (p : ι → ℝ) (hp : ∀ i, 0 ≤ p i) (a : ι → EReal)
77    (ha : ∀ i, a i ≠ ⊥) (c : EReal) (hc1 : c ≠ ⊥) (hc2 : c ≠ ⊤)
78    (h : ∑ i, ((p i : ℝ) : EReal) * a i + c < ⊤) : ∀ i, p i ≠ 0 → a i < ⊤ := by
79  classical
80  intro i0 hi0
81  by_contra hcon
82  have hat : a i0 = ⊤ := by
83    rcases (not_lt.1 hcon).eq_or_lt with h' | h'
84    · exact (top_le_iff.1 (not_lt.1 hcon))
85    · exact absurd h' (not_lt.2 le_top)
86  have hpos : 0 < p i0 := lt_of_le_of_ne (hp i0) (Ne.symm hi0)
87  have hterm : ((p i0 : ℝ) : EReal) * a i0 = ⊤ := by
88    rw [hat]; exact EReal.coe_mul_top_of_pos hpos
89  have hne : ∀ i, ((p i : ℝ) : EReal) * a i ≠ ⊥ := by
90    intro i
91    by_cases hz : p i = 0
92    · simp [hz]
93    · exact (EReal.mul_ne_bot _ _).2 ⟨Or.inl (EReal.coe_ne_bot _), Or.inr (ha i),
94        Or.inl (EReal.coe_ne_top _), Or.inl (EReal.coe_nonneg.2 (hp i))⟩
95  rw [← Finset.add_sum_erase Finset.univ _ (Finset.mem_univ i0), hterm,
96    EReal.top_add_of_ne_bot (h5_sum_ne_bot _ _ hne), EReal.top_add_of_ne_bot hc1] at h
97  exact lt_irrefl _ h

100/-- **B2** (recombination of the entropy inequalities).  If for the weights `p_i ≥ 0` and each `i` with `p_i ≠ 0`
101`Ht_i + F_i ≤ H0_i < ⊤` with `Ht_i, H0_i ≠ ⊥`, then for any real `c`
102`∑ p_i Ht_i + c + ↑(∑ ofReal p_i * F_i) ≤ ∑ p_i H0_i + c`.  Proof: for `p_i ≠ 0` the three numbers `Ht_i, F_i, H0_i`
103are finite (`Ht_i ≠ ⊥`, `Ht_i + F_i ≤ H0_i < ⊤`), so everything is real arithmetic; terms with `p_i = 0` vanish. -/
104theorem h5_recombine {ι : Type*} [Fintype ι] (p : ι → ℝ) (hp : ∀ i, 0 ≤ p i) (Ht H0 : ι → EReal)
105    (F : ι → ℝ≥0∞) (hHt : ∀ i, Ht i ≠ ⊥) (hH0 : ∀ i, H0 i ≠ ⊥)
106    (h : ∀ i, p i ≠ 0 → H0 i < ⊤ ∧ Ht i + (F i : EReal) ≤ H0 i) (c : EReal) (hc1 : c ≠ ⊥) (hc2 : c ≠ ⊤) :
107    ∑ i, ((p i : ℝ) : EReal) * Ht i + c + ((∑ i, ENNReal.ofReal (p i) * F i : ℝ≥0∞) : EReal)
108      ≤ ∑ i, ((p i : ℝ) : EReal) * H0 i + c := by
109  classical
110  obtain ⟨c', rfl⟩ : ∃ c' : ℝ, c = (c' : EReal) := ⟨c.toReal, (EReal.coe_toReal hc2 hc1).symm⟩
111  -- real shadows
112  have key : ∀ i, ((p i : ℝ) : EReal) * Ht i = (((p i * (Ht i).toReal : ℝ)) : EReal) := by
113    intro i
114    by_cases hz : p i = 0
115    · simp [hz]
116    · obtain ⟨h1, h2⟩ := h i hz
117      have hHtt : Ht i ≠ ⊤ := by
118        intro ht
119        rw [ht] at h2
120        have : (⊤ : EReal) + (F i : EReal) = ⊤ := EReal.top_add_of_ne_bot (by simp)
121        rw [this] at h2
122        exact absurd (lt_of_le_of_lt h2 h1) (lt_irrefl _)
123      rw [EReal.coe_mul, EReal.coe_toReal hHtt (hHt i)]
124  have key0 : ∀ i, ((p i : ℝ) : EReal) * H0 i = (((p i * (H0 i).toReal : ℝ)) : EReal) := by
125    intro i
126    by_cases hz : p i = 0
127    · simp [hz]
128    · obtain ⟨h1, h2⟩ := h i hz
129      rw [EReal.coe_mul, EReal.coe_toReal h1.ne (hH0 i)]
130  have keyF : ∀ i, ENNReal.ofReal (p i) * F i = ENNReal.ofReal (p i * (F i).toReal) := by
131    intro i
132    by_cases hz : p i = 0
133    · simp [hz]
134    · obtain ⟨h1, h2⟩ := h i hz
135      have hFt : F i ≠ ⊤ := by
136        intro ft
137        have hh : Ht i + (F i : EReal) = ⊤ := by
138          rw [ft]; simp; exact EReal.add_top_of_ne_bot (hHt i)
139        rw [hh] at h2
140        exact absurd (lt_of_le_of_lt h2 h1) (lt_irrefl _)
141      rw [ENNReal.ofReal_mul (hp i), ENNReal.ofReal_toReal hFt]
142  have hineq : ∑ i, p i * (Ht i).toReal + c' + ∑ i, p i * (F i).toReal ≤ ∑ i, p i * (H0 i).toReal + c' := by
143    have : ∑ i, p i * (Ht i).toReal + ∑ i, p i * (F i).toReal ≤ ∑ i, p i * (H0 i).toReal := by
144      rw [← Finset.sum_add_distrib]
145      refine Finset.sum_le_sum fun i _ => ?_
146      by_cases hz : p i = 0
147      · simp [hz]
148      · obtain ⟨h1, h2⟩ := h i hz
149        have hHtt : Ht i ≠ ⊤ := by
150          intro ht
151          rw [ht] at h2
152          have : (⊤ : EReal) + (F i : EReal) = ⊤ := EReal.top_add_of_ne_bot (by simp)
153          rw [this] at h2
154          exact absurd (lt_of_le_of_lt h2 h1) (lt_irrefl _)
155        have hFt : F i ≠ ⊤ := by
156          intro ft
157          have hh : Ht i + (F i : EReal) = ⊤ := by
158            rw [ft]; simp; exact EReal.add_top_of_ne_bot (hHt i)
159          rw [hh] at h2
160          exact absurd (lt_of_le_of_lt h2 h1) (lt_irrefl _)
161        rw [← mul_add]
162        refine mul_le_mul_of_nonneg_left ?_ (hp i)
163        have e1 : (Ht i) = ((Ht i).toReal : EReal) := (EReal.coe_toReal hHtt (hHt i)).symm
164        have e2 : (F i : EReal) = (((F i).toReal : ℝ) : EReal) := by
165          rw [← EReal.coe_ennreal_toReal hFt]
166        have e0 : (H0 i) = ((H0 i).toReal : EReal) := (EReal.coe_toReal h1.ne (hH0 i)).symm
167        rw [e1, e2, e0, ← EReal.coe_add] at h2
168        exact EReal.coe_le_coe_iff.1 h2
169    linarith
170  simp only [key, key0, keyF]
171  rw [h5_coe_sum, h5_coe_sum, ← ENNReal.ofReal_sum_of_nonneg
172    (fun i _ => mul_nonneg (hp i) ENNReal.toReal_nonneg), EReal.coe_ennreal_ofReal,
173    ← EReal.coe_add, ← EReal.coe_add, ← EReal.coe_add]
174  refine EReal.coe_le_coe_iff.2 ?_
175  have := le_max_left (∑ i, p i * (F i).toReal) 0
176  rw [max_eq_left (Finset.sum_nonneg fun i _ => mul_nonneg (hp i) ENNReal.toReal_nonneg)]
177  exact hineq

188/-- The conditional selected law of the class `σ` at time `t` (with a harmless probability measure supported on
189`C σ` for classes of `P`-measure `0`, whose weight is `0` in every mixture). -/

316        ∃ y : Fin N → Euc 1, (fun i => Torus.mk (y i)) = X₀ ω ∧ cep_StrictlyOrdered σ.1 y := by
