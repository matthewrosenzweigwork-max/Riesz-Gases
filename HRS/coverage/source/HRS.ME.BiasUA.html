19/-- P1: uniform lower bound and `C^k` bound of the thermal densities (constants chosen before `β`, `ρβ`). -/
20theorem thermal_lower_cbNorm_unif (hd : 2 ≤ d) (hs0 : 0 ≤ (d : ℝ) / 2) (hsd : (d : ℝ) / 2 < d)
21    (V : Torus d → ℝ) (hV : Torus.IsSmooth V) (βm βp : ℝ) (hβm : 0 < βm) (hββ : βm ≤ βp) (k : ℕ) :
22    ∃ c₀ Λ : ℝ, 0 < c₀ ∧ 0 ≤ Λ ∧ ∀ β ∈ Set.Icc βm βp, ∀ ρβ : Torus d → ℝ,
23      IsThermalDensity (periodicRiesz d ((d : ℝ) / 2)) V β ρβ →
24      (∀ x, c₀ ≤ ρβ x ∧ ρβ x ≤ Λ) ∧ Torus.cbNorm k ρβ ≤ Λ := by
25  have hd1 : 1 ≤ d := by omega
26  have hdr : (0 : ℝ) ≤ d := Nat.cast_nonneg d
27  set K : ℕ := 6 * d + 30 + k with hKdef
28  obtain ⟨Λ₀, c₀', lam, Lam, B, -, hlam, hB0, hmain⟩ :=
29    HRS.MD.EQEvoUnif.eqevo_thermal_unif hd ((d : ℝ) / 2) βm βp hs0 hsd hβm hββ V hV 0 K
30      (by omega)
31  have hKr : (d : ℝ) / 2 + (k : ℝ) < (K : ℝ) := by
32    have : (K : ℝ) = 6 * d + 30 + k := by simp [hKdef]
33    rw [this]; linarith
34  obtain ⟨C, hC⟩ := cbNorm_le_sobolevNorm (d := d) k (K : ℝ) hKr
35  have hVc : Continuous V :=
36    HRS.PDE.continuous_of_isCk (by exact_mod_cast HRS.MD.EQEvoUnif.eqevo_isCk_nat hV 1)
37  refine ⟨lam, max Lam (max C 0 * Real.sqrt B), hlam, ?_, fun β hβ ρβ hρβ => ?_⟩
38  · exact le_max_of_le_right (mul_nonneg (le_max_right _ _) (Real.sqrt_nonneg _))
39  · have hβ0 : 0 < β := lt_of_lt_of_le hβm hβ.1
40    obtain ⟨hpos, hc⟩ := HRS.Q41.q41_thermal_pos_cont hd hs0 hsd hβ0 hVc hρβ
41    have hSob : MemSobolev (K : ℝ) ρβ :=
42      HRS.PDE.MFThermalAssembly.thermal_memSobolev_all hd1 ((d : ℝ) / 2) hs0 hsd K V
43        (by exact_mod_cast HRS.MD.EQEvoUnif.eqevo_isCk_nat hV (K + 2)) β hβ0 ρβ hρβ hc hpos
44    obtain ⟨hlamB, hLamB⟩ := (hmain β hβ ρβ hρβ).2
45    have hB' : hsNormSq (K : ℝ) (coeffOf ρβ) ≤ B := hLamB
46    have hEq := (hsNormSq_coeffOf (K : ℝ) ρβ hSob).2
47    have hsn : sobolevNorm (K : ℝ) ρβ ≤ Real.sqrt B := by
48      have h2 : sobolevNorm (K : ℝ) ρβ ^ 2 ≤ B := hEq ▸ hB'
49      calc sobolevNorm (K : ℝ) ρβ = Real.sqrt (sobolevNorm (K : ℝ) ρβ ^ 2) :=
50            (Real.sqrt_sq (Real.sqrt_nonneg _)).symm
51        _ ≤ Real.sqrt B := Real.sqrt_le_sqrt h2
52    have hsm : Torus.IsSmooth ρβ := (smoothPos_of_thermal hd hs0 hsd hβ0 hV hρβ).smooth
53    have hCk : Torus.IsCk (k : WithTop ENat) ρβ := HRS.MD.EQEvoUnif.eqevo_isCk_nat hsm k
54    have hcb := hC ρβ hSob (by exact_mod_cast hCk)
55    refine ⟨fun x => ⟨?_, ?_⟩, ?_⟩
56    · exact hlamB x |>.1
57    · exact (hlamB x).2.trans (le_max_left _ _)
58    · refine hcb.trans ((mul_le_mul_of_nonneg_right (le_max_left C 0) (by
59        unfold sobolevNorm; exact Real.sqrt_nonneg _)).trans ?_) |>.trans (le_max_right _ _)
60      exact mul_le_mul_of_nonneg_left hsn (le_max_right _ _)
