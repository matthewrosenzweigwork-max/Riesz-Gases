51theorem wr8_fam_pos (hd : 1 ≤ d) {s : ℝ} (hs : 0 < s) (hsd : s < d) (L ε : ℝ) (hε : 0 < ε) :
52    ∃ (q P : Euc d → ℝ) (Kw κ m : ℝ), Wr8Split d s L 1 ε q P Kw κ m := by
53  have hα : 0 < ((d : ℝ) - s) / 2 := by linarith

59  have hτα : τ ^ (((d : ℝ) - s) / 2) = ε / (c + 1) := by
60    rw [hτdef, ← Real.rpow_mul (by positivity), one_div_mul_cancel hα.ne', Real.rpow_one]

62  have hqε : ∫ z, qTau d s τ z ≤ ε := by
63    rw [hqI, ← hcdef, hτα]

65      _ ≤ ε * 1 := mul_le_mul_of_nonneg_left (by rw [div_le_one (by positivity)]; linarith) hε.le
66      _ = ε := mul_one _
67  have hk0 : 0 ≤ kTau d s τ 0 := kTau_nonneg hs hτ 0
68  have hkm : Measurable (fun p : Euc d × Euc d => kTau d s τ (p.1 - p.2)) :=
69    (kTau_continuous hs hτ).measurable.comp (measurable_fst.sub measurable_snd)
70  have hW : ∀ z : Euc d, rieszWeight s z = rieszKernel d s z := by
71    intro z; unfold rieszWeight rieszKernel; rw [if_neg hs.ne', if_neg hs.ne']
72  refine ⟨qTau d s τ, kTau d s τ, kTau d s τ 0, kTau d s τ 0, 0, ?_⟩
73  refine ⟨wr8_qTau_meas, wr8_qTau_even, qTau_nonneg hs, hqi, hqε, ?_, ?_, ?_, hk0, ?_, ?_, ?_⟩
74  · intro z hz; exact riesz_heat_split hs hτ hz
75  · intro z hz
76    have h1 := riesz_heat_split hs hτ hz
77    have h2 := kTau_nonneg hs hτ z
78    rw [hW]; linarith
79  · intro z hz

81    have h2 := kTau_le_zero hs hτ z
82    rw [hW]; linarith
83  · intro μ _ _
84    refine ⟨?_, fun x => ?_, ?_⟩
85    · exact hkm
86    · exact Integrable.of_bound
87        ((kTau_continuous hs hτ).measurable.comp (measurable_const.sub measurable_id)).aestronglyMeasurable
88        (kTau d s τ 0) (wr8_kTau_bound hs hτ x)
89    · exact Integrable.of_bound hkm.aestronglyMeasurable (kTau d s τ 0)

94    exact integral_nonneg fun p => kTau_nonneg hs hτ _

100/-- The case `s > 0` of `HRS.Whole.euclidean_pressure` (the `s = 0` hypothesis is vacuous). -/
101theorem euclidean_pressure_pos (hd : 2 ≤ d) {s : ℝ} (hs : 0 < s) (hsd : s < d)
102    (M L bm bp B : ℝ) (hbm : 0 < bm) :
103    ∃ θ₀ > 0, ∃ C : ℝ, ∀ ρ : Euc d → ℝ, HRS.Whole.IsProbDensity ρ → (∀ x, ρ x ≤ M) →
104      (s = 0 → HasLogMoment (HRS.Whole.withDensityReal ρ) L) →
105      ∀ h : Euc d → Euc d → ℝ, IsBoundedPairKernel s B h →
106      ∀ N, 2 ≤ N → ∀ b ∈ Set.Icc bm bp, ∀ θ : ℝ, |θ| ≤ θ₀ →
107        GibbsWellDefined (fun x y => rieszKernel d s (x - y) - θ * h x y) N b
108          (HRS.Whole.withDensityReal ρ) ∧
109        |Real.log (pairNormalizer (fun x y => rieszKernel d s (x - y) - θ * h x y) N b
110          (HRS.Whole.withDensityReal ρ))| ≤ C * N * deltaN d s N :=
111  wr8_core hd hs.le hsd M L bm bp B 1 hbm zero_le_one
112    (fun ε hε => wr8_fam_pos (by omega) hs hsd L ε hε)

167theorem wr8_kLog_bound {τ : ℝ} (hτ : 0 < τ) {μ : Measure (Euc d)} [IsProbabilityMeasure μ]
168    (hlog : Integrable (fun x : Euc d => Real.log (2 + ‖x‖)) μ) :
169    -(1 / (2 * τ) + τ / 2 + 2 * ∫ x, Real.log (2 + ‖x‖) ∂μ) ≤
170      ∫ p : Euc d × Euc d, kTauLog d τ (p.1 - p.2) ∂(μ.prod μ) := by

172  have hk := wr7_kTauLog_int hτ hlog hlog
173  have h1 : Integrable (fun z : Euc d × Euc d => M z.1 * 1) (μ.prod μ) :=
174    hlog.mul_prod (integrable_const (1 : ℝ) : Integrable (fun _ : Euc d => (1 : ℝ)) μ)
175  have h2 : Integrable (fun z : Euc d × Euc d => 1 * M z.2) (μ.prod μ) :=
176    (integrable_const (1 : ℝ) : Integrable (fun _ : Euc d => (1 : ℝ)) μ).mul_prod hlog
177  have hA : Integrable (fun z : Euc d × Euc d => (1 / (2 * τ) + τ / 2 + M z.1 * 1) + 1 * M z.2) (μ.prod μ) :=
178    ((integrable_const _).add h1).add h2
179  have hmono : ∫ z : Euc d × Euc d, -((1 / (2 * τ) + τ / 2 + M z.1 * 1) + 1 * M z.2) ∂(μ.prod μ) ≤
180      ∫ p : Euc d × Euc d, kTauLog d τ (p.1 - p.2) ∂(μ.prod μ) := by
181    refine integral_mono hA.neg hk (fun z => ?_)
182    have h1 := kTauLog_abs_le hτ (z.1 - z.2)
183    have h2 := psd3_log_sub_le z.1 z.2
184    have := neg_abs_le (kTauLog d τ (z.1 - z.2))
185    simp only [hM] at *
186    linarith
187  refine le_trans (le_of_eq ?_) hmono
188  have e1 : ∫ z : Euc d × Euc d, ((1 / (2 * τ) + τ / 2 + M z.1 * 1) + 1 * M z.2) ∂(μ.prod μ) =
189      (∫ z : Euc d × Euc d, (1 / (2 * τ) + τ / 2 + M z.1 * 1) ∂(μ.prod μ)) +
190        ∫ z : Euc d × Euc d, 1 * M z.2 ∂(μ.prod μ) :=
191    integral_add ((integrable_const _).add h1) h2
192  have e2 : ∫ z : Euc d × Euc d, (1 / (2 * τ) + τ / 2 + M z.1 * 1) ∂(μ.prod μ) =
193      (∫ _z : Euc d × Euc d, (1 / (2 * τ) + τ / 2) ∂(μ.prod μ)) +
194        ∫ z : Euc d × Euc d, M z.1 * 1 ∂(μ.prod μ) :=

197  have e4 := integral_prod_mul (μ := μ) (ν := μ) (fun _ => (1 : ℝ)) M
198  rw [integral_neg, e1, e2, e3, e4]
199  simp [Measure.real]
200  ring

202theorem wr8_fam_zero (hd : 1 ≤ d) (L ε : ℝ) (hε : 0 < ε) :
203    ∃ (q P : Euc d → ℝ) (Kw κ m : ℝ), Wr8Split d 0 L 1 ε q P Kw κ m := by
204  have hd0 : (0 : ℝ) < d := by exact_mod_cast hd
205  obtain ⟨c, hcdef⟩ : ∃ c : ℝ, c = Real.pi ^ ((d : ℝ) / 2) / d := ⟨_, rfl⟩
206  have hc0 : 0 ≤ c := by rw [hcdef]; positivity
207  obtain ⟨τ0, hτ0def⟩ : ∃ τ0 : ℝ, τ0 = (ε / (c + 1)) ^ (2 / (d : ℝ)) := ⟨_, rfl⟩
208  have hτ0 : 0 < τ0 := by rw [hτ0def]; exact Real.rpow_pos_of_pos (by positivity) _
209  have hτ0α : τ0 ^ ((d : ℝ) / 2) = ε / (c + 1) := by
210    rw [hτ0def, ← Real.rpow_mul (by positivity)]
211    rw [show 2 / (d : ℝ) * ((d : ℝ) / 2) = 1 by field_simp, Real.rpow_one]
212  obtain ⟨τ, hτdef⟩ : ∃ τ : ℝ, τ = min 1 τ0 := ⟨_, rfl⟩
213  have hτ : 0 < τ := by rw [hτdef]; exact lt_min one_pos hτ0
214  have hτ1 : τ ≤ 1 := by rw [hτdef]; exact min_le_left _ _
215  have hτα : τ ^ ((d : ℝ) / 2) ≤ ε / (c + 1) := by
216    rw [← hτ0α]
217    exact Real.rpow_le_rpow hτ.le (by rw [hτdef]; exact min_le_right _ _) (by positivity)
218  obtain ⟨hqi, hqI⟩ := qTauLog_integrable_integral hd hτ
219  have hqε : ∫ z, qTauLog d τ z ≤ ε := by
220    rw [hqI]
221    have e : Real.pi ^ ((d : ℝ) / 2) * τ ^ ((d : ℝ) / 2) / d = c * τ ^ ((d : ℝ) / 2) := by
222      rw [hcdef]; ring
223    rw [e]
224    calc c * τ ^ ((d : ℝ) / 2) ≤ c * (ε / (c + 1)) := mul_le_mul_of_nonneg_left hτα hc0
225      _ = ε * (c / (c + 1)) := by ring
226      _ ≤ ε * 1 := mul_le_mul_of_nonneg_left (by rw [div_le_one (by positivity)]; linarith) hε.le
227      _ = ε := mul_one _
228  have hKw : 0 ≤ 1 / (2 * τ) + τ / 2 := by positivity
229  have hW0 : ∀ z : Euc d, rieszWeight 0 z = max (-Real.log ‖z‖) 0 := by
230    intro z; unfold rieszWeight; rw [if_pos rfl]
231  refine ⟨qTauLog d τ, kTauLog d τ, 1 / (2 * τ) + τ / 2, kTauLog d τ 0,
232    1 / (2 * τ) + τ / 2 + 2 * L, ?_⟩
233  refine ⟨wr8_qLog_meas, wr8_qLog_even, qTauLog_nonneg hτ, hqi, hqε, ?_, ?_, ?_, hKw, ?_, ?_, ?_⟩
234  · intro z hz; exact riesz_log_heat_split hτ hz
235  · intro z hz; exact wr8_qLog_env hτ hτ1 hz
236  · intro z hz
237    have hq0 := qTauLog_nonneg (d := d) hτ z
238    rcases le_total ‖z‖ 1 with h | h
239    · have hk := riesz_log_heat_split (d := d) hτ hz
240      have hku := kTauLog_abs_le_unit hτ h
241      have hn : 0 < ‖z‖ := norm_pos_iff.2 hz
242      have hlog : Real.log ‖z‖ ≤ 0 := Real.log_nonpos hn.le h
243      have hr : rieszKernel d 0 z = -Real.log ‖z‖ := by simp [rieszKernel]
244      rw [hW0, max_eq_left (by linarith)]
245      rw [abs_le] at hku
246      linarith [hku.1, hku.2]
247    · have hn : 0 < ‖z‖ := by linarith
248      have hlog : 0 ≤ Real.log ‖z‖ := Real.log_nonneg h
249      rw [hW0, max_eq_right (by linarith)]
250      linarith
251  · intro μ _ hlog
252    have hl := (hlog rfl).1
253    refine ⟨(kTauLog_continuous hτ).measurable.comp (measurable_fst.sub measurable_snd), fun x => ?_,
254      kTauLog_pair_integrable hτ μ hl⟩
255    exact kTauLog_rows_integrable hτ μ hl x
256  · intro μ _ hlog
257    have hl := hlog rfl
258    have := wr8_kLog_bound hτ hl.1
259    have h2 := hl.2
260    linarith
261  · intro μ _ hlog N hN X
262    have := kTauLog_diag_bound hτ hN μ (hlog rfl).1 X
263    linarith

266/-- **Proposition S3.1, main bound** (`HRS.Whole.euclidean_pressure`): proof by the heat split
267`g = q_τ + k_τ`, the perturbation `q_θ = q_τ - θ h̃ + |θ| D`, the Euclidean weak reference
268(`weak_reference_pressure_pairNormalizer_E`), the diagonal bound (`kTau_diag_bound`, `kTauLog_diag_bound`) and the
269Jensen lower bound.  Constants `θ₀, C` are chosen before `ρ, h`. -/
270theorem euclidean_pressure_proof : type_of% @HRS.Whole.euclidean_pressure := by
271  intro d s hd hs0 hsd M L bm bp B hbm hb
272  rcases hs0.eq_or_lt with h0 | hpos
273  · subst h0
274    exact wr8_core hd le_rfl hsd M L bm bp B 1 hbm zero_le_one
275      (fun ε hε => wr8_fam_zero (by omega) L ε hε)
276  · exact euclidean_pressure_pos hd hpos hsd M L bm bp B hbm
