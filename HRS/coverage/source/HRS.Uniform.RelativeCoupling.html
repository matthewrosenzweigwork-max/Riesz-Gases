31theorem relative_covariance_coupling_proof {Ω : Type u} {mΩ : MeasurableSpace Ω} (P : Measure Ω)

42  haveI := hκ

58        ((ℱ s).comap π).comap (Prod.fst : Ω' × (ιe → WienerSeq) → Ω') := by
59      rw [MeasurableSpace.comap_comp]; rfl
60    rw [h1]
61    exact (MeasurableSpace.comap_mono (hfilt s)).trans (le_sup_of_le_left le_sup_left)
62  have hπ''m : ∀ s, Measurable[prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe) s, ℱ s]
63      (fun p : Ω' × (ιe → WienerSeq) => π p.1) := fun s => Measurable.of_comap_le (hfilt'' s)
64  have hcomplete : IsCompleteFiltration (P'.prod (wMeasure ιe))
65      (prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe)) := HRS.MRep.prodFiltration_complete P' ℱ'
66  have hlift := hMimm.comp_fst (ι := ιe)
67  -- the difference density
68  have hZtc : ∀ r p, IsTraceClassCov (hsCovForm (Z r p)) := fun r p =>
69    isTraceClassCov_hsCovForm ((hZhs 0).summable r p)
70  -- joint measurability and adaptedness of the ingredients of `Sd`
71  have hCm : ∀ x y : H, Measurable fun a : ℝ × (Ω' × (ιe → WienerSeq)) =>
72      hsCovForm (Φ a.1 a.2.1) x y := fun x y => by

74      HRS.MRep.isProgressive_measurable_uncurry (isProgressive_hsCov hΦ x y)
75    have hf : Measurable fun a : ℝ × (Ω' × (ιe → WienerSeq)) => (a.1, a.2.1) :=
76      measurable_fst.prodMk (measurable_fst.comp measurable_snd)
77    exact Measurable.comp (g := fun p : ℝ × Ω' => hsCovForm (Φ p.1 p.2) x y)
78      (f := fun a : ℝ × (Ω' × (ιe → WienerSeq)) => (a.1, a.2.1)) h hf

80      hsCovForm (Z a.1 a.2) x y := fun x y =>
81    HRS.MRep.isProgressive_measurable_uncurry (isProgressive_hsCov hZhs x y)

83      Sg a.1 (π a.2.1) x y := fun x y => by
84    have h : Measurable (Function.uncurry fun σ ω => Sg σ ω x y) := hM.sigma_meas x y
85    have hf : Measurable fun a : ℝ × (Ω' × (ιe → WienerSeq)) => (a.1, π a.2.1) :=
86      measurable_fst.prodMk (hπ.measurable.comp (measurable_fst.comp measurable_snd))
87    exact Measurable.comp (g := Function.uncurry fun σ ω => Sg σ ω x y)
88      (f := fun a : ℝ × (Ω' × (ιe → WienerSeq)) => (a.1, π a.2.1)) h hf
89  have hDm : ∀ x y : H, Measurable fun a : ℝ × (Ω' × (ιe → WienerSeq)) => Db a.1 x y := fun x y =>
90    (hDb.meas x y).comp measurable_fst

93      Sd r p = SdDef e0 (fun a : ℝ × (Ω' × (ιe → WienerSeq)) => Db a.1)
94        (fun a => hsCovForm (Φ a.1 a.2.1)) (fun a => hsCovForm (Z a.1 a.2))
95        (fun a => Sg a.1 (π a.2.1)) (r, p) := ⟨_, fun _ _ => rfl⟩
96  have hSdm : ∀ x y : H, Measurable (Function.uncurry fun r p => Sd r p x y) := fun x y => by
97    have h := measurable_SdDef e0 hCm hZm hSm hDm x y
98    have e : (Function.uncurry fun r p => Sd r p x y) = fun a : ℝ × (Ω' × (ιe → WienerSeq)) =>
99        SdDef e0 (fun a : ℝ × (Ω' × (ιe → WienerSeq)) => Db a.1)
100          (fun a => hsCovForm (Φ a.1 a.2.1)) (fun a => hsCovForm (Z a.1 a.2))
101          (fun a => Sg a.1 (π a.2.1)) a x y := by
102      funext ⟨r, p⟩; simp only [Function.uncurry_apply_pair, hSdDef]
103    rw [e]; exact h
104  have hSda : ∀ σ (x y : H), Measurable[prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe) σ]
105      (fun p => Sd σ p x y) := by
106    intro σ x y
107    have h1 : ∀ x y : H, Measurable[prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe) σ]
108        fun p : Ω' × (ιe → WienerSeq) => hsCovForm (Φ σ p.1) x y := fun x y =>
109      (adapted_hsCov hΦ σ x y).comp (hfst σ)
110    have h2 : ∀ x y : H, Measurable[prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe) σ]
111        fun p : Ω' × (ιe → WienerSeq) => hsCovForm (Z σ p) x y := fun x y =>
112      adapted_hsCov hZhs σ x y
113    have h3 : ∀ x y : H, Measurable[prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe) σ]
114        fun p : Ω' × (ιe → WienerSeq) => Sg σ (π p.1) x y := fun x y =>
115      (hM.sigma_adapted σ x y).comp (hπ''m σ)
116    have := measurable_SdDef (mα := prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe) σ) e0
117      (D := fun _ => Db σ) (C := fun p : Ω' × (ιe → WienerSeq) => hsCovForm (Φ σ p.1))
118      (Z := fun p => hsCovForm (Z σ p)) (S := fun p => Sg σ (π p.1)) h1 h2 h3
119      (fun x y => measurable_const) x y
120    have e : (fun p : Ω' × (ιe → WienerSeq) => Sd σ p x y) = fun p =>
121        SdDef e0 (fun _ => Db σ) (fun p : Ω' × (ιe → WienerSeq) => hsCovForm (Φ σ p.1))
122          (fun p => hsCovForm (Z σ p)) (fun p => Sg σ (π p.1)) p x y := by
123      funext p; rw [hSdDef]; rfl
124    rw [e]; exact this
125  have hSdtc : ∀ σ p, IsTraceClassCov (Sd σ p) := fun σ p => by
126    rw [hSdDef]; exact SdDef_tc e0 (fun a => hZtc _ _) _
127  have hgoodae : ∀ᵐ p ∂(P'.prod (wMeasure ιe)), ∀ᵐ σ ∂(volume.restrict (Set.Icc (0 : ℝ) T)),
128      hsCovForm (Φ σ p.1) = Sg σ (π p.1) :=

135      · have hz := isHM_ae_zero_nonpos hMimm (not_le.1 h0).le
136        exact hz.mono fun ω h => by
137          have h1 : M τ' (π ω) = 0 := h
138          rw [h1, hM'I.zero τ' (not_le.1 h0).le ω]
139    exact hπp.quasiMeasurePreserving.ae hz'

142  refine ⟨hπ''p, hfilt'', ?_, ?_, ?_, ?_, ?_⟩

166      show M (min τ S) (π p.1) - Gb (min τ S) p = M' (min (min τ S) T) p.1 - Gb (min τ S) p
167      rw [hp, min_eq_left hτ']

178        InitialSigma (prodFiltration P' (wMeasure ιe) ℱ' (wFiltration ιe)))
179    have hset : {τ : ℝ | 0 ≤ τ ∧ ENNReal.ofReal τ ≤ ENNReal.ofReal T} =

183    exact h
