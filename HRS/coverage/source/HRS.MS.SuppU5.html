20lemma u5_fieldAmp_bounds (N : ℕ) (hN : 1 ≤ N) (invβ : ℝ) (h0 : 0 ≤ invβ) :
21    0 ≤ fieldAmp N invβ ∧ fieldAmp N invβ ≤ 1 := by
22  have hNpos : (0 : ℝ) < N := by exact_mod_cast hN
23  have hs : 0 < Real.sqrt N := Real.sqrt_pos.2 hNpos
24  unfold fieldAmp Ascale sigmaScale
25  by_cases hb : invβ = 0

27  · simp only [hb, if_false]
28    refine ⟨div_nonneg (le_min hs.le (Real.sqrt_nonneg _)) hs.le, ?_⟩
29    rw [div_le_one hs]
30    exact min_le_left _ _
