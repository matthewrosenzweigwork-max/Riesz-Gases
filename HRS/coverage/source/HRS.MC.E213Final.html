73theorem uniform_over_realisations_final (hE : EdgeworthHyp S β₀ m) (T : ℝ) (hT : 0 < T) (gd : GaussData S β₀ hE) :
74    ∀ η : ℝ, 0 < η → ∃ N₀ : ℕ, ∀ x : ExpData S β₀ m T, N₀ ≤ x.N → x.expr (gd.Kproj T) ≤ η := by
75  intro η hη
76  by_contra hcon

78  choose f hf1 hf2 using hcon

80  have hxs : ∀ k, xs (k + 1) = f ((xs k).N + 1) := fun k => rfl
81  have hbad : ∀ k, η < (xs k).expr (gd.Kproj T) := by
82    intro k
83    cases k with
84    | zero => exact hf2 0
85    | succ k => rw [hxs]; exact hf2 _

87    refine strictMono_nat_of_lt_succ fun k => ?_

93  have h2 : 2 ≤ (xs N₀).N := (xs N₀).hN
94  have h := hN₀ (xs N₀).N hk h2 (xs N₀).a (xs N₀).ha (xs N₀).Φ (xs N₀).hΦ
