27/-- Uniform `L²(μ)` bound for the response: `∫ h² dμ ≤ Λ²` when `ρ ≤ Λ`. -/
28theorem p8_resp_L2_le (hd : 2 ≤ d) {ρ : Torus d → ℝ} (hρ : SmoothPosDensity ρ) {β : ℝ} (hβ : 0 ≤ β)
29    {Λ : ℝ} (hΛ : ∀ x, ρ x ≤ Λ) :
30    ∫ x, responseInverse (periodicRiesz d ((d : ℝ) / 2)) (withDensityReal ρ) β ρ x ^ 2
31      ∂(withDensityReal ρ) ≤ Λ ^ 2 := by
32  have hs0 : (0 : ℝ) ≤ (d : ℝ) / 2 := by positivity
33  have hdR : (2 : ℝ) ≤ d := by exact_mod_cast hd
34  have hsd : (d : ℝ) / 2 < d := by linarith
35  have hP := PoleCavityAux.isProbabilityMeasure_mu hρ
36  set μ := withDensityReal ρ with hμ
37  set h := responseInverse (periodicRiesz d ((d : ℝ) / 2)) μ β ρ with hh
38  obtain ⟨hm, hp, he⟩ : MemLp h 2 μ ∧ pair μ h = 0 ∧
39      ∀ᵐ x ∂μ, h x + β * Kop (periodicRiesz d ((d : ℝ) / 2)) μ h x = projMean μ ρ x :=
40    eqbias_response hd hs0 hsd hρ hβ
41  have hcoer := eqbias_coercive hd hs0 hsd hρ hβ hm hp
42  have hρm := eqbias_memLp_rho hρ
43  have hhi : Integrable h μ := hm.integrable one_le_two
44  have hhρ : Integrable (fun x => h x * ρ x) μ := hm.integrable_mul hρm
45  have hl2 : l2Inner μ h (fun x => h x + β * Kop (periodicRiesz d ((d : ℝ) / 2)) μ h x) =
46      ∫ x, h x * ρ x ∂μ := by
47    unfold l2Inner
48    have : ∫ x, h x * (h x + β * Kop (periodicRiesz d ((d : ℝ) / 2)) μ h x) ∂μ =
49        ∫ x, h x * projMean μ ρ x ∂μ := by
50      refine integral_congr_ae ?_
51      filter_upwards [he] with x hx
52      rw [hx]
53    rw [this]
54    have hpt : ∀ x, h x * projMean μ ρ x = h x * ρ x - pair μ ρ * h x := fun x => by
55      unfold projMean; ring
56    simp_rw [hpt]
57    rw [integral_sub hhρ (hhi.const_mul _), integral_const_mul]
58    have : ∫ x, h x ∂μ = 0 := hp
59    rw [this]; ring
60  rw [hl2] at hcoer
61  have hh2 : Integrable (fun x => h x ^ 2) μ := hm.integrable_sq
62  have hpt : ∀ x, h x * ρ x ≤ (1 / 2) * h x ^ 2 + (1 / 2) * Λ ^ 2 := by
63    intro x
64    have h1 := hΛ x
65    have h2 := hρ.pos x
66    nlinarith [sq_nonneg (h x - ρ x), sq_nonneg (Λ - ρ x), mul_nonneg h2.le (sub_nonneg.2 h1)]
67  have hint : ∫ x, h x * ρ x ∂μ ≤ ∫ x, ((1 / 2) * h x ^ 2 + (1 / 2) * Λ ^ 2) ∂μ :=
68    integral_mono hhρ ((hh2.const_mul _).add (integrable_const _)) hpt
69  rw [integral_add (hh2.const_mul _) (integrable_const _), integral_const_mul,
70    integral_const, probReal_univ, one_smul] at hint
71  linarith

73/-- Lipschitz/sup membership: a smooth density with `ρ ≥ c₀`, `cbNorm 1 ρ ≤ Λ` lies in `UFam d c₀ (max Λ 1)`. -/
74theorem p8_ufam {c₀ Λ : ℝ} {ρ : Torus d → ℝ} (hρ : SmoothPosDensity ρ) (hlow : ∀ x, c₀ ≤ ρ x)
75    (hΛ : Torus.cbNorm 1 ρ ≤ Λ) : UFam d c₀ (max Λ 1) ρ := by

80/-- Proposition 2.10 `M:H:prop:critical`, uniform marginal expansion (proof of the frozen statement
81`HRS.Main.critical_marginal_expansion`). -/
82theorem critical_marginal_expansion_unif_proof (hd : 2 ≤ d) (βm βp : ℝ) (hβm : 0 < βm)
83    (hββ : βm ≤ βp) :
84    ∃ k₀ : ℕ, ∀ c₀ Λ : ℝ, 0 < c₀ → ∃ C : ℝ, ∀ β : ℝ, β ∈ Set.Icc βm βp →
85      ∀ ρ : Torus d → ℝ, SmoothPosDensity ρ → (∀ x, c₀ ≤ ρ x) → Torus.cbNorm k₀ ρ ≤ Λ →
86      ∀ N : ℕ, 2 ≤ N →
87      firstMarginal (modulatedGibbs (periodicRiesz d ((d : ℝ) / 2)) N β (withDensityReal ρ)) ≪
88          withDensityReal ρ ∧
89        eLpNorm (fun y =>
90            ((firstMarginal (modulatedGibbs (periodicRiesz d ((d : ℝ) / 2)) N β
91                (withDensityReal ρ))).rnDeriv (withDensityReal ρ) y).toReal - 1 -
92              Real.log N / N * (β ^ 2 * sphereArea d / d *
93                responseInverse (periodicRiesz d ((d : ℝ) / 2)) (withDensityReal ρ) β ρ y))
94          2 (withDensityReal ρ) ≤ ENNReal.ofReal (C * Real.sqrt (Real.log N) / N) := by
95  refine ⟨1, fun c₀ Λ hc₀ => ?_⟩
96  have hs0 : (0 : ℝ) ≤ (d : ℝ) / 2 := by positivity
97  have hdR : (2 : ℝ) ≤ d := by exact_mod_cast hd
98  have hsd : (d : ℝ) / 2 < d := by linarith
99  obtain ⟨B, hB⟩ := periodicRiesz_ge (d := d) hd hs0 hsd
100  have hc : ∀ z, 0 ≤ periodicRiesz d ((d : ℝ) / 2) z + B := fun z => by linarith [hB z]
101  set Λ' : ℝ := max Λ 1 with hΛ'
102  have hΛ'1 : 1 ≤ Λ' := le_max_right _ _
103  obtain ⟨C₁, hC₁⟩ := cavity_response_unif (s := (d : ℝ) / 2) hd rfl βm βp hβm hββ Λ' B hc
104  obtain ⟨C₂, hC₂⟩ := insertion_coefficient_unif_proof (s := (d : ℝ) / 2) hd rfl B hc βm βp hβm hββ
105    c₀ Λ' hc₀ hΛ'1
106  have hS : 0 ≤ sphereArea d / d := by
107    have := HRS.StaticLimitsAux.sphereArea_pos hd
108    positivity
109  refine ⟨|C₁| + |C₂| + βp ^ 2 * (sphereArea d / d) * Λ' / Real.sqrt (Real.log 2),
110    fun β hβI ρ hρ hlow hlip N hN => ?_⟩
111  have hβ : 0 < β := lt_of_lt_of_le hβm hβI.1
112  have hfam := p8_ufam hρ hlow hlip
113  have hup := hfam.up
114  obtain ⟨hx, ⟨hP, -, -⟩, hF, -, hz⟩ := hC₁ β hβI ρ hρ hup N hN
115  refine ⟨hP, ?_⟩
116  have hglue := HRS.ME.Marginal.me_marg_glue hd rfl hβ hρ B hc hN hP hF hz
117    (fun x => ⟨(hx x).2.2.1, (hx x).2.2.2.2⟩) (hC₂ β hβI ρ hfam N hN)
118  refine hglue.trans (ENNReal.ofReal_le_ofReal ?_)
119  have hN0 : (0 : ℝ) < N := by exact_mod_cast (by omega : 0 < N)
120  have hlog2 : 0 < Real.log 2 := Real.log_pos (by norm_num)
121  have hsq : 0 < Real.sqrt (Real.log 2) := Real.sqrt_pos.2 hlog2
122  have hA := p8_resp_L2_le hd hρ hβ.le hup
123  have hsA : Real.sqrt (∫ x, responseInverse (periodicRiesz d ((d : ℝ) / 2)) (withDensityReal ρ) β ρ x ^ 2
124      ∂(withDensityReal ρ)) ≤ Λ' := by
125    calc _ ≤ Real.sqrt (Λ' ^ 2) := Real.sqrt_le_sqrt hA
126      _ = Λ' := Real.sqrt_sq (by linarith)
127  have hcoef : |β ^ 2 * sphereArea d / d| ≤ βp ^ 2 * (sphereArea d / d) := by
128    rw [mul_div_assoc, abs_mul, abs_of_nonneg (sq_nonneg β), abs_of_nonneg hS]
129    exact mul_le_mul_of_nonneg_right (pow_le_pow_left₀ hβ.le hβI.2 2) hS
130  have hmid : |C₁| + |C₂| + |β ^ 2 * sphereArea d / d| *
131      Real.sqrt (∫ x, responseInverse (periodicRiesz d ((d : ℝ) / 2)) (withDensityReal ρ) β ρ x ^ 2
132        ∂(withDensityReal ρ)) / Real.sqrt (Real.log 2) ≤
133      |C₁| + |C₂| + βp ^ 2 * (sphereArea d / d) * Λ' / Real.sqrt (Real.log 2) := by
134    have : |β ^ 2 * sphereArea d / d| *
135        Real.sqrt (∫ x, responseInverse (periodicRiesz d ((d : ℝ) / 2)) (withDensityReal ρ) β ρ x ^ 2
136          ∂(withDensityReal ρ)) ≤ βp ^ 2 * (sphereArea d / d) * Λ' :=
137      mul_le_mul hcoef hsA (Real.sqrt_nonneg _) (by positivity)
138    have := div_le_div_of_nonneg_right this hsq.le
139    linarith
140  exact div_le_div_of_nonneg_right
141    (mul_le_mul_of_nonneg_right hmid (Real.sqrt_nonneg _)) hN0.le
