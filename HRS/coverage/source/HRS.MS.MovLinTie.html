9`mov_linear_proof : type_of% @HRS.Whole.mov_linear`.  Existence of `T` with `MovIsSemigroup` is `mov2d_isSemigroup`
10(global Volterra flow on the scale `OscSpace d r`, transported to `L²_0`); uniqueness at every `t ≥ 0` is `mov2d_uniq_core`
11(Schwartz core) plus the density of the core `mov2d_dense_core` (`movlin_uniq`); the decay rate `c = λ` is the `L²`-decay of
12`T` on `L²_0` (`movlin_L2_decay`, from `mov2e_decay` and density); the two bounds of `MovSmoothingBounds` follow from
13`movlin_abstract` (`L²` decay + short-time smoothing with arbitrary loss + the semigroup law) for `T` (`movlin_Hs_T`, from
14`movlin_smooth`) and for the `L²_0`-adjoint (`movlin_Hs_adj`, by duality on the scale; the projection onto `L²_0` costs the
15rank-one term `κ e₁`).

71lemma movlin_part_adj (H : Mov2dTh d s β q ρ) (ψ0 : SchwartzMap (Euc d) ℝ)
72    (hψ0 : (⇑ψ0 : Euc d → ℝ) = fun x => Real.sqrt (ρ x)) (lam C0 : ℝ) (hlam : 0 < lam) (hC0 : 1 ≤ C0)
73    (hdec : ∀ t : ℝ, 0 ≤ t → ∀ v : MovL2Zero ρ,
74      ‖mov2d_T H ψ0 hψ0 t v‖ ≤ C0 * Real.exp (-lam * t) * ‖v‖) {m : ℝ} (hm : movA d s ≤ m) :
75    ∃ C : ℝ, MovSmoothingBounds d s ρ m lam C (fun t => ContinuousLinearMap.adjoint (mov2d_T H ψ0 hψ0 t)) := by
76  have hTs := mov2d_isSemigroup H ψ0 hψ0
77  have hU0 : ContinuousLinearMap.adjoint (mov2d_T H ψ0 hψ0 0) = 1 :=
78    (congrArg ContinuousLinearMap.adjoint hTs.zero).trans movlin_adjoint_one
79  have hUadd : ∀ a b : ℝ, 0 ≤ a → 0 ≤ b →
80      ContinuousLinearMap.adjoint (mov2d_T H ψ0 hψ0 (a + b)) =
81        ContinuousLinearMap.adjoint (mov2d_T H ψ0 hψ0 a) * ContinuousLinearMap.adjoint (mov2d_T H ψ0 hψ0 b) := by
82    intro a b ha hb
83    have e2 : mov2d_T H ψ0 hψ0 (a + b) = mov2d_T H ψ0 hψ0 b * mov2d_T H ψ0 hψ0 a := by
84      rw [add_comm]; exact hTs.add b a hb ha
85    exact (congrArg ContinuousLinearMap.adjoint e2).trans (movlin_adjoint_mul _ _)
86  have hUdec : ∀ t : ℝ, 0 ≤ t → ∀ v : MovL2Zero ρ,
87      ‖ContinuousLinearMap.adjoint (mov2d_T H ψ0 hψ0 t) v‖ ≤ C0 * Real.exp (-lam * t) * ‖v‖ := by
88    intro t ht v
89    have hop : ‖mov2d_T H ψ0 hψ0 t‖ ≤ C0 * Real.exp (-lam * t) :=
90      ContinuousLinearMap.opNorm_le_bound _ (by positivity) (hdec t ht)
91    exact movlin_adjoint_dec hop v
92  exact movlin_abstract s (fun t => ContinuousLinearMap.adjoint (mov2d_T H ψ0 hψ0 t)) lam C0 hlam hC0
93    hU0 hUadd hUdec (movlin_Hs_adj H ψ0 hψ0) hm

110theorem mov_linear_proof {d : ℕ} {s : ℝ} (hd : 2 ≤ d) (hs0 : 0 ≤ s) (hsd : s < d) (β q : ℝ) (hβ : 0 < β)
111    (hq : 0 < q) (ρ : Euc d → ℝ) (hρ : IsThermalEquilibrium d s q β ρ) :
112    ∃ T : ℝ → (MovL2Zero ρ →L[ℝ] MovL2Zero ρ), MovIsSemigroup d s β ρ T ∧
113      (∀ T' : ℝ → (MovL2Zero ρ →L[ℝ] MovL2Zero ρ), MovIsSemigroup d s β ρ T' →
114        ∀ t : ℝ, 0 ≤ t → T' t = T t) ∧
115      ∃ c : ℝ, 0 < c ∧ ∀ m : ℝ, movA d s ≤ m → ∃ C : ℝ,
116        MovSmoothingBounds d s ρ m c C T ∧
117        MovSmoothingBounds d s ρ m c C (fun t => ContinuousLinearMap.adjoint (T t)) := by
118  have H : Mov2dTh d s β q ρ := ⟨hd, hs0, hsd, hβ, hq, hρ⟩
119  obtain ⟨ψ0, hψ0⟩ := mov2d_sqrt_schwartz H
120  obtain ⟨lam, hlam, C0, hC0, hdec⟩ := movlin_L2_decay H ψ0 hψ0
121  refine ⟨mov2d_T H ψ0 hψ0, mov2d_isSemigroup H ψ0 hψ0, fun T' hT' t ht => movlin_uniq H ψ0 hψ0 hT' ht,
122    lam, hlam, fun m hm => ?_⟩
123  exact movlin_part_both H ψ0 hψ0 lam C0 hlam hC0 hdec hm
