3/-!
4# Sharpness of the root-`N` rate: proof of `HRS.Main.edgeworth_sharpness`
5
6The first Edgeworth correction `c = (1/6) E⟨D³Φ(Z), 𝒦_{a,T}⟩` is bounded by
7`√N · d_5(ζ_N, Z) + ε_N` for every `N ≥ 2` (Theorem 2.13 applied to the test function `Φ`
8and the bound `|∫Φ d(P - Q)| ≤ d_5(P, Q)` on the `C_b^5` unit ball), and `ε_N → 0`.
9-/

18theorem edgeworth_sharpness_proof : type_of% @HRS.Main.edgeworth_sharpness := by
19  intro d S β₀ m T hE hT K ε hK a ha ν hν Ω' _ P' _ Z₀ Z hZ Φ hΦ Ω _ P _ X₀ W X hflow hfilt hinit
20  -- the first correction, as a real number
21  set c : ℝ := (1 / 6) * ∫ ω, ((K a : projTensor (PathSpace d m T) 3) :
22      TensorDual (PathSpace d m T) 3) (iteratedFDeriv ℝ 3 Φ (windowPath m T a Z ω)) ∂P' with hc
23  -- a triangle-type inequality
24  have key : ∀ (x c e : ℝ), |x - c| ≤ e → |c| - e ≤ |x| := by
25    intro x c e h
26    have h' := abs_sub_abs_le_abs_sub c x
27    rw [abs_sub_comm c x] at h'
28    linarith
29  have hbound : ∀ N : ℕ, 2 ≤ N →
30      |c| - ε N ≤ Real.sqrt N * smoothDist 5 (law (P N) (windowZeta m T a (P N) (X N)))
31        (law P' (windowPath m T a Z)) := by
32    intro N hN
33    have h1 := hK.expansion a ha N hN (P N) (X₀ N) (W N) (X N) (hflow N hN) (hfilt N hN)
34      (hinit N hN) ν hν P' Z₀ Z hZ Φ hΦ
35    have h2 : |∫ z, Φ z ∂(law (P N) (windowZeta m T a (P N) (X N))) -
36        ∫ z, Φ z ∂(law P' (windowPath m T a Z))| ≤
37        smoothDist 5 (law (P N) (windowZeta m T a (P N) (X N))) (law P' (windowPath m T a Z)) :=
38      abs_integral_sub_le_smoothDist hΦ _ _
39    have hsq : (0 : ℝ) ≤ Real.sqrt N := Real.sqrt_nonneg _
40    have hx := key _ _ _ h1
41    rw [abs_mul (Real.sqrt N), abs_of_nonneg hsq, ← hc] at hx
42    have hm := mul_le_mul_of_nonneg_left h2 hsq
43    linarith
44  have hlim : Tendsto (fun N : ℕ => ENNReal.ofReal (|c| - ε N)) atTop (𝓝 (ENNReal.ofReal |c|)) := by
45    apply ENNReal.tendsto_ofReal
46    have h0 := (tendsto_const_nhds (x := |c|)).sub hK.eps_tendsto
47    rw [sub_zero] at h0
48    exact h0
49  rw [← hlim.liminf_eq]
50  exact liminf_le_liminf (Filter.eventually_atTop.2 ⟨2, fun N hN => ENNReal.ofReal_le_ofReal (hbound N hN)⟩)
