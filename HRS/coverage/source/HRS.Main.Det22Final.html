20theorem deterministic_comparison_proof (b : GeneralBounds d) :
21    ∃ C₀min : ℝ, ∀ C₀ : ℝ, C₀min ≤ C₀ → ∃ C : ℝ, 0 < C ∧
22      ∀ S : GeneralSetup d, S.Within { b with C₀ := C₀ } → S.MFHyp → S.PosDensityWith b.c →
23      (∀ N, S.invβ N = 0) →
24      ∀ (N : ℕ) (_ : 2 ≤ N) {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω)
25      [IsProbabilityMeasure P] (x₀ : Config d N)
26      (W : ℝ≥0 → Ω → (Fin N → Euc d)) (_ : IsVectorBrownian P W)
27      (X : ℝ≥0 → Ω → Config d N)
28      (_ : IsStrongSolution P (S.dyn N) S.T (fun _ => x₀) W X)
29      (_ : ∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
30        (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X)
31      (_ : S.EnergyMomentFinite P N X) (F : NegSobolev d (S.γ : ℝ)),
32      ∀ᵐ ω ∂P, ∀ t ∈ Set.Icc (0 : ℝ) S.T,
33        ‖S.κ N • NegSobolev.ofMeasureDiff (S.γ : ℝ) (empirical (X (Real.toNNReal t) ω))
34            (withDensityReal (S.ρ N t)) - transportField (S.dyn N) (S.ρ N) t (S.γ : ℝ) F‖ ≤
35          C * (‖S.κ N • NegSobolev.ofMeasureDiff (S.γ : ℝ) (empirical x₀)
36              (withDensityReal (S.ρ N 0)) - F‖ + S.κ N * S.energyMoment P N X) := by

39  obtain ⟨K, hK0, hev⟩ := HRS.GF.gf_ev_grad_proof b
40  refine ⟨C₀min, fun C₀ hC₀ => ?_⟩
41  obtain ⟨C, hC, h⟩ := hmain C₀ hC₀
42  refine ⟨C, hC, ?_⟩
43  intro S hS hMF hPos hβ N hN Ω _ P _ x₀ W hW X hX hflt hE F
44  have hSw : S.Within { b with C₀ := S.C₀ } := by
45    obtain ⟨h1, h2, h3, h4, h5, h6, h7⟩ := hS
46    exact ⟨h1, h2, h3, h4, h5, rfl, h7⟩
47  obtain ⟨EV, hEV⟩ := hev S hSw hMF hPos N
48  have hself : ∀ t ∈ Set.Icc (0 : ℝ) S.T, EV.VE t t = 1 := hEV.maxHyp.1.self_E
49  refine HRS.Det22.det22_assemble S (-1) hMF N hN _ le_rfl P x₀ W hW X hX hflt hE (hβ N) EV.VE
50    hEV.dual hself hEV.maxHyp.1.cont_E F F C hC ?_
51  intro Y hY
52  have hfin : S.InitialErrorFinite P N 1 (fun _ => x₀) (fun _ => F) := integrable_const _
53  have hind : ProbabilityTheory.IndepFun (fun ω => ((fun _ => x₀) ω, (fun _ => F) ω))
54      (fun ω => fun t => W t ω) P := indepFun_const_left _ _
55  obtain ⟨-, hW1, -⟩ := h S hS hMF hPos N hN P (fun _ => x₀) W hW X hX hflt hE (fun _ => F)
56    measurable_const hind hfin Y hY
57  exact hW1
