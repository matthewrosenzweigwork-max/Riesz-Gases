25/-- **P1** (measurability of the Gibbs-centred window; scope note 1.4, "JUNK/MEASURABILITY").
26`windowGibbs` (junk value `0` for discontinuous windows) is `P`-a.e. measurable for any strong solution
27`X` of any particle system (`IsStrongSolution` supplies the measurability of each `X t` and the a.s.
28continuity of every coordinate), centred at any probability measure `μ`; `d < 2m` makes `η ↦ etaE m η μ`
29continuous.  Route: `eqlaw_pathOf`, `eqlaw_aemeasurable_pathOf`, `eqlaw_continuous_ofMeasureDiff_empirical`. -/
30theorem windowGibbs_aemeasurable {N : ℕ} (hN : 1 ≤ N) {m : ℝ} (hm : (d : ℝ) < 2 * m) (T u : ℝ)
31    (μ : Measure (Torus d)) [IsProbabilityMeasure μ]
32    {Ω : Type*} [MeasurableSpace Ω] (P : Measure Ω) (D : DynData d) (H : ℝ)
33    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N)
34    (h : IsStrongSolution P D H X₀ W X) :
35    AEMeasurable (windowGibbs m T u μ X) P := by
36  have hη : Continuous fun Y : Config d N => etaE m Y μ :=
37    (HRS.MD.EQLawPath.eqlaw_continuous_ofMeasureDiff_empirical hm μ hN).const_smul (Real.sqrt N)
38  have heq : windowGibbs m T u μ X = fun ω => HRS.MD.EQLawPath.eqlaw_pathOf d m T (fun τ =>
39      etaE m (X (Real.toNNReal (u + τ)) ω) μ) := by
40    funext ω; rfl
41  rw [heq]
42  refine HRS.MD.EQLawPath.eqlaw_aemeasurable_pathOf P T _ (fun τ => ?_) ?_
43  · exact hη.measurable.comp (h.adapted (Real.toNNReal (u + τ)))
44  · filter_upwards [h.cont] with ω hω
45    refine hη.comp ?_
46    refine continuous_pi fun i => ?_
47    exact (hω i).comp (continuous_real_toNNReal.comp (continuous_const.add continuous_subtype_val))
