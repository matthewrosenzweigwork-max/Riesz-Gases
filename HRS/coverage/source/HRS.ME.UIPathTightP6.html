24/-- arithmetic: `L * (ε / (2 L)) ≤ ε / 2` in `ℝ≥0∞` for `L > 0`. -/
25theorem p_up6_mul_div (ε : ℝ≥0∞) {L : ℝ} (hL : 0 < L) :
26    ENNReal.ofReal L * (ε / (2 * ENNReal.ofReal L)) ≤ ε / 2 := by
27  have h0 : ENNReal.ofReal L ≠ 0 := (ENNReal.ofReal_pos.2 hL).ne'
28  have ht : ENNReal.ofReal L ≠ ⊤ := ENNReal.ofReal_ne_top
29  rw [div_eq_mul_inv, div_eq_mul_inv,
30    ENNReal.mul_inv (Or.inl (by norm_num)) (Or.inl (by norm_num))]
31  calc ENNReal.ofReal L * (ε * (2⁻¹ * (ENNReal.ofReal L)⁻¹))
32      = ε * 2⁻¹ * (ENNReal.ofReal L * (ENNReal.ofReal L)⁻¹) := by ring
33    _ = ε * 2⁻¹ := by rw [ENNReal.mul_inv_cancel h0 ht, mul_one]
34    _ ≤ ε * 2⁻¹ := le_rfl
