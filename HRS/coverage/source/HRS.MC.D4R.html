23/-- A bounded kernel has a bounded modulated energy (`μ` a probability measure). -/
24lemma d4r_energy_bound {g : Torus d → ℝ} {M : ℝ} (hg : ∀ x, |g x| ≤ M) {N : ℕ} (hN : 1 ≤ N)
25    (X : Config d N) (μ : Measure (Torus d)) [IsProbabilityMeasure μ] :
26    |modulatedEnergyKernel g X μ| ≤ 2 * M := by
27  have hM : 0 ≤ M := le_trans (abs_nonneg _) (hg 0)
28  have hN0 : (0 : ℝ) < N := by exact_mod_cast hN
29  have hint : ∀ a : Torus d, |∫ y, g (a - y) ∂μ| ≤ M := by
30    intro a
31    have := norm_integral_le_of_norm_le_const (μ := μ) (f := fun y => g (a - y)) (C := M)
32      (Filter.Eventually.of_forall fun y => by simpa using hg (a - y))
33    simpa using this
34  have h1 : |∑ i : Fin N, ∑ j ∈ Finset.univ.erase i, g (X i - X j)| ≤ (N : ℝ) ^ 2 * M := by
35    calc |∑ i : Fin N, ∑ j ∈ Finset.univ.erase i, g (X i - X j)|
36        ≤ ∑ i : Fin N, |∑ j ∈ Finset.univ.erase i, g (X i - X j)| := Finset.abs_sum_le_sum_abs _ _
37      _ ≤ ∑ i : Fin N, (N : ℝ) * M := by
38          refine Finset.sum_le_sum fun i _ => ?_
39          calc |∑ j ∈ Finset.univ.erase i, g (X i - X j)|
40              ≤ ∑ j ∈ Finset.univ.erase i, |g (X i - X j)| := Finset.abs_sum_le_sum_abs _ _
41            _ ≤ ∑ j ∈ Finset.univ.erase i, M := Finset.sum_le_sum fun j _ => hg _
42            _ = ((Finset.univ.erase i).card : ℝ) * M := by simp
43            _ ≤ N * M := by
44                apply mul_le_mul_of_nonneg_right _ hM
45                have : (Finset.univ.erase i).card ≤ N := by
46                  have := Finset.card_erase_le (s := (Finset.univ : Finset (Fin N))) (a := i)
47                  simpa using this
48                exact_mod_cast this
49      _ = (N : ℝ) ^ 2 * M := by simp; ring
50  have h2 : |∑ i : Fin N, ∫ y, g (X i - y) ∂μ| ≤ N * M := by
51    calc |∑ i : Fin N, ∫ y, g (X i - y) ∂μ| ≤ ∑ i : Fin N, |∫ y, g (X i - y) ∂μ| :=
52          Finset.abs_sum_le_sum_abs _ _
53      _ ≤ ∑ i : Fin N, M := Finset.sum_le_sum fun i _ => hint _
54      _ = N * M := by simp
55  have h3 : |∫ x, ∫ y, g (x - y) ∂μ ∂μ| ≤ M := by
56    have := norm_integral_le_of_norm_le_const (μ := μ) (f := fun x => ∫ y, g (x - y) ∂μ) (C := M)
57      (Filter.Eventually.of_forall fun x => by simpa using hint x)
58    simpa using this
59  unfold modulatedEnergyKernel modulatedEnergy
60  have hpos : (0 : ℝ) < 2 * (N : ℝ) ^ 2 := by positivity
61  have hA : |1 / (2 * (N : ℝ) ^ 2) * ∑ i : Fin N, ∑ j ∈ Finset.univ.erase i, g (X i - X j)|
62      ≤ M / 2 := by
63    rw [abs_mul, abs_of_pos (by positivity : (0 : ℝ) < 1 / (2 * (N : ℝ) ^ 2))]
64    calc 1 / (2 * (N : ℝ) ^ 2) * |∑ i : Fin N, ∑ j ∈ Finset.univ.erase i, g (X i - X j)|
65        ≤ 1 / (2 * (N : ℝ) ^ 2) * ((N : ℝ) ^ 2 * M) :=
66          mul_le_mul_of_nonneg_left h1 (by positivity)
67      _ = M / 2 := by field_simp
68  have hB : |1 / (N : ℝ) * ∑ i : Fin N, ∫ y, g (X i - y) ∂μ| ≤ M := by
69    rw [abs_mul, abs_of_pos (by positivity : (0 : ℝ) < 1 / (N : ℝ))]
70    calc 1 / (N : ℝ) * |∑ i : Fin N, ∫ y, g (X i - y) ∂μ| ≤ 1 / (N : ℝ) * (N * M) :=
71          mul_le_mul_of_nonneg_left h2 (by positivity)
72      _ = M := by field_simp
73  have hC : |1 / 2 * ∫ x, ∫ y, g (x - y) ∂μ ∂μ| ≤ M / 2 := by
74    rw [abs_mul, abs_of_pos (by norm_num : (0 : ℝ) < 1 / 2)]
75    linarith
76  rw [abs_le] at hA hB hC ⊢
77  constructor <;> linarith [hA.1, hA.2, hB.1, hB.2, hC.1, hC.2]

115      Real.log_le_log (by positivity) (one_div_le_one_div_of_le htpos hxe)
116    nlinarith
117  · obtain ⟨c, C1, hc, -, h4⟩ := hiv1 hs
118    have ht : Tendsto (fun ε : ℝ => c * ε ^ (-s) - C1) (𝓝[>] 0) atTop :=
119      tendsto_atTop_add_const_right _ _
120        ((tendsto_rpow_neg_nhdsGT_zero (by linarith : -s < 0)).const_mul_atTop hc)
121    filter_upwards [ht.eventually_ge_atTop R] with ε hε hε0 hε2 x hx0 hxe

193theorem fdl_compare_reverse_false_proof (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {ψ : Euc d → ℝ}
194    (hψ : IsFDLProfile ψ) {b : ℝ} (hb : IsFDLConst ψ s b) (C₀ C : ℝ) {N : ℕ} (hN : 2 ≤ N)
195    {Λ : ℝ} (hΛ : 1 ≤ Λ) (hlam : smearScale d N Λ < 1 / 2)
196    {ρ : Torus d → ℝ} (hρ : IsProbDensity ρ) (hρΛ : ∀ x, ρ x ≤ Λ) :
197    ∃ X : Config d N, X ∉ collisionSet d N ∧
198      C * (fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ) + oNΛ d s C₀ Λ N)
199        < rieszEnergy d s X (withDensityReal ρ) := by
200  have hN1 : 1 ≤ N := by omega
201  haveI := isProb_withDensityReal hρ
202  obtain ⟨C₀', C', hC₀', hC', hcmp⟩ := fdl_compare_proof hd hs0 hsd hψ hb
203  have hlam0 : 0 < smearScale d N Λ := d47_lam_pos hN1 hΛ
204  have hlam1 : smearScale d N Λ < 1 := by linarith
205  obtain ⟨M, hM, hMb⟩ := d4r_trunc_bdd hd hs0 hsd hψ hb hlam0 hlam1
206  have hb0 : 0 ≤ b := hb.1.le
207  have hΛ0 : 0 ≤ Λ := by linarith
208  have hNpos : (0 : ℝ) < N := by exact_mod_cast hN1
209  set o := oNΛ d s C₀ Λ N with ho
210  set E0 : ℝ := C' * Λ * smearScale d N Λ ^ ((d : ℝ) - s) with hE0def
211  have hE0 : 0 ≤ E0 := by
212    have : 0 ≤ smearScale d N Λ ^ ((d : ℝ) - s) := Real.rpow_nonneg hlam0.le _
213    positivity
214  set T : ℝ := |C| * (2 * M + |o|) with hT
215  set R : ℝ := (N : ℝ) ^ 2 * (2 * (T + 2 * M + E0)) + M + 1 with hR
216  have hev := d4r_g_large hd hs0 hsd hψ hb R
217  have hev2 : ∀ᶠ ε in 𝓝[>] (0 : ℝ), ε ∈ Set.Ioo (0 : ℝ) (1 / (2 * N)) :=
218    Ioo_mem_nhdsGT (by positivity)

220  obtain ⟨hε0, hεN⟩ := hεmem
221  have hεN' : ε * N ≤ 1 / 2 := by
222    have : ε * N < 1 / (2 * N) * N := mul_lt_mul_of_pos_right hεN hNpos
223    have h2 : 1 / (2 * (N : ℝ)) * N = 1 / 2 := by field_simp
224    linarith
225  have hε2 : ε ≤ 1 / 2 := by
226    have hN1' : (1 : ℝ) ≤ N := by exact_mod_cast hN1
227    nlinarith
228  have hnc := d4r_X_notcol hd hε0 hεN'
229  refine ⟨d4r_X hd N ε, hnc, ?_⟩
230  set X := d4r_X hd N ε with hX
231  obtain ⟨hxne, hxd⟩ := d4r_pair_tdist hd hN hε0 hε2
232  set x := X ⟨0, by omega⟩ - X ⟨1, by omega⟩ with hxdef
233  have hg : R ≤ periodicRiesz d s x := hRε hε0 hε2 x hxne hxd.le
234  have hw : periodicRiesz d s x - M ≤ fdlWT ψ s b (smearScale d N Λ) x := by
235    have h1 : fdlTruncT ψ s b (smearScale d N Λ) x =
236        periodicRiesz d s x - fdlWT ψ s b (smearScale d N Λ) x := by
237      unfold fdlTruncT; simp [hxne]
238    have h2 := (abs_le.mp (hMb x)).2
239    linarith
240  have hP : 1 / (N : ℝ) ^ 2 * fdlWT ψ s b (smearScale d N Λ) x ≤
241      fdlPairW ψ s b (smearScale d N Λ) X := d4r_pairW_ge hψ hb0 hN X
242  have hcm := (hcmp Λ hΛ N hN1 hlam ρ hρ hρΛ X hnc).2.2.1
243  have hFl : |fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ)| ≤ 2 * M :=
244    d4r_energy_bound hMb hN1 X _
245  have hkey : 1 / (N : ℝ) ^ 2 * (R - M) = 2 * (T + 2 * M + E0) + 1 / (N : ℝ) ^ 2 := by
246    rw [hR]; field_simp; ring
247  have hwR : 1 / (N : ℝ) ^ 2 * (R - M) ≤ 1 / (N : ℝ) ^ 2 * fdlWT ψ s b (smearScale d N Λ) x :=
248    mul_le_mul_of_nonneg_left (by linarith) (by positivity)
249  have hpos2 : 0 < 1 / (N : ℝ) ^ 2 := by positivity
250  have hE0' : C' * Λ * smearScale d N Λ ^ ((d : ℝ) - s) = E0 := rfl
251  rw [hE0', abs_le] at hcm
252  have hFl' := abs_le.mp hFl
253  -- the right-hand side
254  have hrhs : C * (fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ) + o) ≤ T := by
255    calc C * (fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ) + o)
256        ≤ |C * (fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ) + o)| := le_abs_self _
257      _ = |C| * |fdlEnergy ψ s b (smearScale d N Λ) X (withDensityReal ρ) + o| := abs_mul _ _

260          exact (abs_add_le _ _).trans (by linarith)

262  linarith [hcm.1, hcm.2, hFl'.1, hFl'.2]
