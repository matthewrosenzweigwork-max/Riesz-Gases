26theorem p_up2_sum_pair_identity {N : ℕ} (H : Fin N → Fin N → ℝ) (hH : ∀ i j, H i j = H j i) :
27    ∑ i, ∑ j, H i j =
28      2 * ∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j), H i j + ∑ i, H i i := by
29  simp only [Finset.sum_filter]
30  have hpt : ∀ i j, H i j = (if i < j then H i j else 0) + (if j < i then H i j else 0) +
31      (if i = j then H i j else 0) := by
32    intro i j
33    rcases lt_trichotomy i j with h | rfl | h
34    · rw [if_pos h, if_neg (not_lt.2 h.le), if_neg h.ne]; ring
35    · simp
36    · rw [if_neg (not_lt.2 h.le), if_pos h, if_neg h.ne']; ring
37  have hB : ∑ i, ∑ j, (if j < i then H i j else 0) = ∑ i, ∑ j, (if i < j then H i j else 0) := by
38    rw [Finset.sum_comm]
39    refine Finset.sum_congr rfl fun j _ => Finset.sum_congr rfl fun i _ => ?_
40    rw [hH]
41  have hD : ∑ i, ∑ j, (if i = j then H i j else 0) = ∑ i, H i i := by
42    refine Finset.sum_congr rfl fun i _ => ?_
43    simp
44  have hsplit : ∑ i, ∑ j, H i j = (∑ i, ∑ j, (if i < j then H i j else 0)) +
45      (∑ i, ∑ j, (if j < i then H i j else 0)) + ∑ i, ∑ j, (if i = j then H i j else 0) := by
46    have hrow : ∀ i, ∑ j, H i j = (∑ j, (if i < j then H i j else 0)) +
47        (∑ j, (if j < i then H i j else 0)) + ∑ j, (if i = j then H i j else 0) := fun i => by
48      rw [← Finset.sum_add_distrib, ← Finset.sum_add_distrib]
49      exact Finset.sum_congr rfl fun j _ => hpt i j
50    simp only [hrow, Finset.sum_add_distrib]
51  rw [hsplit, hB, hD]
52  ring

54theorem p_up2_pairSum_perm {N : ℕ} (H : Fin N → Fin N → ℝ) (hH : ∀ i j, H i j = H j i)
55    (σ : Equiv.Perm (Fin N)) :
56    ∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j), H (σ i) (σ j) =
57      ∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j), H i j := by
58  have h1 := p_up2_sum_pair_identity (fun i j => H (σ i) (σ j)) (fun i j => hH _ _)
59  have h2 := p_up2_sum_pair_identity H hH
60  have hF : ∑ i, ∑ j, H (σ i) (σ j) = ∑ i, ∑ j, H i j := by
61    calc ∑ i, ∑ j, H (σ i) (σ j) = ∑ i, ∑ j, H (σ i) j :=
62          Finset.sum_congr rfl fun i _ => Equiv.sum_comp σ (H (σ i))
63      _ = ∑ i, ∑ j, H i j := Equiv.sum_comp σ (fun i => ∑ j, H i j)
64  have hD : ∑ i, H (σ i) (σ i) = ∑ i, H i i := Equiv.sum_comp σ (fun i => H i i)
65  linarith

67theorem p_up2_canonicalHamiltonian_comp_perm {d N : ℕ} (g V : Torus d → ℝ) (hg : ∀ x, g (-x) = g x)
68    (X : Config d N) (σ : Equiv.Perm (Fin N)) :
69    HRS.canonicalHamiltonian g V (X ∘ σ) = HRS.canonicalHamiltonian g V X := by
70  unfold HRS.canonicalHamiltonian
71  have hH : ∀ i j : Fin N, g (X i - X j) = g (X j - X i) := fun i j => by
72    rw [← hg (X i - X j), neg_sub]
73  have hp := p_up2_pairSum_perm (fun i j => g (X i - X j)) hH σ
74  have hv : ∑ i, V ((X ∘ σ) i) = ∑ i, V (X i) := Equiv.sum_comp σ (fun i => V (X i))
75  have hp' : ∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j), g ((X ∘ σ) i - (X ∘ σ) j) =
76      ∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j), g (X i - X j) := hp
77  rw [hv, hp']

79theorem p_up2_canonicalGibbs_isExchangeable {d N : ℕ} (g V : Torus d → ℝ) (hg : ∀ x, g (-x) = g x)
80    (β : ℝ) : HRS.IsExchangeable (HRS.canonicalGibbs g V N β) := by
81  intro σ
82  have hmeas : AEMeasurable (fun X : Config d N => X ∘ σ)
83      ((volume : Measure (Config d N)).withDensity fun X =>
84        ENNReal.ofReal (Real.exp (-β * HRS.canonicalHamiltonian g V X))) :=
85    (HRS.permMeasEquiv σ).measurable.aemeasurable
86  have hvol : (volume : Measure (Config d N)) =
87      Measure.pi fun _ : Fin N => (volume : Measure (Torus d)) := rfl
88  unfold HRS.canonicalGibbs
89  rw [Measure.map_smul _ hmeas]
90  congr 1
91  rw [hvol]
92  refine HRS.map_withDensity_perm _ σ _ fun X => ?_
93  simp only [p_up2_canonicalHamiltonian_comp_perm g V hg X σ]

95/-- Pointwise identity: for any law `P` on configurations whose mean empirical measure is `μ`,
96`η_N(X; μ) = ζ_N(X; P)`. -/
97theorem p_up2_etaE_eq_zetaE {N : ℕ} (m : ℝ) (P : Measure (Config d N)) (μ : Measure (Torus d))
98    (hμ : HRS.meanEmpirical P = μ) (X : Config d N) :
99    HRS.etaE m X μ = HRS.zetaE m P X := by
100  unfold HRS.etaE HRS.flucE HRS.zetaE

103/-- **P2** (stationary identification; scope note 1.1 step 1).  Along a stationary Gibbs flow on the
104horizon `u + T` the window centred at the FIXED Gibbs first marginal `μ_N^*` is, pointwise in `ω`, the window
105`windowZeta` centred at the actual mean empirical measure (both centres are `μ_N^*`, since
106`P.map (X (u+t)) = G_N` for `0 ≤ t ≤ T` and `G_N` is exchangeable).  Route:
107`md_stat_gibbs_flow_stationary_proof`, `canonicalGibbs_isExchangeable`, `meanEmpirical_eq_firstMarginal`,
108then `dite` congruence on the equal functions of `t ∈ [0,T]`. -/
109theorem windowGibbs_eq_windowZeta (hd : 2 ≤ d) {s β : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) (hβ : 0 < β)
110    (V : Torus d → ℝ) (hV : Torus.IsSmooth V) {m : ℝ} (hm : (d : ℝ) < 2 * m) (T u : ℝ) (hu : 0 ≤ u)
111    {N : ℕ} (hN : 2 ≤ N)
112    {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
113    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N)
114    (hX : IsStationaryGibbsFlow P s hs0 hsd V β hβ (u + T) X₀ W X) :
115    windowGibbs m T u (firstMarginal (canonicalGibbs (periodicRiesz d s) V N β)) X =
116      Main.windowZeta m T u P X := by
117  have hN1 : 1 ≤ N := by omega
118  haveI hG : IsProbabilityMeasure (canonicalGibbs (periodicRiesz d s) V N β) :=
119    HRS.MD.EQLawFlow.eqlaw_gibbs_prob hd hs0 hsd hV hβ hN1
120  have hExch : HRS.IsExchangeable (canonicalGibbs (periodicRiesz d s) V N β) :=
121    p_up2_canonicalGibbs_isExchangeable (periodicRiesz d s) V (fun x => periodicRiesz_neg x) β
122  have hmean : HRS.meanEmpirical (canonicalGibbs (periodicRiesz d s) V N β) =
123      firstMarginal (canonicalGibbs (periodicRiesz d s) V N β) :=
124    HRS.meanEmpirical_eq_firstMarginal _ hExch hN1
125  have hstat := HRS.MD.Stat.md_stat_gibbs_flow_stationary_proof hd hs0 hsd V hV β hβ (u + T) hN
126    P X₀ W X hX
127  have hpt : ∀ t : Set.Icc (0 : ℝ) T,
128      P.map (X (Real.toNNReal (u + t))) = canonicalGibbs (periodicRiesz d s) V N β := by
129    intro t
130    apply hstat
131    rw [Real.coe_toNNReal _ (by linarith [t.2.1])]
132    linarith [t.2.2]
133  have hfun : ∀ ω : Ω,
134      (fun t : Set.Icc (0 : ℝ) T =>
135        HRS.etaE m (X (Real.toNNReal (u + t)) ω) (firstMarginal (canonicalGibbs (periodicRiesz d s) V N β))) =
136      (fun t : Set.Icc (0 : ℝ) T =>
137        HRS.zetaE m (P.map (X (Real.toNNReal (u + t)))) (X (Real.toNNReal (u + t)) ω)) := by
138    intro ω
139    funext t
140    have hμ : HRS.meanEmpirical (P.map (X (Real.toNNReal (u + t)))) =
141        firstMarginal (canonicalGibbs (periodicRiesz d s) V N β) := by
142      rw [hpt t]
143      exact hmean
144    exact (p_up2_etaE_eq_zetaE m (P.map (X (Real.toNNReal (u + t))))
145      (firstMarginal (canonicalGibbs (periodicRiesz d s) V N β)) hμ (X (Real.toNNReal (u + t)) ω))
146  funext ω
147  unfold HRS.windowGibbs HRS.Main.windowZeta
148  rw [hfun ω]
