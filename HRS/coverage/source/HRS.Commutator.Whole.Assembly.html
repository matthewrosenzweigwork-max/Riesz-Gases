63/-- `(5.riesz-scaling)`, `(5.logscaling)` (`[N] L744-759`): `F_N(X̃, μ̃) = η^s F_N(X, μ)` for `s > 0`
64and `F_N(X̃, μ̃) = F_N(X, μ) - log η/(2N)` for `s = 0` (`g(z/η) = g(z) + log η`, and
65`∫_{△ᶜ} dν^{⊗2} = -1/N`, `(5.massoffdiag)`); the two cases read uniformly
66`η^s F_N - 1_{s=0} log η/(2N)`.  Needs the absolute convergence of the integrals (automatic for
67bounded compactly supported `μ`). -/

83/-- The right-hand side bracket of `(1.scale)` (`[N] L144-145`):
84`Q = F_N - 1_{s=0} log η/(2N) + C_p(η^{-s}/N + ‖μ‖_{L^p} η^{d(1-1/p)-s})`. -/
85def scaleQ (s Cp : ℝ) (p : ℝ≥0∞) (η : ℝ) {N : ℕ} (X : Fin N → Euc d) (ρ : Euc d → ℝ) : ℝ :=
86  modEnergy s X ρ - (if s = 0 then Real.log η / (2 * (N : ℝ)) else 0) +
87    Cp * (η ^ (-s) / (N : ℝ) + normE p ρ * η ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s))

89/-- Proposition 1.2 for bounded compactly supported `μ` (`[N] L761`: "Apply `(5.unitestimate)` to the
90rescaled data and multiply by `η^{-s}` … yields `(1.scale)` for bounded compactly supported `μ`.  The
91nonnegativity assertion follows at the same time from `(2.Ecoercive)` for the rescaled data").
92Proved from `unit_estimate`, `energy_control` and the scaling identities. -/
93theorem scale_estimate (hd : 3 ≤ d) (hs0 : 0 ≤ s) (hsd : s < (d : ℝ) - 2) :
94    ∃ C : ℝ, 0 < C ∧ ∀ p : ℝ≥0∞, ENNReal.ofReal ((d : ℝ) / (d - s)) < p →
95      ∃ Cp : ℝ, 0 < Cp ∧ ∀ ρ : Euc d → ℝ, BddCptDensity ρ →
96      ∀ (N : ℕ) (X : Fin N → Euc d), 1 ≤ N → Function.Injective X →
97      ∀ (v : Euc d → Euc d) (K : ℝ), SubTransport d s v K →
98      ∀ η : ℝ, 0 < η →
99        0 ≤ scaleQ s Cp p η X ρ ∧ |commA s X ρ v| ≤ C * K * scaleQ s Cp p η X ρ := by
100  obtain ⟨C, hC, U⟩ := unit_estimate hd hs0 hsd
101  refine ⟨C, hC, fun p hp => ?_⟩
102  have hd0 : (0 : ℝ) < (d : ℝ) - s := by
103    have : (3 : ℝ) ≤ d := by exact_mod_cast hd
104    linarith
105  have h1p : 1 ≤ p := by
106    have h1 : (1 : ℝ) ≤ (d : ℝ) / (d - s) := by
107      rw [le_div_iff₀ hd0]; linarith
108    exact le_trans (by simpa using ENNReal.ofReal_le_ofReal h1) hp.le
109  set q : ℝ≥0∞ := (1 - p⁻¹)⁻¹ with hq
110  have hpq : p⁻¹ + q⁻¹ = 1 := by
111    have : p⁻¹ ≤ 1 := ENNReal.inv_le_one.2 h1p
112    rw [hq, inv_inv, add_tsub_cancel_of_le this]
113  obtain ⟨Cp0, hCp0, Uc⟩ := U p q hp hpq
114  obtain ⟨Cb, hCb, E⟩ := energy_control hd hs0 hsd
115  obtain ⟨Cp1, hCp1, Ec⟩ := E p q hp hpq
116  refine ⟨max Cp0 Cp1, lt_max_of_lt_left hCp0, fun ρ hρ N X hN hX v K hv η hη => ?_⟩
117  have hρt := scaleDens_bdd η hη ρ hρ
118  have hXt := scaleConf_injective η hη X hX
119  have hvt := subTransport_scale hd hs0 hsd η hη v K hv
120  obtain ⟨L, hL⟩ : ∃ L : ℝ≥0, LipschitzWith L v := by
121    obtain ⟨L, F, -, hL, -, -⟩ := hv
122    exact ⟨L, hL⟩
123  have u := Uc (max Cp0 Cp1) (le_max_left _ _) (scaleDens η ρ) hρt N (scaleConf η X) hN hXt
124    (scaleVec η v) K hvt
125  have c := (Ec (max Cp0 Cp1) (le_max_right _ _) (scaleDens η ρ) hρt N (scaleConf η X) hN hXt).1
126  have hηs : 0 < η ^ s := Real.rpow_pos_of_pos hη s
127  -- the rescaled energy is `η^s * Q`
128  have key : unitE s (max Cp0 Cp1) p (scaleConf η X) (scaleDens η ρ) =
129      η ^ s * scaleQ s (max Cp0 Cp1) p η X ρ := by
130    unfold unitE scaleQ
131    rw [modEnergy_scale hd hs0 hsd η hη ρ hρ hN X hX, normE_scaleDens η hη p ρ]
132    have e1 : η ^ s * (η ^ (-s) / (N : ℝ)) = 1 / (N : ℝ) := by
133      rw [← mul_div_assoc, ← Real.rpow_add hη]; simp
134    have e2 : η ^ s * (normE p ρ * η ^ ((d : ℝ) * (1 - (p⁻¹).toReal) - s)) =
135        η ^ ((d : ℝ) * (1 - (p⁻¹).toReal)) * normE p ρ := by
136      have := Real.rpow_add hη s ((d : ℝ) * (1 - (p⁻¹).toReal) - s)
137      rw [show s + ((d : ℝ) * (1 - (p⁻¹).toReal) - s) = (d : ℝ) * (1 - (p⁻¹).toReal) by ring] at this
138      rw [mul_left_comm, ← this]; ring
139    have e3 : η ^ s * (if s = 0 then Real.log η / (2 * (N : ℝ)) else 0) =
140        (if s = 0 then Real.log η / (2 * (N : ℝ)) else 0) := by
141      split_ifs with h0
142      · rw [h0]; simp
143      · simp
144    linear_combination e3 - (max Cp0 Cp1) * e1 - (max Cp0 Cp1) * e2
145  rw [commA_scale hd hs0 hsd η hη ρ hρ hN X hX v L hL, key] at u
146  have hM : 0 ≤ normE p (scaleDens η ρ) := ENNReal.toReal_nonneg
147  have hMq : 0 ≤ normE q (fTrunc s 1 : Euc d → ℝ) := ENNReal.toReal_nonneg
148  have hNpos : (0 : ℝ) < N := by exact_mod_cast hN
149  have hunit : 0 ≤ unitE s (max Cp0 Cp1) p (scaleConf η X) (scaleDens η ρ) := by
150    refine le_trans ?_ c
151    positivity
152  rw [key] at hunit
153  have hQ : 0 ≤ scaleQ s (max Cp0 Cp1) p η X ρ := (mul_nonneg_iff_of_pos_left hηs).1 hunit
154  refine ⟨hQ, ?_⟩
155  rw [abs_mul, abs_of_pos hηs] at u
156  have : η ^ s * |commA s X ρ v| ≤ η ^ s * (C * K * scaleQ s (max Cp0 Cp1) p η X ρ) := by
157    calc η ^ s * |commA s X ρ v| ≤ C * K * (η ^ s * scaleQ s (max Cp0 Cp1) p η X ρ) := u
158      _ = η ^ s * (C * K * scaleQ s (max Cp0 Cp1) p η X ρ) := by ring
159  exact le_of_mul_le_mul_left this hηs

176/-- The approximation `(5.density-approximation)`, `(5.approximation-limits)` (`[N] L770-801`):
177`μ̃_m = μ 1_{|x|≤m, |μ|≤m}`, `c_m = ∫ μ̃_m → 1`, `μ_m = μ̃_m/c_m` (for `m` large; the sequence is

179finitely many small `m`); `μ_m` is a bounded compactly supported real density of mass one with
180`|μ_m| ≤ 2|μ|`, `‖μ_m‖_{L^p} → ‖μ‖_{L^p}`, and for every configuration and Lipschitz `v`

186    ∃ ρm : ℕ → Euc d → ℝ, (∀ m, BddCptDensity (ρm m)) ∧

191  exact cmxSA_density_approx hd hs0 hsd p hp ρ h1 hpρ hmass hlog

195/-- Proposition 1.2 `thm:scale` (`[N] L138-148`): let `d ≥ 3`, `0 ≤ s < d-2`, `d/(d-s) < p ≤ ∞`;
196`μ ∈ L¹ ∩ L^p` real of mass one (and `(1.loghyp)` if `s = 0`); `v` Lipschitz with
197`|∇|^{(d-s)/2} v ∈ L^{2d/(d-s-2)}` (`SubTransport d s v K`: `C(v) ≤ K`).  For every `η > 0` and
198every pairwise distinct `X_N`,
199`|𝖠₁[X_N, μ, v]| ≤ C(‖∇v‖_∞ + ‖|∇|^{(d-s)/2}v‖_{L^{2d/(d-s-2)}})
200  (F_N - (log η)/(2N) 1_{s=0} + C_p(η^{-s}/N + ‖μ‖_{L^p} η^{d(1-1/p)-s}))`,
201with `C` depending only on `d, s` (chosen before `p`) and `C_p` only on `d, s, p` (chosen before `μ,
202v, X_N, η`); the bracket (`scaleQ`) is nonnegative.  The conclusion also asserts the absolute
203convergence of the integrals defining `F_N` and `𝖠₁` (`[N] L154`, `integrable_of_Lp`).  The norm
204`C(v)` is represented by an upper bound `K` as in `HRS.Whole.whole_commutator_sub_sharp`
205(`‖∇v‖_∞ ≤ L` for a Lipschitz constant `L`, `‖|∇|^α v‖_{L^{p₁}} ≤ F`, `L + F ≤ K`).
206Technical hypotheses: `1 ≤ N` (the TeX has `N ∈ ℕ`).  Proved from `scale_estimate`,
207`integrable_of_Lp`, `density_approx` (`[N] L801`). -/
208theorem subcoulomb_commutator_scale (hd : 3 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < (d : ℝ) - 2) :
209    ∃ C : ℝ, 0 < C ∧ ∀ p : ℝ≥0∞, ENNReal.ofReal ((d : ℝ) / (d - s)) < p →
210      ∃ Cp : ℝ, 0 < Cp ∧ ∀ ρ : Euc d → ℝ, Integrable ρ → MemLp ρ p volume → ∫ x, ρ x = 1 →
211      (s = 0 → LogIntegrable ρ) →
212      ∀ (v : Euc d → Euc d) (K : ℝ), SubTransport d s v K →
213      ∀ (N : ℕ) (X : Fin N → Euc d), 1 ≤ N → Function.Injective X → ∀ η : ℝ, 0 < η →
214        EnergyIntegrable s X ρ ∧ CommIntegrable s X ρ v ∧
215        0 ≤ scaleQ s Cp p η X ρ ∧ |commA s X ρ v| ≤ C * K * scaleQ s Cp p η X ρ := by
216  obtain ⟨C, hC, S⟩ := scale_estimate hd hs0 hsd
217  refine ⟨C, hC, fun p hp => ?_⟩
218  obtain ⟨Cp, hCp, S'⟩ := S p hp
219  refine ⟨Cp, hCp, fun ρ h1 hpρ hmass hlog v K hv N X hN hX η hη => ?_⟩
220  obtain ⟨L, hL⟩ : ∃ L : ℝ≥0, LipschitzWith L v := by
221    obtain ⟨L, F, -, hL, -, -⟩ := hv
222    exact ⟨L, hL⟩
223  obtain ⟨hE, hCI⟩ := integrable_of_Lp hd hs0 hsd p hp ρ h1 hpρ hlog v L hL X
224  obtain ⟨ρm, hbdd, hnorm, hlim⟩ := density_approx hd hs0 hsd p hp ρ h1 hpρ hmass hlog
225  obtain ⟨hF, hA⟩ := hlim N X v L hL
226  have hm := fun m => S' (ρm m) (hbdd m) N X hN hX v K hv η hη
227  have hQ : Tendsto (fun m => scaleQ s Cp p η X (ρm m)) atTop (𝓝 (scaleQ s Cp p η X ρ)) := by
228    unfold scaleQ
229    exact (hF.sub tendsto_const_nhds).add
230      (tendsto_const_nhds.mul (tendsto_const_nhds.add (hnorm.mul tendsto_const_nhds)))
231  refine ⟨hE, hCI, ge_of_tendsto hQ (Eventually.of_forall fun m => (hm m).1), ?_⟩
232  exact le_of_tendsto_of_tendsto hA.abs (tendsto_const_nhds.mul hQ)
233    (Eventually.of_forall fun m => (hm m).2)

248/-- Theorem 1.1 `thm:main` (`[N] L100-119`): let `d ≥ 3`, `0 ≤ s < d-2`.  There are constants
249`C, C₀ > 0` depending only on `d, s` such that: for every real `μ ∈ L¹ ∩ L^∞` with `∫ μ = 1` (and
250`(1.loghyp)` if `s = 0`), every Lipschitz `v` with `|∇|^{(d-s)/2} v ∈ L^{2d/(d-s-2)}`
251(`C(v) ≤ K`), `λ = (N‖μ‖_{L^∞})^{-1/d}` and every pairwise distinct `X_N`,
252`|𝖠₁[X_N, μ, v]| ≤ C(‖∇v‖_∞ + ‖|∇|^{(d-s)/2}v‖_{L^{2d/(d-s-2)}})
253  (F_N(X_N, μ) - (log λ)/(2N) 1_{s=0} + C₀‖μ‖_{L^∞} λ^{d-s})`,
254and the second bracket is nonnegative.  The conclusion also asserts the absolute convergence of the
255integrals defining `F_N` and `𝖠₁` (so that no Bochner integral is a junk value).
256Technical hypotheses: `1 ≤ N`.  Proved from Proposition 1.2 with `p = ∞`, `η = λ`
257(`[N] L150`, L761-766): `λ^{-s}/N = ‖μ‖_{L^∞}λ^{d-s}` and `C₀ = 2C_∞`. -/
258theorem subcoulomb_commutator (hd : 3 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < (d : ℝ) - 2) :
259    ∃ C C₀ : ℝ, 0 < C ∧ 0 < C₀ ∧ ∀ ρ : Euc d → ℝ, Integrable ρ → MemLp ρ ⊤ volume →
260      ∫ x, ρ x = 1 → (s = 0 → LogIntegrable ρ) →
261      ∀ (v : Euc d → Euc d) (K : ℝ), SubTransport d s v K →
262      ∀ (N : ℕ) (X : Fin N → Euc d), 1 ≤ N → Function.Injective X →
263        EnergyIntegrable s X ρ ∧ CommIntegrable s X ρ v ∧
264        0 ≤ modEnergy s X ρ - (if s = 0 then Real.log (lamN d N ρ) / (2 * (N : ℝ)) else 0) +
265              C₀ * normE ⊤ ρ * lamN d N ρ ^ ((d : ℝ) - s) ∧
266        |commA s X ρ v| ≤ C * K *
267          (modEnergy s X ρ - (if s = 0 then Real.log (lamN d N ρ) / (2 * (N : ℝ)) else 0) +
268            C₀ * normE ⊤ ρ * lamN d N ρ ^ ((d : ℝ) - s)) := by
269  obtain ⟨C, hC, S⟩ := subcoulomb_commutator_scale hd hs0 hsd
270  obtain ⟨Cp, hCp, S'⟩ := S ⊤ (by simp)
271  refine ⟨C, 2 * Cp, hC, by positivity, fun ρ h1 hinf hmass hlog v K hv N X hN hX => ?_⟩
272  have hM := normE_top_pos ρ h1 hinf hmass
273  have hNpos : (0 : ℝ) < N := by exact_mod_cast hN
274  have hd0 : (d : ℝ) ≠ 0 := by
275    have : (3 : ℝ) ≤ d := by exact_mod_cast hd
276    linarith
277  have hNM : 0 < (N : ℝ) * normE ⊤ ρ := mul_pos hNpos hM
278  have hlam : 0 < lamN d N ρ := Real.rpow_pos_of_pos hNM _
279  have hlamd : lamN d N ρ ^ (d : ℝ) = ((N : ℝ) * normE ⊤ ρ)⁻¹ := by
280    unfold lamN
281    rw [← Real.rpow_mul hNM.le, show (-(1 : ℝ) / d) * d = -1 by field_simp, Real.rpow_neg_one]
282  have hkey : lamN d N ρ ^ (-s) / (N : ℝ) = normE ⊤ ρ * lamN d N ρ ^ ((d : ℝ) - s) := by
283    rw [sub_eq_add_neg, Real.rpow_add hlam, hlamd]
284    field_simp
285  obtain ⟨hE, hCI, hQ, hA⟩ := S' ρ h1 hinf hmass hlog v K hv N X hN hX (lamN d N ρ) hlam
286  have hQe : scaleQ s Cp ⊤ (lamN d N ρ) X ρ =
287      modEnergy s X ρ - (if s = 0 then Real.log (lamN d N ρ) / (2 * (N : ℝ)) else 0) +
288        2 * Cp * normE ⊤ ρ * lamN d N ρ ^ ((d : ℝ) - s) := by
289    unfold scaleQ
290    simp only [ENNReal.inv_top, ENNReal.toReal_zero, sub_zero, mul_one]
291    rw [hkey]
292    ring
293  rw [hQe] at hQ hA
294  exact ⟨hE, hCI, hQ, hA⟩
