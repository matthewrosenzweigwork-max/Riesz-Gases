241      Λ * C * (‖z‖ ^ (β - s - 1) + ‖z‖ ^ β) + B * C * ((1 / 4 : ℝ) ^ (-s - 1) + 1) := by
242  have hz0 : 0 < ‖z‖ := norm_pos_iff.mpr hz
