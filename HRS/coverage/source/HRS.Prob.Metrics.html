6# Probability metrics on Banach spaces: `C_b^k` smooth distances and Wasserstein distances
7
8Source: HRS periodic main paper, display `H:eq:metrics` (source lines 507-516):
9
10* `‖Φ‖_{C_b^k(B)} = ∑_{j=0}^k sup_{z∈B} ‖D^jΦ(z)‖` (SUM over orders, multilinear
11  operator norms);
12* `d_k^B(P,Q) = sup_{‖Φ‖_{C_b^k(B)} ≤ 1} |∫Φ d(P-Q)|`;
13* `W_p^B`, `1 ≤ p < ∞`, uses the norm of `B` and the usual infimum over couplings.
14
15Also: Lemma 4.27 `M:R:lem:smooth-translation` (source lines 5660-5682),
16Lemma A.26 `M:B:lem:dc2-to-w1` (10704-10710), Lemma A.31
17`M:B:newprob:gaussian-covariance` (11003-11014), Lemma A.37 `M:B:lem:yurinskii`
18(11573-11579).
19
20## Representation
21
22* Test functions are arbitrary `Φ : B → ℝ`; the `C_b^k` unit ball is the set of
23  `Φ` with `ContDiff ℝ k Φ` and `cbNorm k Φ ≤ 1` (`ContDiff` guarantees the
24  derivatives `iteratedFDeriv ℝ j Φ` exist; `iteratedFDeriv` is the multilinear
25  derivative with operator norm).
26* `smoothDist k P Q` is a real-valued supremum; for finite measures `P, Q` the set
27  is bounded (by `P univ + Q univ`), so the real `iSup` is the true supremum.
28* `wassersteinE p P Q : ℝ≥0∞` is the infimum over couplings of
29  `(∫⁻ ‖x-y‖^p dπ)^{1/p}`; it is `∞` when no coupling has finite cost, exactly as
30  in the source.  `wasserstein p P Q := (wassersteinE p P Q).toReal` is the real
31  version (0 when infinite; theorem statements below use the `ℝ≥0∞` version where
32  finiteness is part of the claim).

46/-- `‖Φ‖_{C_b^k(B)} = ∑_{j=0}^k sup_z ‖D^jΦ(z)‖` (source line 510). -/
47def cbNorm (k : ℕ) (Φ : B → ℝ) : ℝ :=
48  ∑ j ∈ Finset.range (k + 1), ⨆ z : B, ‖iteratedFDeriv ℝ j Φ z‖
49
50/-- The `C_b^k(B)` unit ball: `C^k` functions with `‖Φ‖_{C_b^k} ≤ 1`. The boundedness conjunct
51encodes the subscript `b` of `C_b^k` and prevents the junk value `0` of the real `iSup` in
52`cbNorm` for unbounded derivatives. -/
53def CbUnitBall (B : Type*) [NormedAddCommGroup B] [NormedSpace ℝ B] (k : ℕ) : Set (B → ℝ) :=
54  {Φ | ContDiff ℝ k Φ ∧ (∀ j ≤ k, BddAbove (Set.range fun z => ‖iteratedFDeriv ℝ j Φ z‖)) ∧
55    cbNorm k Φ ≤ 1}
56
57/-- `d_k^B(P,Q) = sup_{‖Φ‖_{C_b^k(B)} ≤ 1} |∫Φ dP - ∫Φ dQ|` (source line 511). -/
58def smoothDist (k : ℕ) (P Q : Measure B) : ℝ :=
59  ⨆ Φ : CbUnitBall B k, |∫ z, Φ.1 z ∂P - ∫ z, Φ.1 z ∂Q|

61/-- `π` is a coupling of `P` and `Q`. -/
62def IsCoupling (π : Measure (B × B)) (P Q : Measure B) : Prop :=
63  π.map Prod.fst = P ∧ π.map Prod.snd = Q
64
65/-- The set of couplings of `P` and `Q`. -/
66def couplings (P Q : Measure B) : Set (Measure (B × B)) := {π | IsCoupling π P Q}
67
68/-- `W_p^B(P,Q)` in `ℝ≥0∞`: `inf_π (∫ ‖x - y‖^p dπ)^{1/p}` over couplings. -/
69def wassersteinE (p : ℝ) (P Q : Measure B) : ℝ≥0∞ :=
70  ⨅ π : couplings P Q, (∫⁻ z, ENNReal.ofReal (‖z.1 - z.2‖ ^ p) ∂π.1) ^ (1 / p)
71
72/-- Real-valued `W_p^B` (equal to `0` when `wassersteinE = ∞`). -/
73def wasserstein (p : ℝ) (P Q : Measure B) : ℝ := (wassersteinE p P Q).toReal

103/-- Hölder/Jensen for `lintegral` on a probability space, measurable case. -/
104theorem lintegral_rpow_mono_meas {α : Type*} [MeasurableSpace α] (μ : Measure α)
105    [IsProbabilityMeasure μ] {h : α → ℝ≥0∞} (hh : Measurable h) {p q : ℝ} (hp : 0 < p)
106    (hpq : p ≤ q) :
107    (∫⁻ a, h a ^ p ∂μ) ^ (1 / p) ≤ (∫⁻ a, h a ^ q ∂μ) ^ (1 / q) := by
108  rcases hpq.eq_or_lt with rfl | hlt
109  · exact le_rfl
110  have hq : 0 < q := hp.trans hlt
111  set r : ℝ := q / p with hr
112  have hr1 : 1 < r := by rw [hr, lt_div_iff₀ hp]; linarith
113  have hconj : r.HolderConjugate (r / (r - 1)) :=
114    Real.HolderConjugate.conjExponent hr1 |>.symm.symm
115  have H := ENNReal.lintegral_mul_le_Lp_mul_Lq μ hconj (f := fun a => h a ^ p) (g := fun _ => 1)
116    (hh.pow_const p).aemeasurable aemeasurable_const
117  have e1 : ∀ a, (h a ^ p) ^ r = h a ^ q := fun a => by
118    rw [← ENNReal.rpow_mul]; congr 1; rw [hr]; field_simp
119  simp only [Pi.mul_apply, mul_one, e1, ENNReal.one_rpow, lintegral_const, measure_univ] at H
120  have e2 : 1 / r = p / q := by rw [hr]; field_simp
121  rw [e2] at H
122  calc (∫⁻ a, h a ^ p ∂μ) ^ (1 / p) ≤ ((∫⁻ a, h a ^ q ∂μ) ^ (p / q)) ^ (1 / p) := by
123        gcongr
124    _ = (∫⁻ a, h a ^ q ∂μ) ^ (1 / q) := by
125        rw [← ENNReal.rpow_mul]; congr 1; field_simp

127/-- Hölder/Jensen for the (lower) `lintegral` of an arbitrary, not necessarily measurable,
128function on a probability space. -/
129theorem lintegral_rpow_mono {α : Type*} [MeasurableSpace α] (μ : Measure α)
130    [IsProbabilityMeasure μ] (c : α → ℝ≥0∞) {p q : ℝ} (hp : 0 < p) (hpq : p ≤ q) :
131    (∫⁻ a, c a ^ p ∂μ) ^ (1 / p) ≤ (∫⁻ a, c a ^ q ∂μ) ^ (1 / q) := by
132  have hq : 0 < q := hp.trans_le hpq
133  obtain ⟨g, hgm, hgc, hint⟩ := exists_measurable_le_lintegral_eq μ (fun a => c a ^ p)
134  set h : α → ℝ≥0∞ := fun a => g a ^ (1 / p) with hh
135  have hhm : Measurable h := hgm.pow_const _
136  have hhp : ∀ a, h a ^ p = g a := fun a => by
137    rw [hh]; simp only; rw [← ENNReal.rpow_mul]
138    rw [one_div, inv_mul_cancel₀ hp.ne', ENNReal.rpow_one]
139  have hhq : ∀ a, h a ^ q ≤ c a ^ q := fun a => by
140    have : h a ≤ c a := by
141      have := hgc a
142      calc h a = g a ^ (1 / p) := rfl
143        _ ≤ (c a ^ p) ^ (1 / p) := by gcongr
144        _ = c a := by rw [← ENNReal.rpow_mul, mul_one_div_cancel hp.ne', ENNReal.rpow_one]
145    gcongr
146  calc (∫⁻ a, c a ^ p ∂μ) ^ (1 / p) = (∫⁻ a, h a ^ p ∂μ) ^ (1 / p) := by
147        rw [hint]; simp only [hhp]
148    _ ≤ (∫⁻ a, h a ^ q ∂μ) ^ (1 / q) := lintegral_rpow_mono_meas μ hhm hp hpq
149    _ ≤ _ := ENNReal.rpow_le_rpow (lintegral_mono fun a => hhq a) (by positivity)
