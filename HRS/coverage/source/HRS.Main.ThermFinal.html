26theorem mixture_limit_proof (S : PreparedData d) (b : Torus d → ℝ) (hb : Torus.IsSmooth b)
27    (hb0 : pair S.μ0 b = 0) (a₀ : ℝ) (m : ℝ) (hm : (d : ℝ) / 2 < m)
28    (γ : Measure (NegSobolev d m)) (hγ : IsCenteredGaussianWithCov m γ (iidCov S.μ0)) :
29    (∀ N : ℕ, 1 ≤ N → (∀ x, |a₀ * b x| < Real.sqrt N) →
30      IsProbabilityMeasure (mixturePrep S.μ0 b a₀ N) ∧
31        firstMarginal (mixturePrep S.μ0 b a₀ N) = S.μ0) ∧
32    ConvergesInLaw
33      (fun N : ℕ => law (mixturePrep S.μ0 b a₀ N) (fun X => etaE m X S.μ0))
34      ((1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
35        (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x))) := by
36  refine ⟨fun N hN hpos => therm_mixture_marginal S hb hb0 a₀ N hN hpos, ?_⟩
37  have hbc : Continuous b := therm_continuous_of_smooth hb
38  have hbm : Measurable b := hbc.measurable
39  have hρc : Continuous S.ρ0 := therm_continuous_of_smooth S.hρ0.smooth
40  have hm2 : (d : ℝ) < 2 * m := by
41    have : (0 : ℝ) ≤ d := Nat.cast_nonneg d
42    linarith
43  -- bounds
44  obtain ⟨B₀, hB₀⟩ := isCompact_univ.exists_bound_of_continuousOn hbc.continuousOn
45  set B : ℝ := |B₀| with hBdef
46  have hB : ∀ x, |b x| ≤ B := fun x => by
47    have := hB₀ x (Set.mem_univ x)
48    rw [Real.norm_eq_abs] at this
49    exact this.trans (le_abs_self _)
50  have hB0 : 0 ≤ B := abs_nonneg _
51  obtain ⟨xmin, -, hxmin⟩ := isCompact_univ.exists_isMinOn Set.univ_nonempty hρc.continuousOn
52  obtain ⟨xmax, -, hxmax⟩ := isCompact_univ.exists_isMaxOn Set.univ_nonempty hρc.continuousOn
53  have hc : 0 < S.ρ0 xmin := S.hρ0.pos xmin
54  have hbL : MemLp b ⊤ (volume : Measure (Torus d)) :=
55    MemLp.of_bound hbc.aestronglyMeasurable B
56      (Eventually.of_forall fun x => by rw [Real.norm_eq_abs]; exact hB x)
57  have hnorm : (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ≤ B := by
58    have h1 : eLpNorm b ⊤ (volume : Measure (Torus d)) ≤ ENNReal.ofReal B := by
59      rw [eLpNorm_exponent_top]
60      exact eLpNormEssSup_le_of_ae_bound
61        (Eventually.of_forall fun x => by rw [Real.norm_eq_abs]; exact hB x)
62      all_goals first | exact hbc.aestronglyMeasurable | skip
63    exact ENNReal.toReal_le_of_le_ofReal hB0 h1
64  obtain ⟨Cm, hCm0, hCm⟩ := HRS.iid_field_mixture_proof S.hd m hm (S.ρ0 xmin) (S.ρ0 xmax) hc a₀ b
65    hbm hbL
66  -- threshold
67  set N0 : ℕ := ⌈4 * a₀ ^ 2 * B ^ 2⌉₊ + 2 with hN0
68  have hNge : ∀ N : ℕ, N0 ≤ N → 2 ≤ N ∧ 4 * a₀ ^ 2 * B ^ 2 + 2 ≤ (N : ℝ) := fun N hN => by
69    refine ⟨by omega, ?_⟩
70    have h1 : (4 * a₀ ^ 2 * B ^ 2 : ℝ) ≤ ⌈4 * a₀ ^ 2 * B ^ 2⌉₊ := Nat.le_ceil _
71    have h2 : ((N0 : ℕ) : ℝ) ≤ N := by exact_mod_cast hN
72    have h3 : ((N0 : ℕ) : ℝ) = ⌈4 * a₀ ^ 2 * B ^ 2⌉₊ + 2 := by rw [hN0]; push_cast; ring
73    linarith
74  have hposN : ∀ N : ℕ, N0 ≤ N → ∀ x, |a₀ * b x| < Real.sqrt N := fun N hN x => by
75    obtain ⟨-, h⟩ := hNge N hN
76    rw [abs_mul]
77    have h1 : |a₀| * |b x| ≤ |a₀| * B := mul_le_mul_of_nonneg_left (hB x) (abs_nonneg _)
78    have h2 : (|a₀| * B) ^ 2 < N := by
79      have : (|a₀| * B) ^ 2 = a₀ ^ 2 * B ^ 2 := by rw [mul_pow, sq_abs]
80      have h0 : 0 ≤ a₀ ^ 2 * B ^ 2 := by positivity
81      linarith
82    exact h1.trans_lt (Real.lt_sqrt (by positivity) |>.2 h2)
83  -- the limit measure
84  haveI hμ0 : IsProbabilityMeasure S.μ0 := HRS.isProb_withDensityReal S.hρ0.density
85  haveI hγp : IsProbabilityMeasure γ := hγ.1.1.toIsProbabilityMeasure
86  set ν : Measure (NegSobolev d m) :=
87    (1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
88        (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) with hν
89  haveI hνp : IsProbabilityMeasure ν := by
90    refine ⟨?_⟩
91    have e1 : (γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x))) Set.univ = 1 := by
92      rw [Measure.map_apply (by fun_prop) MeasurableSet.univ]; simp
93    have e2 : (γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x))) Set.univ = 1 := by
94      rw [Measure.map_apply (by fun_prop) MeasurableSet.univ]; simp
95    simp only [hν, Measure.add_apply, Measure.smul_apply, smul_eq_mul, e1, e2, mul_one]
96    rw [one_div, ENNReal.inv_two_add_inv_two]
97  have hmeas : ∀ N : ℕ, 1 ≤ N → Measurable (fun X : Config d N => etaE m X S.μ0) := fun N hN => by
98    have := measurable_ofMeasureDiff_empirical (N := N) hm2 S.μ0 hN
99    unfold etaE flucE
100    exact this.const_smul (Real.sqrt N)
101  have hprob : ∀ᶠ N : ℕ in atTop,
102      IsProbabilityMeasure (law (mixturePrep S.μ0 b a₀ N) (fun X => etaE m X S.μ0)) := by
103    filter_upwards [eventually_ge_atTop N0] with N hN
104    have hN1 : 1 ≤ N := by have := (hNge N hN).1; omega
105    have hP := (therm_mixture_marginal S hb hb0 a₀ N hN1 (hposN N hN)).1
106    haveI := hP
107    infer_instance
108  have hbd : ∀ᶠ N : ℕ in atTop,
109      wassersteinE 1 (law (mixturePrep S.μ0 b a₀ N) (fun X => etaE m X S.μ0)) ν ≤
110        ENNReal.ofReal (Cm * iidRateW d m N) := by
111    filter_upwards [eventually_ge_atTop N0] with N hN
112    obtain ⟨hN2, hN4⟩ := hNge N hN
113    have h4 : 4 * a₀ ^ 2 * (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ^ 2 ≤ N := by
114      have h1 : (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ^ 2 ≤ B ^ 2 :=
115        pow_le_pow_left₀ ENNReal.toReal_nonneg hnorm 2
116      have h2 : 4 * a₀ ^ 2 * (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ^ 2 ≤
117          4 * a₀ ^ 2 * B ^ 2 := by gcongr
118      linarith
119    have := (hCm S.ρ0 S.hρ0.density (fun x => by simpa using hxmin (Set.mem_univ x))
120      (fun x => by simpa using hxmax (Set.mem_univ x)) hb0 γ hγ N hN2 h4).2.1
121    rw [← therm_mixturePrep_eq S b hbm a₀ N] at this
122    simp only [therm_ofFun_smul] at this
123    have e : ν = (2 : ℝ≥0∞)⁻¹ • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
124        (2 : ℝ≥0∞)⁻¹ • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) := by
125      simp [hν, one_div]
126    rw [e]
127    exact this
128  have hW : Tendsto (fun N : ℕ => wassersteinE 1 (law (mixturePrep S.μ0 b a₀ N)
129      (fun X => etaE m X S.μ0)) ν) atTop (𝓝 0) := by
130    have h0 : Tendsto (fun N : ℕ => ENNReal.ofReal (Cm * iidRateW d m N)) atTop (𝓝 0) := by
131      have := (therm_iidRateW_tendsto hm).const_mul Cm
132      rw [mul_zero] at this
133      simpa using ENNReal.tendsto_ofReal this
134    exact tendsto_of_tendsto_of_tendsto_of_le_of_le' tendsto_const_nhds h0
135      (Eventually.of_forall fun _ => zero_le) hbd
136  exact therm_convergesInLaw_of_W1 _ ν hprob hW

158        ENNReal.ofReal C := by
159  haveI hμ0 : IsProbabilityMeasure S.μ0 := HRS.isProb_withDensityReal S.hρ0.density
160  have hm2 : (d : ℝ) < 2 * m := by
161    have : (0 : ℝ) ≤ d := Nat.cast_nonneg d
162    linarith
163  obtain ⟨A, hA0, hA⟩ : ∃ A : ℝ, 0 ≤ A ∧ ∀ N : ℕ, 2 ≤ N → ∀ k : NonzeroLattice d,
164      ∫⁻ X, ENNReal.ofReal (therm_eta S.μ0 k X ^ 2) ∂(modulatedGibbs (periodicRiesz d S.s) N β₀ S.μ0) ≤
165        ENNReal.ofReal A := by
166    rcases hβ₀.eq_or_lt with h | h

168      refine ⟨4, by norm_num, fun N hN k => ?_⟩
169      rw [modulatedGibbs_zero]
170      exact therm_mode_bound_iid S.μ0 (by omega) k
171    · exact therm_mode_bound_gibbs S hs h
172  refine ⟨A * ∑' k : NonzeroLattice d, Lattice.jp k.1 ^ (-(2 * m)), ?_, fun N hN => ?_⟩
173  · exact mul_nonneg hA0 (tsum_nonneg fun k => Real.rpow_nonneg (Torus.jp_pos k.1).le _)

176theorem standard_preparation_norms_proof (S : PreparedData d) (hs : S.s < 2 * (d : ℝ) / 3) (m : ℝ)
177    (hm : (d : ℝ) + 2 < m) (β₀ : ℝ) (hβ₀ : 0 ≤ β₀) :
178    ∃ C : ℝ, ∀ N : ℕ, 2 ≤ N →
179      (∫⁻ X, ‖etaE m X S.μ0‖ₑ ∂(modulatedGibbs (periodicRiesz d S.s) N β₀ S.μ0)) ≤
180        ENNReal.ofReal C ∧
181      ((0 < β₀ ∨ S.s = 0) →
182        (∫⁻ X, ‖etaE m X S.μ0‖ₑ ^ (2 : ℝ) ∂(modulatedGibbs (periodicRiesz d S.s) N β₀ S.μ0)) ≤
183          ENNReal.ofReal C) := by
184  obtain ⟨C₂, hC₂0, hC₂⟩ := therm_prep_sq_bound S hs m hm hβ₀
185  refine ⟨C₂ + 1, fun N hN => ⟨?_, fun _ => (hC₂ N hN).trans (ENNReal.ofReal_le_ofReal (by linarith))⟩⟩
186  haveI := therm_prep_prob S hβ₀ (N := N) (by omega)
187  have h1 := HRS.lintegral_rpow_mono (modulatedGibbs (periodicRiesz d S.s) N β₀ S.μ0)
188    (fun X : Config d N => ‖etaE m X S.μ0‖ₑ) (p := 1) (q := 2) (by norm_num) (by norm_num)
189  simp only [ENNReal.rpow_one, div_one, one_div] at h1
190  refine h1.trans ?_
191  have h2 := ENNReal.rpow_le_rpow (hC₂ N hN) (by norm_num : (0 : ℝ) ≤ 2⁻¹)
192  refine h2.trans ?_
193  rw [ENNReal.ofReal_rpow_of_nonneg hC₂0 (by norm_num), show C₂ ^ (2⁻¹ : ℝ) = Real.sqrt C₂ by
194    rw [Real.sqrt_eq_rpow, one_div]]
195  refine ENNReal.ofReal_le_ofReal ?_
196  nlinarith [Real.sq_sqrt hC₂0, sq_nonneg (Real.sqrt C₂ - 1), Real.sqrt_nonneg C₂]

200/-- `thermalization_recentered` with the repair `d/2 < m` (see `ThermPlan.lean`: for `m ≤ d/2` the objects are
201`ofSeq` junk).  The frozen statement without `hm` is NOT claimed. -/
202theorem thermalization_recentered_hm_proof (S : PreparedData d) (m : ℝ) (hm : (d : ℝ) / 2 < m)
203    (T : ℝ) (hT : 0 < T)
204    (ρβ : Torus d → ℝ) (hρβ : IsThermalDensity (periodicRiesz d S.s) S.V S.β ρβ) (a : ℝ) (ha : 0 ≤ a)
205    (N : ℕ) (hN : 2 ≤ N)
206    {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
207    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N)
208    (hflow : SelectedFlow P S.dyn (a + T) X₀ W X)
209    (Q : Measure (PathSpace d m T)) [IsProbabilityMeasure Q] :
210    let shift : ℝ≥0∞ := ENNReal.ofReal (Real.sqrt N) *
211      ⨆ t : Set.Icc (0 : ℝ) T, ‖NegSobolev.ofMeasureDiff m (withDensityReal (S.ρ (a + t)))
212        (withDensityReal ρβ)‖ₑ
213    let recentered := law P (windowFluc m T a (Real.sqrt N) X (fun _ => ρβ))
214    ENNReal.ofReal (smoothDist 2 recentered Q) ≤
215      ENNReal.ofReal (smoothDist 2 (S.etaLaw m T a P X) Q) + shift ∧
216    wassersteinE 1 recentered Q ≤ wassersteinE 1 (S.etaLaw m T a P X) Q + shift ∧
217    wassersteinE 2 recentered Q ≤ wassersteinE 2 (S.etaLaw m T a P X) Q + shift :=
218  therm_recentered_hm S m hm T hT ρβ hρβ a ha N hN P X₀ W X hflow Q
