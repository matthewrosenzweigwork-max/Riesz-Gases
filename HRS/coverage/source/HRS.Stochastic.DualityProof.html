23/-- The pointwise (in time) algebra: the difference of the empirical Itô integrand and the
24mean-field integrand, both for the test function with time derivative `-𝓛_r f^r`, is
25`flucInteraction`. -/
26theorem fluc_pointwise (hd : 1 ≤ d) (hN : 0 < N) {D : DynData d} {ρ : Torus d → ℝ}
27    (hρ : IsProbDensity ρ) (hρ1 : Torus.IsCk 1 ρ) {Λ' : ℝ≥0}
28    (hLip : LipschitzWith Λ' (Torus.lift ρ)) {ψ : Torus d → ℝ} (hψ : Torus.IsCk 2 ψ)
29    (X : Config d N) :
30    (pair (empirical X) (fun x => -(backwardGenerator D ρ ψ x) + D.invβ * Torus.laplacian ψ x
31        - inner ℝ (Torus.grad D.V x) (Torus.grad ψ x))
32      + interactionPairing D ψ (empirical X) (empirical X))
33    - pair (withDensityReal ρ) (fun x => -(backwardGenerator D ρ ψ x) + D.invβ * Torus.laplacian ψ x
34        + inner ℝ (meanFieldDrift D ρ x) (Torus.grad ψ x))
35      = flucInteraction D ψ X ρ := by
36  rw [flucInteraction_eq hd hψ hρ hρ1 hLip hN X]
37  have h1 : (fun x => -(backwardGenerator D ρ ψ x) + D.invβ * Torus.laplacian ψ x
38        - inner ℝ (Torus.grad D.V x) (Torus.grad ψ x))
39      = fun x => PDE.nonlocalTerm D ρ ψ x - inner ℝ (Torus.grad ψ x)
40          (Torus.matVec D.M (convGradConv (periodicRieszGrad d D.s) ρ x)) := by
41    funext x
42    rw [PDE.backwardGenerator_eq]
43    unfold meanFieldDrift
44    rw [inner_sub_left, real_inner_comm (Torus.grad ψ x)]
45    ring
46  have h2 : (fun x => -(backwardGenerator D ρ ψ x) + D.invβ * Torus.laplacian ψ x
47        + inner ℝ (meanFieldDrift D ρ x) (Torus.grad ψ x)) = PDE.nonlocalTerm D ρ ψ := by
48    funext x
49    rw [PDE.backwardGenerator_eq]
50    ring
51  rw [h1, h2, pair_empirical hN X]
52  have h3 : pair (withDensityReal ρ) (PDE.nonlocalTerm D ρ ψ)
53      = -∫ x, ρ x * inner ℝ (Torus.grad ψ x)
54          (Torus.matVec D.M (convGradConv (periodicRieszGrad d D.s) ρ x)) := by
55    rw [← integral_nonlocalTerm_eq_neg hd D hψ hρ hρ1 hLip]
56    exact PDE.MFPairing.integral_withDensityReal hρ.measurable hρ.nonneg _
57  rw [h3]
58  have h4 : ∑ i, (PDE.nonlocalTerm D ρ ψ (X i) - inner ℝ (Torus.grad ψ (X i))
59        (Torus.matVec D.M (convGradConv (periodicRieszGrad d D.s) ρ (X i))))
60      = -∑ i, (inner ℝ (Torus.grad ψ (X i))
61          (Torus.matVec D.M (convGradConv (periodicRieszGrad d D.s) ρ (X i)))
62          - PDE.nonlocalTerm D ρ ψ (X i)) := by
63    rw [← Finset.sum_neg_distrib]
64    exact Finset.sum_congr rfl fun i _ => by ring
65  rw [h4]
66  ring

68/-- **Proposition 3.7** (`M:B:prop:duality-identity`, main paper L1654-1674). -/
69theorem duality_identity
70    (hd : 1 ≤ d) (hN : 2 ≤ N) [NeZero N] {Ω : Type*} [mΩ : MeasurableSpace Ω] {P : Measure Ω}
71    [IsProbabilityMeasure P] {ℱ : Filtration ℝ mΩ}
72    {W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ} {D : DynData d} (hV : Torus.IsCk 2 D.V)
73    {T Λ : ℝ} {ρ : ℝ → Torus d → ℝ}
74    (hρ : IsMeanFieldSolution D T ρ) (hreg : IsRegularMF 3 Λ T ρ)
75    {X₀ : Ω → Config d N} {W : ℝ≥0 → Ω → (Fin N → Euc d)} {X : ℝ≥0 → Ω → Config d N}
76    (hX : IsContactStrongSolution P D T X₀ W X) (hF : IsFilteredSolution P ℱ W' W X)
77    (hocc : D.M = 0 ∨ NullContactOccupation P T X)
78    {t : ℝ} (ht : t ∈ Ioc 0 T) {φ : Torus d → ℝ} (hφ : Torus.IsCk 2 φ)
79    {f : ℝ → Torus d → ℝ} (hf : IsClassicalAdjointSolution D ρ t φ f) :
80    IsProgressiveL2 P ℱ (empiricalNoiseIntegrand D t f X) ∧
81    ∀ᵐ ω ∂P,
82      IntegrableOn (fun r => flucInteraction D (f r) (X (Real.toNNReal r) ω) (ρ r)) (Icc 0 t) ∧
83      flucPair (X (Real.toNNReal t) ω) (ρ t) φ
84        = flucPair (X 0 ω) (ρ 0) (f 0)
85          + empiricalMartingale P ℱ W' D t f X (Real.toNNReal t) ω
86          + ∫ r in (0 : ℝ)..t, flucInteraction D (f r) (X (Real.toNNReal r) ω) (ρ r) := by
87  have ht' : t ∈ Icc 0 T := ⟨ht.1.le, ht.2⟩
88  have hNpos : 0 < N := by omega
89  have hC := hf.isC12Test
90  have hRg := MFReg.of_regular hρ hreg
91  obtain ⟨hprog, hae⟩ := empirical_ito_of_contact hN hV hX hF hocc ht' hC
92  obtain ⟨hint2, heq2⟩ := meanField_pairing_timeDependent hd hV hρ hreg ht' hC
93  refine ⟨hprog, ?_⟩
94  filter_upwards [hae] with ω hω
95  obtain ⟨hint1, heq1⟩ := hω t ⟨ht.1.le, le_rfl⟩
96  -- the pointwise identity on `[0,t]`
97  have hpt : ∀ r ∈ Icc 0 t,
98      (pair (empirical (X (Real.toNNReal r) ω))
99          (fun x => -(backwardGenerator D (ρ r) (f r) x) + D.invβ * Torus.laplacian (f r) x
100            - inner ℝ (Torus.grad D.V x) (Torus.grad (f r) x))
101        + interactionPairing D (f r) (empirical (X (Real.toNNReal r) ω))
102            (empirical (X (Real.toNNReal r) ω)))
103      - pair (withDensityReal (ρ r))
104          (fun x => -(backwardGenerator D (ρ r) (f r) x) + D.invβ * Torus.laplacian (f r) x
105            + inner ℝ (meanFieldDrift D (ρ r) x) (Torus.grad (f r) x))
106      = flucInteraction D (f r) (X (Real.toNNReal r) ω) (ρ r) := by
107    intro r hr
108    have hrT : r ∈ Icc 0 T := ⟨hr.1, hr.2.trans ht.2⟩
109    exact fluc_pointwise hd hNpos (hRg.dens r hrT) (hRg.c1 r hrT) (hRg.lip r hrT)
110      (hC.c2 r hr) (X (Real.toNNReal r) ω)
111  have hint : IntegrableOn (fun r => flucInteraction D (f r) (X (Real.toNNReal r) ω) (ρ r)) (Icc 0 t) := by
112    refine (hint1.sub hint2).congr_fun (fun r hr => ?_) measurableSet_Icc
113    exact hpt r hr
114  refine ⟨hint, ?_⟩
115  have hii1 : IntervalIntegrable (fun r => pair (empirical (X (Real.toNNReal r) ω))
116          (fun x => -(backwardGenerator D (ρ r) (f r) x) + D.invβ * Torus.laplacian (f r) x
117            - inner ℝ (Torus.grad D.V x) (Torus.grad (f r) x))
118        + interactionPairing D (f r) (empirical (X (Real.toNNReal r) ω))
119            (empirical (X (Real.toNNReal r) ω))) volume 0 t := by
120    rw [intervalIntegrable_iff_integrableOn_Icc_of_le ht.1.le]
121    exact hint1
122  have hii2 : IntervalIntegrable (fun r => pair (withDensityReal (ρ r))
123          (fun x => -(backwardGenerator D (ρ r) (f r) x) + D.invβ * Torus.laplacian (f r) x
124            + inner ℝ (meanFieldDrift D (ρ r) x) (Torus.grad (f r) x))) volume 0 t := by
125    rw [intervalIntegrable_iff_integrableOn_Icc_of_le ht.1.le]
126    exact hint2
127  have hsplit : ∫ r in (0 : ℝ)..t, flucInteraction D (f r) (X (Real.toNNReal r) ω) (ρ r)
128      = (∫ r in (0 : ℝ)..t, (pair (empirical (X (Real.toNNReal r) ω))
129          (fun x => -(backwardGenerator D (ρ r) (f r) x) + D.invβ * Torus.laplacian (f r) x
130            - inner ℝ (Torus.grad D.V x) (Torus.grad (f r) x))
131        + interactionPairing D (f r) (empirical (X (Real.toNNReal r) ω))
132            (empirical (X (Real.toNNReal r) ω))))
133        - ∫ r in (0 : ℝ)..t, pair (withDensityReal (ρ r))
134          (fun x => -(backwardGenerator D (ρ r) (f r) x) + D.invβ * Torus.laplacian (f r) x
135            + inner ℝ (meanFieldDrift D (ρ r) x) (Torus.grad (f r) x)) := by
136    rw [← intervalIntegral.integral_sub hii1 hii2]
137    refine intervalIntegral.integral_congr fun r hr => ?_
138    rw [Set.uIcc_of_le ht.1.le] at hr
139    exact (hpt r hr).symm
140  rw [hsplit]
141  have hφt : f t = φ := hf.terminal
142  unfold flucPair
143  rw [← hφt]
144  linarith

243/-- **Proposition 3.7, final form** (`M:B:eq:duality-identity-final`, TeX L1660-1664), with
244`U^{0,t}φ = propagator D ρ 0 t φ` and the adjoint solution `f^r = propagator D ρ r t φ`
245(existence hypothesis `hex`, cf. Proposition 3.9). The factor `σ_N` of the TeX is a scalar
246multiple of both sides and is omitted, as in `PART01_REST_SPEC.md` §3. -/
247theorem duality_identity_final
248    (hd : 1 ≤ d) (hN : 2 ≤ N) [NeZero N] {Ω : Type*} [mΩ : MeasurableSpace Ω] {P : Measure Ω}
249    [IsProbabilityMeasure P] {ℱ : Filtration ℝ mΩ}
250    {W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ} {D : DynData d} (hV : Torus.IsCk 2 D.V)
251    {T Λ : ℝ} {ρ : ℝ → Torus d → ℝ}
252    (hρ : IsMeanFieldSolution D T ρ) (hreg : IsRegularMF 3 Λ T ρ)
253    {X₀ : Ω → Config d N} {W : ℝ≥0 → Ω → (Fin N → Euc d)} {X : ℝ≥0 → Ω → Config d N}
254    (hX : IsContactStrongSolution P D T X₀ W X) (hF : IsFilteredSolution P ℱ W' W X)
255    (hocc : D.M = 0 ∨ NullContactOccupation P T X)
256    {t : ℝ} (ht : t ∈ Ioc 0 T) {φ : Torus d → ℝ} (hφ : Torus.IsCk 2 φ)
257    (hex : ∃ f, IsClassicalAdjointSolution D ρ t φ f) :
258    ∀ᵐ ω ∂P, flucPair (X (Real.toNNReal t) ω) (ρ t) φ
259      = flucPair (X 0 ω) (ρ 0) (propagator D ρ 0 t φ)
260        + empiricalMartingale P ℱ W' D t (fun r => propagator D ρ r t φ) X (Real.toNNReal t) ω
261        + ∫ r in (0 : ℝ)..t, flucInteraction D (propagator D ρ r t φ) (X (Real.toNNReal r) ω) (ρ r) := by
262  have hf : IsClassicalAdjointSolution D ρ t φ (fun r => propagator D ρ r t φ) := by
263    have hp : (fun r => propagator D ρ r t φ) = hex.choose := by
264      funext r
265      unfold propagator
266      rw [dif_pos hex]
267    rw [hp]
268    exact hex.choose_spec
269  have hall := (duality_identity_all hd hN hV hρ hreg hX hF hocc ht hφ hf).2
270  have hterm : propagator D ρ t t φ = φ := hf.terminal
271  filter_upwards [hall] with ω hω
272  have h := (hω t ⟨ht.1.le, le_rfl⟩).2
273  simp only [hterm] at h
274  exact h

318/-- The empirical covariation `[M_N[f], M_N[g]]_τ` for `τ ∈ [0,t]` (pointwise in `ω`). -/
319theorem covProcess_empirical {Ω : Type*} [mΩ : MeasurableSpace Ω] {P : Measure Ω}
320    [IsProbabilityMeasure P] {ℱ : Filtration ℝ mΩ} {W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ}
321    {W : ℝ≥0 → Ω → (Fin N → Euc d)} {X : ℝ≥0 → Ω → Config d N} {D : DynData d}
322    (hF : IsFilteredSolution P ℱ W' W X) {t : ℝ} (ht : 0 ≤ t) {f ft g gt : ℝ → Torus d → ℝ}
323    (hf : IsC12Test t f ft) (hg : IsC12Test t g gt) (ω : Ω) {τ : ℝ} (hτ : τ ∈ Icc 0 t) :
324    covProcess (empiricalNoiseIntegrand D t f X) (empiricalNoiseIntegrand D t g X) τ ω
325      = (2 * D.invβ / (N:ℝ)^2) * ∑ i : Fin N, ∫ r in (0:ℝ)..τ,
326          inner ℝ (Torus.grad (f r) (X (Real.toNNReal r) ω i))
327            (Torus.grad (g r) (X (Real.toNNReal r) ω i)) := by
328  obtain ⟨B, hB⟩ := empiricalNoise_bound (D := D) ht hf (Ω := Ω) X
329  obtain ⟨B', hB'⟩ := empiricalNoise_bound (D := D) ht hg (Ω := Ω) X
330  have hΦ := (isProgressiveL2_empiricalNoiseIntegrand (P := P) (D := D) hF.progressive ht hf).prog
331  have hΦ' := (isProgressiveL2_empiricalNoiseIntegrand (P := P) (D := D) hF.progressive ht hg).prog
332  rw [covProcess_eq_sum_fiber hΦ hΦ' hB hB' ω τ, Finset.mul_sum]
333  refine Finset.sum_congr rfl fun i _ => ?_
334  have hτ0 : (0:ℝ) ≤ τ := hτ.1
335  have hpt : ∀ s ∈ Icc 0 τ, ∑ j : Fin d, empiricalNoiseIntegrand D t f X s ω (i, j)
336        * empiricalNoiseIntegrand D t g X s ω (i, j)
337      = (2 * D.invβ / (N:ℝ)^2) * inner ℝ (Torus.grad (f s) (X (Real.toNNReal s) ω i))
338          (Torus.grad (g s) (X (Real.toNNReal s) ω i)) := by
339    intro s hs
340    have hs' : s ∈ Icc 0 t := ⟨hs.1, hs.2.trans hτ.2⟩
341    have hc : Real.sqrt (2 * D.invβ) / (N:ℝ) * (Real.sqrt (2 * D.invβ) / (N:ℝ))
342        = 2 * D.invβ / (N:ℝ)^2 := by
343      rw [div_mul_div_comm, Real.mul_self_sqrt (by linarith [D.hβ])]
344      ring
345    simp only [empiricalNoiseIntegrand, hs', ite_true, PiLp.inner_apply, RCLike.inner_apply,
346      conj_trivial]
347    rw [← hc, Finset.mul_sum]
348    refine Finset.sum_congr rfl fun j _ => ?_
349    ring
350  rw [setIntegral_congr_fun measurableSet_Icc hpt, integral_const_mul,
351    intervalIntegral.integral_of_le hτ0, integral_Icc_eq_integral_Ioc]

400theorem empiricalMartingale_zero_temp {Ω : Type*} [mΩ : MeasurableSpace Ω] {P : Measure Ω}
401    [IsProbabilityMeasure P] {ℱ : Filtration ℝ mΩ} {W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ}
402    {W : ℝ≥0 → Ω → (Fin N → Euc d)} {X : ℝ≥0 → Ω → Config d N} {D : DynData d}
403    (hF : IsFilteredSolution P ℱ W' W X) (h0 : D.invβ = 0) {t : ℝ} {f : ℝ → Torus d → ℝ}
404    (hprog : IsProgressiveL2 P ℱ (empiricalNoiseIntegrand D t f X)) :
405    ∀ᵐ ω ∂P, ∀ τ ≤ t, empiricalMartingale P ℱ W' D t f X (Real.toNNReal τ) ω = 0 := by
406  have hΦ0 : ∀ (s : ℝ) (ω : Ω) (k : Fin N × Fin d),
407      empiricalNoiseIntegrand D t f X s ω k = 0 := by
408    intro s ω k
409    simp [empiricalNoiseIntegrand, h0]
410  have key : empiricalNoiseIntegrand D t f X
411      = fun s ω => (0:ℝ) • empiricalNoiseIntegrand D t f X s ω := by
412    funext s ω k
413    simp [hΦ0]
414  have h := itoIntegral_smul hF.brownian hprog 0
415  rw [← key] at h
416  filter_upwards [h] with ω hω τ _
417  have hτ := hω (Real.toNNReal τ)
418  rw [zero_mul] at hτ
419  exact hτ
