3/-!
4# `HRS.MH.CondOu.conditioned_ou_third_cumulant`: proof (packet (b))
5
6With `c = e^{-λt}√v > 0`, `ν_t = μ ∗ γ` where `μ` is the law of `c|G₀|` and `γ` is a centred Gaussian.
7The third central moment of `μ ∗ γ` equals that of `μ` (the Gaussian part is symmetric and independent).
8For the half-normal law, `E|G| = 2K`, `E|G|³ = 4K`, `E G² = 1` with `K = (√(2π))⁻¹`, so the third central
9moment is `c³ (16K³ - 2K) = 2c³K(8K² - 1) > 0`, because `8K² = 4/π > 1`.
10-/

142/-- Shifted monomials are integrable. -/
143lemma shift_pow_integrable (μ : Measure ℝ) [IsFiniteMeasure μ]
144    (hμ : ∀ j : ℕ, Integrable (fun x : ℝ => x ^ j) μ) (m : ℝ) (k : ℕ) (hk : k ≤ 3) :
145    Integrable (fun x : ℝ => (x - m) ^ k) μ := by
146  interval_cases k
147  · simpa using (integrable_const (1 : ℝ) : Integrable (fun _ : ℝ => (1 : ℝ)) μ)
148  · exact ((hμ 1).sub (integrable_const m)).congr (ae_of_all _ fun x => by simp)
149  · refine (((hμ 2).sub ((hμ 1).const_mul (2 * m))).add (integrable_const (m ^ 2))).congr
150      (ae_of_all _ fun x => ?_)
151    simp only [Pi.add_apply, Pi.sub_apply]
152    ring
153  · refine ((((hμ 3).sub ((hμ 2).const_mul (3 * m))).add ((hμ 1).const_mul (3 * m ^ 2))).sub
154      (integrable_const (m ^ 3))).congr (ae_of_all _ fun x => ?_)
155    simp only [Pi.add_apply, Pi.sub_apply]

158/-- Expansion of the third central moment in raw moments. -/
159lemma third_central_expand (ρ : Measure ℝ) [IsProbabilityMeasure ρ]
160    (hρ : ∀ j : ℕ, Integrable (fun x : ℝ => x ^ j) ρ) :
161    ∫ x, (x - ∫ y, y ∂ρ) ^ 3 ∂ρ =
162      ∫ x, x ^ 3 ∂ρ - 3 * (∫ x, x ∂ρ) * (∫ x, x ^ 2 ∂ρ) + 2 * (∫ x, x ∂ρ) ^ 3 := by
163  have i1 : Integrable (fun x : ℝ => x) ρ := by simpa using hρ 1
164  have i2 := hρ 2
165  have i3 := hρ 3
166  set m := ∫ y, y ∂ρ with hm
167  have e : ∀ x : ℝ, (x - m) ^ 3 = (x ^ 3 - 3 * m * x ^ 2 + 3 * m ^ 2 * x) - m ^ 3 :=
168    fun x => by ring
169  have iA : Integrable (fun x : ℝ => x ^ 3 - 3 * m * x ^ 2) ρ := i3.sub (i2.const_mul (3 * m))
170  have iB : Integrable (fun x : ℝ => 3 * m ^ 2 * x) ρ := i1.const_mul (3 * m ^ 2)
171  have iAB : Integrable (fun x : ℝ => x ^ 3 - 3 * m * x ^ 2 + 3 * m ^ 2 * x) ρ := iA.add iB
172  simp only [e]
173  rw [integral_sub iAB (integrable_const _), integral_add iA iB,
174    integral_sub i3 (i2.const_mul _), integral_const_mul, integral_const_mul]
175  simp only [integral_const, probReal_univ, one_smul, ← hm, mul_one]
176  ring

178/-- Convolution with an independent centred Gaussian-type summand leaves the third central moment
179unchanged. -/
180theorem conv_third_eq (μ γ : Measure ℝ) [IsProbabilityMeasure μ] [IsProbabilityMeasure γ]
181    (hμ : ∀ j : ℕ, Integrable (fun x : ℝ => x ^ j) μ)
182    (hγ : ∀ j : ℕ, Integrable (fun y : ℝ => y ^ j) γ)
183    (h1 : ∫ y, y ∂γ = 0) (h3 : ∫ y, y ^ 3 ∂γ = 0) :
184    thirdCentralMoment (Measure.conv μ γ) = ∫ x, (x - ∫ y, y ∂μ) ^ 3 ∂μ := by
185  have hconv : ∀ g : ℝ → ℝ, Measurable g →
186      ∫ z, g z ∂(Measure.conv μ γ) = ∫ p : ℝ × ℝ, g (p.1 + p.2) ∂(μ.prod γ) := by
187    intro g hg
188    unfold Measure.conv
189    exact integral_map (measurable_fst.add measurable_snd).aemeasurable hg.aestronglyMeasurable
190  set m := ∫ y, y ∂μ with hm
191  have hx : Integrable (fun x : ℝ => x) μ := by simpa using hμ 1
192  have hy : Integrable (fun y : ℝ => y) γ := by simpa using hγ 1
193  have hz : ∫ x, (x - m) ∂μ = 0 := by
194    rw [integral_sub hx (integrable_const _)]
195    simp [hm]
196  have ip1 : Integrable (fun p : ℝ × ℝ => p.1) (μ.prod γ) := by
197    simpa using hx.mul_prod (integrable_const (1 : ℝ))
198  have ip2 : Integrable (fun p : ℝ × ℝ => p.2) (μ.prod γ) := by
199    simpa using (integrable_const (1 : ℝ)).mul_prod hy
200  have f1 : ∫ p : ℝ × ℝ, p.1 ∂(μ.prod γ) = m := by
201    have := integral_prod_mul (μ := μ) (ν := γ) (fun x : ℝ => x) (fun _ : ℝ => (1 : ℝ))
202    simpa using this
203  have f2 : ∫ p : ℝ × ℝ, p.2 ∂(μ.prod γ) = 0 := by
204    have := integral_prod_mul (μ := μ) (ν := γ) (fun _ : ℝ => (1 : ℝ)) (fun y : ℝ => y)
205    simpa [h1] using this
206  have hmean : ∫ y, y ∂(Measure.conv μ γ) = m := by
207    rw [hconv (fun y => y) measurable_id]
208    rw [integral_add ip1 ip2, f1, f2, add_zero]
209  unfold thirdCentralMoment
210  rw [hmean, hconv (fun z => (z - m) ^ 3) (by fun_prop)]
211  have expand : ∀ p : ℝ × ℝ, (p.1 + p.2 - m) ^ 3 =
212      (p.1 - m) ^ 3 * 1 + (3 * (p.1 - m) ^ 2) * p.2 + (3 * (p.1 - m)) * p.2 ^ 2 + 1 * p.2 ^ 3 :=
213    fun p => by ring
214  have sh3 := shift_pow_integrable μ hμ m 3 (by norm_num)
215  have sh2 := shift_pow_integrable μ hμ m 2 (by norm_num)
216  have sh1 : Integrable (fun x : ℝ => 3 * (x - m)) μ :=
217    ((shift_pow_integrable μ hμ m 1 (by norm_num)).const_mul 3).congr (ae_of_all _ fun x => by simp)
218  have J0 : Integrable (fun p : ℝ × ℝ => (p.1 - m) ^ 3 * 1) (μ.prod γ) :=
219    sh3.mul_prod (integrable_const (1 : ℝ))
220  have J1 : Integrable (fun p : ℝ × ℝ => (3 * (p.1 - m) ^ 2) * p.2) (μ.prod γ) :=
221    (sh2.const_mul 3).mul_prod hy
222  have J2 : Integrable (fun p : ℝ × ℝ => (3 * (p.1 - m)) * p.2 ^ 2) (μ.prod γ) :=
223    sh1.mul_prod (hγ 2)
224  have J3 : Integrable (fun p : ℝ × ℝ => 1 * p.2 ^ 3) (μ.prod γ) :=
225    (integrable_const (1 : ℝ)).mul_prod (hγ 3)
226  have Q : ∀ f g : ℝ → ℝ, ∫ p : ℝ × ℝ, f p.1 * g p.2 ∂(μ.prod γ) =
227      (∫ x, f x ∂μ) * ∫ y, g y ∂γ := fun f g => integral_prod_mul f g
228  have K01 : Integrable (fun p : ℝ × ℝ => (p.1 - m) ^ 3 * 1 + 3 * (p.1 - m) ^ 2 * p.2) (μ.prod γ) :=
229    J0.add J1
230  have K012 : Integrable (fun p : ℝ × ℝ => (p.1 - m) ^ 3 * 1 + 3 * (p.1 - m) ^ 2 * p.2 +
231      3 * (p.1 - m) * p.2 ^ 2) (μ.prod γ) := K01.add J2
232  simp only [expand]
233  rw [integral_add K012 J3, integral_add K01 J2, integral_add J0 J1,
234    Q (fun x => (x - m) ^ 3) (fun _ => 1), Q (fun x => 3 * (x - m) ^ 2) (fun y => y),
235    Q (fun x => 3 * (x - m)) (fun y => y ^ 2), Q (fun _ => 1) (fun y => y ^ 3)]
236  have hz3 : ∫ x, 3 * (x - m) ∂μ = 0 := by
237    rw [integral_const_mul, hz, mul_zero]
238  rw [h1, h3, hz3]
239  simp

246theorem conditioned_ou_third_cumulant_proof :
247    type_of% @HRS.MH.CondOu.conditioned_ou_third_cumulant := by
248  intro d hd s hs0 hsd β hβ k hk t _ht
249  have hnn : 0 ≤ rieszCoeff d s k := HRS.PDE.rieszCoeff_nonneg (by omega) hs0 hsd k
250  have hpos1 : 0 < 1 + β * rieszCoeff d s k := by nlinarith [mul_nonneg hβ.le hnn]
251  have hc : 0 < Real.exp (-PDE.transientRate d s β k * t) * Real.sqrt (1 + β * rieszCoeff d s k)⁻¹ :=
252    mul_pos (Real.exp_pos _) (Real.sqrt_pos.2 (inv_pos.2 hpos1))
253  unfold halfNormalMixtureLaw
254  generalize hcdef : Real.exp (-PDE.transientRate d s β k * t) *
255    Real.sqrt (1 + β * rieszCoeff d s k)⁻¹ = c at hc ⊢
256  obtain ⟨hP, hγ, hγ1, hγ3⟩ := gaussianLaw_facts
257    ((1 + β * rieszCoeff d s k)⁻¹ * (1 - Real.exp (-2 * PDE.transientRate d s β k * t)))
258  have hφ : Measurable (fun z : ℝ => c * |z|) := by fun_prop
259  rw [conv_third_eq _ _ (map_pow_integrable c) hγ hγ1 hγ3, third_central_expand _ (map_pow_integrable c)]
260  have hm1 : ∫ x, x ∂((gaussianReal 0 1).map (fun z : ℝ => c * |z|)) = c * (2 * gaussK) := by
261    rw [integral_map hφ.aemeasurable (by fun_prop : Measurable fun x : ℝ => x).aestronglyMeasurable,
262      integral_const_mul, gauss_abs_one']
263  have hm2 : ∫ x, x ^ 2 ∂((gaussianReal 0 1).map (fun z : ℝ => c * |z|)) = c ^ 2 := by
264    rw [integral_map hφ.aemeasurable (by fun_prop : Measurable fun x : ℝ => x ^ 2).aestronglyMeasurable]
265    simp only [mul_pow, sq_abs]
266    rw [integral_const_mul, gauss_sq_one, mul_one]
267  have hm3 : ∫ x, x ^ 3 ∂((gaussianReal 0 1).map (fun z : ℝ => c * |z|)) = c ^ 3 * (4 * gaussK) := by
268    rw [integral_map hφ.aemeasurable (by fun_prop : Measurable fun x : ℝ => x ^ 3).aestronglyMeasurable]
269    simp only [mul_pow]
270    rw [integral_const_mul, gauss_abs_three]
271  rw [hm1, hm2, hm3]
272  have h8 := gaussK_eight
273  have hK := gaussK_pos
274  have hcub : 0 < 16 * gaussK ^ 3 - 2 * gaussK := by nlinarith [mul_pos hK (sub_pos.2 h8)]
275  have e : c ^ 3 * (4 * gaussK) - 3 * (c * (2 * gaussK)) * c ^ 2 + 2 * (c * (2 * gaussK)) ^ 3 =
276      c ^ 3 * (16 * gaussK ^ 3 - 2 * gaussK) := by ring
277  rw [e]
278  exact mul_pos (pow_pos hc 3) hcub
