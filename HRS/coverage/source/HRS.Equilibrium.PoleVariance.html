268/-- A tilt of a probability measure by a bounded measurable `a ≤ A` has density at most
269`e^{A - ∫ a dν}` (Jensen for the normalizer). -/
270theorem tilted_le_smul_jensen {ν : Measure α} [IsProbabilityMeasure ν] {a : α → ℝ}
271    (ha : Measurable a) {A B : ℝ} (hA : ∀ y, a y ≤ A) (hB : ∀ y, |a y| ≤ B) :
272    ν.tilted a ≤ ENNReal.ofReal (Real.exp (A - ∫ y, a y ∂ν)) • ν := by
273  have hint : Integrable a ν := integrable_of_abs_le ha B hB
274  have hintE : Integrable (fun y => Real.exp (a y)) ν :=
275    integrable_of_abs_le (Real.measurable_exp.comp ha) (Real.exp B) fun y => by
276      rw [abs_of_pos (Real.exp_pos _)]
277      exact Real.exp_le_exp.2 ((le_abs_self _).trans (hB y))
278  have hJ := log_integral_exp_ge hint hintE
279  have hpos := integral_exp_pos hintE
280  have hlow : Real.exp (∫ y, a y ∂ν) ≤ ∫ y, Real.exp (a y) ∂ν :=
281    (Real.le_log_iff_exp_le hpos).1 hJ
282  have hdens : ∀ y, Real.exp (a y) / ∫ z, Real.exp (a z) ∂ν ≤ Real.exp (A - ∫ y, a y ∂ν) := by
283    intro y
284    rw [div_le_iff₀ hpos]
285    calc Real.exp (a y) ≤ Real.exp A := Real.exp_le_exp.2 (hA y)
286      _ = Real.exp (A - ∫ y, a y ∂ν) * Real.exp (∫ y, a y ∂ν) := by
287          rw [← Real.exp_add]; ring_nf
288      _ ≤ _ := mul_le_mul_of_nonneg_left hlow (Real.exp_pos _).le
289  intro S
290  show (ν.withDensity fun x => ENNReal.ofReal (Real.exp (a x) / ∫ z, Real.exp (a z) ∂ν)) S ≤ _
291  rw [withDensity_apply' _ S, Measure.smul_apply, smul_eq_mul]
292  calc ∫⁻ x in S, ENNReal.ofReal (Real.exp (a x) / ∫ z, Real.exp (a z) ∂ν) ∂ν
293      ≤ ∫⁻ _x in S, ENNReal.ofReal (Real.exp (A - ∫ y, a y ∂ν)) ∂ν :=
294        lintegral_mono fun x => ENNReal.ofReal_le_ofReal (hdens x)
295    _ = _ := by rw [setLIntegral_const]

401    rw [Finset.sum_sub_distrib, Finset.sum_add_distrib, hS, inner_sum]

403  have he : ∑ i, inner ℝ ((-β) • Xb) (Xi μ Ψ (y i)) =

407    rw [Finset.sum_sub_distrib, hS, inner_sum]

417    rw [inner_smul_left, real_inner_comm]
418    simp
