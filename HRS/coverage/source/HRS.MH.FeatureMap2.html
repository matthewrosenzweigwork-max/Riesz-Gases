35/-- The two halves of the coordinate sum, each over the positive half-lattice. -/
36theorem fm2_inner_split {a : Lattice d → ℝ} (ha0 : ∀ k, 0 ≤ a k) (has : Summable a) (x y : Torus d) :
37    inner ℝ (featureMap ha0 has x) (featureMap ha0 has y) =
38      ∑' k : {k : Lattice d // Lattice.IsPosHalf k}, featCoord a (Sum.inl k) x * featCoord a (Sum.inl k) y
39        + ∑' k : {k : Lattice d // Lattice.IsPosHalf k}, featCoord a (Sum.inr k) x * featCoord a (Sum.inr k) y := by
40  rw [fm2_inner_coord ha0 has x y]
41  exact Summable.tsum_sum (f := fun i : HalfIdx d => featCoord a i x * featCoord a i y)
42    ((fm2_summable_half ha0 has x y).comp_injective Sum.inl_injective)
43    ((fm2_summable_half ha0 has x y).comp_injective Sum.inr_injective)

45/-- Per-frequency identity: the cos and sin coordinates of `k` contribute `2 a_k cos_k(x - y)`. -/
46theorem fm2_term {a : Lattice d → ℝ} (ha0 : ∀ k, 0 ≤ a k) (x y : Torus d)
47    (k : {k : Lattice d // Lattice.IsPosHalf k}) :
48    featCoord a (Sum.inl k) x * featCoord a (Sum.inl k) y
49        + featCoord a (Sum.inr k) x * featCoord a (Sum.inr k) y =
50      2 * (a k.1 * Torus.cosMode k.1 (x - y)) := by
51  have hsq : Real.sqrt (2 * a k.1) * Real.sqrt (2 * a k.1) = 2 * a k.1 :=
52    Real.mul_self_sqrt (mul_nonneg zero_le_two (ha0 k.1))
53  simp only [featCoord]
54  rw [StaticLimitAux.cosMode_sub']
55  linear_combination (Torus.cosMode k.1 x * Torus.cosMode k.1 y
56    + Torus.sinMode k.1 x * Torus.sinMode k.1 y) * hsq

58/-- The inner product of the feature map is the spectral kernel at `x - y`. -/
59theorem fm2_inner_spec {a : Lattice d → ℝ} (hsym : IsSpectralSymbol a) (x y : Torus d) :
60    inner ℝ (featureMap hsym.nonneg hsym.summable x) (featureMap hsym.nonneg hsym.summable y) =
61      StaticLimitAux.specKernel a (x - y) := by
62  have ha0 := hsym.nonneg
63  have has := hsym.summable
64  have hL : Summable fun k : Lattice d => a k * Torus.cosMode k (x - y) :=
65    Summable.of_norm_bounded has fun k => norm_mul_cos_le ha0 k (x - y)
66  have hh : Summable fun k : {k : Lattice d // Lattice.IsPosHalf k} => a k.1 * Torus.cosMode k.1 (x - y) :=
67    fm1_summable_pos ha0 has (fun k => Torus.cosMode k (x - y)) (fun k => by
68      have := Torus.norm_cosMode_le k (x - y)
69      rwa [Real.norm_eq_abs] at this)
70  have h0 : (fun k : Lattice d => a k * Torus.cosMode k (x - y)) 0 = 0 := by
71    simp [hsym.zero]
72  have hev : ∀ k : Lattice d, (fun k : Lattice d => a k * Torus.cosMode k (x - y)) (-k) =
73      (fun k : Lattice d => a k * Torus.cosMode k (x - y)) k := by
74    intro k
75    show a (-k) * Torus.cosMode (-k) (x - y) = a k * Torus.cosMode k (x - y)
76    rw [hsym.even k, Torus.cosMode_neg]
77  have hl : Summable fun k : {k : Lattice d // Lattice.IsPosHalf k} =>
78      featCoord a (Sum.inl k) x * featCoord a (Sum.inl k) y :=
79    (fm2_summable_half ha0 has x y).comp_injective Sum.inl_injective
80  have hr : Summable fun k : {k : Lattice d // Lattice.IsPosHalf k} =>
81      featCoord a (Sum.inr k) x * featCoord a (Sum.inr k) y :=
82    (fm2_summable_half ha0 has x y).comp_injective Sum.inr_injective
83  have e1 : ∑' k : {k : Lattice d // Lattice.IsPosHalf k}, featCoord a (Sum.inl k) x * featCoord a (Sum.inl k) y
84      + ∑' k : {k : Lattice d // Lattice.IsPosHalf k}, featCoord a (Sum.inr k) x * featCoord a (Sum.inr k) y
85      = ∑' k : {k : Lattice d // Lattice.IsPosHalf k}, 2 * (a k.1 * Torus.cosMode k.1 (x - y)) := by
86    rw [← Summable.tsum_add hl hr]
87    exact tsum_congr fun k => fm2_term ha0 x y k
88  calc _ = _ := fm2_inner_split ha0 has x y
89    _ = _ := e1
90    _ = 2 * ∑' k : {k : Lattice d // Lattice.IsPosHalf k}, a k.1 * Torus.cosMode k.1 (x - y) :=
91        hh.tsum_mul_left 2
92    _ = ∑' k, a k * Torus.cosMode k (x - y) :=
93        (fm1_tsum_pair (fun k => a k * Torus.cosMode k (x - y)) hL h0 hev).symm

113/-- `‖Ψ(x)‖ = √(g(0))` for every `x`. -/
114theorem fm2_norm_eq (a : Lattice d → ℝ) (hsym : IsSpectralSymbol a) (x : Torus d) :
115    ‖featureMap hsym.nonneg hsym.summable x‖ = Real.sqrt (StaticLimitAux.specKernel a 0) := by
116  calc ‖featureMap hsym.nonneg hsym.summable x‖
117      = Real.sqrt (‖featureMap hsym.nonneg hsym.summable x‖ ^ 2) :=
118        (Real.sqrt_sq (norm_nonneg _)).symm
119    _ = Real.sqrt (StaticLimitAux.specKernel a 0) := by rw [featureMap_norm_sq_proof hsym x]
