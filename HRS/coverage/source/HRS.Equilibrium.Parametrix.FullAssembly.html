6# Proposition 4.13 with all clauses (packet PX-FID)
7
8`pair_poisson_full_proof` is the frozen spec `HRS.pair_poisson_full`.
9
10* The conclusions of `pair_poisson` are those of `pair_poisson_proof` (same `Ψ`).
11* The distributional clause (`IsWeakPairPoisson`) is proved for **every** pair corrector, by
12  `HRS.Parx.PXFID.weak_of_corrector` (`PXFIDAuxQ.lean`): off the diagonal the equation holds
13  pointwise, and the integrations by parts have no diagonal flux (`PXFIDAuxG.lean`).
14* The tangential clause (`PsiSize` of `tanIter es (Ψ f)`, `|es| ≤ 4`) is proved by `PXFIDTAuxA-E`
15  (packet PX-FID-T) for `Ψ f = ψ^loc + Ψ_{R_L}` with the Sobolev level `a = ⌊d/2⌋ + 7` of the pair
16  inverse (`C^6` output needs `d + 12 < 2a`, input `C^{2a}`), the gap lemma at `J = 2a - 1`
17  (`R_L ∈ C^{J+1} = C^{2a}` with one more conormal derivative, `σ_L = -J-3`), and the closure of the
18  conormal classes under `∂_X` (a tangential derivative costs one conormal derivative).  The index is
19  `K = parxK d J + 1 = 10⌊d/2⌋ + 2d + 81 ≤ 7d + 81 ≤ 8d + 80`.
20-/

32theorem pair_poisson_full_proof (hd : 2 ≤ d) (hcrit : s = (d : ℝ) / 2) (β : ℝ) (hβ : 0 < β)
33    (V : Torus d → ℝ) (hV : Torus.IsSmooth V) (ρ : Torus d → ℝ)
34    (hρ : IsThermalDensity (periodicRiesz d s) V β ρ) (hρs : SmoothPosDensity ρ) :
35    ∃ (Ψ : (Torus d → ℝ) → Torus d → Torus d → ℝ) (C : ℝ),
36      (∀ (f f' : Torus d → ℝ) (a b : ℝ), Torus.IsSmooth f → Torus.IsSmooth f' →
37        Ψ (a • f + b • f') = a • Ψ f + b • Ψ f') ∧
38      ∀ f : Torus d → ℝ, Torus.IsSmooth f →
39        IsPairCorrector d s V β ρ f (C * posSeminorm (8 * d + 80) f) (Ψ f) ∧
40        (∃ M : ℝ, ∀ x, |aCoef d s ρ f x| ≤ M) ∧
41        eLpNorm (aCoef d s ρ f) 2 (withDensityReal ρ) ≤
42          ENNReal.ofReal (C * posSeminorm (8 * d + 80) f) ∧
43        (∀ es : List (Euc d), es.length ≤ 4 →
44          PsiSize s (C * posSeminorm (8 * d + 80) f * (es.map fun e => ‖e‖).prod)
45            (tanIter es (Ψ f))) ∧
46        IsWeakPairPoisson d s β ρ f (Ψ f) := by
47  have H : Parx.PXFID.FidHyp d s V β ρ := ⟨hd, hcrit, hβ, hV, hρ, hρs⟩
48  have H' : Parx.ParxHyp d s V β ρ := ⟨hd, hcrit, hβ, hV, hρ, hρs⟩
49  -- Sobolev level `a = ⌊d/2⌋ + 7` and `J = 2a - 1`
50  obtain ⟨q, hq⟩ : ∃ q : ℕ, q = ⌊(d : ℝ) / 2⌋₊ := ⟨_, rfl⟩
51  have hq1 : 2 * q ≤ d := by
52    have h : (q : ℝ) ≤ (d : ℝ) / 2 := by rw [hq]; exact Nat.floor_le (by positivity)
53    have : (2 * q : ℝ) ≤ d := by linarith
54    exact_mod_cast this
55  have hq2 : (d : ℝ) / 2 < (q : ℝ) + 1 := by rw [hq]; exact Nat.lt_floor_add_one _
56  have hpm : PDE.PX7.px7_m d = q + 5 := by rw [hq]; rfl
57  set J : ℕ := 2 * q + 13 with hJ
58  have hspec : Parx.PairInverseSpec H' J :=
59    PDE.PX7.px7_spec H'.two_le s H'.s_nonneg H'.s_lt β H'.beta_pos V H'.V_smooth ρ H'.thermal
60      H'.rho_smooth.smooth H'.rho_smooth.pos J (by rw [hpm]; omega)
61  have h6 : ∃ C6 : ℝ, 0 ≤ C6 ∧ ∀ R : Torus d → Torus d → ℝ, PDE.IsCk2 (J + 1) R →
62      PDE.IsCk2 6 (PDE.pairInverse H'.dyn ρ R) ∧
63      PDE.cbNorm2 6 (PDE.pairInverse H'.dyn ρ R) ≤ C6 * PDE.cbNorm2 (J + 1) R := by
64    obtain ⟨Cinv, hC0, hC⟩ := PDE.PX7.pxfidt_inv_Cn_spec (d := d) (by omega) s H'.s_nonneg H'.s_lt β
65      H'.beta_pos V H'.V_smooth ρ H'.thermal H'.rho_smooth.smooth H'.rho_smooth.pos (q + 7)
66      (by rw [hpm]; omega) 6 (by push_cast; linarith)
67    refine ⟨Cinv, hC0, fun R hR => ?_⟩
68    have e : q + 7 + (q + 7) = J + 1 := by rw [hJ]; omega
69    rw [e] at hC
70    exact hC R hR
71  obtain ⟨Ψ, C, hlin, hf⟩ := Parx.PXFIDT.fidt_gen H' J hspec h6 (by rw [hJ]; omega) (8 * d + 80)
72    (by unfold Parx.parxK; rw [hJ]; omega)
73  refine ⟨Ψ, C, hlin, fun f hfs => ⟨(hf f hfs).1, (hf f hfs).2.1, (hf f hfs).2.2.1,
74    (hf f hfs).2.2.2, Parx.PXFID.weak_of_corrector H (hf f hfs).1⟩⟩
