403/-- A kernel domination constant depending only on a bound for `cbNorm 2 ψ`. -/
404theorem interKernel_dom (hd : 1 ≤ d) (D : DynData d) (Cb : ℝ) :
405    ∃ K0 : ℝ, 0 ≤ K0 ∧ ∀ ψ : Torus d → ℝ, Torus.IsCk 2 ψ → Torus.cbNorm 2 ψ ≤ Cb →
406      ∀ x y : Torus d, |interKernel D ψ x y| ≤ K0 * (‖x - y‖ ^ (-D.s) + d) := by
407  obtain ⟨C, hG⟩ := KernelGradD1.goodKernel_periodicRieszGrad_of_one_le hd D.hs0 D.hsd
408  obtain ⟨CM, hCM0, hCM⟩ := Duality.exists_matVec_bound D.M
409  have hC0 := hG.hC0
410  have hCb : 0 ≤ Cb ∨ True := Or.inr trivial
411  refine ⟨CM * (max Cb 0 * d) * C, by positivity, fun ψ hψ hcb x y => ?_⟩
412  have hK0 : 0 ≤ CM * (max Cb 0 * d) * C := by positivity
413  unfold interKernel
414  split_ifs with hxy
415  · rw [abs_zero]
416    exact mul_nonneg hK0 (add_nonneg (Real.rpow_nonneg (norm_nonneg _) _) (Nat.cast_nonneg _))
417  · have hz : x - y ≠ 0 := sub_ne_zero.mpr hxy
418    have hG' := hG.norm_mul_tdist_le (x - y) hz
419    rw [← KernelGrad.tdist_eq_sub_zero] at hG'
420    have hL := grad_lip_explicit hψ x y
421    have hcb0 : Torus.cbNorm 2 ψ ≤ max Cb 0 := hcb.trans (le_max_left _ _)
422    have hL0 : 0 ≤ max Cb 0 * d := by positivity
423    have hL' : ‖Torus.grad ψ x - Torus.grad ψ y‖ ≤ (max Cb 0 * d) * KernelGrad.tdist x y :=
424      hL.trans (mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_right hcb0 (Nat.cast_nonneg _))
425        (KernelGrad.tdist_nonneg x y))
426    have e1 : ‖Torus.grad ψ x - Torus.grad ψ y‖ * ‖Torus.matVec D.M (periodicRieszGrad d D.s (x - y))‖
427        ≤ ((max Cb 0 * d) * KernelGrad.tdist x y) * (CM * ‖periodicRieszGrad d D.s (x - y)‖) :=
428      mul_le_mul hL' (hCM _) (norm_nonneg _) (mul_nonneg hL0 (KernelGrad.tdist_nonneg x y))
429    calc |inner ℝ (Torus.grad ψ x - Torus.grad ψ y)
430          (Torus.matVec D.M (periodicRieszGrad d D.s (x - y)))|
431        ≤ ‖Torus.grad ψ x - Torus.grad ψ y‖ * ‖Torus.matVec D.M (periodicRieszGrad d D.s (x - y))‖ :=
432          abs_real_inner_le_norm _ _
433      _ ≤ ((max Cb 0 * d) * KernelGrad.tdist x y) * (CM * ‖periodicRieszGrad d D.s (x - y)‖) := e1
434      _ = CM * (max Cb 0 * d) * (KernelGrad.tdist x y * ‖periodicRieszGrad d D.s (x - y)‖) := by ring
435      _ ≤ CM * (max Cb 0 * d) * (C * (‖x - y‖ ^ (-D.s) + d)) :=
436          mul_le_mul_of_nonneg_left hG' (mul_nonneg hCM0 hL0)
437      _ = _ := by ring
