21theorem fdl_kernel_torus_proof (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) {ψ : Euc d → ℝ}
22    (hψ : IsFDLProfile ψ) {b : ℝ} (hb : IsFDLConst ψ s b) :
23    -- (i)
24    (∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
25      (∀ x : Torus d, x ≠ 0 → 0 ≤ fdlWT ψ s b lam x) ∧
26      (∀ x : Torus d, lam ≤ KernelGrad.tdist x 0 →
27        fdlWT ψ s b lam x = 0 ∧ fdlTruncT ψ s b lam x = periodicRiesz d s x) ∧
28      (∀ x : Torus d, x ≠ 0 → fdlWT ψ s b lam x ≤ periodicRiesz d s x + C) ∧
29      ∫⁻ x : Torus d, ENNReal.ofReal (fdlWT ψ s b lam x) ≤
30        ENNReal.ofReal (C * lam ^ ((d : ℝ) - s))) ∧
31    -- (ii), `j + s > 0`
32    (∀ j : ℕ, 0 < (j : ℝ) + s → ∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
33      ∀ y : Euc d, Torus.mk y ≠ 0 →
34        ‖iteratedFDeriv ℝ j (Torus.lift (fdlWT ψ s b lam)) y‖ ≤
35          (if KernelGrad.tdist (Torus.mk y) 0 ≤ lam then
36            C * KernelGrad.tdist (Torus.mk y) 0 ^ (-s - (j : ℝ)) else 0) ∧
37        ‖iteratedFDeriv ℝ j (Torus.lift (fdlTruncT ψ s b lam)) y‖ ≤
38          C * (max lam (KernelGrad.tdist (Torus.mk y) 0)) ^ (-s - (j : ℝ)) + C) ∧
39    -- (ii), `s = j = 0`
40    (s = 0 → ∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ x : Torus d, x ≠ 0 →
41      |fdlWT ψ s b lam x| ≤
42        (if KernelGrad.tdist x 0 ≤ lam then C * Real.log (lam / KernelGrad.tdist x 0) else 0)) ∧
43    (∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ x : Torus d,
44      s = 0 → |fdlTruncT ψ s b lam x| ≤ C * (1 + |Real.log lam|)) ∧
45    -- (ii), smoothness and sup bounds ("in particular")
46    (∀ lam : ℝ, 0 < lam → lam < 1 → Torus.IsSmooth (fdlTruncT ψ s b lam)) ∧
47    (∀ j : ℕ, 0 < (j : ℝ) + s → ∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ y : Euc d,
48      ‖iteratedFDeriv ℝ j (Torus.lift (fdlTruncT ψ s b lam)) y‖ ≤ C * lam ^ (-s - (j : ℝ))) ∧
49    (∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
50      |fdlTruncT ψ s b lam 0| ≤
51        C * (lam ^ (-s) + (if s = 0 then |Real.log lam| else 0))) ∧
52    -- (iii)
53    (∀ lam : ℝ, 0 < lam → lam < 1 → ∀ k : Lattice d,
54      Torus.fcoef (fdlWT ψ s b lam) k = ((fdlWhat ψ s b lam k : ℝ) : ℂ) ∧
55        0 ≤ fdlWhat ψ s b lam k) ∧
56    (∀ lam : ℝ, 0 < lam → lam < 1 → ∀ k : Lattice d, k ≠ 0 →
57      IntegrableOn (fun t : ℝ => t ^ ((d : ℝ) - s) * fdlPsiHat ψ (t • Lattice.toEuc k) ^ 2 / t)
58        (Set.Ioi lam) ∧
59      Torus.fcoef (fdlTruncT ψ s b lam) k = ((fdlGhat ψ s b lam k : ℝ) : ℂ) ∧
60        0 ≤ fdlGhat ψ s b lam k) ∧
61    (∃ C : ℝ, 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 →
62      Summable (fun k : Lattice d => Lattice.enorm k ^ 2 * fdlGhat ψ s b lam k) ∧
63      |Torus.laplacian (fdlTruncT ψ s b lam) 0| ≤
64        4 * Real.pi ^ 2 * ∑' k : Lattice d, Lattice.enorm k ^ 2 * fdlGhat ψ s b lam k ∧
65      4 * Real.pi ^ 2 * ∑' k : Lattice d, Lattice.enorm k ^ 2 * fdlGhat ψ s b lam k ≤
66        C * lam ^ (-s - 2)) ∧
67    -- (iv)
68    (0 < s → ∃ c C : ℝ, 0 < c ∧ 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ x : Torus d,
69      lam ≤ KernelGrad.tdist x 0 →
70      c * KernelGrad.tdist x 0 ^ (-s) - C ≤ fdlTruncT ψ s b lam x) ∧
71    (s = 0 → ∃ c C : ℝ, 0 < c ∧ 0 ≤ C ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ x : Torus d,
72      lam ≤ KernelGrad.tdist x 0 → KernelGrad.tdist x 0 ≤ 1 / 2 →
73      c * Real.log (1 / KernelGrad.tdist x 0) - C ≤ fdlTruncT ψ s b lam x) ∧
74    (∀ K : ℝ, fdlK ≤ K → ∃ c : ℝ, 0 < c ∧ ∀ lam : ℝ, 0 < lam → lam < 1 → ∀ x : Torus d, x ≠ 0 →
75      KernelGrad.tdist x 0 ≤ 2 * lam →
76      c * (if s = 0 then 1 else lam ^ (-s)) ≤ fdlWT ψ s b (K * lam) x) := by
77  obtain ⟨hiv1, hiv2, hiv3⟩ := D3T1.d3t_part_iv hd hs0 hsd hψ hb
78  refine ⟨D3T1.d3t_part_i hd hs0 hsd hψ hb, ?_, D3TW.d3tw_log hd hs0 hsd hψ hb, D3TG.d3tg_log hd hs0 hsd hψ hb,
79    D3TG.d3tg_smooth hd hs0 hsd hψ hb, D3TG.d3tg_sup hd hs0 hsd hψ hb, D3TG.d3tg_zero hd hs0 hsd hψ hb,
80    D3T3.d3f_part_iii_a hd hs0 hsd hψ hb, D3T3.d3f_part_iii_b hd hs0 hsd hψ hb,
81    D3T3.d3f_part_iii_c hd hs0 hsd hψ hb, hiv1, hiv2, hiv3⟩
82  intro j hj
83  obtain ⟨C1, hC1, h1⟩ := D3TW.d3tw_deriv hd hs0 hsd hψ hb j hj
84  obtain ⟨C2, hC2, h2⟩ := D3TG.d3tg_deriv hd hs0 hsd hψ hb j hj
85  refine ⟨max C1 C2, le_max_of_le_left hC1, fun lam hl hl1 y hy => ⟨?_, ?_⟩⟩
86  · have := h1 lam hl hl1 y hy
87    split_ifs at this ⊢
88    · exact this.trans (mul_le_mul_of_nonneg_right (le_max_left _ _)
89        (Real.rpow_nonneg (KernelGrad.tdist_nonneg _ _) _))
90    · exact this
91  · have h := h2 lam hl hl1 y hy
92    have hm : 0 ≤ (max lam (KernelGrad.tdist (Torus.mk y) 0)) ^ (-s - (j : ℝ)) :=
93      Real.rpow_nonneg (le_trans hl.le (le_max_left _ _)) _
94    calc _ ≤ _ := h
95      _ ≤ _ := by
96        have := mul_le_mul_of_nonneg_right (le_max_right C1 C2) hm
97        linarith [le_max_right C1 C2]
