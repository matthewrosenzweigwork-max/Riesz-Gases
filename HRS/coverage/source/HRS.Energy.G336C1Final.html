26theorem g336c1_ae_bound (hd : 1 ≤ d) {s : ℝ} {C₀ C : ℝ} (hC : 0 < C)
27    (hcc : ∀ (Λ : ℝ), 1 ≤ Λ → ∀ (N : ℕ), 2 ≤ N →
28      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
29      ∀ X : Config d N, X ∉ collisionSet d N →
30        0 ≤ rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N ∧
31        (∀ (v : Torus d → Euc d) (K : ℝ), HasTransportBound d s v K →
32          |commQuantity d s v X (withDensityReal ρ)| ≤
33            C * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N)))
34    (D : DynData d) (hDs : D.s = s) (hinv : D.invβ = 0) (hS : Torus.SymNonpos D.M)
35    (hV : Torus.IsCk 2 D.V) (hN : 2 ≤ N) {T : ℝ} (hT : 0 ≤ T)
36    {ρ : ℝ → Torus d → ℝ} {r : ℕ} {Λ : ℝ} (hr : 3 ≤ r) (hΛ : 1 ≤ Λ)
37    (hmf : IsMeanFieldSolution D T ρ) (hregr : IsRegularMF r Λ T ρ)
38    {Λ' : ℝ} (hΛ' : ∀ t ∈ Set.Icc 0 T, HasTransportBound d s (meanFieldDrift D (ρ t)) Λ')
39    {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω)
40    (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N)
41    (hsol : IsStrongSolution P D T X₀ W X) :
42    ∀ᵐ ω ∂P, ∀ t ∈ Set.Icc 0 T,
43      correctedEnergy D C₀ Λ ρ (X (Real.toNNReal t) ω) t ≤
44        Real.exp (C / 2 * T * Λ') * correctedEnergy D C₀ Λ ρ (X 0 ω) 0 := by
45  subst hDs
46  have hreg3 : IsRegularMF 3 Λ T ρ := fun t ht =>
47    ⟨(hregr t ht).1.of_le (by exact_mod_cast hr), (KernelGrad.cbNorm_mono hr).trans (hregr t ht).2⟩
48  have c : HRS.KID.MFctx D T Λ ρ := ⟨hd, hT, hV, hmf, hreg3⟩
49  have hcontA := hsol.cont
50  have hcolA := hsol.no_collision
51  obtain ⟨Y, hY, rfl⟩ := hsol.lift
52  filter_upwards [hY, hcontA, hcolA] with ω hω hcont hcol
53  set Y' : ℝ≥0 → Fin N → Euc d := fun t => Y t ω with hY'
54  have hp : G336C1Path D T Y' :=
55    { hint := fun t htT i => (hω t htT).1 i
56      hy := fun t htT i => by
57        have := (hω t htT).2 i
58        rw [hinv] at this
59        simp only [mul_zero, Real.sqrt_zero, zero_smul, add_zero] at this
60        exact this
61      hcont := hcont
62      hcol := fun t htT => hcol t htT }
63  have hA : ∀ r ∈ Set.Icc (0 : ℝ) T,
64      0 ≤ rieszEnergy d D.s (g336c1_x Y' r) (withDensityReal (ρ r)) + oNΛ d D.s C₀ Λ N ∧
65      |commQuantity d D.s (meanFieldDrift D (ρ r)) (g336c1_x Y' r) (withDensityReal (ρ r))| ≤
66        C * Λ' * (rieszEnergy d D.s (g336c1_x Y' r) (withDensityReal (ρ r)) + oNΛ d D.s C₀ Λ N) := by
67    intro r hrT
68    obtain ⟨h1, h2⟩ := hcc Λ hΛ N hN (ρ r) (hmf.density r hrT)
69      (fun x => (le_abs_self _).trans (c.abs_le hrT x)) (g336c1_x Y' r) (g336c1_x_not_collision hp hrT)
70    exact ⟨h1, h2 _ _ (hΛ' r hrT)⟩
71  intro t ht
72  have key := g336c1_det_gronwall hd D hinv hS hN hV hT hmf hreg3 hΛ' hp hC (fun r hr => (hA r hr).1)
73    (fun r hr => (hA r hr).2) t ht
74  have e0 : g336c1_x Y' 0 = projectPath Y 0 ω := by
75    funext i; simp [g336c1_x, projectPath, hY']
76  unfold correctedEnergy
77  rw [e0] at key
78  exact key

80theorem gronwall_zero_temperature_proof (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) :
81    ∃ C₀ C : ℝ, 0 ≤ C₀ ∧ 0 < C ∧
82      ∀ (D : DynData d), D.s = s → D.invβ = 0 → Torus.SymNonpos D.M → Torus.IsCk 2 D.V →
83      ∀ (N : ℕ), 2 ≤ N → ∀ (T : ℝ), 0 ≤ T →
84      ∀ (ρ : ℝ → Torus d → ℝ) (r : ℕ) (Λ : ℝ), 3 ≤ r → 1 ≤ Λ →
85        IsMeanFieldSolution D T ρ → IsRegularMF r Λ T ρ →
86      ∀ Λ' : ℝ, (∀ t ∈ Set.Icc 0 T, HasTransportBound d s (meanFieldDrift D (ρ t)) Λ') →
87      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
88        (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N),
89        IsStrongSolution P D T X₀ W X →
90        ∀ᵐ ω ∂P, ∀ t ∈ Set.Icc 0 T,
91          correctedEnergy D C₀ Λ ρ (X (Real.toNNReal t) ω) t ≤
92            Real.exp (C * T * Λ') * correctedEnergy D C₀ Λ ρ (X 0 ω) 0 := by
93  obtain ⟨C₀, C, hC₀, hC, hcc⟩ := commutator_coercivity_common_proof hd hs0 hsd
94  refine ⟨C₀, C / 2, hC₀, half_pos hC, ?_⟩
95  intro D hDs hinv hS hV N hN T hT ρ r Λ hr hΛ hmf hregr Λ' hΛ' Ω _ P _ X₀ W X hsol
96  exact g336c1_ae_bound hd hC (fun Λ hΛ N hN ρ hρ hρΛ X hX =>
97    ⟨(hcc Λ hΛ N hN ρ hρ hρΛ X hX).1, (hcc Λ hΛ N hN ρ hρ hρΛ X hX).2.1⟩) D hDs hinv hS hV hN hT hr hΛ
98    hmf hregr hΛ' P X₀ W X hsol

100theorem gronwall_zero_temperature_moment_proof (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) :
101    ∃ C₀ C : ℝ, 0 ≤ C₀ ∧ 0 < C ∧
102      ∀ (D : DynData d), D.s = s → D.invβ = 0 → Torus.SymNonpos D.M → Torus.IsCk 2 D.V →
103      ∀ (N : ℕ), 2 ≤ N → ∀ (T : ℝ), 0 ≤ T →
104      ∀ (ρ : ℝ → Torus d → ℝ) (r : ℕ) (Λ : ℝ), 3 ≤ r → 1 ≤ Λ →
105        IsMeanFieldSolution D T ρ → IsRegularMF r Λ T ρ →
106      ∀ Λ' : ℝ, (∀ t ∈ Set.Icc 0 T, HasTransportBound d s (meanFieldDrift D (ρ t)) Λ') →
107      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
108        (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N),
109        IsStrongSolution P D T X₀ W X →
110        Integrable (fun ω => correctedEnergy D C₀ Λ ρ (X 0 ω) 0) P →
111        ∀ t ∈ Set.Icc 0 T,
112          ∫⁻ ω, ENNReal.ofReal (correctedEnergy D C₀ Λ ρ (X (Real.toNNReal t) ω) t) ∂P ≤
113            ENNReal.ofReal (Real.exp (C * T * Λ') *
114              ∫ ω, correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P) := by
115  obtain ⟨C₀, C, hC₀, hC, hcc⟩ := commutator_coercivity_common_proof hd hs0 hsd
116  refine ⟨C₀, C / 2, hC₀, half_pos hC, ?_⟩
117  intro D hDs hinv hS hV N hN T hT ρ r Λ hr hΛ hmf hregr Λ' hΛ' Ω _ P _ X₀ W X hsol hint t ht
118  have hcc' : ∀ (Λ : ℝ), 1 ≤ Λ → ∀ (N : ℕ), 2 ≤ N →
119      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
120      ∀ X : Config d N, X ∉ collisionSet d N →
121        0 ≤ rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N ∧
122        (∀ (v : Torus d → Euc d) (K : ℝ), HasTransportBound d s v K →
123          |commQuantity d s v X (withDensityReal ρ)| ≤
124            C * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N)) :=
125    fun Λ hΛ N hN ρ hρ hρΛ X hX =>
126      ⟨(hcc Λ hΛ N hN ρ hρ hρΛ X hX).1, (hcc Λ hΛ N hN ρ hρ hρΛ X hX).2.1⟩
127  have hae := g336c1_ae_bound hd hC hcc' D hDs hinv hS hV hN hT hr hΛ hmf hregr hΛ' P X₀ W X hsol
128  have hreg3 : IsRegularMF 3 Λ T ρ := fun t ht =>
129    ⟨(hregr t ht).1.of_le (by exact_mod_cast hr), (KernelGrad.cbNorm_mono hr).trans (hregr t ht).2⟩
130  have c : HRS.KID.MFctx D T Λ ρ := ⟨hd, hT, hV, hmf, hreg3⟩
131  have h0T : (0 : ℝ) ∈ Set.Icc 0 T := ⟨le_rfl, hT⟩
132  have hnn : ∀ᵐ ω ∂P, 0 ≤ correctedEnergy D C₀ Λ ρ (X 0 ω) 0 := by
133    filter_upwards [hsol.no_collision] with ω hω
134    have h1 := (hcc Λ hΛ N hN (ρ 0) (hmf.density 0 h0T)
135      (fun x => (le_abs_self _).trans (c.abs_le h0T x)) (X 0 ω) (hω 0 (by simpa using hT))).1
136    subst hDs
137    exact h1
138  set K : ℝ := Real.exp (C / 2 * T * Λ') with hK
139  have hK0 : 0 ≤ K := (Real.exp_pos _).le
140  calc ∫⁻ ω, ENNReal.ofReal (correctedEnergy D C₀ Λ ρ (X (Real.toNNReal t) ω) t) ∂P
141      ≤ ∫⁻ ω, ENNReal.ofReal (K * correctedEnergy D C₀ Λ ρ (X 0 ω) 0) ∂P := by
142        refine lintegral_mono_ae ?_
143        filter_upwards [hae] with ω hω
144        exact ENNReal.ofReal_le_ofReal (hω t ht)
145    _ = ENNReal.ofReal (∫ ω, K * correctedEnergy D C₀ Λ ρ (X 0 ω) 0 ∂P) := by
146        rw [ofReal_integral_eq_lintegral_ofReal (hint.const_mul K)
147          (hnn.mono fun ω h => mul_nonneg hK0 h)]
148    _ = _ := by rw [integral_const_mul]
