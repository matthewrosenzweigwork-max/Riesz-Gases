18/-- `L²` norm of `c • h` from the second moment bound. -/
19theorem p_biasub_eLpNorm_le {μ : Measure (Torus d)} {h : Torus d → ℝ} (hm : MemLp h 2 μ)
20    {c B : ℝ} (hc : 0 ≤ c) (hB : 0 ≤ B) (hint : ∫ x, h x ^ 2 ∂μ ≤ B ^ 2) :
21    eLpNorm (fun x => c * h x) 2 μ ≤ ENNReal.ofReal (c * B) := by
22  have hm' : MemLp (fun x => c * h x) 2 μ := hm.const_mul c
23  rw [hm'.eLpNorm_eq_integral_rpow_norm (by norm_num) (by norm_num)]
24  refine ENNReal.ofReal_le_ofReal ?_
25  have e : ∫ x, ‖c * h x‖ ^ (ENNReal.toReal 2) ∂μ = c ^ 2 * ∫ x, h x ^ 2 ∂μ := by
26    rw [← integral_const_mul]
27    refine integral_congr_ae (Filter.Eventually.of_forall fun x => ?_)
28    simp only [ENNReal.toReal_ofNat, Real.norm_eq_abs]
29    rw [Real.rpow_two, sq_abs, mul_pow]
30  have h1 : c ^ 2 * ∫ x, h x ^ 2 ∂μ ≤ (c * B) ^ 2 := by
31    rw [mul_pow]; exact mul_le_mul_of_nonneg_left hint (sq_nonneg c)
32  have hcB : 0 ≤ c * B := mul_nonneg hc hB
33  rw [e]
34  rw [ENNReal.toReal_ofNat, ← one_div, ← Real.sqrt_eq_rpow]
35  calc Real.sqrt (c ^ 2 * ∫ x, h x ^ 2 ∂μ) ≤ Real.sqrt ((c * B) ^ 2) := Real.sqrt_le_sqrt h1
36    _ = c * B := Real.sqrt_sq hcB

38/-- P2: norm bound of the critical bias for densities bounded above by `Λ`. -/
39theorem criticalBias_norm_le (hd : 2 ≤ d) {m : ℝ} (hm : (d : ℝ) < 2 * m) {β : ℝ} (hβ : 0 < β)
40    {ρ : Torus d → ℝ} (hρ : SmoothPosDensity ρ) {Λ : ℝ} (hΛ : ∀ x, ρ x ≤ Λ) :
41    ‖criticalBias d ((d : ℝ) / 2) β m ρ‖ ≤
42      Real.sqrt 2 * (β ^ 2 * sphereArea d / d) * Λ *
43        Real.sqrt (∑' k : NonzeroLattice d, Lattice.jp k.1 ^ (-(2 * m))) := by
44  have hs0 : (0 : ℝ) ≤ (d : ℝ) / 2 := by positivity
45  have hdR : (2 : ℝ) ≤ d := by exact_mod_cast hd
46  have hsd : (d : ℝ) / 2 < d := by linarith
47  have hμP : IsProbabilityMeasure (withDensityReal ρ) := PoleCavityAux.isProbabilityMeasure_mu hρ
48  obtain ⟨hmh, -, -⟩ := eqbias_response hd hs0 hsd hρ hβ.le
49  set h := responseInverse (periodicRiesz d ((d : ℝ) / 2)) (withDensityReal ρ) β ρ with hhdef
50  set c : ℝ := β ^ 2 * sphereArea d / d with hcdef
51  have hA : 0 < sphereArea d := HRS.StaticLimitsAux.sphereArea_pos hd
52  have hdpos : (0 : ℝ) < d := by linarith
53  have hc : 0 ≤ c := by positivity
54  have hΛ0 : 0 ≤ Λ := (hρ.pos (0 : Torus d)).le.trans (hΛ _)
55  have hL2 := HRS.ME.Unif.p8_resp_L2_le hd hρ hβ.le hΛ
56  have hFe := p_biasub_eLpNorm_le hmh hc hΛ0 hL2
57  have hGaesm : AEStronglyMeasurable (fun y => c * h y) (withDensityReal ρ) :=
58    (hmh.const_mul c).aestronglyMeasurable
59  obtain ⟨hIG, hIGb⟩ := integral_abs_mul_le_aesm hρ.density hGaesm (mul_nonneg hc hΛ0) hFe
60  have hcbk : ∀ k : NonzeroLattice d,
61      NegSobolev.coord (criticalBias d ((d : ℝ) / 2) β m ρ) k =
62        Lattice.jp k.1 ^ (-m) * Torus.funRealCoeff (fun y => c * h y * ρ y) k :=
63    fun k => coord_ofFun hm hIG k
64  have hK : 0 ≤ Real.sqrt 2 * (c * Λ) := mul_nonneg (Real.sqrt_nonneg 2) (mul_nonneg hc hΛ0)
65  have hbnd : ∀ k : NonzeroLattice d,
66      |NegSobolev.coord (criticalBias d ((d : ℝ) / 2) β m ρ) k| ≤
67        (Real.sqrt 2 * (c * Λ)) * Lattice.jp k.1 ^ (-m) := by
68    intro k
69    have hjp : 0 < Lattice.jp k.1 ^ (-m) := Real.rpow_pos_of_pos (Torus.jp_pos k.1) _
70    rw [hcbk k, abs_mul, abs_of_pos hjp]
71    have := (abs_funRealCoeff_le_integral hIG k).trans
72      (mul_le_mul_of_nonneg_left hIGb (Real.sqrt_nonneg 2))
73    nlinarith
74  have := norm_le_of_coord_bound hm _ hK hbnd
75  calc _ ≤ _ := this
76    _ = _ := by ring
