94      ((∑ i : Fin d, movb2_T (movb2_gloc d s)
95          (mova3b1_Dj i (mova3b1_Dj i (fun y => Real.sqrt (ρ y) * v y))) x) +
96        (∑ i : Fin d, movb2_T (mova3b1_Dj i (movb2_kfar d s i)) (fun y => Real.sqrt (ρ y) * v y) x) +

98          (movb2_T (movb2_gloc d s) (mova3b1_Dj i (fun y => Real.sqrt (ρ y) * v y)) x +
99            movb2_T (movb2_kfar d s i) (fun y => Real.sqrt (ρ y) * v y) x)) := by

101  have hρd : MovRhoData d ρ := ⟨hpos, hρs, fun k => by
102    obtain ⟨-, c, hc, hk⟩ := hψ
103    obtain ⟨C, hC⟩ := hk k
104    exact ⟨C, c, hc, hC⟩⟩
105  rw [mv1_P_formula hd hs0 hsd hρd v x]
106  have hh : MovcG h := movc1_schwartz_G_mul hψ v
107  have hh1 : ∀ i, MovcG (mova3b1_Dj i h) := fun i => movc1_G_Dj hh i
108  have hh2 : ∀ i, MovcG (mova3b1_Dj i (mova3b1_Dj i h)) := fun i => movc1_G_Dj (hh1 i) i
109  -- Gaussian data

111  have hu0 : ContDiff ℝ 1 h := hh.1.of_le (by exact_mod_cast le_top)
112  -- the gradient terms
113  have hgrad : ∀ i : Fin d, HRS.Whole.convFun (rieszKernel d s) (fun y => gradient h y i) x =
114      movb2_T (movb2_gloc d s) (mova3b1_Dj i h) x + movb2_T (movb2_kfar d s i) h x := by
115    intro i
116    have e : (fun y => gradient h y i) = mova3b1_Dj i h := by
117      funext y; exact movtr_gradient_apply h y i
118    rw [e]
119    obtain ⟨c1, E1, hc1, hE1, hga1⟩ := movc1_G_GA (hh1 i) 1
120    obtain ⟨h01, -⟩ := movb2_GA_01 hga1
121    rw [movc1_split_T hd hs0 hsd (hh1 i).1.continuous hc1 hE1 h01 x]
122    congr 1
123    exact movb2_ibp hd hs0 hsd hc0 hu0 hga0 i x
124  -- the Laplacian term
125  have hlap : HRS.Whole.convFun (rieszKernel d s) (fun y => Laplacian.laplacian h y) x =
126      (∑ i : Fin d, movb2_T (movb2_gloc d s) (mova3b1_Dj i (mova3b1_Dj i h)) x) +
127        ∑ i : Fin d, movb2_T (mova3b1_Dj i (movb2_kfar d s i)) h x := by
128    have e : (fun y => Laplacian.laplacian h y) = fun y => ∑ i : Fin d, mova3b1_Dj i (mova3b1_Dj i h) y := by
129      funext y
130      exact mv1_laplacian_eq (hh.1.of_le (by norm_num)) y
131    rw [e, mv1_convFun_eq]
132    have hint : ∀ i : Fin d, Integrable (fun z => rieszKernel d s z * mova3b1_Dj i (mova3b1_Dj i h) (x - z)) volume := by
133      intro i
134      obtain ⟨c2, E2, hc2, hE2, hga2⟩ := movc1_G_GA (hh2 i) 1
135      obtain ⟨h02, -⟩ := movb2_GA_01 hga2
136      exact movc1_conv_int hd hs0 hsd (hh2 i).1.continuous hc2 hE2 h02 x
137    simp_rw [Finset.mul_sum]
138    rw [integral_finsetSum _ fun i _ => hint i, ← Finset.sum_add_distrib]
139    refine Finset.sum_congr rfl fun i _ => ?_
140    obtain ⟨c2, E2, hc2, hE2, hga2⟩ := movc1_G_GA (hh2 i) 1
141    obtain ⟨h02, -⟩ := movb2_GA_01 hga2
142    obtain ⟨c1, E1, hc1, hE1, hga1⟩ := movc1_G_GA (hh1 i) 1
143    have hsp := movc1_split_T hd hs0 hsd (hh2 i).1.continuous hc2 hE2 h02 x
144    rw [mv1_convFun_eq] at hsp
145    rw [hsp]
146    congr 1
147    -- ∂_i ∂_i h against g_far
148    have hu1 : ContDiff ℝ 1 (mova3b1_Dj i h) := (hh1 i).1.of_le (by exact_mod_cast le_top)
149    have s1 := movb2_ibp hd hs0 hsd hc1 hu1 hga1 i x
150    have hκ : MovcB (movb2_kfar d s i) := movc1_K_B (movc1_kfar_K hs0 i)
151    obtain ⟨A, hA0, hA⟩ := movc1_B_Bd hκ 1

157  rw [hlap]
