4/-!
5# S5.2 `orbit_maximal_identification`: the identification clause (all-horizon form)
6
7**Status.**  `orbit_maximal_identification_allhorizon` below is the identification clause of Lemma S5.2 with exactly the
8conclusion of the frozen statement `HRS.Whole.orbit_maximal_identification` and the following single change of
9hypothesis: `hΦ : ∀ T', IsHSIntegrand P ℱ T' Φ` (Hilbert-Schmidt integrand on every finite horizon `T'`) in place of
10`hΦ : IsHSIntegrand P ℱ T Φ` (horizon `T` only).
11
12**Why the change is necessary.**  `IsHilbertItoIntegral.approx` quantifies over all times `τ ≥ 0` and is phrased
13through `itoIntegral`, which is `Classical.choose` of an `IsItoIntegral` when one exists and `0` otherwise
14(`IsItoIntegral.approx` is vacuous for times at which the integrand is not `L²`).  With the integrand square
15integrable only on `[0,T]` the coordinate integrals `⟨Φ,x⟩` can have no continuous version on `[0,∞)` (the integrand
16beyond `T` is the base part `c(Dφ_α(X_r))_α`, controlled only pointwise in `x` by `OscAdjointInput.feature`, not
17uniformly), so `itoIntegral = 0` and `hM` holds for a `M` with `⟨M_t,x⟩ = 0`; then the conclusion fails.  This is
18the defect already recorded for `hilbert_polar_coupling` (attempt 4, `HRS/Prob/HilbertMartingale.lean`).  Whenever the
19base part is bounded uniformly in `x` (as under the hypotheses of `orbit_feature_bounded`, `b = 0`), `hΦ` on `[0,T]`
20implies the all-horizon form.  The originally frozen statement therefore could not be tied; approved (2026-10-23)
21the repair that adds the uniform bound `hB` of the base part to `HRS.Whole.orbit_maximal_identification`, and
22`orbit_maximal_identification_bdd` below is tied to the repaired statement with `type_of%`.
23
24Proof: `⟨M_τ, y⟩` is the Itô integral of `⟨Φ, y⟩` (`s52_pair_ito`); the coordinate `α` of `(𝒯M)(t)` is
25`⟨M_T,y1⟩ - ⟨M_T,y2⟩ + ⟨M_t,y2⟩` (`s52_coord_recon`); by the Duhamel identity of `OscAdjointInput`
26`⟨Φ, y1⟩ - ⟨Φ, y2⟩` vanishes for `r > t` (`s52_pair_grad_y1`), so the integral of the difference stops at `t`
27(`s52_lc`, `s52_kill`), and `⟨Φ, y1⟩` is the truncated integrand of the conclusion on `[0,t]`.
28-/

49  refine ⟨isProgressive_pi (u := fun (k : κ) r ω => inner ℝ (Φ r ω k) y) fun k i =>

104    rw [s52_pair_Φ, s52_pair_grad_y1 hC ht.2 k.2 hr]

107    (pc_isProgressiveL2_inner hΦ (s52y1 m T t α) Finset.univ) hT
108  have hp2T := s52_pair_ito hW' hM (s52y2 m T α) (pc_isProgressiveL2_inner hΦ (s52y2 m T α) Finset.univ) hT
109  have hp2t := s52_pair_ito hW' hM (s52y2 m T α) (pc_isProgressiveL2_inner hΦ (s52y2 m T α) Finset.univ) ht.1
110  have hsub := itoIntegral_sub hW' hL1 hL2
111  -- the integral of the difference stops at `t`
112  have hagreeE : ∀ r ∈ Set.Icc (0 : ℝ) T, ∀ ω, ud r ω = (fun r ω k => if r ≤ t then ud r ω k else 0) r ω := by
113    intro r hr ω
114    funext k
115    by_cases hrt : r ≤ t
116    · simp [hrt]
117    · simp only [hrt, if_false, hud, hu2val r ω k, hu1val r hr ω k]
118      simp
119  have hLdt : IsProgressiveL2 P ℱ (fun r ω k => if r ≤ t then ud r ω k else 0) := by

122  have hlcE := s52_lc hW' hLd hLdt hT hagreeE
123  have hkillE := s52_kill hW' hLd ht.1
124  -- the target integrand
125  set ψ : ℝ → Ω → Fin N × Fin d → ℝ := fun r ζ' ia => if r ∈ Set.Icc (0 : ℝ) t then Real.sqrt (2 * q / N) *
126    gradComp ia.2 (propagatorW C r t (oscTestFun m α)) (X (Real.toNNReal r) ζ' ia.1) else 0 with hψ
127  have hψeq : ψ = fun r ω k => if r ∈ Set.Icc (0 : ℝ) t then u1 r ω k else 0 := by
128    funext r ω k
129    by_cases hr : r ∈ Set.Icc (0 : ℝ) t
130    · have hrT : r ∈ Set.Icc (0 : ℝ) T := ⟨hr.1, hr.2.trans ht.2⟩
131      simp only [hψ, hr, if_true, hu1val r hrT ω k, hr.2]
132      rfl
133    · simp only [hψ, if_neg hr]

135    rw [hψeq]; exact s52_isProgressiveL2_ite hL1 measurableSet_Icc
136  have hagreeG : ∀ r ∈ Set.Icc (0 : ℝ) t, ∀ ω, u1 r ω = ψ r ω := by
137    intro r hr ω
138    rw [hψeq]
139    simp [hr]
140  have hlcG := s52_lc hW' hL1 hLψ ht.1 hagreeG
141  filter_upwards [hp1T, hp2T, hp2t, hsub, hlcE, hkillE, hlcG] with ζ h1 h2 h3 h4 h5 h6 h7
142  rw [s52_coord_recon ht.1 ht.2]
143  have k0a : inner ℝ (M T ζ) (s52y1 m T t α) = itoIntegral P ℱ W u1 T ζ := h1
144  have k0b : inner ℝ (M T ζ) (s52y2 m T α) = itoIntegral P ℱ W u2 T ζ := h2
145  have k0c : inner ℝ (M t ζ) (s52y2 m T α) = itoIntegral P ℱ W u2 t ζ := h3
146  have k1 : itoIntegral P ℱ W ud T ζ = itoIntegral P ℱ W u1 T ζ - itoIntegral P ℱ W u2 T ζ := h4 T
147  have k2 : itoIntegral P ℱ W ud t ζ = itoIntegral P ℱ W u1 t ζ - itoIntegral P ℱ W u2 t ζ := h4 t
148  have k3 : itoIntegral P ℱ W ud (min T T) ζ =
149      itoIntegral P ℱ W (fun r ω k => if r ≤ t then ud r ω k else 0) (min T T) ζ := h5 T hT
150  have k4 : itoIntegral P ℱ W (fun r ω k => if r ≤ t then ud r ω k else 0) T ζ =
151      itoIntegral P ℱ W ud (min T t) ζ := h6 T hT
152  have k5 : itoIntegral P ℱ W u1 (min t t) ζ = itoIntegral P ℱ W ψ (min t t) ζ := h7 t ht.1
153  rw [min_self] at k3 k5
154  rw [min_eq_right ht.2] at k4
155  rw [k0a, k0b, k0c]

213    IsHSIntegrand P ℱ T' (empiricalNoiseIntegrand C m T q X) := by

218  · refine lt_of_le_of_lt ?_ hΦ.sq

221  · have hTT' : T ≤ T' := (not_le.1 hTT).le

223      (ENNReal.ofReal (2 * |Real.sqrt (2 * q / N)| * max B 0)) ^ 2 with hK
224    have hpt : ∀ r : ℝ, T ≤ r → ∀ ω, ∑' k : Fin N × Fin d,
225        ‖empiricalNoiseIntegrand C m T q X r ω k‖ₑ ^ 2 ≤ K := by
226      intro r hr ω
227      rw [tsum_fintype]
228      calc ∑ k : Fin N × Fin d, ‖empiricalNoiseIntegrand C m T q X r ω k‖ₑ ^ 2
229          ≤ ∑ _k : Fin N × Fin d, (ENNReal.ofReal (2 * |Real.sqrt (2 * q / N)| * max B 0)) ^ 2 := by
230            refine Finset.sum_le_sum fun k _ => ?_
231            gcongr
232            rw [← ofReal_norm]
233            apply ENNReal.ofReal_le_ofReal
234            unfold empiricalNoiseIntegrand
235            rw [norm_smul, Real.norm_eq_abs]
236            have h1 := s52_norm_feature_tail (C := C) (m := m) (T := T) hr (gradComp k.2)
237              (X (Real.toNNReal r) ω k.1)
238            have h2 := hB k.2 (X (Real.toNNReal r) ω k.1)
239            have h3 : ‖gradFeature C m T k.2 r (X (Real.toNNReal r) ω k.1)‖ ≤ 2 * max B 0 := by
240              unfold gradFeature
241              exact h1.trans (by linarith [le_max_left B 0])
242            calc |Real.sqrt (2 * q / N)| * ‖gradFeature C m T k.2 r (X (Real.toNNReal r) ω k.1)‖
243                ≤ |Real.sqrt (2 * q / N)| * (2 * max B 0) := by gcongr
244              _ = _ := by ring
245        _ = K := by simp [hK]
246    unfold hsSqIntegralOn
247    have hsplit : ∀ ω : Ω, (∫⁻ r in Set.Icc 0 T', ∑' k, ‖empiricalNoiseIntegrand C m T q X r ω k‖ₑ ^ 2) ≤
248        (∫⁻ r in Set.Icc 0 T, ∑' k, ‖empiricalNoiseIntegrand C m T q X r ω k‖ₑ ^ 2) +
249          ENNReal.ofReal (T' - T) * K := by
250      intro ω
251      rw [← Set.Icc_union_Ioc_eq_Icc hT0 hTT', lintegral_union measurableSet_Ioc

253      refine add_le_add_right ?_ _
254      calc (∫⁻ r in Set.Ioc T T', ∑' k, ‖empiricalNoiseIntegrand C m T q X r ω k‖ₑ ^ 2)
255          ≤ ∫⁻ r in Set.Ioc T T', K :=
256            setLIntegral_mono' measurableSet_Ioc fun r hr => hpt r hr.1.le ω
257        _ = ENNReal.ofReal (T' - T) * K := by
258            rw [setLIntegral_const, Real.volume_Ioc, mul_comm]
259    have hKtop : ENNReal.ofReal (T' - T) * K < ⊤ :=
260      ENNReal.mul_lt_top ENNReal.ofReal_lt_top
261        (ENNReal.mul_lt_top (ENNReal.natCast_lt_top _) (ENNReal.pow_lt_top ENNReal.ofReal_lt_top))
262    calc ∫⁻ ω, (∫⁻ r in Set.Icc 0 T', ∑' k, ‖empiricalNoiseIntegrand C m T q X r ω k‖ₑ ^ 2) ∂P
263        ≤ ∫⁻ ω, ((∫⁻ r in Set.Icc 0 T, ∑' k, ‖empiricalNoiseIntegrand C m T q X r ω k‖ₑ ^ 2) +

279    (M : ℝ → Ω → KSpace d m T) (hN : 1 ≤ N) (hW : IsCylindricalBrownian P ℱ W)
280    (hΦ : IsHSIntegrand P ℱ T (empiricalNoiseIntegrand C m T q X))
281    (hM : IsHilbertItoIntegral P ℱ W (empiricalNoiseIntegrand C m T q X) M)
282    (hC : OscAdjointInput C m T)
