26/-- Theorem 2.5 (`H:thm:prepared`), smooth, `W₁` and `E_m`-initial-term bounds. -/
27theorem prepared_fluctuations_proof (S : PreparedData d) (m : ℝ) (hm : (d : ℝ) + 2 < m) (T : ℝ)
28    (hT : 0 < T) :
29    ∃ C c : ℝ, 0 < C ∧ 0 < c ∧
30      ∀ a : ℝ, 0 ≤ a → ∀ N : ℕ, 2 ≤ N →
31      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
32        (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N),
33        SelectedFlow P S.dyn (a + T) X₀ W X →
34        (∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
35          (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X) →
36        relEntropy (law P X₀) (S.refMeasure N 0) < ⊤ →
37      ∀ (ν : Measure (NegSobolev d m)) {Ω' : Type} [MeasurableSpace Ω'] (P' : Measure Ω')
38        [IsProbabilityMeasure P'] (Y₀ : Ω' → NegSobolev d m) (Y : ℝ → Ω' → NegSobolev d m),
39        IsPreparedLinear S m (a + T) ν P' Y₀ Y →
40        (S.initW m 1 P X₀ ν ≠ ⊤ →
41          smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
42            C * (Real.exp (-c * a) * (S.initW m 1 P X₀ ν).toReal +
43              S.B N (law P X₀) / Real.sqrt N + Real.sqrt (S.B N (law P X₀) / N)) ∧
44          wassersteinE 1 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
45            ENNReal.ofReal (C * (Real.exp (-c * a) * (S.initW m 1 P X₀ ν).toReal +
46              S.B N (law P X₀) / Real.sqrt N +
47              (S.B N (law P X₀) / N) ^ ((1 : ℝ) / 4)))) ∧
48        smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
49          C * (smoothDist 2 (law P (fun ω => etaE m (X₀ ω) S.μ0)) ν +
50            S.B N (law P X₀) / Real.sqrt N + Real.sqrt (S.B N (law P X₀) / N)) := by
51  obtain ⟨C1, c1, hC1, hc1, h1⟩ := HRS.T25P.t25p_clause1 S m hm T hT
52  obtain ⟨C3, hC3, h3⟩ := HRS.T25P.t25p_clause3 S m hm T hT
53  refine ⟨max C1 C3, c1, lt_max_of_lt_left hC1, hc1, ?_⟩
54  intro a ha N hN Ω _ P _ X₀ W X hSel hfilt hrel ν Ω' _ P' _ Y₀ Y hY
55  have hB0 : 0 ≤ S.B N (law P X₀) := HRS.T25P.t25p_B_nonneg S hN (law P X₀)
56  have hsN : 0 ≤ Real.sqrt N := Real.sqrt_nonneg _
57  have hf0 : 0 ≤ S.B N (law P X₀) / Real.sqrt N := div_nonneg hB0 hsN
58  have hg0 : 0 ≤ Real.sqrt (S.B N (law P X₀) / N) := Real.sqrt_nonneg _
59  have hh0 : 0 ≤ (S.B N (law P X₀) / N) ^ ((1 : ℝ) / 4) :=
60    Real.rpow_nonneg (div_nonneg hB0 (Nat.cast_nonneg N)) _
61  refine ⟨fun hfin => ?_, ?_⟩
62  · obtain ⟨hs, hw⟩ := h1 a ha N hN P X₀ W X hSel hfilt hrel ν P' Y₀ Y hY hfin
63    have hx0 : 0 ≤ Real.exp (-c1 * a) * (S.initW m 1 P X₀ ν).toReal :=
64      mul_nonneg (Real.exp_pos _).le ENNReal.toReal_nonneg
65    refine ⟨hs.trans (mul_le_mul_of_nonneg_right (le_max_left _ _) (by positivity)),
66      hw.trans (ENNReal.ofReal_le_ofReal (mul_le_mul_of_nonneg_right (le_max_left _ _) (by positivity)))⟩
67  · exact (h3 a ha N hN P X₀ W X hSel hfilt hrel ν P' Y₀ Y hY).trans
68      (mul_le_mul_of_nonneg_right (le_max_right _ _)
69        (add_nonneg (add_nonneg (smoothDist_nonneg 2 _ _) hf0) hg0))
