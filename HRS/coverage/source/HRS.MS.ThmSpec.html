3/-!
4# THM-5: proof of `HRS.Whole.thermalDensity_spec` (the `s > 0` thermal equilibrium)
5
6Instantiate `ThmSetting` with `K x y = rieszKernel d s (x - y)`, `r = γ = gaussianComparison d a β`
7(`ThmSpecAux1.t5_setting`), take `M = max 1 (exp (β B))`, a minimiser `μ` of `G` on `Cset γ M` (THM-3) which is an a.e.
8fixed point of the Euler map (THM-4), and its pointwise representative `ρ₀ = thermalRep μ = Tmap μ`.  Then `ρ₀` is a
9thermal equilibrium bounded by `M c_γ` (`ThmSpecAux2`), every free-energy minimiser equals `ρ₀` a.e. (translation
10`F = G + β⁻¹ log c_γ`), and `thermalDensity` equals `ρ₀` at every point (pointwise rigidity of the thermal equation).
11
12Exports: `thermalDensity_eq_thermalRep` (`thermalDensity x = thermalRep μ x` for all `x`, and `thermalDensity ≤ M γ`).
13-/

26/-- The data produced by THM-3/THM-4 for the Riesz kernel and the Gaussian comparison density. -/
27theorem t5_exists_fixed (hd : 1 ≤ d) {s a β : ℝ} (hs0 : 0 < s) (hsd : s < d) (ha : 0 < a) (hβ : 0 < β) :
28    ∃ (B M : ℝ) (μ : Euc d → ℝ), ThmSetting (t5K d s) (gaussianComparison d a β) B ∧ 1 ≤ M ∧
29      Real.exp (β * B) ≤ M ∧ μ ∈ Cset (gaussianComparison d a β) M ∧
30      μ =ᵐ[volume] Tmap (t5K d s) (gaussianComparison d a β) β μ := by
31  obtain ⟨B, hS⟩ := t5_setting hd hs0 hsd ha hβ
32  have hM : 1 ≤ max 1 (Real.exp (β * B)) := le_max_left _ _
33  have hexp : Real.exp (β * B) ≤ max 1 (Real.exp (β * B)) := le_max_right _ _
34  obtain ⟨μ, hμ, hmin⟩ := exists_min_Cset hS hM hβ
35  exact ⟨B, _, μ, hS, hM, hexp, hμ, fixedpoint_of_min hS hβ hM hexp hμ hmin⟩

37/-- Everything about `thermalDensity` for `0 < s < d`: it is the pointwise representative `thermalRep μ` of an a.e.
38Euler fixed point `μ ∈ Cset γ M` minimising `G`, and `thermalDensity ≤ M γ`. -/
39theorem thermalDensity_eq_thermalRep (hd : 1 ≤ d) {s a β : ℝ} (hs0 : 0 < s) (hsd : s < d) (ha : 0 < a)
40    (hβ : 0 < β) :
41    ∃ (B M : ℝ) (μ : Euc d → ℝ), ThmSetting (t5K d s) (gaussianComparison d a β) B ∧ 1 ≤ M ∧
42      μ ∈ Cset (gaussianComparison d a β) M ∧
43      μ =ᵐ[volume] Tmap (t5K d s) (gaussianComparison d a β) β μ ∧
44      HRS.Whole.IsThermalEquilibrium d s a β (thermalRep (t5K d s) (gaussianComparison d a β) β μ) ∧
45      (∀ x, thermalDensity d s a β x = thermalRep (t5K d s) (gaussianComparison d a β) β μ x) ∧
46      (∀ x, thermalDensity d s a β x ≤ M * gaussianComparison d a β x) ∧
47      (∀ l : Euc d → ℝ, HRS.Whole.IsFreeEnergyMinimizer (rieszKernel d s) (confinement a) β l →
48        l =ᵐ[volume] thermalRep (t5K d s) (gaussianComparison d a β) β μ) := by
49  obtain ⟨B, M, μ, hS, hM, hexp, hμ, hfix⟩ := t5_exists_fixed hd hs0 hsd ha hβ
50  have hρC := thermalRep_mem_Cset hS hβ.le hexp hμ.1
51  have hρfix := thermalRep_fixed hfix
52  have hEq := t5_equilibrium_of_fixed hd hs0 hsd ha hβ hM hS hρC hρfix
53  have hex : ∃ ρ, HRS.Whole.IsThermalEquilibrium d s a β ρ := ⟨_, hEq⟩
54  have hτ : HRS.Whole.IsThermalEquilibrium d s a β (thermalDensity d s a β) := by
55    unfold thermalDensity
56    rw [dif_pos hex]
57    exact hex.choose_spec
58  have hae := t5_minimizer_ae_eq hd hs0 hsd ha hβ hM hS hρC hρfix hτ.1
59  have hpt : ∀ x, thermalDensity d s a β x = thermalRep (t5K d s) (gaussianComparison d a β) β μ x :=
60    thermalDensity_eq_pointwise hτ.2 hEq.2 hae
61  exact ⟨B, M, μ, hS, hM, hμ, hfix, hEq, hpt, fun x => (hpt x).symm ▸ hρC.2 x,
62    fun l hl => t5_minimizer_ae_eq hd hs0 hsd ha hβ hM hS hρC hρfix hl⟩

64/-- **THM-5.**  The thermal equilibrium of the `s > 0` Riesz gas in a quadratic confinement exists, is bounded,
65is the unique free-energy minimiser (a.e.), and is `thermalDensity`. -/
66theorem thermalDensity_spec_proof : type_of% @HRS.Whole.thermalDensity_spec := by
67  intro d s hd hs0 hsd a β ha hβ
68  have hd1 : 1 ≤ d := by omega
69  obtain ⟨B, M, μ, hS, hM, hμ, hfix, hEq, hpt, hle, hmin⟩ := thermalDensity_eq_thermalRep hd1 hs0 hsd ha hβ
70  have hbd : ∃ C : ℝ, ∀ x, thermalRep (t5K d s) (gaussianComparison d a β) β μ x ≤ C := by
71    refine ⟨M * t5c d a β, fun x => ?_⟩
72    have h1 : thermalRep (t5K d s) (gaussianComparison d a β) β μ x ≤ M * gaussianComparison d a β x := by
73      rw [← hpt x]; exact hle x
74    exact h1.trans (mul_le_mul_of_nonneg_left (gaussianComparison_le ha hβ x) (by linarith))
75  refine ⟨⟨_, hEq, hbd⟩, ?_, ?_⟩
76  · intro ρ ρ' h h'
77    exact (hmin ρ h).trans (hmin ρ' h').symm
78  · intro ρ h
79    refine (hmin ρ h).trans (Filter.EventuallyEq.symm ?_)
80    exact ae_of_all _ fun x => hpt x
