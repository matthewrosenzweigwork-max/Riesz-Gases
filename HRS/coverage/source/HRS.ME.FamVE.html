1/-
2# HRS.ME.FamV (packet V4b): upper bound of `critical_family_sharpness`
3
4`rate_W2_pointwise_upper`: one constant `C` with `W₂(rateLaw, rateGaussian) ≤ C N^{-1/2}` for every `θ ∈ [0,R]`, every thermal density
5`ρθ` of `V_θ` and every `N ≥ 2`.  Proof: for given `θ` choose a stationary Gibbs flow (`HRS.MD.EQLawFlow.eq_flow_exists_proof`) and a stationary
6Gaussian field (`HRS.Main.stationaryField_exists`) on the window `[0,1]`; the path bound `rate_family_path_bound` (V2) bounds the `W₂` distance of their
7path laws by `C₀ N^{-1/2}`; the evaluation map `rateEval` (V3a) has operator norm `K` independent of `θ`, carries the path laws to `rateLaw` (V3a) and
8`rateGaussian` (V3b), and does not increase `W₂` by more than the factor `K` (`HRS.Main.wassersteinE_map_clm_le`).
9`critical_family_sharpness_upper` integrates the pointwise bound over `[0,R]`.
10-/

24theorem rate_W2_pointwise_upper {d : ℕ} (hd : 2 ≤ d) (hs0 : 0 ≤ (d : ℝ) / 2) (hsd : (d : ℝ) / 2 < d)
25    (β : ℝ) (hβ : 0 < β) (k : Lattice d) (R : ℝ) :
26    ∃ C : ℝ, 0 < C ∧ ∀ θ ∈ Set.Icc (0 : ℝ) R, ∀ ρθ : Torus d → ℝ,
27      IsThermalDensity (periodicRiesz d ((d : ℝ) / 2)) (rateV k β θ) β ρθ →
28      ∀ N : ℕ, 2 ≤ N →
29        wassersteinE 2 (Main.rateLaw d ((d : ℝ) / 2) k β N θ) (Main.rateGaussian d ((d : ℝ) / 2) k β ρθ) ≤
30          ENNReal.ofReal (C * (N : ℝ) ^ (-(1 : ℝ) / 2)) := by
31  set m : ℝ := 24 * (d : ℝ) + 240 with hmdef
32  have hm : 24 * (d : ℝ) + 240 ≤ m := le_rfl
33  have hd0 : (0 : ℝ) ≤ d := Nat.cast_nonneg d
34  have hm2 : (d : ℝ) < 2 * m := by rw [hmdef]; nlinarith
35  have hmmax : max ((d : ℝ) + 1) ((d : ℝ) / 2 + 2) < m := by
36    rw [hmdef]; refine max_lt ?_ ?_ <;> nlinarith
37  obtain ⟨C₀, hC₀, h⟩ := rate_family_path_bound hd hs0 hsd β hβ k R m hm 1 one_pos
38  set L := rateEval d m 1 zero_le_one k with hL
39  refine ⟨(‖L‖ + 1) * C₀, by positivity, fun θ hθ ρθ hρθ N hN => ?_⟩
40  have hV : Torus.IsSmooth (rateV k β θ) := rateV_isSmooth k β θ
41  obtain ⟨Ω, _, P, _, X₀, W, X, hX⟩ :=
42    HRS.MD.EQLawFlow.eq_flow_exists_proof hd hs0 hsd (rateV k β θ) hV hβ hN (0 + 1) (by norm_num)
43  obtain ⟨Ω', _, P', _, Z, hZ⟩ :=
44    Main.stationaryField_exists hd hs0 hsd hβ (rateV k β θ) hV ρθ hρθ m hmmax
45  have hX' : HRS.LZ.IsStationaryGibbsFlow P ((d : ℝ) / 2) hs0 hsd (rateV k β θ) β hβ 1 X₀ W X := by
46    rwa [zero_add] at hX
47  have hW := h θ hθ ρθ hρθ N hN P X₀ W X hX'.flow hX'.filtered hX'.init P' Z hZ
48  have hz : AEMeasurable (Main.windowZeta m 1 0 P X) P :=
49    HRS.MD.EQInterp.windowZeta_aemeasurable hd hs0 hsd (rateV k β θ) hV hβ hN le_rfl zero_le_one P hX hm2
50  have hp : AEMeasurable (Main.windowPath m 1 0 Z) P' :=
51    HRS.ME.Ret.aemeasurable_windowPath_Icc P' Z hZ.meas
52      (by filter_upwards [hZ.cont] with ω hω using hω.mono Set.Icc_subset_Ici_self)
53  haveI : IsProbabilityMeasure (law P (Main.windowZeta m 1 0 P X)) :=
54    (Measure.isProbabilityMeasure_map_iff hz).2 inferInstance
55  haveI : IsProbabilityMeasure (law P' (Main.windowPath m 1 0 Z)) :=
56    (Measure.isProbabilityMeasure_map_iff hp).2 inferInstance
57  have h1 := rate_zeta_law hd hs0 hsd hβ k θ hN hm2 zero_le_one P hX
58  have h2 := rate_gauss_law hd hs0 hsd hβ k (rateV k β θ) ρθ hV hρθ zero_le_one P' Z hZ
59  rw [← h1, ← h2]
60  refine (Main.wassersteinE_map_clm_le L 2 (by norm_num) _ _).trans ?_
61  calc ENNReal.ofReal ‖L‖ * wassersteinE 2 (law P (Main.windowZeta m 1 0 P X)) (law P' (Main.windowPath m 1 0 Z))
62      ≤ ENNReal.ofReal ‖L‖ * ENNReal.ofReal (C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2)) := by gcongr
63    _ = ENNReal.ofReal (‖L‖ * (C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2))) :=
64        (ENNReal.ofReal_mul (norm_nonneg _)).symm
65    _ ≤ ENNReal.ofReal ((‖L‖ + 1) * C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2)) := by
66        apply ENNReal.ofReal_le_ofReal
67        have : 0 ≤ C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2) := by positivity
68        nlinarith [norm_nonneg L]

70theorem critical_family_sharpness_upper {d : ℕ} (hd : 2 ≤ d) (hs0 : 0 ≤ (d : ℝ) / 2) (hsd : (d : ℝ) / 2 < d)
71    (β : ℝ) (hβ : 0 < β) (k : Lattice d) :
72    ∃ C : ℝ, 0 < C ∧ ∀ ρ : ℝ → Torus d → ℝ,
73      (∀ θ, IsThermalDensity (periodicRiesz d ((d : ℝ) / 2)) (rateV k β θ) β (ρ θ)) →
74      ∀ N : ℕ, 2 ≤ N →
75        ∫⁻ θ in Set.Icc (0 : ℝ) (16 * (1 + β * rieszCoeff d ((d : ℝ) / 2) k)),
76            wassersteinE 2 (Main.rateLaw d ((d : ℝ) / 2) k β N θ)
77              (Main.rateGaussian d ((d : ℝ) / 2) k β (ρ θ)) ≤
78          ENNReal.ofReal (C * (N : ℝ) ^ (-(1 : ℝ) / 2)) := by
79  set R : ℝ := 16 * (1 + β * rieszCoeff d ((d : ℝ) / 2) k) with hR
80  obtain ⟨C₀, hC₀, h⟩ := rate_W2_pointwise_upper hd hs0 hsd β hβ k R
81  refine ⟨C₀ * (max R 0 + 1), by positivity, fun ρ hρ N hN => ?_⟩
82  have hpt : ∀ θ ∈ Set.Icc (0 : ℝ) R, wassersteinE 2 (Main.rateLaw d ((d : ℝ) / 2) k β N θ)
83      (Main.rateGaussian d ((d : ℝ) / 2) k β (ρ θ)) ≤ ENNReal.ofReal (C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2)) :=
84    fun θ hθ => h θ hθ (ρ θ) (hρ θ) N hN
85  calc ∫⁻ θ in Set.Icc (0 : ℝ) R, wassersteinE 2 (Main.rateLaw d ((d : ℝ) / 2) k β N θ)
86          (Main.rateGaussian d ((d : ℝ) / 2) k β (ρ θ))
87      ≤ ∫⁻ θ in Set.Icc (0 : ℝ) R, ENNReal.ofReal (C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2)) :=
88        setLIntegral_mono' measurableSet_Icc hpt
89    _ = ENNReal.ofReal (C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2)) * volume (Set.Icc (0 : ℝ) R) := by
90        rw [setLIntegral_const]
91    _ ≤ ENNReal.ofReal (C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2)) * ENNReal.ofReal (max R 0 + 1) := by
92        gcongr
93        rw [Real.volume_Icc]
94        apply ENNReal.ofReal_le_ofReal
95        linarith [le_max_left R 0, le_max_right R 0]
96    _ = ENNReal.ofReal (C₀ * (N : ℝ) ^ (-(1 : ℝ) / 2) * (max R 0 + 1)) :=
97        (ENNReal.ofReal_mul (by positivity)).symm
98    _ = ENNReal.ofReal (C₀ * (max R 0 + 1) * (N : ℝ) ^ (-(1 : ℝ) / 2)) := by ring_nf
