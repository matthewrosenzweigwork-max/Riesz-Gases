24/-- **P5** (the transfer inequality `ν(A) ≤ Θ(L) + L λ(A)`; scope note 1.1 step 3).  Let `P₁` be a filtered
25particle system with `N` particles on the horizon `u + T` whose initial law is `≪ ℙ_{N,β}`, and `P₂` a stationary
26Gibbs flow on the same horizon.  Then for every measurable set `A` of windows and every `L ≥ 0`, the law of the
27Gibbs-centred window under `P₁` satisfies
28`≤ ν{h > L} + L · (law of the Gibbs-centred window under P₂)(A)`, `ν = law P₁ X₀₁`, `h = dν/dℙ_{N,β}`.
29(Stated for a single `N`; `ν{h > L} ≤ uiTail L` is `le_iSup` in P6.)  Route: P1, P4, `ret_path_law_proof`
30(template `RetGlueB.lean:94-137`), split `h ≤ L` / `h > L`, `Measure.withDensity_rnDeriv_eq`. -/

40    {Ω₂ : Type} [MeasurableSpace Ω₂] (P₂ : Measure Ω₂) [IsProbabilityMeasure P₂]

42    (h₂ : IsStationaryGibbsFlow P₂ s hs0 hsd V β hβ (u + T) X₀₂ W₂ X₂)
43    (A : Set (PathSpace d m T)) (hA : MeasurableSet A) (L : ℝ) (hL : 0 ≤ L) :
44    (P₁.map (windowGibbs m T u (firstMarginal (canonicalGibbs (periodicRiesz d s) V N β)) X₁)) A ≤
45      (law P₁ X₀₁) {x | ENNReal.ofReal L <
46        (law P₁ X₀₁).rnDeriv (canonicalGibbs (periodicRiesz d s) V N β) x} +

48        (P₂.map (windowGibbs m T u (firstMarginal (canonicalGibbs (periodicRiesz d s) V N β)) X₂)) A := by
