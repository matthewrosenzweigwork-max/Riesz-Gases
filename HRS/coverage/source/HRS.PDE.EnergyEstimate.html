136theorem re_hsInner_le_basic {n : ℕ} (D : DynData d) (ρ : Torus d → ℝ) (m : ℕ) (F : Coeff d)
137    (hF : IsTrigPoly n F) (CT : ℝ)
138    (hT : ∀ α ∈ multiIndices d m,
139      |(bilinForm (wtP α) (transportCoeff D ρ F) F).re| ≤ CT * hsNormSq (m : ℝ) F) :

143      (∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ)) * CT * hsNormSq (m : ℝ) F -
144      ∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ) *
145        (bilinForm (wtP α) (nonlocalCoeff D ρ F) F).re := by
146  rw [re_hsInner_forwardGen D ρ m F hF]
147  have hX := sum_mw_gradDerivL2Sq m F hF
148  have hXG : ∑ α ∈ multiIndices d m, gradDerivL2Sq α F ≤
149      ∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ) * gradDerivL2Sq α F := by
150    refine Finset.sum_le_sum fun α hα => ?_
151    have h1 : (1 : ℝ) ≤ multinomialWeight m α := by exact_mod_cast one_le_multinomialWeight m α hα
152    have := gradDerivL2Sq_nonneg α F
153    nlinarith
154  have hTsum : ∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ) *
155      (bilinForm (wtP α) (transportCoeff D ρ F) F).re ≤
156      (∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ)) * CT * hsNormSq (m : ℝ) F := by
157    rw [mul_assoc, Finset.sum_mul]
158    refine Finset.sum_le_sum fun α hα => ?_
159    have h0 : (0 : ℝ) ≤ multinomialWeight m α := Nat.cast_nonneg _
160    exact mul_le_mul_of_nonneg_left ((le_abs_self _).trans (hT α hα)) h0
161  have hν := D.hβ
162  rw [← hX] at *
163  nlinarith [mul_le_mul_of_nonneg_left hXG hν]

167theorem transport_part (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (m : ℕ) (MB Λ VB : ℝ) :
168    ∃ CT : ℝ, 0 ≤ CT ∧ ∀ (D : DynData d), D.s = s → MatBound MB D.M → VHyp m VB D.V →
169      ∀ μ : Torus d → ℝ, IsProbDensity μ → Torus.IsCk (m + 3) μ → Torus.cbNorm (m + 3) μ ≤ Λ →
170      ∀ (n : ℕ) (F : Coeff d), IsRealCoeff F → IsTrigPoly n F → ∀ α ∈ multiIndices d m,
171        |(bilinForm (wtP α) (transportCoeff D μ F) F).re| ≤ CT * hsNormSq (m : ℝ) F := by
172  obtain ⟨B, hB⟩ := exists_driftFuns hd s hs0 hsd m MB Λ VB
173  obtain ⟨C0, hC0⟩ := transport_form_bound hd m B
174  refine ⟨max C0 0, le_max_right _ _, ?_⟩
175  intro D hDs hM hV μ hp hCk hΛ n F hFr hF α hα
176  obtain ⟨b, hb1, hb2, hb3⟩ := hB D hDs hM hV μ hp hCk hΛ
177  have e : transportCoeff D μ F = fun k => ∑ j, mulCoeff (coeffOf (b j)) (derivCoeff j F) k := by
178    funext k; simp only [transportCoeff, hb3]
179  rw [e]
180  exact (hC0 b hb1 hb2 α hα n F hFr hF).trans
181    (mul_le_mul_of_nonneg_right (le_max_left _ _) (hsNormSq_nonnegE _ _))

183theorem r1_part (s : ℝ) (hs : s ≤ (d : ℝ) - 2) (m : ℕ) (MB Λ : ℝ) :
184    ∃ CR : ℝ, 0 ≤ CR ∧ ∀ (D : DynData d), D.s = s → MatBound MB D.M →
185      ∀ μ : Torus d → ℝ, Torus.IsCk (m + 3) μ → Torus.cbNorm (m + 3) μ ≤ Λ →
186      ∀ (n : ℕ) (F : Coeff d), IsTrigPoly n F → ∀ α ∈ multiIndices d m,
187        |(bilinForm (wtP α) (nonlocalCoeff D μ F) F).re| ≤ CR * hsNormSq (m : ℝ) F := by
188  classical
189  refine ⟨(1 + ∑ α' ∈ multiIndices d m, r1Const d s MB Λ α') / 2, by
190    have : 0 ≤ ∑ α' ∈ multiIndices d m, r1Const d s MB Λ α' :=
191      Finset.sum_nonneg fun α' _ => r1Const_nonneg s MB Λ α'
192    positivity, ?_⟩
193  intro D hDs hM μ hμ hΛ n F hF α hα
194  have hαm : ∑ i, α i ≤ m := (Finset.mem_filter.mp hα).2
195  have hμ' : ContDiff ℝ ((m + 3 : ℕ) : WithTop ℕ∞) (Torus.lift μ) := by
196    unfold Torus.IsCk at hμ; push_cast; exact hμ
197  set H := hsNormSq (m : ℝ) F with hH
198  have hA : ∑ k ∈ boxFin d n, ‖multiDerivCoeff α F k‖ ^ 2 ≤ H :=
199    sum_sq_multiDeriv_le α m hαm F (memHs_of_isTrigPolyE hF _) _
200  have hBd : ∑ k ∈ boxFin d n, ‖multiDerivCoeff α (nonlocalCoeff D μ F) k‖ ^ 2 ≤
201      r1Const d s MB Λ α * H :=
202    nonlocal_R1_fixed s hs m MB Λ D hDs hM μ hμ' hΛ α hαm F hF _
203  have hr : r1Const d s MB Λ α ≤ ∑ α' ∈ multiIndices d m, r1Const d s MB Λ α' :=
204    Finset.single_le_sum (f := fun α' => r1Const d s MB Λ α') (fun α' _ => r1Const_nonneg s MB Λ α') hα
205  have hH0 : 0 ≤ H := hsNormSq_nonnegE _ _
206  have e : bilinForm (wtP α) (nonlocalCoeff D μ F) F =
207      ∑ k ∈ boxFin d n, conj (multiDerivCoeff α F k) * multiDerivCoeff α (nonlocalCoeff D μ F) k :=
208    bilinForm_weight_eq α _ F hF
209  calc |(bilinForm (wtP α) (nonlocalCoeff D μ F) F).re|
210      ≤ ‖bilinForm (wtP α) (nonlocalCoeff D μ F) F‖ := Complex.abs_re_le_norm _
211    _ ≤ ∑ k ∈ boxFin d n, ‖multiDerivCoeff α F k‖ * ‖multiDerivCoeff α (nonlocalCoeff D μ F) k‖ := by
212        rw [e]
213        refine (norm_sum_le _ _).trans (le_of_eq (Finset.sum_congr rfl fun k _ => ?_))
214        rw [norm_mul, RCLike.norm_conj]
215    _ ≤ ∑ k ∈ boxFin d n, (‖multiDerivCoeff α F k‖ ^ 2 +
216          ‖multiDerivCoeff α (nonlocalCoeff D μ F) k‖ ^ 2) / 2 := by
217        refine Finset.sum_le_sum fun k _ => ?_
218        nlinarith [sq_nonneg (‖multiDerivCoeff α F k‖ - ‖multiDerivCoeff α (nonlocalCoeff D μ F) k‖)]
219    _ = (∑ k ∈ boxFin d n, ‖multiDerivCoeff α F k‖ ^ 2 +
220          ∑ k ∈ boxFin d n, ‖multiDerivCoeff α (nonlocalCoeff D μ F) k‖ ^ 2) / 2 := by
221        rw [← Finset.sum_div, Finset.sum_add_distrib]
222    _ ≤ (H + r1Const d s MB Λ α * H) / 2 := by linarith
223    _ ≤ _ := by nlinarith

262  ∀ (D : DynData d) (ν₀ T : ℝ) (μ : ℝ → Torus d → ℝ)
263    (_ : AdjointData s m R MB Λ VB D ν₀ T μ)
264    (lam : Fin d → ℝ) (q : Fin d → Euc d) (_ : R.IsR2 → IsSymDiag D.M lam q)
265    (r : ℝ) (_ : r ∈ Set.Icc 0 T) (n : ℕ) (F : Coeff d) (_ : IsRealCoeff F) (_ : IsTrigPoly n F),
266    (R.IsR2 → WeightedDissipationFinite D m q (μ r) F) ∧
267    (hsInner (m : ℝ) (forwardGen D (μ r) F) F).re ≤
268      energyRHS R (R.lambdaM C ν₀ (thetaOrder d s)) D m lam q (μ r) F

270theorem formEst_R1 (hd : 1 ≤ d) (s : ℝ) (hs0 : 0 ≤ s) (hsd : s < d) (m : ℕ) (MB Λ VB : ℝ) :
271    ∃ C : ℝ, 0 < C ∧ FormEst d s m .R1 MB Λ VB C := by
272  by_cases hs : s ≤ (d : ℝ) - 2
273  · obtain ⟨CT, hCT0, hCT⟩ := transport_part hd s hs0 hsd m MB Λ VB
274    obtain ⟨CR, hCR0, hCR⟩ := r1_part s hs m MB Λ
275    have hc0 := sum_mw_pos (d := d) m
276    refine ⟨(∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ)) * (CT + CR) + 1, by positivity, ?_⟩
277    intro D ν₀ T μ A lam q _ r hr n F hFr hF
278    refine ⟨fun h => absurd h (by simp [Regime.IsR2]), ?_⟩
279    have hp := A.hμ.prob r hr
280    have hk := A.hμ.smooth r hr
281    have hb := A.hμ.bound r hr
282    have hbasic := re_hsInner_le_basic D (μ r) m F hF CT
283      (hCT D A.hs A.hM A.hV (μ r) hp hk hb n F hFr hF)
284    have hNl := neg_sum_le_of_abs m _ CR (hsNormSq (m : ℝ) F)
285      (hCR D A.hs A.hM (μ r) hk hb n F hF)
286    have hX := H1_sub_nonneg m F hF
287    have hH := hsNormSq_nonnegE (m : ℝ) F
288    have hν := D.hβ
289    have h1 : 0 ≤ (D.invβ / 2) * (hsNormSq ((m : ℝ) + 1) F - hsNormSq (m : ℝ) F) :=
290      mul_nonneg (by positivity) hX
291    have hno : ¬ Regime.R1.IsR2 := by simp [Regime.IsR2]
292    simp only [energyRHS, Regime.lambdaM]
293    rw [if_neg hno, sub_zero]
294    have h2 : (∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ)) * CT * hsNormSq (m : ℝ) F +
295        (∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ)) * CR * hsNormSq (m : ℝ) F ≤
296        ((∑ α ∈ multiIndices d m, (multinomialWeight m α : ℝ)) * (CT + CR) + 1) *
297          hsNormSq (m : ℝ) F := by nlinarith
298    have h3 : -(D.invβ / 2) * ∑ α ∈ multiIndices d m, gradDerivL2Sq α F =
299        -((D.invβ / 2) * ∑ α ∈ multiIndices d m, gradDerivL2Sq α F) := by ring
300    linarith
301  · refine ⟨1, one_pos, ?_⟩
302    intro D ν₀ T μ A
303    exact absurd A.regime hs
