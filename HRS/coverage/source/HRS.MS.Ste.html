12(coordinates `z = (x, s)`) the map `G(x,s) = s (2x, |x|²-1) = s(1+|x|²) · u(x)` is injective on `s > 0`
13with `|det G'| = (2s)^d (1+|x|²)` (`SteAux2`).  The image of `{x ∈ A, 0 < s < 1/(1+|x|²)}` lies in the cone

16(`ste_pointwise_bound`) gives `ν(S) = ∫_A ρ ≤ (A₀/2^d) · sphereSurface d S`.  No statement was adjusted.

18Exports: `logThermal_stereographic_proof : type_of% @HRS.Whole.logThermal_stereographic`.

203/-- The cone over `S ∩ S^d` contains the image of the parameter region. -/
204lemma ste_image_subset_cone {S : Set (EuclideanSpace ℝ (Fin (d + 1)))} :
205    steG '' steB {y : Fin d → ℝ | stereoInv (WithLp.toLp 2 y) ∈ S} ⊆
206      (fun w : Fin (d + 1) → ℝ => (WithLp.toLp 2 w : EuclideanSpace ℝ (Fin (d + 1)))) ⁻¹'
207        steCone S := by

210  have hQ : ‖steX z‖ ^ 2 = steQ z := steX_norm_sq z
211  have hmem : stereoInv (steX z) ∈ Metric.sphere (0 : EuclideanSpace ℝ (Fin (d + 1))) 1 := by
212    simp [ste_norm_stereoInv]
213  refine Set.mem_smul.2 ⟨z (Fin.last d) * (1 + ‖steX z‖ ^ 2), ⟨by positivity, ?_⟩,

229/-- Conjunct (1) of `logThermal_stereographic`. -/
230lemma ste_pushforward (hd : 2 ≤ d) {a β : ℝ} (ha : 0 < a) (hβ : 0 < β) :
231    ∃ C : ℝ≥0, (thermalMeasure d 0 a β).map stereoInv ≤ C • sphereSurface d := by
232  obtain ⟨A0, hA0, hρ⟩ := ste_pointwise_bound hd ha hβ
233  refine ⟨Real.toNNReal (A0 / 2 ^ d), ?_⟩
234  rw [Measure.le_iff]
235  intro S hS
236  have hmeas := ste_continuous_stereoInv (d := d)
237  have hA : MeasurableSet (stereoInv ⁻¹' S : Set (Euc d)) := hmeas.measurable hS
238  set A : Set (Euc d) := stereoInv ⁻¹' S with hAdef
239  have hmpt := PiLp.volume_preserving_toLp (Fin d)
240  have hembt := (MeasurableEquiv.toLp 2 (Fin d → ℝ)).measurableEmbedding
241  set A' : Set (Fin d → ℝ) := {y | stereoInv (WithLp.toLp 2 y) ∈ S} with hA'def
242  have hA'eq : A' = (WithLp.toLp 2 : (Fin d → ℝ) → Euc d) ⁻¹' A := rfl
243  have hA' : MeasurableSet A' := by
244    rw [hA'eq]; exact hmpt.measurable hA
245  -- the surface measure of `S`
246  have hσ : sphereSurface d S = (d + 1 : ℝ≥0∞) * volume (steCone S) := by
247    unfold sphereSurface
248    rw [Measure.map_apply measurable_subtype_coe hS,
249      Measure.toSphere_apply' _ (measurable_subtype_coe hS)]
250    simp [steCone]
251  -- lower bound for the cone volume
252  have hcone : volume (steG '' steB A') ≤ volume (steCone S) := by
253    calc volume (steG '' steB A') ≤ volume ((WithLp.toLp 2 : (Fin (d + 1) → ℝ) →
254          EuclideanSpace ℝ (Fin (d + 1))) ⁻¹' steCone S) := measure_mono ste_image_subset_cone
255      _ = volume (steCone S) := (PiLp.volume_preserving_toLp (Fin (d + 1))).measure_preimage_emb
256          (MeasurableEquiv.toLp 2 (Fin (d + 1) → ℝ)).measurableEmbedding _
257  -- the image volume as an integral over `A`
258  have hvol : volume (steG '' steB A') =
259      ∫⁻ x in A, ENNReal.ofReal ((2 / (1 + ‖x‖ ^ 2)) ^ d / (d + 1)) := by
260    rw [ste_volume_image hA', ste_lintegral_B hA']
261    have := hmpt.setLIntegral_comp_preimage_emb hembt
262      (fun x : Euc d => ENNReal.ofReal ((2 / (1 + ‖x‖ ^ 2)) ^ d / (d + 1))) A
263    rw [← this]
264    refine setLIntegral_congr_fun hA' fun y _ => ?_
265    have : ‖(WithLp.toLp 2 y : Euc d)‖ ^ 2 = steQ' y := by
266      rw [EuclideanSpace.norm_sq_eq]; simp [steQ']
267    rw [this]
268  -- the density bound
269  have hbd : ∀ x : Euc d, ENNReal.ofReal (thermalDensity d 0 a β x) ≤
270      ENNReal.ofReal (A0 / 2 ^ d) * ((d + 1 : ℝ≥0∞) *
271        ENNReal.ofReal ((2 / (1 + ‖x‖ ^ 2)) ^ d / (d + 1))) := by
272    intro x
273    have hpos : (0 : ℝ) < 1 + ‖x‖ ^ 2 := by positivity
274    have h1 := hρ x
275    have h2 : A0 * (1 + ‖x‖ ^ 2) ^ (-(d : ℝ)) = A0 / 2 ^ d * (2 / (1 + ‖x‖ ^ 2)) ^ d := by
276      rw [Real.rpow_neg hpos.le, Real.rpow_natCast, div_pow]
277      field_simp
278    have h3 : (d + 1 : ℝ≥0∞) = ENNReal.ofReal ((d : ℝ) + 1) := by
279      rw [ENNReal.ofReal_add (by positivity) (by norm_num)]; simp
280    rw [h3, ← ENNReal.ofReal_mul (by positivity), ← ENNReal.ofReal_mul (by positivity)]
281    refine ENNReal.ofReal_le_ofReal ?_
282    rw [h2] at h1
283    refine h1.trans (le_of_eq ?_)
284    field_simp
285  calc (thermalMeasure d 0 a β).map stereoInv S = ∫⁻ x in A, ENNReal.ofReal (thermalDensity d 0 a β x) := by
286        rw [Measure.map_apply hmeas.measurable hS]
287        exact withDensity_apply _ hA
288    _ ≤ ∫⁻ x in A, ENNReal.ofReal (A0 / 2 ^ d) * ((d + 1 : ℝ≥0∞) *
289        ENNReal.ofReal ((2 / (1 + ‖x‖ ^ 2)) ^ d / (d + 1))) := lintegral_mono hbd
290    _ = ENNReal.ofReal (A0 / 2 ^ d) * ((d + 1 : ℝ≥0∞) *
291        ∫⁻ x in A, ENNReal.ofReal ((2 / (1 + ‖x‖ ^ 2)) ^ d / (d + 1))) := by
292        have hm : Measurable fun x : Euc d => ENNReal.ofReal ((2 / (1 + ‖x‖ ^ 2)) ^ d / (d + 1)) := by
293          refine ENNReal.measurable_ofReal.comp ?_
294          fun_prop
295        rw [lintegral_const_mul _ (hm.const_mul _), lintegral_const_mul _ hm]
296    _ ≤ ENNReal.ofReal (A0 / 2 ^ d) * ((d + 1 : ℝ≥0∞) * volume (steCone S)) := by
297        rw [← hvol]; gcongr
298    _ = _ := by
299        rw [Measure.smul_apply, hσ, ENNReal.smul_def, smul_eq_mul]
