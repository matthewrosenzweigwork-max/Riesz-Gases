21theorem iid_field_comparison_proof (hd : 2 ≤ d) (m : ℝ) (hm : (d : ℝ) / 2 < m) (c Cbar : ℝ)
22    (hc : 0 < c) :
23    ∃ Cm : ℝ, 0 ≤ Cm ∧ ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, c ≤ ρ x) →
24      (∀ x, ρ x ≤ Cbar) → ∀ γ : Measure (NegSobolev d m),
25      IsCenteredGaussianWithCov m γ (iidCov (withDensityReal ρ)) →
26      ∀ N : ℕ, 1 ≤ N →
27        smoothDist 2 (law (Measure.pi fun _ : Fin N => withDensityReal ρ)
28            (fun X : Config d N => etaE m X (withDensityReal ρ))) γ ≤ Cm * iidRateSmooth d m N ∧
29        wassersteinE 1 (law (Measure.pi fun _ : Fin N => withDensityReal ρ)
30            (fun X : Config d N => etaE m X (withDensityReal ρ))) γ ≤

32        wassersteinE 2 (law (Measure.pi fun _ : Fin N => withDensityReal ρ)

40theorem iid_field_mixture_proof (hd : 2 ≤ d) (m : ℝ) (hm : (d : ℝ) / 2 < m) (c Cbar : ℝ) (hc : 0 < c)
41    (a : ℝ) (b : Torus d → ℝ) (hbm : Measurable b) (hbL : MemLp b ⊤ volume) :
42    ∃ Cm : ℝ, 0 ≤ Cm ∧ ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, c ≤ ρ x) →
43      (∀ x, ρ x ≤ Cbar) → pair (withDensityReal ρ) b = 0 →
44      ∀ γ : Measure (NegSobolev d m),
45      IsCenteredGaussianWithCov m γ (iidCov (withDensityReal ρ)) →
46      ∀ N : ℕ, 2 ≤ N → 4 * a ^ 2 * (eLpNorm b ⊤ volume).toReal ^ 2 ≤ N →
47        smoothDist 2
48            (law ((2 : ℝ≥0∞)⁻¹ • Measure.pi (fun _ : Fin N =>

51                  withDensityReal fun x => (1 - a * b x / Real.sqrt N) * ρ x))

55          Cm * iidRateSmooth d m N ∧
56        wassersteinE 1
57            (law ((2 : ℝ≥0∞)⁻¹ • Measure.pi (fun _ : Fin N =>
58                  withDensityReal fun x => (1 + a * b x / Real.sqrt N) * ρ x) +
59                (2 : ℝ≥0∞)⁻¹ • Measure.pi (fun _ : Fin N =>
60                  withDensityReal fun x => (1 - a * b x / Real.sqrt N) * ρ x))

62            ((2 : ℝ≥0∞)⁻¹ • γ.map (fun F => F + NegSobolev.ofFun m (fun x => a * b x * ρ x)) +
63              (2 : ℝ≥0∞)⁻¹ • γ.map (fun F => F - NegSobolev.ofFun m (fun x => a * b x * ρ x))) ≤
64          ENNReal.ofReal (Cm * iidRateW d m N) ∧
65        wassersteinE 2
66            (law ((2 : ℝ≥0∞)⁻¹ • Measure.pi (fun _ : Fin N =>
67                  withDensityReal fun x => (1 + a * b x / Real.sqrt N) * ρ x) +
68                (2 : ℝ≥0∞)⁻¹ • Measure.pi (fun _ : Fin N =>
69                  withDensityReal fun x => (1 - a * b x / Real.sqrt N) * ρ x))
