4/-!
5# Lipschitz-only regime `d - 2 ≤ s < d`: assembly (packet CM-S5)

7`super_core` (Serfaty smearing, `Renorm.lean`) bounds `|Comm_N|` by
8`C L (F_N + logPart + N⁻² ∑ selfExcess(r_i) + a_N^Λ)`; `nearField_control` (`NearField.lean`) bounds
9the self-energy sum by `C₃ (F_N + o_N(C₀'))`.  Choosing `C₀ = C₃ C₀' + K` closes the estimate.

22/-- **Regime `d - 2 ≤ s < d`**: `|Comm_N(v)| ≤ C L (F_N + o_N)` for `L`-Lipschitz `v`
23(no fractional norm).  `C₀, C` depend only on `d, s`. -/
24theorem commutator_estimate_super (hd : 1 ≤ d) (hs0 : 0 ≤ s) (hs : (d : ℝ) - 2 ≤ s) (hsd : s < d) :
25    ∃ C₀ C : ℝ, 0 ≤ C₀ ∧ 0 < C ∧ ∀ Λ : ℝ, 1 ≤ Λ → ∀ N : ℕ, 2 ≤ N →
26      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
27      ∀ X : Config d N, X ∉ collisionSet d N →
28      ∀ (L : ℝ) (v : Torus d → Euc d), 0 ≤ L → IsLipschitzVF L v →
29        |commQuantity d s v X (withDensityReal ρ)| ≤
30          C * L * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N) := by
31  obtain ⟨C₁, K, hC₁, hK, hcore⟩ := super_core hd hs0 hs hsd
32  obtain ⟨C₀', C₃, hC₀', hC₃, hNF⟩ := nearField_control hd hs0 hsd
33  refine ⟨C₃ * C₀' + K, C₁ * (1 + C₃), by positivity, by positivity, ?_⟩
34  intro Λ hΛ N hN ρ hρ hρΛ X hX L v hL hv
35  have h1 := hcore Λ hΛ N hN ρ hρ hρΛ X hX L v hL hv
36  have h2 := hNF Λ hΛ N hN ρ hρ hρΛ X hX
37  have ha := aNΛ_nonneg d s Λ N (by linarith)
38  rw [oNΛ_eq] at h2 ⊢
39  set F := rieszEnergy d s X (withDensityReal ρ) with hF
40  set S := (1 / (N : ℝ) ^ 2) * ∑ i, selfExcess d s Λ N (smearRadius Λ X i) with hS
41  set a := aNΛ d s Λ N with ha_def
42  set lp := logPart d s Λ N with hlp
43  have hT : F + lp + S + K * a ≤ (1 + C₃) * (F + (lp + (C₃ * C₀' + K) * a)) := by
44    have hc : 0 ≤ C₃ * C₀' + K := by positivity
45    nlinarith [mul_nonneg ha (mul_nonneg hC₃.le hc), mul_nonneg ha hK]
46  calc |commQuantity d s v X (withDensityReal ρ)|
47      ≤ C₁ * L * (F + lp + S + K * a) := h1
48    _ ≤ C₁ * L * ((1 + C₃) * (F + (lp + (C₃ * C₀' + K) * a))) := by
49        have : 0 ≤ C₁ * L := by positivity
50        exact mul_le_mul_of_nonneg_left hT this
51    _ = C₁ * (1 + C₃) * L * (F + (lp + (C₃ * C₀' + K) * a)) := by ring
