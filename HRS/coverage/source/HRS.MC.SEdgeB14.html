6For `F in C_b^5`, `f` is the Stein solution of B11b.  B12 gives the exact decomposition
7`sqrt N (E F(W_N) - E F(Z)) = e1 - e2 + e3` with
8* `e1 = sum_ab sqrt N (C_ab - E A_ab) E g_ab(W_N) -> 0` (B13a),
9* `e2 = sum_ab E[Y_{w_ab} g_ab(W_N)] -> sum mu(w_ab r_c) E d_cba f(Z)` (B13b),
10* `e3 = beta sum_a E[B_a d_a f(W_N)] -> (beta/2) sum Q_{a;bc} E d_abc f(Z)` (B13c; absent for `beta = 0`),
11and the cubic cancellation B9cancel together with the symmetry of `D^3 f` and B11b give the limit.

230theorem static_edge_assembly_proof (hd : 2 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hs : s < (d : ℝ) / 2) (β : ℝ)
231    (hβ : 0 ≤ β) (ρ : Torus d → ℝ) (hρ : SmoothPosDensity ρ) (p : ℕ) (φ : Fin p → Torus d → ℝ)
232    (hφ : ∀ a, Torus.IsSmooth (φ a)) :
233    let μ := withDensityReal ρ
234    let r : Fin p → Torus d → ℝ := fun a => responseInverse (periodicRiesz d s) μ β (φ a)
235    let C : Matrix (Fin p) (Fin p) ℝ := Matrix.of fun a b => ∫ x, φ a x * r b x ∂μ
236    let κ : Fin p → Fin p → Fin p → ℝ := fun a b c => ∫ x, r a x * r b x * r c x ∂μ
237    ∀ F : EuclideanSpace ℝ (Fin p) → ℝ, MemCb 5 F →
238      Tendsto (fun N : ℕ => Real.sqrt N *
239        (∫ X, F (centeredStat (modulatedGibbs (periodicRiesz d s) N β μ) φ X)
240            ∂(modulatedGibbs (periodicRiesz d s) N β μ) -
241          ∫ z, F z ∂(multivariateGaussian 0 C) -
242          (1 / (6 * Real.sqrt N)) * ∑ a, ∑ b, ∑ c, κ a b c *
243            ∫ z, iteratedFDeriv ℝ 3 F z ![stdVec a, stdVec b, stdVec c] ∂(multivariateGaussian 0 C)))
244        atTop (𝓝 0) := by
245  intro μ r C κ F hF
246  exact b14_main hd hs0 hs hβ hρ hφ hF
