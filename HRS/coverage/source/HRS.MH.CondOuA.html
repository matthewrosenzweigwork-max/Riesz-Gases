21/-- Finitely many unit complex numbers `z` with `(z w).re = c`, for a unit `w`. -/
22theorem unit_re_set_finite (c : ℝ) (w : ℂ) (hw : ‖w‖ = 1) :
23    {z : ℂ | (z * w).re = c ∧ ‖z‖ = 1}.Finite := by
24  have hw0 : w ≠ 0 := by
25    intro h; rw [h, norm_zero] at hw; exact zero_ne_one hw
26  have hB : {v : ℂ | v.re = c ∧ ‖v‖ = 1}.Finite := by
27    refine (Set.toFinite {(⟨c, Real.sqrt (1 - c^2)⟩ : ℂ), ⟨c, -Real.sqrt (1 - c^2)⟩}).subset ?_
28    rintro v ⟨h1, h2⟩
29    have h3 : v.im ^ 2 = 1 - c ^ 2 := by
30      have := Complex.sq_norm v
31      rw [h2, Complex.normSq_apply, h1] at this
32      nlinarith
33    have h4 : 0 ≤ 1 - c ^ 2 := by rw [← h3]; positivity
34    have h5 : v.im = Real.sqrt (1 - c^2) ∨ v.im = -Real.sqrt (1 - c^2) := by
35      have : v.im ^ 2 = (Real.sqrt (1 - c^2)) ^ 2 := by rw [Real.sq_sqrt h4, h3]
36      exact sq_eq_sq_iff_eq_or_eq_neg.mp this
37    rcases h5 with h | h
38    · left; exact Complex.ext h1 h
39    · right; exact Complex.ext h1 h
40  refine (hB.image (fun v : ℂ => v * w⁻¹)).subset ?_
41  rintro z ⟨h1, h2⟩
42  refine ⟨z * w, ⟨h1, ?_⟩, mul_inv_cancel_right₀ hw0 z⟩
43  rw [norm_mul, h2, hw, mul_one]

45/-- Level sets of `t ↦ Re(e_1(t) w)` are null on the circle, for a unit `w`. -/
46theorem circle_mul_re_null (w : ℂ) (hw : ‖w‖ = 1) (c : ℝ) :
47    (volume : Measure UnitAddCircle) {u : UnitAddCircle | (fourier (1:ℤ) u * w).re = c} = 0 := by
48  apply circle_finite_null
49  have hinj : Function.Injective (fun t : UnitAddCircle => fourier (1:ℤ) t) := by
50    intro a b h
51    simp only [fourier_one] at h
52    exact AddCircle.injective_toCircle one_ne_zero (Subtype.ext h)
53  refine ((unit_re_set_finite c w hw).preimage hinj.injOn).subset fun u hu => ⟨hu, ?_⟩
54  exact Circle.norm_coe _

56/-- The same for `t ↦ Re(e_n(t) w)`, `n ≠ 0`, by the measure-preserving map `t ↦ n • t`. -/
57theorem circle_mul_re_null_zsmul (n : ℤ) (hn : n ≠ 0) (w : ℂ) (hw : ‖w‖ = 1) (c : ℝ) :
58    (volume : Measure UnitAddCircle) {t : UnitAddCircle | (fourier n t * w).re = c} = 0 := by
59  have hset : {t : UnitAddCircle | (fourier n t * w).re = c} =
60      (fun t : UnitAddCircle => n • t) ⁻¹' {u : UnitAddCircle | (fourier (1:ℤ) u * w).re = c} := by
61    ext t
62    simp only [Set.mem_setOf_eq, Set.mem_preimage]
63    rw [fourier_apply, fourier_apply, one_smul]
64  have hs : MeasurableSet {u : UnitAddCircle | (fourier (1:ℤ) u * w).re = c} :=
65    (isClosed_eq (Complex.continuous_re.comp ((fourier (1:ℤ)).continuous.mul continuous_const))
66      continuous_const).measurableSet
67  have hpm := MeasureTheory.Measure.measurePreserving_zsmul (volume : Measure UnitAddCircle) hn
68  have hmp := hpm.measure_preimage hs.nullMeasurableSet
69  rw [hset, hmp]
70  exact circle_mul_re_null w hw c

72/-- Every level set of `cos(2π k·x)` on the torus is Haar-null, for `k ≠ 0`. -/
73theorem torus_cos_level_null (m : ℕ) (k : Lattice (m+1)) (hk : k ≠ 0) (c : ℝ) :
74    (volume : Measure (Torus (m+1))) {x | Torus.cosMode k x = c} = 0 := by
75  obtain ⟨j, hj⟩ : ∃ j, k j ≠ 0 := by
76    by_contra h
77    push_neg at h
78    exact hk (funext h)
79  have mp := measurePreserving_piFinSuccAbove (fun _ : Fin (m+1) => (volume : Measure UnitAddCircle)) j
80  set W : (Fin m → UnitAddCircle) → ℂ :=
81    fun y => ∏ i : Fin m, fourier (k (j.succAbove i)) (y i) with hW
82  set S : Set (UnitAddCircle × (Fin m → UnitAddCircle)) :=
83    {p | (fourier (k j) p.1 * W p.2).re = c} with hS
84  have hmeas : MeasurableSet S := by
85    have hc : Continuous (fun p : UnitAddCircle × (Fin m → UnitAddCircle) =>
86        (fourier (k j) p.1 * W p.2).re) := by
87      rw [hW]
88      fun_prop
89    exact (isClosed_eq hc continuous_const).measurableSet
90  have hpre : {x : Torus (m+1) | Torus.cosMode k x = c} =
91      (MeasurableEquiv.piFinSuccAbove (fun _ : Fin (m+1) => UnitAddCircle) j) ⁻¹' S := by
92    ext x
93    have hmode : Torus.mode k x = fourier (k j) (x j) *
94        ∏ i : Fin m, fourier (k (j.succAbove i)) (x (j.succAbove i)) := by
95      simp only [Torus.mode, UnitAddTorus.mFourier, ContinuousMap.coe_mk]
96      exact Fin.prod_univ_succAbove (fun i => fourier (k i) (x i)) j
97    simp only [Set.mem_setOf_eq, Set.mem_preimage, hS, hW, MeasurableEquiv.piFinSuccAbove_apply,
98      Torus.cosMode, hmode]
99    exact Iff.rfl
100  have hfib : ∀ y : Fin m → UnitAddCircle,
101      (volume : Measure UnitAddCircle) ((fun a => (a, y)) ⁻¹' S) = 0 := by
102    intro y
103    have hw : ‖W y‖ = 1 := by
104      rw [hW]
105      simp only
106      rw [norm_prod]
107      exact Finset.prod_eq_one fun i _ => Circle.norm_coe _
108    have e1 : (fun a => (a, y)) ⁻¹' S = {a : UnitAddCircle | (fourier (k j) a * W y).re = c} := rfl
109    rw [e1]
110    exact circle_mul_re_null_zsmul (k j) hj (W y) hw c
111  rw [hpre]
112  show (Measure.pi fun _ : Fin (m+1) => (volume : Measure UnitAddCircle)) _ = 0
113  rw [mp.measure_preimage_equiv, Measure.prod_apply_symm hmeas]
114  simp [hfib]

116/-- The sum-level set `{Σ_i cos(2π k·x_i) = 0}` is null in `N = n+1` particles. -/
117theorem null_cos_sum_gen (n : ℕ) {m : ℕ} (k : Lattice (m+1)) (hk : k ≠ 0) :
118    (volume : Measure (Config (m+1) (n+1)))
119      {X | ∑ i, Torus.cosMode k (X i) = 0} = 0 := by
120  have mp := measurePreserving_piFinSuccAbove (fun _ : Fin (n+1) => (volume : Measure (Torus (m+1)))) 0
121  set S' : Set (Torus (m+1) × (Fin n → Torus (m+1))) :=
122    {p | Torus.cosMode k p.1 + ∑ i, Torus.cosMode k (p.2 i) = 0} with hS'
123  have hmeas : MeasurableSet S' := by
124    have hc : Continuous (Torus.cosMode k) := Complex.continuous_re.comp (Torus.continuous_mode _)
125    refine measurableSet_eq_fun ?_ measurable_const
126    exact (hc.measurable.comp measurable_fst).add
127      (Finset.measurable_sum _ fun i _ => hc.measurable.comp ((measurable_pi_apply i).comp measurable_snd))
128  have hpre : {X : Config (m+1) (n+1) | ∑ i, Torus.cosMode k (X i) = 0}
129      = (MeasurableEquiv.piFinSuccAbove (fun _ : Fin (n+1) => Torus (m+1)) 0) ⁻¹' S' := by
130    ext X
131    simp only [Set.mem_setOf_eq, Set.mem_preimage, hS', MeasurableEquiv.piFinSuccAbove_apply,
132      Fin.sum_univ_succ]
133    rfl
134  rw [hpre]
135  show (Measure.pi fun _ : Fin (n+1) => (volume : Measure (Torus (m+1)))) _ = 0
136  rw [mp.measure_preimage_equiv, Measure.prod_apply_symm hmeas]
137  have hfib : ∀ y : Fin n → Torus (m+1),
138      (volume : Measure (Torus (m+1))) ((fun x => (x, y)) ⁻¹' S') = 0 := by
139    intro y
140    set c := ∑ i, Torus.cosMode k (y i) with hc
141    have e1 : (fun x => (x, y)) ⁻¹' S' = {x : Torus (m+1) | Torus.cosMode k x = -c} := by
142      ext x
143      simp only [hS', Set.mem_preimage, Set.mem_setOf_eq]
144      constructor <;> intro h <;> linarith
145    rw [e1]
146    exact torus_cos_level_null m k hk (-c)
147  simp [hfib]

155/-- The shift `t = mk(e_j / (2 k_j))` flips the sign of `cos(2π k·x)`. -/
156theorem cos_shift {d : ℕ} (k : Lattice d) (j : Fin d) (hj : k j ≠ 0) (x : Torus d) :
157    Torus.cosMode k (Torus.mk (EuclideanSpace.single j (1 / (2 * (k j : ℝ)))) + x) =
158      - Torus.cosMode k x := by
159  have hj' : (k j : ℝ) ≠ 0 := by exact_mod_cast hj
160  conv_lhs => rw [← Torus.mk_rep x, ← Torus.mk_add, Torus.cosMode_mk]
161  conv_rhs => rw [← Torus.mk_rep x, Torus.cosMode_mk]
162  rw [Torus.dot_add]
163  have : Lattice.dot k (EuclideanSpace.single j (1 / (2 * (k j : ℝ)))) = 1/2 := by
164    simp only [Lattice.dot, EuclideanSpace.single_apply, mul_ite, mul_zero, Finset.sum_ite_eq',
165      Finset.mem_univ, if_true]
166    field_simp
167  rw [this, mul_add, show 2 * Real.pi * (1/2 : ℝ) = Real.pi by ring, add_comm Real.pi,
168    Real.cos_add_pi]

170/-- `∫ cos(2π k·x) dx = 0` for `k ≠ 0`. -/
171theorem integral_cos_mode_zero {d : ℕ} (k : Lattice d) (hk : k ≠ 0) :
172    ∫ x : Torus d, Torus.cosMode k x = 0 := by
173  obtain ⟨j, hj⟩ := exists_coord_ne_zero k hk
174  have h1 : ∫ x : Torus d, Torus.cosMode k
175      (Torus.mk (EuclideanSpace.single j (1 / (2 * (k j : ℝ)))) + x) =
176      ∫ x : Torus d, Torus.cosMode k x :=
177    integral_add_left_eq_self (μ := (volume : Measure (Torus d))) _ _
178  simp only [cos_shift k j hj, integral_neg] at h1
179  linarith
