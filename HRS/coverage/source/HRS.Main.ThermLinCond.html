28theorem thermlin_wassersteinE_self {B : Type*} [NormedAddCommGroup B] [NormedSpace ℝ B] [MeasurableSpace B]
29    [BorelSpace B] (p : ℝ) (hp : 1 ≤ p) (μ : Measure B) [IsProbabilityMeasure μ] :
30    wassersteinE p μ μ = 0 := by
31  have hp0 : 0 < p := by linarith
32  have h := wassersteinE_le_of_coupling μ p hp id id measurable_id.aemeasurable measurable_id.aemeasurable
33  simp only [law, Measure.map_id, sub_self, norm_zero] at h
34  rw [Real.zero_rpow hp0.ne'] at h
35  refine le_antisymm (h.trans ?_) zero_le
36  simp [ENNReal.zero_rpow_of_pos (inv_pos.2 hp0)]

41theorem thermlin_ofReal_mul_le {C₁ C : ℝ} (hC₁ : 0 ≤ C₁) (h : C₁ ≤ C) (x : ℝ) :
42    ENNReal.ofReal (C₁ * x) ≤ ENNReal.ofReal (C * x) := by
43  rcases le_total 0 x with hx | hx
44  · exact ENNReal.ofReal_le_ofReal (mul_le_mul_of_nonneg_right h hx)
45  · rw [ENNReal.ofReal_of_nonpos (mul_nonpos_of_nonneg_of_nonpos hC₁ hx)]
46    exact zero_le

58theorem thermalization_proof (S : PreparedData d) (m : ℝ) (hm : (d : ℝ) + 2 < m) (T : ℝ) (hT : 0 < T)
59    (ρβ : Torus d → ℝ) (hρβ : IsThermalDensity (periodicRiesz d S.s) S.V S.β ρβ) :
60    ∃ C c : ℝ, 0 < C ∧ 0 < c ∧
61      ∀ a : ℝ, 0 ≤ a → ∀ N : ℕ, 2 ≤ N →
62      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
63        (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N),
64        SelectedFlow P S.dyn (a + T) X₀ W X →
65        (∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
66          (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X) →
67        relEntropy (law P X₀) (S.refMeasure N 0) < ⊤ →
68      ∀ {Ω'' : Type} [MeasurableSpace Ω''] (P'' : Measure Ω'') [IsProbabilityMeasure P'']
69        (Z : ℝ → Ω'' → NegSobolev d m), IsStationaryGaussianField P'' S.s S.β S.hs0 S.hsd S.hβ S.V ρβ m Z →
70        ENNReal.ofReal (smoothDist 2 (S.etaLaw m T a P X) (law P'' (windowPath m T 0 Z))) ≤
71          ENNReal.ofReal (C * (S.B N (law P X₀) / Real.sqrt N +
72              Real.sqrt (S.B N (law P X₀) / N))) +
73            ENNReal.ofReal (C * Real.exp (-c * a)) *
74              (1 + ∫⁻ ω, ‖etaE m (X₀ ω) S.μ0‖ₑ ∂P) ∧
75        wassersteinE 1 (S.etaLaw m T a P X) (law P'' (windowPath m T 0 Z)) ≤
76          ENNReal.ofReal (C * (S.B N (law P X₀) / Real.sqrt N +
77              (S.B N (law P X₀) / N) ^ ((1 : ℝ) / 4))) +
78            ENNReal.ofReal (C * Real.exp (-c * a)) *
79              (1 + ∫⁻ ω, ‖etaE m (X₀ ω) S.μ0‖ₑ ∂P) := by
80  classical
81  obtain ⟨C1, c1, hC1, hc1, hpf⟩ := prepared_fluctuations_proof S m hm T hT
82  obtain ⟨C2, c2, hC2, hc2, hgc⟩ := thermlin_gaussian_comparison_proof S m hm T hT ρβ hρβ
83  refine ⟨max C1 C2, min c1 c2, lt_max_of_lt_left hC1, lt_min hc1 hc2, ?_⟩
84  intro a ha N hN Ω _ P _ X₀ W X hflow hfilt hent Ω'' _ P'' _ Z hZ
85  have hηm := thermlin_eta_measurable S m hm hN
86  have hη0 : Measurable (fun ω => etaE m (X₀ ω) S.μ0) := hηm.comp (hflow.init_meas)
87  set ν : Measure (NegSobolev d m) := law P (fun ω => etaE m (X₀ ω) S.μ0) with hν
88  haveI : IsProbabilityMeasure ν := by rw [hν]; unfold law; infer_instance
89  obtain ⟨Ω', mΩ', P', hP', Y₀, Y, hY⟩ := preparedLinear_exists_proof S m hm (a + T) (by linarith) ν
90  have h1 := hpf a ha N hN P X₀ W X hflow hfilt hent ν P' Y₀ Y hY
91  have hinit : S.initW m 1 P X₀ ν = 0 := by
92    unfold PreparedData.initW
93    exact thermlin_wassersteinE_self 1 le_rfl ν
94  obtain ⟨h1s, h1w⟩ := h1.1 (by rw [hinit]; exact ENNReal.zero_ne_top)
95  rw [hinit, ENNReal.toReal_zero, mul_zero, zero_add] at h1s h1w
96  obtain ⟨g1, -⟩ := hgc a ha ν P' Y₀ Y hY P'' Z hZ
97  haveI : IsProbabilityMeasure (S.etaLaw m T a P X) := by unfold PreparedData.etaLaw; infer_instance
98  have hmom : ∫⁻ y, ‖y‖ₑ ∂ν = ∫⁻ ω, ‖etaE m (X₀ ω) S.μ0‖ₑ ∂P := by
99    rw [hν, law, lintegral_map (f := fun y : NegSobolev d m => ‖y‖ₑ) (by fun_prop) hη0]
100  have hexp : ENNReal.ofReal (C2 * Real.exp (-c2 * a)) ≤ ENNReal.ofReal (max C1 C2 * Real.exp (-min c1 c2 * a)) := by
101    refine ENNReal.ofReal_le_ofReal (mul_le_mul (le_max_right _ _) ?_ (Real.exp_pos _).le
102      (le_trans hC1.le (le_max_left _ _)))
103    refine Real.exp_le_exp.2 ?_
104    nlinarith [min_le_right c1 c2]
105  have hrhs : ENNReal.ofReal (C2 * Real.exp (-c2 * a)) * (1 + ∫⁻ y, ‖y‖ₑ ∂ν) ≤
106      ENNReal.ofReal (max C1 C2 * Real.exp (-min c1 c2 * a)) *
107        (1 + ∫⁻ ω, ‖etaE m (X₀ ω) S.μ0‖ₑ ∂P) := by
108    rw [hmom]
109    gcongr
110  constructor
111  · calc ENNReal.ofReal (smoothDist 2 (S.etaLaw m T a P X) (law P'' (windowPath m T 0 Z)))
112        ≤ ENNReal.ofReal (smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) +
113            smoothDist 2 (law P' (windowPath m T a Y)) (law P'' (windowPath m T 0 Z))) :=
114          ENNReal.ofReal_le_ofReal (smoothDist_triangle 2 _ _ _)
115      _ = ENNReal.ofReal (smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y))) +
116            ENNReal.ofReal (smoothDist 2 (law P' (windowPath m T a Y)) (law P'' (windowPath m T 0 Z))) :=
117          ENNReal.ofReal_add (smoothDist_nonneg _ _ _) (smoothDist_nonneg _ _ _)
118      _ ≤ ENNReal.ofReal (max C1 C2 * (S.B N (law P X₀) / Real.sqrt N +
119              Real.sqrt (S.B N (law P X₀) / N))) +
120            ENNReal.ofReal (max C1 C2 * Real.exp (-min c1 c2 * a)) *
121              (1 + ∫⁻ ω, ‖etaE m (X₀ ω) S.μ0‖ₑ ∂P) := by
122          refine add_le_add ?_ (le_trans ?_ hrhs)
123          · exact (ENNReal.ofReal_le_ofReal h1s).trans
124              (thermlin_ofReal_mul_le hC1.le (le_max_left _ _) _)
125          · exact (ENNReal.ofReal_le_ofReal (smoothDist_succ_le 1 _ _)).trans
126              ((smoothDist_one_le_wasserstein _ _).trans g1)
127  · calc wassersteinE 1 (S.etaLaw m T a P X) (law P'' (windowPath m T 0 Z))
128        ≤ wassersteinE 1 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) +
129            wassersteinE 1 (law P' (windowPath m T a Y)) (law P'' (windowPath m T 0 Z)) :=
130          wassersteinE_triangle 1 le_rfl _ _ _
131      _ ≤ ENNReal.ofReal (max C1 C2 * (S.B N (law P X₀) / Real.sqrt N +
132              (S.B N (law P X₀) / N) ^ ((1 : ℝ) / 4))) +
133            ENNReal.ofReal (max C1 C2 * Real.exp (-min c1 c2 * a)) *
134              (1 + ∫⁻ ω, ‖etaE m (X₀ ω) S.μ0‖ₑ ∂P) := by
135          refine add_le_add ?_ (le_trans g1 hrhs)
136          exact h1w.trans (thermlin_ofReal_mul_le hC1.le (le_max_left _ _) _)
