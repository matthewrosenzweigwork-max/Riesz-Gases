30/-! ### (2) Fisher information with a general potential -/
31
32/-- `fisherInfo` with the potential `U` and the inverse temperature factor `invβ` (`β = invβ⁻¹`) as
33arguments instead of `totalPotential D`, `D.invβ`: the same variational formula
34`B:eq:selected-fisher-variational` (TeX M L12205-12209),
35`sup_ζ ∫ (-2 Div ζ + 2 β DU[ζ] - |ζ|²) dP` over `ζ ∈ C_c^∞(Ω_N; ℝ^{dN})`, with `DU[ζ]` computed by
36differentiating the lift of `U`.  It allows a regularised potential `U_ε` (B.15, Step 2).
37`fisherInfo D N P = b15_fisherInfoGen D.invβ (totalPotential D) P` is `b15_fisherInfo_eq_gen`. -/
38def b15_fisherInfoGen (invβ : ℝ) (U : Config d N → ℝ) (P : Measure (Config d N)) : ℝ≥0∞ :=
39  ⨆ (ζ : Config d N → (Fin N → Euc d)) (_ : IsFisherTestField ζ),
40    ENNReal.ofReal (∫ X : Config d N,
41      (-2 * configDiv ζ X
42        + 2 * invβ⁻¹ * fderiv ℝ (fun Z : Fin N → Euc d => U (fun i => Torus.mk (Z i)))
43            (fun i => Torus.rep (X i)) (ζ X)
44        - ∑ i, ‖ζ X i‖ ^ 2) ∂P)
45
46/-- `fisherInfo` is `b15_fisherInfoGen` at the total potential. -/
47theorem b15_fisherInfo_eq_gen (D : DynData d) (N : ℕ) (P : Measure (Config d N)) :
48    fisherInfo D N P = b15_fisherInfoGen D.invβ (totalPotential D) P := rfl

50/-! ### (1) The conclusion of B.15 and the collision-free (strong) branch -/
51
52/-- The conclusion of Proposition B.15 (`M:B:prop:sde-entropy-law`, TeX M L12140-12152), literally the
53conclusion of `sde_entropy_law` (CepaFlow.lean): for every deterministic `t ∈ [0,T]` the time marginal
54`P_N^t` has a density, the Gibbs entropy inequality `B:eq:selected-gradient-entropy`
55`𝓗_G(P_N^t) + β_N⁻¹ ∫_0^t 𝓘_G(P_N^r) dr ≤ 𝓗_G(P_N^0)` holds in `EReal` (`gibbsEntropy`, `fisherInfo`,
56`β_N⁻¹ = invβ`, the time integral a `lintegral` over `Icc 0 t`), and the family solves the selected-law
57forward equation of B.3 (`SolvesSelectedFKE`, distributional on `Ω_N`).  A named `Prop` so that the
58free statement, the CL statement and the dispatcher share one text. -/
59def b15_EntropyLawConcl {Ω : Type*} [mΩ : MeasurableSpace Ω] (P : Measure Ω) (D : DynData d) (T : ℝ)
60    (X : ℝ≥0 → Ω → Config d N) : Prop :=
61  (∀ t ∈ Set.Icc 0 T, HasSelectedDensity P X t ∧
62    gibbsEntropy D (selectedLaw P X t)
63      + ((ENNReal.ofReal D.invβ *
64          ∫⁻ r in Set.Icc (0 : ℝ) t, fisherInfo D N (selectedLaw P X r) : ℝ≥0∞) : EReal)
65      ≤ gibbsEntropy D (selectedLaw P X 0)) ∧
66  SolvesSelectedFKE D T (selectedLaw P X)
67
68/-- Proposition B.15 (`M:B:prop:sde-entropy-law`, TeX M L12140-12152; proof L12153-12235), the
69collision-free (strong-solution) branch only.  STATEMENT, proof open (target of the later packets
70C (radial cutoff `g_ε`, `X^ε = X` eventually), B (lower semicontinuity of `klDiv` and of the Fisher
71information), D (parabolic theory on `𝕋^{dN}` for the smooth coefficients, Itô duality, entropy
72dissipation), E (assembly)).
73
74Representation decisions.
75* `SelectedFlow` (`flow` field) is `(IsStrongSolution ∧ PathwiseUnique) ∨ (d = 1 ∧ s = 0 ∧ … CL)`; the
76  present statement is the first disjunct, with the four standing fields of `SelectedFlow`
77  (`init_meas`, `init_distinct`, `brownian`, `indep`) as the explicit hypotheses `hX₀ hdist hW hind`
78  (names as in `cepa_entropy_confined`), so `SelectedFlow P D T X₀ W X` is rebuilt as
79  `⟨hX₀, hdist, hW, hind, Or.inl ⟨hS, hU⟩⟩`.  The FKE conjunct then follows from `selected_law_fke`
80  (`b15_solvesSelectedFKE`).
81* Collision-freeness of `X` is the field `IsStrongSolution.no_collision` (TeX: B:eq:particle-uniform-separation
82  for the strong solution, L12163-12165, gives the a.s. positive minimal pair distance);
83  `PathwiseUnique` is kept as a hypothesis because it is what the strong disjunct supplies (the TeX
84  Step 1 uses *stopped* pathwise uniqueness to show `X^ε = X` on `[0,T]` for `ε` small, L12162-12166).
85* All other hypotheses, and the conclusion, are exactly those of `sde_entropy_law`:
86  `hN : 2 ≤ N`, `ζ_BM = -I` (`D.M = -1`), `0 < invβ` (`β_N < ∞`), `V ∈ C²`, `0 < T`, the Brownian
87  filtration `hF : IsFilteredSolution`, a density at time 0 (`hdens0`), and
88  `𝓗_G(P_N^0) < ⊤` (`B:eq:selected-initial-gibbs-entropy`; `gibbsEntropy` is `EReal`-valued).
89* Valid for every `d`, including `d = 1` (then both branches of `SelectedFlow` can apply; B.14 covers the
90  Cépa-Lépingle one).  The TeX proof states `g_ε ≥ 0` for `s > 0` (L12159; false for the mean-zero torus kernel,
91  only the uniform lower bound `b15c_gReg_ge` is used, ISSUES "B15-C") and needs the `d = 1`, `0 < s < 1`
92  integrability of the Gibbs weight (see `b15_periodicRiesz_ge_d1`).  It also covers strong collision-free
93  solutions for `d = 1`, `s = 0`, which the TeX sends to B.14 (L12154); Step 1 does not use `g_ε ≥ 0`
94  when `s = 0`, so this extra case is harmless. -/

114/-- Dispatcher for Proposition B.15: `sde_entropy_law`'s conclusion follows from the strong-branch
115statement `b15_sde_entropy_law_free` (called directly, so this theorem depends exactly on that statement
116and on `cepa_entropy_confined`; statement review 2026-10-16, ISSUES "B15-A0") and from `cepa_entropy_confined` (the
117Cépa-Lépingle branch, after `subst d = 1`; `M = -I` there is exactly the `hM` of the statement, so the
118constant `a` of the disjunct is not needed). -/
119theorem b15_sde_entropy_law_of_free
120    {Ω : Type*} [mΩ : MeasurableSpace Ω] {P : Measure Ω} [IsProbabilityMeasure P]
121    (hN : 2 ≤ N) [NeZero N]
122    {ℱ : Filtration ℝ mΩ} {W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ}
123    {D : DynData d} (hM : D.M = -(1 : Matrix (Fin d) (Fin d) ℝ)) (hβ : 0 < D.invβ)
124    (hV : Torus.IsCk 2 D.V) {T : ℝ} (hT : 0 < T)
125    {X₀ : Ω → Config d N} {W : ℝ≥0 → Ω → (Fin N → Euc d)} {X : ℝ≥0 → Ω → Config d N}
126    (hX : SelectedFlow P D T X₀ W X) (hF : IsFilteredSolution P ℱ W' W X)
127    (hdens0 : HasSelectedDensity P X 0)
128    (hH0 : gibbsEntropy D (selectedLaw P X 0) < ⊤) :
129    (∀ t ∈ Set.Icc 0 T, HasSelectedDensity P X t ∧
130      gibbsEntropy D (selectedLaw P X t)
131        + ((ENNReal.ofReal D.invβ *
132            ∫⁻ r in Set.Icc (0 : ℝ) t, fisherInfo D N (selectedLaw P X r) : ℝ≥0∞) : EReal)
133        ≤ gibbsEntropy D (selectedLaw P X 0)) ∧
134    SolvesSelectedFKE D T (selectedLaw P X) := by
135  rcases hX with ⟨hX₀, hdist, hW, hind, hflow⟩
136  rcases hflow with ⟨hS, hU⟩ | ⟨hd1, hs0, _, _, hCL⟩
137  · exact b15_sde_entropy_law_free hN hM hβ hV hT hX₀ hdist hW hind hS hU hF hdens0 hH0
138  · subst hd1
139    exact cepa_entropy_confined hN hs0 hβ hM hV hT hX₀ hdist hW hind hCL hF hdens0 hH0

155/-- A collision-free strong solution has a selected law that does not charge the collision set
156`𝔇_N`, at every deterministic time `t ∈ [0,T]`
157(`P_N^t(𝔇_N) = 0`, from `IsStrongSolution.no_collision`). -/
158theorem b15_selectedLaw_collisionSet
159    {Ω : Type*} [mΩ : MeasurableSpace Ω] {P : Measure Ω}
160    {D : DynData d} {T : ℝ} {X₀ : Ω → Config d N} {W : ℝ≥0 → Ω → (Fin N → Euc d)}
161    {X : ℝ≥0 → Ω → Config d N} (hS : IsStrongSolution P D T X₀ W X) {t : ℝ} (ht0 : 0 ≤ t)
162    (htT : t ≤ T) :
163    selectedLaw P X t (collisionSet d N) = 0 := by
164  have hcoll : MeasurableSet (collisionSet d N) := (isClosed_collisionSet d N).measurableSet
165  unfold selectedLaw
166  rw [Measure.map_apply (hS.adapted _) hcoll]
167  have hle : ((Real.toNNReal t : ℝ≥0) : ℝ) ≤ T := by
168    rw [Real.coe_toNNReal t ht0]; exact htT
169  rw [measure_eq_zero_iff_ae_notMem]
170  filter_upwards [hS.no_collision] with ω hω using hω _ hle
