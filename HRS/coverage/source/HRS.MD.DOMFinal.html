30theorem domp_isOpen_noncollision : IsOpen ((collisionSet d N)ᶜ) := by
31  have : (collisionSet d N) = ⋃ a : Fin N, ⋃ b : Fin N, ⋃ (_ : a ≠ b), {X : Config d N | X a = X b} := by
32    ext X; simp [collisionSet]

66    (hρ : IsThermalDensity (periodicRiesz d s) V β ρ) (hρs : SmoothPosDensity ρ) :
67    ∀ (f : Torus d → ℝ) (K₁ : ℝ) (ψ : Torus d → Torus d → ℝ), Torus.IsSmooth f →
68      IsPairCorrector d s V β ρ f K₁ ψ → ∀ N : ℕ, 2 ≤ N →
69        IsInClosedGenerator d s V β (canonicalGibbs (periodicRiesz d s) V N β)
70          (centeredUnder (canonicalGibbs (periodicRiesz d s) V N β) (Fstat ψ))

78  set c : ℝ := ∫ X, Fstat (N := N) ψ X ∂μ with hc
79  obtain ⟨C, hC0, hC⟩ := domp_diff_bounds hd hcrit hV hβ hN1 hψ c
80  obtain ⟨Xc, hXc0, hXc⟩ := domp_chi_deriv hs1 (d := d) (N := N)

87    hac.ae_le (q84d_ae_not_collision (d := d) (N := N) (by omega))

89    domp_integrable_Pw_pow hd hcrit hV hβ hN1 8

95      (fun a b hab hXab => hX ⟨a, b, hab, hXab⟩) i
96  have hslFe : ∀ n, ∀ X : Config d N, X ∉ collisionSet d N → ∀ i,
97      ContDiffAt ℝ 2 (slice (Fe (ε n) c (Fstat ψ)) X i) 0 := fun n X hX i => by
98    have h1 := (hXc (ε n) (hεpos n) X hX i).1
99    have h2 := hslF X hX i
100    exact h1.mul (h2.sub contDiffAt_const)
101  have hGm : AEStronglyMeasurable (particleGen d s V β (Fstat (N := N) ψ)) μ :=
102    domp_aestronglyMeasurable_particleGen V β hsd hnc hFm hslF
103  have hGem : ∀ n, AEStronglyMeasurable (particleGen d s V β (Fe (ε n) c (Fstat (N := N) ψ))) μ :=
104    fun n => domp_aestronglyMeasurable_particleGen V β hsd hnc (hFem n) (hslFe n)
105  refine ⟨fun n => Fe (ε n) c (Fstat ψ), fun n => ⟨domp_contDiff_Fe_lift hψ.size.smooth (hεpos n) c,
106    Kset (ε n) d N, domp_isCompact_Kset _, domp_Kset_disjoint (hεpos n),
107    fun X hX => domp_Fe_zero_outside (hεpos n) c _ X hX⟩, ?_, ?_, ?_⟩
