183. the local part is absorbed pointwise by `X Y ≤ η X² + Y²/(4η)` into `η μ(x) 𝔡_p[W](x) + (4η)^{-1}|W|^p J`
19   (`elemb_local_bound`), the far part by `a^{p-1} G ≤ a^p + G^p` (`elemb_far_bound`);
204. integrate in `x`: only the term `ofReal(μ x) 𝔡_p(x)` is not known to be measurable (`μ` is a bare function), it
21   is integrated against a constant (`lintegral_const_mul'`); the other terms are measurable
22   (`lintegral_add_right`); Young's inequality (`elemb_Gx_young`) bounds `∫ Gx^p` by `C ∫ |W|^p`.

46/-- the real chain: `|V^{p-2} ∑_i W_i (√a ∑_j B_ij P_j)| ≤ (card · b) (√a V^{p-1}) ∑_j |P_j|` -/
47lemma elemb_real_chain (Wx : ι → ℝ) (Bm : Matrix ι ι ℝ) (P : ι → ℝ) {a b p : ℝ} (hp : 2 ≤ p) (ha : 0 ≤ a)
48    (hb : ∀ i j, |Bm i j| ≤ b) :
49    |vnorm Wx ^ (p - 2) * ∑ i, Wx i * (√a * ∑ j, Bm i j * P j)| ≤
50      ((Fintype.card ι : ℝ) * b) * (√a * vnorm Wx ^ (p - 1)) * ∑ j, |P j| := by
51  have hV := elemb_vnorm_nonneg Wx
52  have hb0 : ∀ i j, 0 ≤ b := fun i j => (abs_nonneg _).trans (hb i j)
53  have hS : 0 ≤ ∑ j, |P j| := Finset.sum_nonneg fun j _ => abs_nonneg _
54  -- each row
55  have hrow : ∀ i, |∑ j, Bm i j * P j| ≤ b * ∑ j, |P j| := fun i => by
56    calc |∑ j, Bm i j * P j| ≤ ∑ j, |Bm i j * P j| := Finset.abs_sum_le_sum_abs _ _
57      _ ≤ ∑ j, b * |P j| := Finset.sum_le_sum fun j _ => by
58          rw [abs_mul]
59          exact mul_le_mul_of_nonneg_right (hb i j) (abs_nonneg _)
60      _ = b * ∑ j, |P j| := by rw [Finset.mul_sum]
61  have hinner : |∑ i, Wx i * (√a * ∑ j, Bm i j * P j)| ≤
62      (∑ i, |Wx i|) * (√a * (b * ∑ j, |P j|)) := by
63    calc |∑ i, Wx i * (√a * ∑ j, Bm i j * P j)| ≤ ∑ i, |Wx i * (√a * ∑ j, Bm i j * P j)| :=
64          Finset.abs_sum_le_sum_abs _ _
65      _ ≤ ∑ i, |Wx i| * (√a * (b * ∑ j, |P j|)) := Finset.sum_le_sum fun i _ => by
66          rw [abs_mul, abs_mul, abs_of_nonneg (Real.sqrt_nonneg _)]
67          exact mul_le_mul_of_nonneg_left (mul_le_mul_of_nonneg_left (hrow i) (Real.sqrt_nonneg _))
68            (abs_nonneg _)
69      _ = _ := by rw [Finset.sum_mul]
70  have hsum := elemb_sum_abs_le Wx
71  have hq : 0 ≤ √a * (b * ∑ j, |P j|) := by
72    have : ∀ i : ι, 0 ≤ b := fun i => hb0 i i
73    by_cases hne : Nonempty ι
74    · obtain ⟨i⟩ := hne
75      exact mul_nonneg (Real.sqrt_nonneg _) (mul_nonneg (hb0 i i) hS)
76    · have : ∑ j : ι, |P j| = 0 := by
77        haveI : IsEmpty ι := not_nonempty_iff.1 hne
78        simp
79      rw [this]; simp
80  have hVp : vnorm Wx ^ (p - 2) * vnorm Wx = vnorm Wx ^ (p - 1) := by
81    rw [← Real.rpow_add_one' hV (by linarith)]
82    congr 1; ring
83  rw [abs_mul, abs_of_nonneg (Real.rpow_nonneg hV _)]
84  calc vnorm Wx ^ (p - 2) * |∑ i, Wx i * (√a * ∑ j, Bm i j * P j)|
85      ≤ vnorm Wx ^ (p - 2) * (((Fintype.card ι : ℝ) * vnorm Wx) * (√a * (b * ∑ j, |P j|))) :=
86        mul_le_mul_of_nonneg_left (hinner.trans (mul_le_mul_of_nonneg_right hsum hq))
87          (Real.rpow_nonneg hV _)
88    _ = ((Fintype.card ι : ℝ) * b) * (√a * (vnorm Wx ^ (p - 2) * vnorm Wx)) * ∑ j, |P j| := by ring
89    _ = _ := by rw [hVp]
