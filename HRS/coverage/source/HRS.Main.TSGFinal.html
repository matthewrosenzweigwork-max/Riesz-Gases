16/-- Squared directional derivative bound for `a` with `‖∇²a‖ ≤ L`: `(∂_h a)² ≤ 2 L a ‖h‖²`. -/
17theorem tsg_sq_fderiv_le {d : ℕ} {a : Torus d → ℝ} {L : ℝ} (hL : 0 ≤ L) (ha : ∀ x, 0 ≤ a x)
18    (hC11 : PDE.IsC11 L a) (x h : Euc d) :
19    (fderiv ℝ (Torus.lift a) x h) ^ 2 ≤ 2 * L * (Torus.lift a) x * ‖h‖ ^ 2 :=
20  HRS.sqrtLip_grad (Torus.lift a) hC11.1 (fun _ => ha _) L hL hC11.2 x h

22/-- Pointwise gradient bound: `‖∇√a(x)‖ ≤ √(L/2)` at every point, with `√a` the lift. -/
23theorem tsg_norm_fderiv_le {d : ℕ} {a : Torus d → ℝ} {L : ℝ} (hL : 0 ≤ L) (ha : ∀ x, 0 ≤ a x)
24    (hC11 : PDE.IsC11 L a) (x : Euc d) :
25    ‖fderiv ℝ (Torus.lift fun y => Real.sqrt (a y)) x‖ ≤ Real.sqrt (L / 2) := by
26  have hCd : ContDiff ℝ 1 (Torus.lift a) := hC11.1
27  have hAd : HasFDerivAt (Torus.lift a) (fderiv ℝ (Torus.lift a) x) x :=
28    ((hCd.differentiable (by norm_num)) x).hasFDerivAt
29  have hnn : 0 ≤ (Torus.lift a) x := ha _
30  rcases hnn.eq_or_lt with h0 | hpos
31  · -- a(x) = 0: `x` is a global minimum of `√(a ∘ mk)`, so the derivative vanishes
32    have hmin : IsLocalMin (fun y => Real.sqrt ((Torus.lift a) y)) x := by
33      refine Filter.Eventually.of_forall fun y => ?_
34      show Real.sqrt ((Torus.lift a) x) ≤ Real.sqrt ((Torus.lift a) y)
35      rw [← h0, Real.sqrt_zero]
36      exact Real.sqrt_nonneg _
37    have hz : fderiv ℝ (Torus.lift fun y => Real.sqrt (a y)) x = 0 :=
38      IsLocalMin.fderiv_eq_zero hmin
39    rw [hz, norm_zero]
40    exact Real.sqrt_nonneg _
41  · -- a(x) > 0: chain rule for `√`, then the Taylor bound on `∇a`
42    have hs_pos : 0 < Real.sqrt ((Torus.lift a) x) := Real.sqrt_pos.mpr hpos
43    have hs_ne : Real.sqrt ((Torus.lift a) x) ≠ 0 := hs_pos.ne'
44    have hs_sq : Real.sqrt ((Torus.lift a) x) ^ 2 = (Torus.lift a) x := Real.sq_sqrt hpos.le
45    have hr_sq : Real.sqrt (L / 2) ^ 2 = L / 2 := Real.sq_sqrt (by positivity)
46    have hs : HasFDerivAt (fun y => Real.sqrt ((Torus.lift a) y))
47        ((1 / (2 * Real.sqrt ((Torus.lift a) x))) • fderiv ℝ (Torus.lift a) x) x :=
48      hAd.sqrt hpos.ne'
49    have hfg : fderiv ℝ (Torus.lift fun y => Real.sqrt (a y)) x =
50        (1 / (2 * Real.sqrt ((Torus.lift a) x))) • fderiv ℝ (Torus.lift a) x := hs.fderiv
51    have hc : 0 < 1 / (2 * Real.sqrt ((Torus.lift a) x)) := by
52      apply div_pos one_pos
53      linarith
54    have hD : ‖fderiv ℝ (Torus.lift a) x‖ ≤ 2 * Real.sqrt ((Torus.lift a) x) * Real.sqrt (L / 2) := by
55      refine ContinuousLinearMap.opNorm_le_bound _ (by positivity) fun h => ?_
56      have hsq := tsg_sq_fderiv_le hL ha hC11 x h
57      have h2 : (fderiv ℝ (Torus.lift a) x h) ^ 2 ≤
58          (2 * Real.sqrt ((Torus.lift a) x) * Real.sqrt (L / 2) * ‖h‖) ^ 2 := by
59        have e : (2 * Real.sqrt ((Torus.lift a) x) * Real.sqrt (L / 2) * ‖h‖) ^ 2 =
60            2 * L * (Torus.lift a) x * ‖h‖ ^ 2 := by
61          linear_combination (4 * Real.sqrt ((Torus.lift a) x) ^ 2 * ‖h‖ ^ 2) * hr_sq
62            + (2 * L * ‖h‖ ^ 2) * hs_sq
63        rw [e]
64        exact hsq
65      rw [Real.norm_eq_abs]
66      exact abs_le_of_sq_le_sq h2 (by positivity)
67    rw [hfg, norm_smul, Real.norm_eq_abs, abs_of_pos hc]
68    calc 1 / (2 * Real.sqrt ((Torus.lift a) x)) * ‖fderiv ℝ (Torus.lift a) x‖
69        ≤ 1 / (2 * Real.sqrt ((Torus.lift a) x)) *
70            (2 * Real.sqrt ((Torus.lift a) x) * Real.sqrt (L / 2)) :=
71          mul_le_mul_of_nonneg_left hD hc.le
72      _ = Real.sqrt (L / 2) := by field_simp

74/-- **Lemma C.6 `M:B:lem:sqrt-lipschitz`, gradient clause on `𝕋^d`** (proof of
75`HRS.Main.SpecSW2.torus_sqrt_gradient_le`; statement copied verbatim). -/
76theorem torus_sqrt_gradient_le_proof {d : ℕ} {a : Torus d → ℝ} {L : ℝ} (hL : 0 ≤ L) (ha : ∀ x, 0 ≤ a x)
77    (hC11 : PDE.IsC11 L a) :
78    (∀ x : Euc d, DifferentiableAt ℝ (Torus.lift fun y => Real.sqrt (a y)) x →
79        ‖fderiv ℝ (Torus.lift fun y => Real.sqrt (a y)) x‖ ≤ Real.sqrt (L / 2)) ∧
80    (∀ᵐ x : Euc d ∂volume, DifferentiableAt ℝ (Torus.lift fun y => Real.sqrt (a y)) x) ∧
81    eLpNormEssSup (fun x => fderiv ℝ (Torus.lift fun y => Real.sqrt (a y)) x) volume
82      ≤ ENNReal.ofReal (Real.sqrt (L / 2)) := by
83  have hLip : LipschitzWith (Real.toNNReal (Real.sqrt (L / 2)))
84      (Torus.lift fun y => Real.sqrt (a y)) :=
85    HRS.sqrt_lipschitz (E := Euc d) (Torus.lift a) hC11.1 (fun y => ha _) L hL hC11.2
86  have hae : ∀ᵐ x : Euc d ∂volume, DifferentiableAt ℝ (Torus.lift fun y => Real.sqrt (a y)) x :=
87    hLip.ae_differentiableAt
88  refine ⟨fun x _ => tsg_norm_fderiv_le hL ha hC11 x, hae, ?_⟩
89  refine eLpNormEssSup_le_of_ae_bound ?_
90  filter_upwards [hae] with x _
91  exact tsg_norm_fderiv_le hL ha hC11 x
