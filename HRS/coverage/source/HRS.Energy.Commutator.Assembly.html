33/-- **Proposition 3.32 (commutator estimate), assembled from the packets.**  The statement is a
34verbatim copy of `HRS.commutator_estimate`. -/
35theorem commutator_estimate_of_packets (hd : 1 ≤ d) {s : ℝ} (hs0 : 0 ≤ s) (hsd : s < d) :
36    ∃ C₀ C : ℝ, 0 ≤ C₀ ∧ 0 < C ∧
37      ∀ (Λ : ℝ), 1 ≤ Λ → ∀ (N : ℕ), 2 ≤ N →
38      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
39      ∀ X : Config d N, X ∉ collisionSet d N →
40      ∀ (v : Torus d → Euc d) (K : ℝ), HasTransportBound d s v K →
41        OffDiagIntegrable (commKernel (periodicRieszGrad d s) v) X (withDensityReal ρ) ∧
42        |commQuantity d s v X (withDensityReal ρ)| ≤
43          C * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N) := by
44  obtain ⟨Cn, hCn, hnn⟩ := energy_add_oN_nonneg hd hs0 hsd
45  by_cases hsub : s < (d : ℝ) - 2
46  · -- fractional regime
47    have hd3 : 3 ≤ d := by
48      have h2 : (2 : ℝ) < d := by linarith
49      have : 2 < d := by exact_mod_cast h2
50      omega
51    obtain ⟨C₀', C', hC₀', hC', hreg⟩ := TorusNote.commutator_estimate_sub_of_torus hd3 hs0 hsub
52    refine ⟨max C₀' Cn, C', le_max_of_le_left hC₀', hC', ?_⟩
53    intro Λ hΛ N hN ρ hρ hρΛ X hX v K hK
54    obtain ⟨L, F, hL, hF, hlip, hfrac, hLK⟩ := hK
55    rw [if_pos hsub] at hLK
56    have hΛ0 : (0 : ℝ) ≤ Λ := by linarith
57    refine ⟨offDiagIntegrable_of_lipschitz hd hs0 hsd hρ hρΛ X hL hlip, ?_⟩
58    have hmain := hreg Λ hΛ N hN ρ hρ hρΛ X hX L F v hL hF hlip (hfrac hsub)
59    have hY0 : 0 ≤ rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max C₀' Cn) Λ N :=
60      le_trans (hnn Λ hΛ N hN ρ hρ hρΛ X hX)
61        (by have := oNΛ_mono (d := d) (s := s) (N := N) hΛ0 (le_max_right C₀' Cn); linarith)
62    have hmono := oNΛ_mono (d := d) (s := s) (N := N) hΛ0 (le_max_left C₀' Cn)
63    have hCLF : 0 ≤ C' * (L + F) := by positivity
64    calc |commQuantity d s v X (withDensityReal ρ)|
65        ≤ C' * (L + F) * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀' Λ N) := hmain
66      _ ≤ C' * (L + F) * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max C₀' Cn) Λ N) :=
67          mul_le_mul_of_nonneg_left (by linarith) hCLF
68      _ ≤ C' * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max C₀' Cn) Λ N) := by
69          have : C' * (L + F) ≤ C' * K := mul_le_mul_of_nonneg_left hLK hC'.le
70          exact mul_le_mul_of_nonneg_right this hY0
71  · -- Lipschitz-only regime
72    push_neg at hsub
73    obtain ⟨C₀', C', hC₀', hC', hreg⟩ := commutator_estimate_super hd hs0 hsub hsd
74    refine ⟨max C₀' Cn, C', le_max_of_le_left hC₀', hC', ?_⟩
75    intro Λ hΛ N hN ρ hρ hρΛ X hX v K hK
76    obtain ⟨L, F, hL, hF, hlip, hfrac, hLK⟩ := hK
77    rw [if_neg (not_lt.mpr hsub)] at hLK
78    have hΛ0 : (0 : ℝ) ≤ Λ := by linarith
79    refine ⟨offDiagIntegrable_of_lipschitz hd hs0 hsd hρ hρΛ X hL hlip, ?_⟩
80    have hmain := hreg Λ hΛ N hN ρ hρ hρΛ X hX L v hL hlip
81    have hY0 : 0 ≤ rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max C₀' Cn) Λ N :=
82      le_trans (hnn Λ hΛ N hN ρ hρ hρΛ X hX)
83        (by have := oNΛ_mono (d := d) (s := s) (N := N) hΛ0 (le_max_right C₀' Cn); linarith)
84    have hmono := oNΛ_mono (d := d) (s := s) (N := N) hΛ0 (le_max_left C₀' Cn)
85    have hCL : 0 ≤ C' * L := by positivity
86    have hLK' : L ≤ K := by linarith
87    calc |commQuantity d s v X (withDensityReal ρ)|
88        ≤ C' * L * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀' Λ N) := hmain
89      _ ≤ C' * L * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max C₀' Cn) Λ N) :=
90          mul_le_mul_of_nonneg_left (by linarith) hCL
91      _ ≤ C' * K * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s (max C₀' Cn) Λ N) := by
92          have : C' * L ≤ C' * K := mul_le_mul_of_nonneg_left hLK' hC'.le
93          exact mul_le_mul_of_nonneg_right this hY0
