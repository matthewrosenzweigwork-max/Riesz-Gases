7# The torus Theorem 1.1 implies the `s < d-2` branch of `HRS.commutator_estimate`
8
9`HRS.Commutator.commutator_estimate_sub` (`HRS/Energy/Commutator/SubAssembly.lean`) is the branch used by
10`commutator_estimate_of_packets` (`Assembly.lean`) for `0 ≤ s < d-2`; its proof there uses the packets H1, H3, H4, H5 (Route H,
11now **replaced**).  `commutator_estimate_sub_of_torus` below has the same statement (checked by `rfl`) and is proved from

13‖|∇|^{(d-s)/2}v‖_{L^{2d/(d-s-2)}} ≲ F`) and three elementary lemmas (identification of the repository's `rieszEnergy`,
14`commQuantity` for `μ = ρ dx` with `modEnergyT`, `commAT`).

28/-- `∫ f d(ρ dx) = ∫ f ρ dx` for a probability density `ρ ≥ 0` (unconditional: Mathlib's
29`integral_withDensity_eq_integral_toReal_smul` holds without integrability hypotheses). -/
30theorem tbr_integral_withDensity (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (f : Torus d → ℝ) :
31    ∫ x, f x ∂(withDensityReal ρ) = ∫ x, f x * ρ x := by
32  unfold withDensityReal
33  rw [integral_withDensity_eq_integral_toReal_smul hρ.measurable.ennreal_ofReal
34    (Filter.Eventually.of_forall fun x => ENNReal.ofReal_lt_top)]
35  have h : ∀ x, (ENNReal.ofReal (ρ x)).toReal • f x = f x * ρ x := fun x => by
36    rw [ENNReal.toReal_ofReal (hρ.nonneg x), smul_eq_mul, mul_comm]
37  simp only [h]

39/-- The iterated integral against `ρ(x) ρ(y)`, with the factor `ρ(x)` pulled out of the inner integral. -/
40theorem tbr_double (ρ : Torus d → ℝ) (k : Torus d → Torus d → ℝ) :
41    ∫ x, ∫ y, k x y * ρ x * ρ y = ∫ x, (∫ y, k x y * ρ y) * ρ x := by
42  congr 1
43  funext x
44  rw [← integral_mul_const]
45  congr 1
46  funext y
47  ring

49/-- `rieszEnergy d s X (ρ dx) = modEnergyT s X ρ` for a probability density (`∫ f d(ρ dx) = ∫ ρ f dx`, `ρ ≥ 0`;
50unconditional, Bochner integrals on both sides). -/
51theorem rieszEnergy_eq_T (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) {N : ℕ} (X : Config d N) :
52    rieszEnergy d s X (withDensityReal ρ) = modEnergyT s X ρ := by
53  unfold rieszEnergy modulatedEnergyKernel modulatedEnergy modEnergyT pairT
54  simp only [tbr_integral_withDensity ρ hρ, tbr_double ρ]
55  ring

57/-- `commQuantity d s v X (ρ dx) = commAT s X ρ v` (same). -/
58theorem commQuantity_eq_T (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) {N : ℕ} (X : Config d N)
59    (v : Torus d → Euc d) :
60    commQuantity d s v X (withDensityReal ρ) = commAT s X ρ v := by
61  unfold commQuantity commQuantityWith offDiagPairing commAT pairT
62  simp only [tbr_integral_withDensity ρ hρ, tbr_double ρ]

64/-- `‖ρ‖_{L^∞} ≤ Λ` for `0 ≤ ρ ≤ Λ`. -/
65theorem normT_top_le (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ) (hΛ : ∀ x, ρ x ≤ Λ) :
66    normT ⊤ ρ ≤ Λ := by
67  have h : eLpNorm ρ ⊤ volume ≤ ENNReal.ofReal Λ := by
68    rw [eLpNorm_exponent_top hρ.measurable.aestronglyMeasurable]
69    exact eLpNormEssSup_le_of_ae_bound (Filter.Eventually.of_forall fun x => by
70      rw [Real.norm_eq_abs, abs_of_nonneg (hρ.nonneg x)]; exact hΛ x)
71  have hΛ0 : 0 ≤ Λ := le_trans (hρ.nonneg 0) (hΛ 0)
72  exact ENNReal.toReal_le_of_le_ofReal hΛ0 h

74/-- **The torus Theorem 1.1 implies the `0 ≤ s < d-2` branch of `HRS.commutator_estimate`**
75(statement identical to `HRS.Commutator.commutator_estimate_sub`, checked by `rfl` in `BridgeCheck.lean`).  Constants: `C₀ = C₀^{torus}`, `C = C^{torus}·max(1, C_emb)`. -/
76theorem commutator_estimate_sub_of_torus (hd : 3 ≤ d) (hs0 : 0 ≤ s) (hs : s < (d : ℝ) - 2) :
77    ∃ C₀ C : ℝ, 0 ≤ C₀ ∧ 0 < C ∧ ∀ Λ : ℝ, 1 ≤ Λ → ∀ N : ℕ, 2 ≤ N →
78      ∀ ρ : Torus d → ℝ, IsProbDensity ρ → (∀ x, ρ x ≤ Λ) →
79      ∀ X : Config d N, X ∉ collisionSet d N →
80      ∀ (L F : ℝ) (v : Torus d → Euc d), 0 ≤ L → 0 ≤ F → IsLipschitzVF L v → HasFracBound d v F →
81        |commQuantity d s v X (withDensityReal ρ)| ≤
82          C * (L + F) * (rieszEnergy d s X (withDensityReal ρ) + oNΛ d s C₀ Λ N) := by
83  obtain ⟨C, C₀, hC, hC₀, H⟩ := torus_subcoulomb_commutator hd hs0 hs
84  obtain ⟨Ce, hCe, E⟩ := embedding_T hd hs0 hs
85  refine ⟨C₀, C * max 1 Ce, hC₀.le, by positivity, ?_⟩
86  intro Λ hΛ N hN ρ hρ hΛρ X hX L F v hL hF hLip hfrac
87  have h1 : Integrable ρ := by
88    by_contra h
89    have := hρ.integral_one
90    rw [integral_undef h] at this
91    exact zero_ne_one this
92  have hinf : MemLp ρ ⊤ volume :=
93    memLp_top_of_bound h1.aestronglyMeasurable |Λ| (Filter.Eventually.of_forall fun x => by
94      rw [Real.norm_eq_abs, abs_of_nonneg (hρ.nonneg x)]
95      exact le_trans (hΛρ x) (le_abs_self Λ))
96  obtain ⟨w, hw, hwn⟩ := E v F hF hfrac
97  have hv' : TorusSubTransport d s v (L + Ce * F) :=
98    ⟨L, Ce * F, hL, by positivity, hLip, ⟨w, hw, hwn⟩, le_refl _⟩
99  obtain ⟨-, -, hQ, hA⟩ := H ρ h1 hinf hρ.integral_one v _ hv' N X (by omega) hX
100  rw [commQuantity_eq_T ρ hρ X v, rieszEnergy_eq_T ρ hρ X]
101  have hM1 := normT_top_ge_one ρ h1 hinf hρ.integral_one
102  have hM : 0 < normT ⊤ ρ := lt_of_lt_of_le one_pos hM1
103  have hMΛ := normT_top_le ρ hρ Λ hΛρ
104  have hNpos : (0 : ℝ) < N := by exact_mod_cast (by omega : 0 < N)
105  have hd0 : (0 : ℝ) < d := by exact_mod_cast (by omega : 0 < d)
106  have hNM : 0 < (N : ℝ) * normT ⊤ ρ := mul_pos hNpos hM
107  have hlam : 0 < lamT d N ρ := Real.rpow_pos_of_pos hNM _
108  have hlamd : lamT d N ρ ^ (d : ℝ) = ((N : ℝ) * normT ⊤ ρ)⁻¹ := by
109    unfold lamT
110    rw [← Real.rpow_mul hNM.le, show (-(1 : ℝ) / d) * d = -1 by field_simp, Real.rpow_neg_one]
111  have hlams : lamT d N ρ ^ (-s) = ((N : ℝ) * normT ⊤ ρ) ^ (s / d) := by
112    unfold lamT
113    rw [← Real.rpow_mul hNM.le]
114    congr 1
115    field_simp
116  have hMl : normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s) ≤ Λ ^ (s / d) * (N : ℝ) ^ (s / d - 1) := by
117    rw [sub_eq_add_neg, Real.rpow_add hlam, hlamd, hlams, Real.mul_rpow hNpos.le hM.le]
118    have e : normT ⊤ ρ * (((N : ℝ) * normT ⊤ ρ)⁻¹ * ((N : ℝ) ^ (s / d) * normT ⊤ ρ ^ (s / d))) =
119        (N : ℝ) ^ (s / d - 1) * normT ⊤ ρ ^ (s / d) := by
120      rw [Real.rpow_sub_one hNpos.ne']
121      field_simp
122    rw [e, mul_comm]
123    gcongr
124  have hlogterm : (if s = 0 then -(Real.log (lamT d N ρ) / (2 * (N : ℝ))) else 0) ≤
125      (if s = 0 then Real.log ((N : ℝ) * Λ) / (2 * d * N) else 0) := by
126    split_ifs with h0
127    · have h2 : Real.log (lamT d N ρ) = (-(1 : ℝ) / d) * Real.log ((N : ℝ) * normT ⊤ ρ) := by
128        unfold lamT; rw [Real.log_rpow hNM]
129      rw [h2]
130      have h3 : Real.log ((N : ℝ) * normT ⊤ ρ) ≤ Real.log ((N : ℝ) * Λ) :=
131        Real.log_le_log hNM (mul_le_mul_of_nonneg_left hMΛ hNpos.le)
132      have : -((-(1 : ℝ) / d * Real.log ((N : ℝ) * normT ⊤ ρ)) / (2 * (N : ℝ))) =
133          Real.log ((N : ℝ) * normT ⊤ ρ) / (2 * d * N) := by field_simp
134      rw [this]
135      gcongr
136    · exact le_rfl
137  have hQle : modEnergyT s X ρ - (if s = 0 then Real.log (lamT d N ρ) / (2 * (N : ℝ)) else 0) +
138        C₀ * normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s) ≤
139      modEnergyT s X ρ + oNΛ d s C₀ Λ N := by
140    unfold oNΛ
141    have h5 : C₀ * normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s) ≤ C₀ * Λ ^ (s / d) * (N : ℝ) ^ (s / d - 1) := by
142      calc C₀ * normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s)
143          = C₀ * (normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s)) := by ring
144        _ ≤ C₀ * (Λ ^ (s / d) * (N : ℝ) ^ (s / d - 1)) := by gcongr
145        _ = _ := by ring
146    have h7 : -(if s = 0 then Real.log (lamT d N ρ) / (2 * (N : ℝ)) else 0) ≤
147        (if s = 0 then Real.log ((N : ℝ) * Λ) / (2 * d * N) else 0) := by
148      split_ifs at hlogterm ⊢ with h0
149      · simpa [neg_div] using hlogterm
150      · simp
151    linarith
152  have hL1 : 0 ≤ L + Ce * F := by positivity
153  calc |commAT s X ρ v| ≤ C * (L + Ce * F) *
154        (modEnergyT s X ρ - (if s = 0 then Real.log (lamT d N ρ) / (2 * (N : ℝ)) else 0) +
155          C₀ * normT ⊤ ρ * lamT d N ρ ^ ((d : ℝ) - s)) := hA
156    _ ≤ C * (L + Ce * F) * (modEnergyT s X ρ + oNΛ d s C₀ Λ N) := by
157        have : 0 ≤ C * (L + Ce * F) := by positivity
158        exact mul_le_mul_of_nonneg_left hQle this
159    _ ≤ C * max 1 Ce * (L + F) * (modEnergyT s X ρ + oNΛ d s C₀ Λ N) := by
160        have hY : 0 ≤ modEnergyT s X ρ + oNΛ d s C₀ Λ N := le_trans hQ hQle
161        have hineq : C * (L + Ce * F) ≤ C * max 1 Ce * (L + F) := by
162          have : L + Ce * F ≤ max 1 Ce * (L + F) := by
163            have h1' : 1 ≤ max 1 Ce := le_max_left _ _
164            have h2' : Ce ≤ max 1 Ce := le_max_right _ _
165            nlinarith [mul_nonneg hL (sub_nonneg.2 h1'), mul_nonneg hF (sub_nonneg.2 h2')]
166          calc C * (L + Ce * F) ≤ C * (max 1 Ce * (L + F)) := mul_le_mul_of_nonneg_left this hC.le
167            _ = _ := by ring
168        exact mul_le_mul_of_nonneg_right hineq hY
