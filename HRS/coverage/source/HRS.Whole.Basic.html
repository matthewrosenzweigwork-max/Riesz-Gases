37## Representation decisions
38
39* The kernel `rieszKernel d s` is defined pointwise with the exact normalisation; at `x = 0`
40  Lean's `0^(-s) = 0` and `-log 0 = 0` are junk values, never load-bearing (the energy deletes
41  the diagonal and integrals are against measures with densities).
42* Convolutions, energies and the Gibbs normaliser are Bochner integrals (junk `0` when not
43  integrable; integrability is a hypothesis where needed), as on the torus.
44* `R_{β,μ}` is represented by the response equation `h + βK_μ h = P_μ ψ`, `μ(h) = 0`
45  (`IsResponse`) and `Classical.choose` (`responseInverse`), as on the torus (ISSUES K4).
46* The thermal equilibrium is a *predicate* (`IsThermalDensity`: probability density solving
47  the thermal equation) and a predicate `IsFreeEnergyMinimizer` (minimiser of
48  `W:eq:euc-free-energy` among finite-free-energy probability densities); `thermalDensity`
49  picks a solution by choice when one exists (`0` otherwise).  Nothing here assumes
50  existence, uniqueness, tails or smoothness: these are the S3.4 targets.
51* The smallness class is a `Prop`-structure with exactly the source's conditions
52  (`ψ ∈ C_c^∞`, `ψ ≠ 0`, `μ_β(ψ) = 0`, the `C_b^{100d+1001}` norm (project SUM convention,
53  `HRS.cbNorm`) plus the `Σ^{102d+1020}` norm of `√ρ_β ψ` at most `ε_*`); `ε = 0` is allowed.
54* Definition S.A.1 is an inductive type with the two source choices (A)/(B) as constructors
55  carrying exactly the stated parameter constraints; `r_𝔠, p_𝔠, e_N^𝔠, m_𝔠, Γ_𝔠` are functions
56  of it, and the three embedding claims of the last sentence are separate sorried theorems.

98/-- `g(x) = |x|^{-s}` for `s ≠ 0`, `g(x) = -log|x|` for `s = 0`: no factor `1/s`, no zero-mode
99adjustment.  Junk value at `x = 0`. -/

115/-! ### Modulated energy and modulated Gibbs law -/
116
117/-- The modulated energy `F_N^g(X_N, μ) = (2N²)⁻¹ ∑_{i≠j} g(x_i-x_j) - μ_N(g*μ) + ½ μ^{⊗2}(g)`
118(`W:r2:normalization`, S lines 221-226), with the literal deleted particle diagonal and the
119factor `½` on the background term. -/
120def modulatedEnergy (g : Euc d → ℝ) (X : EucConfig d N) (μ : Measure (Euc d)) : ℝ :=
121  (1 / (2 * (N : ℝ) ^ 2)) * ∑ i, ∑ j ∈ Finset.univ.erase i, g (X i - X j)
122    - (1 / (N : ℝ)) * ∑ i, convMeasure g μ (X i)
123    + (1 / 2) * energyPairing g μ μ
124
125/-- `F_N` for the whole-space Riesz/log kernel with parameters `d, s`. -/
126def rieszModulatedEnergy (d : ℕ) (s : ℝ) (X : EucConfig d N) (μ : Measure (Euc d)) : ℝ :=
127  modulatedEnergy (rieszKernel d s) X μ

129/-- The normaliser `K_{N,β}(μ) = ∫ e^{-βN F_N^g} dμ^{⊗N}` (`S:eq:modulated-law`, S lines
1305285-5288; M lines 446-452). -/
131def gibbsNormalizer (g : Euc d → ℝ) (N : ℕ) (β : ℝ) (μ : Measure (Euc d)) : ℝ :=
132  ∫ X : EucConfig d N, Real.exp (-β * N * modulatedEnergy g X μ) ∂(Measure.pi fun _ => μ)
133
134/-- The modulated Gibbs law `Q_{N,β}(μ) = K_{N,β}(μ)⁻¹ e^{-βN F_N^g(X_N,μ)} μ^{⊗N}`
135(`S:eq:modulated-law`); `Q_{N,0}(μ) = μ^{⊗N}` (S line 395). -/
136def modulatedGibbs (g : Euc d → ℝ) (N : ℕ) (β : ℝ) (μ : Measure (Euc d)) :
137    Measure (EucConfig d N) :=
138  (ENNReal.ofReal (gibbsNormalizer g N β μ))⁻¹ •
139    (Measure.pi fun _ => μ).withDensity
140      fun X => ENNReal.ofReal (Real.exp (-β * N * modulatedEnergy g X μ))

142/-- The centered pair functional `D_N(h;ν) = N⁻² ∑_{i≠j} h(x_i,x_j) - 2 μ_N(∫ h(·,y) dν(y))
143+ ∬ h dν^{⊗2}` of `W:eq:euc-D` (S lines 940-944), "with the original deleted particle
144diagonal". -/
145def centeredPairFunctional (h : Euc d → Euc d → ℝ) (X : EucConfig d N) (ν : Measure (Euc d)) :
146    ℝ :=
147  (1 / (N : ℝ) ^ 2) * ∑ i, ∑ j ∈ Finset.univ.erase i, h (X i) (X j)
148    - 2 * ((1 / (N : ℝ)) * ∑ i, ∫ y, h (X i) y ∂ν)
149    + ∫ x, ∫ y, h x y ∂ν ∂ν

151/-! ### Canonical Gibbs ensemble with quadratic confinement -/
152
153/-- The confinement `V(x) = a|x|²/2` (S lines 166, 318). -/
154def confinement (a : ℝ) (x : Euc d) : ℝ := a * ‖x‖ ^ 2 / 2

156/-- `H_N(X_N) = ∑_i V(x_i) + N⁻¹ ∑_{i<j} g(x_i-x_j)` (S lines 1375-1376). -/
157def canonicalHamiltonian (g V : Euc d → ℝ) (X : EucConfig d N) : ℝ :=
158  ∑ i, V (X i) +
159    (1 / (N : ℝ)) * ∑ i, ∑ j ∈ Finset.univ.filter (fun j => i < j), g (X i - X j)

161/-- `Z_{N,β} = ∫ e^{-β H_N} dX_N` (Lebesgue measure on `(ℝ^d)^N`). -/
162def canonicalPartition (g V : Euc d → ℝ) (N : ℕ) (β : ℝ) : ℝ :=
163  ∫ X : EucConfig d N, Real.exp (-β * canonicalHamiltonian g V X)

165/-- The canonical Gibbs ensemble `P_{N,β} = Z⁻¹ e^{-β H_N} dX_N`. -/
166def canonicalGibbs (g V : Euc d → ℝ) (N : ℕ) (β : ℝ) : Measure (EucConfig d N) :=
167  (ENNReal.ofReal (canonicalPartition g V N β))⁻¹ •
168    (volume : Measure (EucConfig d N)).withDensity
169      fun X => ENNReal.ofReal (Real.exp (-β * canonicalHamiltonian g V X))

217/-! ### Thermal equilibrium (S3.4, `W:prop:euc-thermal`, S lines 1348-1384) -/
218
219/-- The mean-field free energy
220`𝓕(ν) = β⁻¹ ∫ ρ log ρ dx + ∫ V dν + ½ ∬ g(x-y) dν(x) dν(y)`, `ν = ρ dx`
221(`W:eq:euc-free-energy`, S lines 1352-1356). -/
222def freeEnergy (g V : Euc d → ℝ) (β : ℝ) (ρ : Euc d → ℝ) : ℝ :=
223  β⁻¹ * (∫ x, ρ x * Real.log (ρ x)) + (∫ x, V x * ρ x) +
224    (1 / 2) * (∫ x, ∫ y, g (x - y) * ρ x * ρ y)
225
226/-- Finiteness of the three terms of the free energy (so that the Bochner integrals in
227`freeEnergy` are the source's integrals). -/
228def HasFiniteFreeEnergy (g V : Euc d → ℝ) (ρ : Euc d → ℝ) : Prop :=
229  Integrable (fun x => ρ x * Real.log (ρ x)) ∧ Integrable (fun x => V x * ρ x) ∧
230    Integrable (fun p : Euc d × Euc d => g (p.1 - p.2) * ρ p.1 * ρ p.2)
231      ((volume : Measure (Euc d)).prod volume)
232
233/-- `ρ` is a minimiser of the free energy among probability densities with finite free
234energy (`W:prop:euc-thermal`: "has a unique minimizer `μ_β`"). -/
235def IsFreeEnergyMinimizer (g V : Euc d → ℝ) (β : ℝ) (ρ : Euc d → ℝ) : Prop :=
236  IsProbDensity ρ ∧ HasFiniteFreeEnergy g V ρ ∧
237    ∀ ρ' : Euc d → ℝ, IsProbDensity ρ' → HasFiniteFreeEnergy g V ρ' →
238      freeEnergy g V β ρ ≤ freeEnergy g V β ρ'
239
240/-- The thermal equation `ρ_β(x) = Z⁻¹ e^{-β(V(x) + g*μ_β(x))}`, `μ_β = ρ_β dx`
241(`W:eq:euc-thermal-bounds` first display, S line 1360; M `H:eq:thermal-equation`). -/
242def IsThermalDensity (g V : Euc d → ℝ) (β : ℝ) (ρ : Euc d → ℝ) : Prop :=
243  IsProbDensity ρ ∧
244    ∀ x, ρ x = Real.exp (-β * (V x + convMeasure g (withDensityReal ρ) x)) /
245      ∫ y, Real.exp (-β * (V y + convMeasure g (withDensityReal ρ) y))
246
247/-- "The actual interacting thermal equilibrium measure" of the quadratic confinement
248`V = a|x|²/2` at inverse temperature `β` (S lines 318-320): the minimiser of the mean-field
249free energy, which solves the thermal equation for the whole-space kernel
250(`W:prop:euc-thermal`, S lines 1357-1360).  Both clauses are what the source *proves* about
251`μ_β`; neither existence nor uniqueness is assumed here. -/
252def IsThermalEquilibrium (d : ℕ) (s a β : ℝ) (ρ : Euc d → ℝ) : Prop :=
253  IsFreeEnergyMinimizer (rieszKernel d s) (confinement a) β ρ ∧
254    IsThermalDensity (rieszKernel d s) (confinement a) β ρ

256open Classical in
257/-- A thermal equilibrium density chosen by `Classical.choose` when one exists (`0`
258otherwise).  Existence and uniqueness are the S3.4 target `thermalDensity_spec`, not
259assumed here. -/
260def thermalDensity (d : ℕ) (s a β : ℝ) : Euc d → ℝ :=
261  if h : ∃ ρ, IsThermalEquilibrium d s a β ρ then h.choose else 0

263/-- `μ_β = ρ_β dx`. -/
264def thermalMeasure (d : ℕ) (s a β : ℝ) : Measure (Euc d) :=

274/-- `δ_N` and `o_N` of `W:r2:rate-summary` (S lines 302-309) coincide literally with the main

281      else if s = (d : ℝ) / 2 then (N : ℝ)⁻¹ * Real.log (Real.exp 1 * N)

328/-- Definition S.A.1: a choice of commutator regularity and error, `𝔠 = (r_𝔠, p_𝔠, e_N^𝔠)`,

331* (A) sharp-rate density: `d < a < min{d+2, 2s+4}`, `r = a/2`, `p = 2d/(a-2)`, `e_N = 0`;
332* (B) lower-regularity, slower-rate density: `s > (d-4)/3`, `r = (d-s)/2`,
