5# POIS-FINAL: the three Poisson rows from the master theorem `pois_resolvent`

11All three are read off `HRS.MS.PoisRes.pois_resolvent` (the resolvent integral `χ = ∫_0^∞ P_t r dt`).
12For S4.2/S4.3 the data `r ∈ C_b^m` is used with `D = cbNorm m r` (derivatives of order `1..m`) and
13`D' = cbNorm 2 r` (orders `≤ 2`); the master theorem provides the low-order bounds for the same `χ`.
14The hypothesis `Integrable r μ` of the frozen statements is not needed (it is automatic).
15`d ≥ 1` is not needed either (the proof of the master theorem treats `d = 0` uniformly).
16Helper lemmas carry the prefix `rf_`.
17-/

29theorem rf_cb_nonneg {k : ℕ} (r : Euc d → ℝ) : 0 ≤ cbNorm k r :=
30  Finset.sum_nonneg fun j _ => Real.iSup_nonneg fun z => norm_nonneg _

32theorem rf_cb_le {k : ℕ} {r : Euc d → ℝ} (h : MemCb k r) {j : ℕ} (hj : j ≤ k) (y : Euc d) :
33    ‖iteratedFDeriv ℝ j r y‖ ≤ cbNorm k r := by
34  unfold cbNorm
35  calc ‖iteratedFDeriv ℝ j r y‖ ≤ ⨆ z, ‖iteratedFDeriv ℝ j r z‖ := le_ciSup (h.2 j hj) y
36    _ ≤ _ := Finset.single_le_sum (f := fun j => ⨆ z : Euc d, ‖iteratedFDeriv ℝ j r z‖)
37        (fun _ _ => Real.iSup_nonneg fun z => norm_nonneg _) (Finset.mem_range.2 (by omega))

39theorem rf_memCb_two {m : ℕ} (hm : 2 ≤ m) {r : Euc d → ℝ} (h : MemCb m r) : MemCb 2 r :=
40  ⟨h.1.of_le (by exact_mod_cast hm), fun j hj => h.2 j (hj.trans hm)⟩

46/-- the `IsPoissonSolution` conclusions from the master theorem data at level `2`. -/
47theorem rf_isSol {q : ℝ} {B : Euc d → Euc d} {μ : Measure (Euc d)} {r : Euc d → ℝ} {χ : Euc d → ℝ}
48    {C2 Cl2 : ℝ} (hr : MemCb 2 r) (h2 : ContDiff ℝ 2 χ) (hint : Integrable χ μ)
49    (hmean : ∫ x, χ x ∂μ = 0) (heq : ∀ x, poissonOp (fun x => -q • x + B x) χ x = r x)
50    (hder : ∀ j : ℕ, 1 ≤ j → j ≤ 2 → ∀ x, ‖iteratedFDeriv ℝ j χ x‖ ≤ C2 * cbNorm 2 r)
51    (hlog : ∀ x, |χ x| ≤ Cl2 * (cbNorm 2 r + cbNorm 2 r) * (1 + Real.log (2 + ‖x‖))) :
52    IsPoissonSolution (fun x => -q • x + B x) μ r (2 * |Cl2| + 2 * |C2|) χ := by
53  have hcb := rf_cb_nonneg (k := 2) r
54  refine ⟨h2, hint, hmean, heq, fun x => ?_, fun x y => ?_⟩
55  · refine (hlog x).trans ?_
56    have hl : 0 ≤ 1 + Real.log (2 + ‖x‖) := by
57      have := Real.log_nonneg (show (1 : ℝ) ≤ 2 + ‖x‖ by linarith [norm_nonneg x])
58      linarith
59    have : Cl2 * (cbNorm 2 r + cbNorm 2 r) ≤ (2 * |Cl2| + 2 * |C2|) * cbNorm 2 r := by
60      nlinarith [le_abs_self Cl2, abs_nonneg C2, mul_nonneg hcb (abs_nonneg Cl2),
61        mul_nonneg hcb (abs_nonneg C2)]
62    nlinarith [mul_le_mul_of_nonneg_right this hl]
63  · have h1 : ‖gradient χ x‖ = ‖iteratedFDeriv ℝ 1 χ x‖ := by
64      rw [norm_iteratedFDeriv_one]
65      unfold gradient
66      exact (InnerProductSpace.toDual ℝ (Euc d)).symm.norm_map _
67    rw [h1]
68    have a1 := hder 1 le_rfl (by norm_num) x
69    have a2 := hder 2 (by norm_num) le_rfl y
70    nlinarith [le_abs_self C2, abs_nonneg Cl2, mul_nonneg hcb (abs_nonneg Cl2),
71      mul_nonneg hcb (abs_nonneg C2), mul_le_mul_of_nonneg_right (le_abs_self C2) hcb]

74theorem affine_poisson_proof : type_of% @HRS.Whole.affine_poisson := by
75  intro d hd q Mb m₂ hq
76  obtain ⟨Cm, Cl, C2, Cl2, h⟩ := HRS.MS.PoisRes.pois_resolvent (d := d) q Mb m₂ hq 2 le_rfl
77  refine ⟨2 * |Cl2| + 2 * |C2|, fun B hB μ hμ hμ2 hm2 r hr _ hcen => ?_⟩
78  have hD : ∀ j : ℕ, 1 ≤ j → j ≤ 2 → ∀ x, ‖iteratedFDeriv ℝ j r x‖ ≤ cbNorm 2 r :=
79    fun j h1 h2 x => rf_cb_le hr h2 x
80  obtain ⟨χ, h1, h2, h3, h4, -, -, h7⟩ := h B hB μ hμ hμ2 hm2 r hr.1 (cbNorm 2 r) hD hcen
81  obtain ⟨g1, g2⟩ := h7 (cbNorm 2 r) hD
82  exact ⟨χ, rf_isSol hr h1 h2 h3 h4 g1 (g2 (cbNorm 2 r) (fun x => rf_cb_le hr (Nat.zero_le _) x |>
83    fun h => by simpa using h))⟩

86theorem affine_poisson_higher_proof : type_of% @HRS.Whole.affine_poisson_higher := by
87  intro d hd q Mb m₂ hq m hm
88  obtain ⟨Cm, Cl, C2, Cl2, h⟩ := HRS.MS.PoisRes.pois_resolvent (d := d) q Mb m₂ hq m hm
89  refine ⟨2 * |Cl2| + 2 * |C2|, (m : ℝ) * |Cm|, fun B hB μ hμ hμ2 hm2 r hr _ hcen => ?_⟩
90  have hr2 := rf_memCb_two hm hr
91  have hD : ∀ j : ℕ, 1 ≤ j → j ≤ m → ∀ x, ‖iteratedFDeriv ℝ j r x‖ ≤ cbNorm m r :=
92    fun j h1 h2 x => rf_cb_le hr h2 x
93  have hD2 : ∀ j : ℕ, 1 ≤ j → j ≤ 2 → ∀ x, ‖iteratedFDeriv ℝ j r x‖ ≤ cbNorm 2 r :=
94    fun j h1 h2 x => rf_cb_le hr2 h2 x
95  obtain ⟨χ, h1, h2, h3, h4, h5, -, h7⟩ := h B hB μ hμ hμ2 hm2 r hr.1 (cbNorm m r) hD hcen
96  obtain ⟨g1, g2⟩ := h7 (cbNorm 2 r) hD2
97  have hM0 : ∀ x, ‖r x‖ ≤ cbNorm 2 r := fun x => by
98    have := rf_cb_le hr2 (Nat.zero_le _) x
99    simpa using this
100  refine ⟨χ, rf_isSol hr2 (h1.of_le (by exact_mod_cast hm)) h2 h3 h4 g1 (g2 _ hM0), h1, fun x => ?_⟩
101  have hcb := rf_cb_nonneg (k := m) r
102  have hb : ∀ j : Fin m, ‖iteratedFDeriv ℝ ((j : ℕ) + 1) χ (x j)‖ ≤ |Cm| * cbNorm m r := fun j =>
103    (h5 ((j : ℕ) + 1) (by omega) (by omega) (x j)).trans
104      (mul_le_mul_of_nonneg_right (le_abs_self _) hcb)
105  calc ∑ j : Fin m, ‖iteratedFDeriv ℝ ((j : ℕ) + 1) χ (x j)‖ ≤ ∑ _j : Fin m, |Cm| * cbNorm m r :=
106        Finset.sum_le_sum fun j _ => hb j
107    _ = (m : ℝ) * |Cm| * cbNorm m r := by simp; ring
