26theorem q40b_adjoint_all (hd : 2 ≤ d) (htr : IsSmoothPositiveTrajectory d s β V ρ₀ ρ)

32      (∀ a, ∀ τ : ℝ, 0 ≤ τ → ∃ F : ℝ → Torus d → ℝ,
33        IsClassicalAdjointSolution htr.dyn ρ τ (φ a) F ∧
34          ∀ r ∈ Set.Icc 0 τ, propagator htr.dyn ρ r τ (φ a) = F r) ∧
35      (∀ a, ∀ τ : ℝ, 0 ≤ τ → ∀ r ∈ Set.Icc 0 τ, ∀ x,
36        ‖Torus.grad (propagator htr.dyn ρ r τ (φ a)) x‖ ≤ Ca a * Real.exp (-m * (τ - r))) := by
37  have H := fun a => q40b_adjoint hd htr hρβ hρS hρpos hj hjd (hφ a)
38  choose C m0 hC0 hm0 hbd using fun a => (H a).2
39  choose B hB using fun a => Parx.PXFID.exists_norm_grad_le_of_smooth (hφ a)
40  have hS0 : 0 ≤ ∑ a, 1 / m0 a := Finset.sum_nonneg fun a _ => (one_div_pos.2 (hm0 a)).le
41  set m : ℝ := 1 / (1 + ∑ a, 1 / m0 a) with hm
42  have hmpos : 0 < m := one_div_pos.2 (by linarith)
43  have hmle : ∀ a, m ≤ m0 a := by
44    intro a
45    have hS : 1 / m0 a ≤ ∑ b, 1 / m0 b :=
46      Finset.single_le_sum (f := fun b => 1 / m0 b) (fun b _ => (one_div_pos.2 (hm0 b)).le)
47        (Finset.mem_univ a)
48    have h1 : 1 / (1 + ∑ b, 1 / m0 b) ≤ 1 / (1 / m0 a) :=
49      one_div_le_one_div_of_le (one_div_pos.2 (hm0 a)) (by linarith)
50    rwa [one_div_one_div] at h1
51  refine ⟨fun a => max (C a) (B a), m, fun a => (hC0 a).trans (le_max_left _ _), hmpos,
52    fun a τ hτ => (H a).1 τ hτ, ?_⟩
53  intro a τ hτ r hr x
54  rcases hτ.eq_or_lt with h0 | hpos
55  · subst h0
56    have hr0 : r = 0 := le_antisymm hr.2 hr.1
57    subst hr0
58    obtain ⟨F, hF, hFp⟩ := (H a).1 0 le_rfl
59    have : propagator htr.dyn ρ 0 0 (φ a) = φ a := by
60      rw [hFp 0 ⟨le_rfl, le_rfl⟩]; exact hF.terminal
61    rw [this]
62    simp only [sub_self, mul_zero, Real.exp_zero, mul_one]
63    exact (hB a x).trans (le_max_right _ _)
64  · refine (hbd a τ hpos r hr x).trans ?_
65    have h1 : Real.exp (-m0 a * (τ - r)) ≤ Real.exp (-m * (τ - r)) :=
66      Real.exp_le_exp.2 (by nlinarith [hmle a, hr.2])
67    exact mul_le_mul (le_max_left _ _) h1 (Real.exp_pos _).le ((hC0 a).trans (le_max_left _ _))
