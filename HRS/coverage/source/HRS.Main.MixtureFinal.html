27/-- The initial rates of Proposition 5.9 for the mixture preparation: all three clauses, in the form used by
28`mixture_full_bounds` (limit law `ν_mix` with the factor `1 / 2`). -/
29theorem mxf_init (S : PreparedData d) (m : ℝ) (hmd : (d : ℝ) / 2 < m) (b : Torus d → ℝ)
30    (hb : Torus.IsSmooth b) (hb0 : pair S.μ0 b = 0) (a₀ : ℝ)
31    (γ : Measure (NegSobolev d m)) (hγ : IsCenteredGaussianWithCov m γ (iidCov S.μ0)) :
32    ∃ Cm : ℝ, 0 ≤ Cm ∧ ∀ N : ℕ, 2 ≤ N → 4 * a₀ ^ 2 * (⨆ x, |b x|) ^ 2 ≤ (N : ℝ) →
33      smoothDist 2 (law (mixturePrep S.μ0 b a₀ N) (fun X => etaE m X S.μ0))
34          ((1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
35            (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x))) ≤
36        Cm * iidRateSmooth d m N ∧
37      wassersteinE 1 (law (mixturePrep S.μ0 b a₀ N) (fun X => etaE m X S.μ0))
38          ((1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
39            (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x))) ≤
40        ENNReal.ofReal (Cm * iidRateW d m N) ∧
41      wassersteinE 2 (law (mixturePrep S.μ0 b a₀ N) (fun X => etaE m X S.μ0))
42          ((1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
43            (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x))) ≤
44        ENNReal.ofReal (Cm * iidRateW d m N ^ ((1 : ℝ) / 3)) := by
45  have hbc : Continuous b := therm_continuous_of_smooth hb
46  have hbm : Measurable b := hbc.measurable
47  have hρc : Continuous S.ρ0 := therm_continuous_of_smooth S.hρ0.smooth
48  set B : ℝ := ⨆ x, |b x| with hBdef
49  have hB : ∀ x, |b x| ≤ B := fun x => mx_abs_le_iSup hb x
50  have hB0 : 0 ≤ B := (abs_nonneg _).trans (hB 0)
51  obtain ⟨xmin, -, hxmin⟩ := isCompact_univ.exists_isMinOn Set.univ_nonempty hρc.continuousOn
52  obtain ⟨xmax, -, hxmax⟩ := isCompact_univ.exists_isMaxOn Set.univ_nonempty hρc.continuousOn
53  have hc : 0 < S.ρ0 xmin := S.hρ0.pos xmin
54  have hbL : MemLp b ⊤ (volume : Measure (Torus d)) :=
55    MemLp.of_bound hbc.aestronglyMeasurable B
56      (Eventually.of_forall fun x => by rw [Real.norm_eq_abs]; exact hB x)
57  have hnorm : (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ≤ B := by
58    have h1 : eLpNorm b ⊤ (volume : Measure (Torus d)) ≤ ENNReal.ofReal B := by
59      rw [eLpNorm_exponent_top]
60      exact eLpNormEssSup_le_of_ae_bound
61        (Eventually.of_forall fun x => by rw [Real.norm_eq_abs]; exact hB x)
62      all_goals first | exact hbc.aestronglyMeasurable | skip
63    exact ENNReal.toReal_le_of_le_ofReal hB0 h1
64  obtain ⟨Cm, hCm0, hCm⟩ := HRS.iid_field_mixture_proof S.hd m hmd (S.ρ0 xmin) (S.ρ0 xmax) hc a₀ b
65    hbm hbL
66  refine ⟨Cm, hCm0, fun N hN2 hN4' => ?_⟩
67  have h4 : 4 * a₀ ^ 2 * (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ^ 2 ≤ N := by
68    have h1 : (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ^ 2 ≤ B ^ 2 :=
69      pow_le_pow_left₀ ENNReal.toReal_nonneg hnorm 2
70    have h2 : 4 * a₀ ^ 2 * (eLpNorm b ⊤ (volume : Measure (Torus d))).toReal ^ 2 ≤
71        4 * a₀ ^ 2 * B ^ 2 := by gcongr
72    linarith
73  have := (hCm S.ρ0 S.hρ0.density (fun x => by simpa using hxmin (Set.mem_univ x))
74    (fun x => by simpa using hxmax (Set.mem_univ x)) hb0 γ hγ N hN2 h4)
75  rw [← therm_mixturePrep_eq S b hbm a₀ N] at this
76  simp only [therm_ofFun_smul] at this
77  simp only [one_div (2 : ℝ≥0∞)]
78  exact this

80/-- Clauses 1 and 2 (smooth and `W_1`) of `mixture_full_bounds`. -/
81theorem mxf_part12 (S : PreparedData d) (m : ℝ) (hm : (d : ℝ) + 2 < m) (T : ℝ)
82    (hT : 0 < T) (b : Torus d → ℝ) (hb : Torus.IsSmooth b) (hb0 : pair S.μ0 b = 0) (a₀ : ℝ)
83    (γ : Measure (NegSobolev d m)) (hγ : IsCenteredGaussianWithCov m γ (iidCov S.μ0)) :
84    ∃ C : ℝ, 0 < C ∧
85      ∀ a : ℝ, 0 ≤ a → ∀ N : ℕ, 2 ≤ N → 4 * a₀ ^ 2 * (⨆ x, |b x|) ^ 2 ≤ (N : ℝ) →
86      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
87        (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N),
88        SelectedFlow P S.dyn (a + T) X₀ W X →
89        (∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
90          (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X) →
91        law P X₀ = mixturePrep S.μ0 b a₀ N →
92      ∀ {Ω' : Type} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
93        (Y₀ : Ω' → NegSobolev d m) (Y : ℝ → Ω' → NegSobolev d m),
94        IsPreparedLinear S m (a + T)
95          ((1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
96            (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)))
97          P' Y₀ Y →
98        smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
99          C * (Real.sqrt N * deltaN d S.s N + iidRateSmooth d m N) ∧
100        wassersteinE 1 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
101          ENNReal.ofReal (C * (Real.sqrt N * deltaN d S.s N + deltaN d S.s N ^ ((1 : ℝ) / 4) +
102            iidRateW d m N)) := by
103  have hmd : (d : ℝ) / 2 < m := by
104    have : (0 : ℝ) ≤ d := Nat.cast_nonneg d
105    linarith
106  obtain ⟨Cm, hCm0, hCm⟩ := mxf_init S m hmd b hb hb0 a₀ γ hγ
107  obtain ⟨Cb, hCb0, hCb⟩ := mx_B_mixture S b hb hb0 a₀
108  obtain ⟨C1, hC1, h1⟩ := fb_smooth_W1 S m hm T hT Cb Cm Cm hCb0.le hCm0 hCm0
109    (iidRateSmooth d m) (iidRateW d m) (fun N _ => fb_iidRateSmooth_nonneg d m N)
110    (fun N hN => fb_iidRateW_nonneg d m (by omega))
111  refine ⟨C1, hC1, ?_⟩
112  intro a ha N hN hN4 Ω _ P _ X₀ W X hSel hfilt hinit Ω' _ P' _ Y₀ Y hY
113  have hlaw : law P (fun ω => etaE m (X₀ ω) S.μ0) =
114      law (mixturePrep S.μ0 b a₀ N) (fun X => etaE m X S.μ0) := by
115    rw [fb_law_eta S m hmd (by omega) P hSel.init_meas, hinit]
116  have hBk := hCb N hN hN4
117  rw [← hinit] at hBk
118  obtain ⟨hs1, hw1, -⟩ := hCm N hN hN4
119  exact h1 a ha N hN P X₀ W X hSel hfilt hBk.1 hBk.2 _ P' Y₀ Y hY (by rw [hlaw]; exact hs1)
120    (by rw [hlaw]; exact hw1)

126    (hs : S.s = 0) :
127    ∃ C : ℝ, 0 < C ∧
128      ∀ a : ℝ, 0 ≤ a → ∀ N : ℕ, 2 ≤ N → 4 * a₀ ^ 2 * (⨆ x, |b x|) ^ 2 ≤ (N : ℝ) →

134        law P X₀ = mixturePrep S.μ0 b a₀ N →
135      ∀ {Ω' : Type} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
136        (Y₀ : Ω' → NegSobolev d m) (Y : ℝ → Ω' → NegSobolev d m),
137        IsPreparedLinear S m (a + T)
138          ((1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
139            (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)))
140          P' Y₀ Y →
141        wassersteinE 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
142          ENNReal.ofReal (C * ((N : ℝ) ^ (-(1 : ℝ) / 4) + iidRateW d m N ^ ((1 : ℝ) / 3))) := by
143  have hd1 : 1 ≤ d := by have := S.hd; omega
144  have hmd : (d : ℝ) / 2 < m := by
145    have : (0 : ℝ) ≤ d := Nat.cast_nonneg d
146    linarith
147  obtain ⟨Cm, hCm0, hCm⟩ := mxf_init S m hmd b hb hb0 a₀ γ hγ
148  obtain ⟨Cb, hCb0, hCb⟩ := mx_B_mixture S b hb hb0 a₀
149  obtain ⟨p₀, hp₀, hLp⟩ := mx_Lp_mixture S hs
150  obtain ⟨Cj, hCj0, hCj⟩ := hLp b hb hb0 a₀
151  have hrw3 : ∀ N : ℕ, 2 ≤ N → 0 ≤ iidRateW d m N ^ ((1 : ℝ) / 3) := fun N hN =>
152    Real.rpow_nonneg (fb_iidRateW_nonneg d m (by omega)) _
153  obtain ⟨C2, hC2, h2⟩ := fb_W2 S m hm T hT p₀ hp₀ Cb Cj Cm hCb0.le hCj0.le hCm0
154    (fun N => iidRateW d m N ^ ((1 : ℝ) / 3)) hrw3

159    rw [fb_law_eta S m hmd (by omega) P hSel.init_meas, hinit]
160  have hδ := therm_deltaN_pos S.hd S.hs0 S.hsd N (by omega)
161  have hN0 : (0 : ℝ) < N := by exact_mod_cast (by omega : 0 < N)
162  have hBk := hCb N hN hN4
163  have hJk := hCj N hN hN4
164  rw [← hinit] at hBk hJk
165  have hmain := h2 a ha N hN P X₀ W X hSel hfilt hBk.1 hJk.1 hBk.2 hJk.2 _ P' Y₀ Y hY
166    (by rw [hlaw]; exact (hCm N hN hN4).2.2)
167  refine hmain.trans (ENNReal.ofReal_le_ofReal ?_)
168  have e : deltaN d S.s N = deltaN d 0 N := by rw [hs]
169  rw [e]
170  obtain ⟨hr1, hr2⟩ := fb_zero_rates hd1 (by omega : 1 ≤ N)
171  have hr := hrw3 N hN
172  have hq : 0 ≤ (N : ℝ) ^ (-(1 : ℝ) / 4) := Real.rpow_nonneg hN0.le _
173  rw [hr2]

179theorem mixture_full_bounds_proof (S : PreparedData d) (m : ℝ) (hm : (d : ℝ) + 2 < m) (T : ℝ)
180    (hT : 0 < T) (b : Torus d → ℝ) (hb : Torus.IsSmooth b) (hb0 : pair S.μ0 b = 0) (a₀ : ℝ)
181    (γ : Measure (NegSobolev d m)) (hγ : IsCenteredGaussianWithCov m γ (iidCov S.μ0)) :
182    ∃ C : ℝ, 0 < C ∧
183      ∀ a : ℝ, 0 ≤ a → ∀ N : ℕ, 2 ≤ N → 4 * a₀ ^ 2 * (⨆ x, |b x|) ^ 2 ≤ (N : ℝ) →
184      ∀ {Ω : Type} [MeasurableSpace Ω] (P : Measure Ω) [IsProbabilityMeasure P]
185        (X₀ : Ω → Config d N) (W : ℝ≥0 → Ω → (Fin N → Euc d)) (X : ℝ≥0 → Ω → Config d N),
186        SelectedFlow P S.dyn (a + T) X₀ W X →
187        (∃ (ℱ : MeasureTheory.Filtration ℝ ‹MeasurableSpace Ω›)
188          (W' : ℝ≥0 → Ω → (Fin N × Fin d) → ℝ), IsFilteredSolution P ℱ W' W X) →
189        law P X₀ = mixturePrep S.μ0 b a₀ N →
190      ∀ {Ω' : Type} [MeasurableSpace Ω'] (P' : Measure Ω') [IsProbabilityMeasure P']
191        (Y₀ : Ω' → NegSobolev d m) (Y : ℝ → Ω' → NegSobolev d m),
192        IsPreparedLinear S m (a + T)
193          ((1 / 2 : ℝ≥0∞) • γ.map (fun F => F + a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)) +
194            (1 / 2 : ℝ≥0∞) • γ.map (fun F => F - a₀ • NegSobolev.ofFun m (fun x => b x * S.ρ0 x)))
195          P' Y₀ Y →
196        smoothDist 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
197          C * (Real.sqrt N * deltaN d S.s N + iidRateSmooth d m N) ∧
198        wassersteinE 1 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
199          ENNReal.ofReal (C * (Real.sqrt N * deltaN d S.s N + deltaN d S.s N ^ ((1 : ℝ) / 4) +
200            iidRateW d m N)) ∧
201        (S.s = 0 →
202          wassersteinE 2 (S.etaLaw m T a P X) (law P' (windowPath m T a Y)) ≤
203            ENNReal.ofReal (C * ((N : ℝ) ^ (-(1 : ℝ) / 4) + iidRateW d m N ^ ((1 : ℝ) / 3)))) := by
204  obtain ⟨C1, hC1, h1⟩ := mxf_part12 S m hm T hT b hb hb0 a₀ γ hγ
205  by_cases hs : S.s = 0
206  · obtain ⟨C2, hC2, h2⟩ := mxf_part3 S m hm T hT b hb hb0 a₀ γ hγ hs
207    refine ⟨max C1 C2, lt_max_of_lt_left hC1, ?_⟩
208    intro a ha N hN hN4 Ω _ P _ X₀ W X hSel hfilt hinit Ω' _ P' _ Y₀ Y hY
209    have hδ := therm_deltaN_pos S.hd S.hs0 S.hsd N (by omega)
210    have hN0 : (0 : ℝ) < N := by exact_mod_cast (by omega : 0 < N)
211    have hx : 0 ≤ Real.sqrt N * deltaN d S.s N := by positivity
212    have hq : 0 ≤ deltaN d S.s N ^ ((1 : ℝ) / 4) := Real.rpow_nonneg hδ.le _
213    have hq2 : 0 ≤ (N : ℝ) ^ (-(1 : ℝ) / 4) := Real.rpow_nonneg hN0.le _
214    have hj := fb_iidRateW_nonneg d m (by omega : 1 ≤ N)
215    have hj3 : 0 ≤ iidRateW d m N ^ ((1 : ℝ) / 3) := Real.rpow_nonneg hj _
216    obtain ⟨hs1, hw1⟩ := h1 a ha N hN hN4 P X₀ W X hSel hfilt hinit P' Y₀ Y hY
217    refine ⟨hs1.trans (mul_le_mul_of_nonneg_right (le_max_left _ _)
218        (add_nonneg hx (fb_iidRateSmooth_nonneg d m N))),
219      hw1.trans (ENNReal.ofReal_le_ofReal (mul_le_mul_of_nonneg_right (le_max_left _ _)
220        (add_nonneg (add_nonneg hx hq) hj))), fun _ => ?_⟩
221    exact (h2 a ha N hN hN4 P X₀ W X hSel hfilt hinit P' Y₀ Y hY).trans
222      (ENNReal.ofReal_le_ofReal (mul_le_mul_of_nonneg_right (le_max_right _ _)
223        (add_nonneg hq2 hj3)))
224  · refine ⟨C1, hC1, ?_⟩
225    intro a ha N hN hN4 Ω _ P _ X₀ W X hSel hfilt hinit Ω' _ P' _ Y₀ Y hY
226    obtain ⟨hs1, hw1⟩ := h1 a ha N hN hN4 P X₀ W X hSel hfilt hinit P' Y₀ Y hY
227    exact ⟨hs1, hw1, fun h => absurd h hs⟩
