25/-- the lattice rate `κ⁻¹ + κ N^{-2/d}` tends to zero along `N = n^d`. -/
26theorem lat39fd_rate_tendsto (hd : 1 ≤ d) (κ : ℕ → ℝ) (hκpos : ∀ N, 0 < κ N)
27    (hκ1 : Tendsto (fun n : ℕ => κ (n ^ d)) atTop atTop)
28    (hκ2 : Tendsto (fun n : ℕ => κ (n ^ d) / (n : ℝ)) atTop (𝓝 0)) :

30      κ (n ^ d) * ((n ^ d : ℕ) : ℝ) ^ (-(2 : ℝ) / (d : ℝ))) atTop (𝓝 0) := by
31  have hd0 : (d : ℝ) ≠ 0 := by exact_mod_cast (show d ≠ 0 by omega)
32  have h1 : Tendsto (fun n : ℕ => (κ (n ^ d))⁻¹) atTop (𝓝 0) :=
33    tendsto_inv_atTop_zero.comp hκ1
34  have h2 : Tendsto (fun n : ℕ => κ (n ^ d) * ((n ^ d : ℕ) : ℝ) ^ (-(2 : ℝ) / (d : ℝ)))
35      atTop (𝓝 0) := by
36    refine squeeze_zero' ?_ ?_ hκ2
37    · exact Eventually.of_forall fun n => by have := hκpos (n ^ d); positivity
38    · filter_upwards [eventually_ge_atTop 1] with n hn
39      have hnr : (1 : ℝ) ≤ n := by exact_mod_cast hn
40      have hpow : ((n ^ d : ℕ) : ℝ) ^ (-(2 : ℝ) / (d : ℝ)) = (n : ℝ) ^ (-(2 : ℝ)) := by
41        rw [Nat.cast_pow, ← Real.rpow_natCast, ← Real.rpow_mul (by positivity)]
42        congr 1
43        field_simp
44      have hle : (n : ℝ) ^ (-(2 : ℝ)) ≤ (n : ℝ)⁻¹ := by
45        rw [← Real.rpow_neg_one]
46        exact Real.rpow_le_rpow_of_exponent_le hnr (by norm_num)
47      rw [hpow, div_eq_mul_inv]
48      exact mul_le_mul_of_nonneg_left hle (hκpos _).le
49  simpa using h1.add h2
