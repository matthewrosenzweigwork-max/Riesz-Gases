22/-- Increment hypothesis on dyadic grid values. -/
23def mkc_Inc (x : ℕ → ℕ → E) (lam r : ℝ) : Prop :=
24  ∀ n k : ℕ, k < 2 ^ n → ‖x n (k + 1) - x n k‖ ≤ lam * r ^ n

26/-- Consistency of the dyadic grids. -/
27def mkc_Cons (x : ℕ → ℕ → E) : Prop := ∀ n k : ℕ, x (n + 1) (2 * k) = x n k

29theorem mkc_chain {x : ℕ → ℕ → E} {lam r : ℝ} (hc : mkc_Cons x) (hi : mkc_Inc x lam r) (m : ℕ) :
30    ∀ (j k : ℕ), k ≤ 2 ^ (m + j) →
31      ‖x (m + j) k - x m (k / 2 ^ j)‖ ≤ ∑ i ∈ Finset.range j, lam * r ^ (m + 1 + i) := by
32  intro j
33  induction j with
34  | zero => intro k hk; simp
35  | succ j ih =>
36    intro k hk
37    have hQ : 2 ^ (m + (j + 1)) = 2 * 2 ^ (m + j) := by ring
38    rw [hQ] at hk
39    have h1 : k / 2 ≤ 2 ^ (m + j) := by omega
40    have ih' := ih (k / 2) h1
41    have hdiv : k / 2 ^ (j + 1) = k / 2 / 2 ^ j := by
42      rw [Nat.div_div_eq_div_mul, pow_succ']
43    rw [hdiv, Finset.sum_range_succ]
44    have hstep : ‖x (m + (j + 1)) k - x (m + j) (k / 2)‖ ≤ lam * r ^ (m + 1 + j) := by
45      have hmj : m + (j + 1) = m + j + 1 := by ring
46      have hpow : r ^ (m + 1 + j) = r ^ (m + j + 1) := by ring_nf
47      rw [hmj, hpow]
48      rcases Nat.even_or_odd' k with ⟨k', hk' | hk'⟩
49      · have : k / 2 = k' := by omega
50        rw [this, hk', hc (m + j) k']
51        simp only [sub_self, norm_zero]
52        have := hi (m + j + 1) 0 (by positivity)
53        have h0 : 0 ≤ ‖x (m + j + 1) (0 + 1) - x (m + j + 1) 0‖ := norm_nonneg _
54        linarith
55      · have : k / 2 = k' := by omega
56        rw [this, hk']
57        have h2 : 2 * k' < 2 ^ (m + j + 1) := by
58          have : 2 ^ (m + j + 1) = 2 * 2 ^ (m + j) := by ring
59          omega
60        have := hi (m + j + 1) (2 * k') h2
61        rw [← hc (m + j) k']
62        exact this
63    calc ‖x (m + (j + 1)) k - x m (k / 2 / 2 ^ j)‖
64        = ‖(x (m + (j + 1)) k - x (m + j) (k / 2)) + (x (m + j) (k / 2) - x m (k / 2 / 2 ^ j))‖ := by
65          congr 1; abel
66      _ ≤ _ := norm_add_le _ _
67      _ ≤ lam * r ^ (m + 1 + j) + ∑ i ∈ Finset.range j, lam * r ^ (m + 1 + i) := add_le_add hstep ih'
68      _ = _ := add_comm _ _

70theorem mkc_geom_le {lam r : ℝ} (hlam : 0 ≤ lam) (hr0 : 0 ≤ r) (hr1 : r < 1) (m j : ℕ) :
71    ∑ i ∈ Finset.range j, lam * r ^ (m + 1 + i) ≤ lam * r ^ (m + 1) / (1 - r) := by
72  have h1r : 0 < 1 - r := by linarith
73  have e : ∑ i ∈ Finset.range j, lam * r ^ (m + 1 + i) =
74      lam * r ^ (m + 1) * ∑ i ∈ Finset.range j, r ^ i := by
75    rw [Finset.mul_sum]; exact Finset.sum_congr rfl fun i _ => by ring
76  rw [e, div_eq_mul_one_div (lam * r ^ (m + 1))]
77  refine mul_le_mul_of_nonneg_left ?_ (mul_nonneg hlam (pow_nonneg hr0 _))
78  have hg : (∑ i ∈ Finset.range j, r ^ i) * (1 - r) = 1 - r ^ j := geom_sum_mul_neg r j
79  rw [le_div_iff₀ h1r, hg]
80  have := pow_nonneg hr0 j
81  linarith

83/-- The constant of the chaining estimate. -/
84def mkc_c (r : ℝ) : ℝ := 1 + 2 * r / (1 - r)

86theorem mkc_incr {x : ℕ → ℕ → E} {lam r : ℝ} (hc : mkc_Cons x) (hi : mkc_Inc x lam r)
87    (hlam : 0 ≤ lam) (hr0 : 0 ≤ r) (hr1 : r < 1) (m j k l : ℕ)
88    (hk : k ≤ 2 ^ (m + j)) (hl : l ≤ 2 ^ (m + j)) (hkl : k ≤ l + 2 ^ j) (hlk : l ≤ k + 2 ^ j) :
89    ‖x (m + j) k - x (m + j) l‖ ≤ mkc_c r * (lam * r ^ m) := by
90  have h1r : 0 < 1 - r := by linarith
91  have hpos : 0 < 2 ^ j := by positivity
92  have hb : ∀ q, q ≤ 2 ^ (m + j) →
93      ‖x (m + j) q - x m (q / 2 ^ j)‖ ≤ lam * r ^ (m + 1) / (1 - r) := fun q hq =>
94    (mkc_chain hc hi m j q hq).trans (mkc_geom_le hlam hr0 hr1 m j)
95  have hk' : k / 2 ^ j ≤ 2 ^ m := by
96    apply Nat.div_le_of_le_mul
97    rw [pow_add] at hk
98    linarith
99  have hl' : l / 2 ^ j ≤ 2 ^ m := by
100    apply Nat.div_le_of_le_mul
101    rw [pow_add] at hl
102    linarith
103  have hkl' : k / 2 ^ j ≤ l / 2 ^ j + 1 := by
104    calc k / 2 ^ j ≤ (l + 2 ^ j) / 2 ^ j := Nat.div_le_div_right hkl
105      _ = l / 2 ^ j + 1 := by rw [Nat.add_div_right _ hpos]
106  have hlk' : l / 2 ^ j ≤ k / 2 ^ j + 1 := by
107    calc l / 2 ^ j ≤ (k + 2 ^ j) / 2 ^ j := Nat.div_le_div_right hlk
108      _ = k / 2 ^ j + 1 := by rw [Nat.add_div_right _ hpos]
109  have hmid : ‖x m (k / 2 ^ j) - x m (l / 2 ^ j)‖ ≤ lam * r ^ m := by
110    rcases Nat.lt_trichotomy (k / 2 ^ j) (l / 2 ^ j) with h | h | h
111    · have e : l / 2 ^ j = k / 2 ^ j + 1 := by omega
112      rw [e, norm_sub_rev]
113      exact hi m (k / 2 ^ j) (by omega)
114    · rw [h]; simp only [sub_self, norm_zero]; positivity
115    · have e : k / 2 ^ j = l / 2 ^ j + 1 := by omega
116      rw [e]
117      exact hi m (l / 2 ^ j) (by omega)
118  calc ‖x (m + j) k - x (m + j) l‖
119      = ‖(x (m + j) k - x m (k / 2 ^ j)) + (x m (k / 2 ^ j) - x m (l / 2 ^ j)) -
120          (x (m + j) l - x m (l / 2 ^ j))‖ := by congr 1; abel
121    _ ≤ ‖(x (m + j) k - x m (k / 2 ^ j)) + (x m (k / 2 ^ j) - x m (l / 2 ^ j))‖ +
122          ‖x (m + j) l - x m (l / 2 ^ j)‖ := norm_sub_le _ _
123    _ ≤ ‖x (m + j) k - x m (k / 2 ^ j)‖ + ‖x m (k / 2 ^ j) - x m (l / 2 ^ j)‖ +
124          ‖x (m + j) l - x m (l / 2 ^ j)‖ := by gcongr; exact norm_add_le _ _
125    _ ≤ lam * r ^ (m + 1) / (1 - r) + lam * r ^ m + lam * r ^ (m + 1) / (1 - r) := by
126        gcongr
127        · exact hb k hk
128        · exact hb l hl
129    _ = mkc_c r * (lam * r ^ m) := by
130        unfold mkc_c; field_simp; ring

156theorem mkc_pt_idx {T : ℝ} (hT : 0 < T) {t : ℝ} (ht0 : 0 ≤ t) (n : ℕ) :
157    mkc_pt T n (mkc_idx T n t) ≤ t ∧ t < mkc_pt T n (mkc_idx T n t) + T / 2 ^ n := by
158  have h2 : (0 : ℝ) < 2 ^ n := by positivity

187theorem mkc_limit [CompleteSpace E] {x : ℕ → ℕ → E} {lam r : ℝ} (hc : mkc_Cons x)
188    (hi : mkc_Inc x lam r) (hlam : 0 ≤ lam) (hr0 : 0 ≤ r) (hr1 : r < 1) {T : ℝ} (hT : 0 < T)
189    {t : ℝ} (ht : t ∈ Set.Icc (0 : ℝ) T) :
190    ∃ L : E, Tendsto (fun n => x n (mkc_idx T n t)) atTop (𝓝 L) ∧
191      ∀ n, ‖x n (mkc_idx T n t) - L‖ ≤ lam * r ^ (n + 1) / (1 - r) := by
192  have h1r : 0 < 1 - r := by linarith
193  have hb : ∀ n j : ℕ, ‖x (n + j) (mkc_idx T (n + j) t) - x n (mkc_idx T n t)‖ ≤
194      lam * r ^ (n + 1) / (1 - r) := by
195    intro n j
196    have := mkc_chain hc hi n j (mkc_idx T (n + j) t) (mkc_idx_le hT ht.2 _)
197    rw [mkc_idx_div hT] at this
198    exact this.trans (mkc_geom_le hlam hr0 hr1 n j)
199  have hcau : CauchySeq (fun n => x n (mkc_idx T n t)) := by
200    rw [Metric.cauchySeq_iff']
201    intro ε hε
202    have hlim : Tendsto (fun n : ℕ => lam * r ^ (n + 1) / (1 - r)) atTop (𝓝 0) := by
203      have := ((tendsto_pow_atTop_nhds_zero_of_lt_one hr0 hr1).comp
204        (tendsto_add_atTop_nat 1)).const_mul lam |>.div_const (1 - r)
205      simpa using this
206    obtain ⟨N, hN⟩ := (hlim.eventually (gt_mem_nhds hε)).exists
207    refine ⟨N, fun n hn => ?_⟩
208    obtain ⟨j, rfl⟩ := Nat.exists_eq_add_of_le hn
209    rw [dist_eq_norm]
210    exact (hb N j).trans_lt hN
211  obtain ⟨L, hL⟩ := cauchySeq_tendsto_of_complete hcau
212  refine ⟨L, hL, fun n => ?_⟩
213  have : Tendsto (fun n' => ‖x n' (mkc_idx T n' t) - x n (mkc_idx T n t)‖) atTop
214      (𝓝 ‖L - x n (mkc_idx T n t)‖) := (hL.sub_const _).norm
215  rw [norm_sub_rev]
216  refine le_of_tendsto this (eventually_atTop.2 ⟨n, fun n' hn' => ?_⟩)
217  obtain ⟨j, rfl⟩ := Nat.exists_eq_add_of_le hn'
218  exact hb n j

244theorem mkc_markov (μ : Measure Ω) {d : Ω → E} (hd : Measurable d) {p : ℝ} (hp : 0 < p)
245    {c M : ℝ} (hc : 0 < c) (hM : ∫⁻ ζ, ‖d ζ‖ₑ ^ p ∂μ ≤ ENNReal.ofReal M) :
246    μ {ζ | c < ‖d ζ‖} ≤ ENNReal.ofReal (M / c ^ p) := by
247  have hcp : 0 < c ^ p := Real.rpow_pos_of_pos hc p
248  have hsub : {ζ | c < ‖d ζ‖} ⊆ {ζ | ENNReal.ofReal (c ^ p) ≤ ‖d ζ‖ₑ ^ p} := by
249    intro ζ hζ
250    simp only [Set.mem_ofPred_eq] at hζ ⊢
251    rw [← ENNReal.ofReal_rpow_of_nonneg hc.le hp.le]
252    exact ENNReal.rpow_le_rpow (by rw [← ofReal_norm]; exact ENNReal.ofReal_le_ofReal hζ.le)
253      hp.le
254  have hmeas : AEMeasurable (fun ζ => ‖d ζ‖ₑ ^ p) μ :=
255    (hd.enorm.pow_const p).aemeasurable
256  calc μ {ζ | c < ‖d ζ‖} ≤ μ {ζ | ENNReal.ofReal (c ^ p) ≤ ‖d ζ‖ₑ ^ p} := measure_mono hsub
257    _ ≤ (∫⁻ ζ, ‖d ζ‖ₑ ^ p ∂μ) / ENNReal.ofReal (c ^ p) :=
258        meas_ge_le_lintegral_div hmeas (by simpa using hcp) ENNReal.ofReal_ne_top
259    _ ≤ ENNReal.ofReal M / ENNReal.ofReal (c ^ p) := by gcongr
260    _ = ENNReal.ofReal (M / c ^ p) := (ENNReal.ofReal_div_of_pos hcp).symm

309theorem mkc_real_bound {T β p r q lam Cp : ℝ} (hT : 0 < T) (hCp : 0 ≤ Cp)

316      Real.rpow_pow_comm (by norm_num) β n]
317  have e2 : (lam * r ^ n) ^ p = lam ^ p * (r ^ p) ^ n := by
318    rw [Real.mul_rpow hlam.le (pow_nonneg hr0.le n), Real.rpow_pow_comm hr0.le p n]

320  have hrp : 0 < r ^ p := Real.rpow_pos_of_pos hr0 p

323  have hApos : 0 < A := by positivity

325  have hKc0 : 0 ≤ Kc := by
326    have : 0 ≤ T ^ (1 + β) := Real.rpow_nonneg hT.le _
327    positivity

329    rw [e1, e2, hKc, hA]
330    field_simp

332    rw [Real.rpow_neg hlam.le, hKc]
333    field_simp
334  rw [hL, hR, div_le_iff₀ hApos]
335  have h1 : 1 ≤ (q * 2 ^ β * r ^ p) ^ n := one_le_pow₀ hqr
336  have h3 : (q * 2 ^ β * r ^ p) ^ n = q ^ n * A := by
337    rw [hA, mul_assoc, mul_pow, mul_pow]
338  rw [h3] at h1
