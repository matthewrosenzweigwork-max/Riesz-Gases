5/-!
6# The attractive side of Proposition 4.25 (`M:O:prop:log-prep`)
7
8For the periodic logarithm and a bounded density there is `a > 0` with
9`sup_{N ≥ 2} sup_{-a ≤ γ ≤ 0} K_{N,γ}(μ) < ∞` (source lines 5588-5593 for `γ ≤ 0`).  The
10proof in the manuscript (lines 5605-5610) cites Delgadino-Gvalani-Rosenzweig, Theorem 2.6 and
11Remark 2.1, together with the exponential-row estimate `periodic_log_row` (Lemma 4.24).
12
13The cited theorem is available, in the abstract form needed here (any symmetric measurable
14kernel with factorial row moments for the centred kernel), as
15`DGRHS.Attractive.source_uniform_partition` in the vendored modules of `external/dgrhs`
16(`external/dgrhs/PROVENANCE.md`).  This file only translates between the two conventions:
17`W(x, y) = -g(x - y)`, the centred kernel of `DGRHS` is minus the centred kernel `g_μ` of
18`HRS.centeredKernel`, and `K_{N,γ}` with `γ = -β ≤ 0` is the partition function
19`∫ exp(-β N F_N^W)`.
20-/

91theorem modulatedEnergyKernel_eq [IsProbabilityMeasure μ] (hg : ∀ z, g (-z) = g z)
92    (hI : Integrable (fun p : Torus d × Torus d => g (p.1 - p.2)) (μ.prod μ))
93    {N : ℕ} (hN : 0 < N) (X : Config d N) :
94    modulatedEnergyKernel g X μ =
95      -DGRHS.CLT.sourceCanonicalModulatedEnergy
96        (fun p => DGRHS.centeredKernel μ (Wk g) p.1 p.2)
97        (DGRHS.potential μ (Wk g)) (DGRHS.interactionEnergy μ (Wk g)) N X := by
98  rw [DGRHS.CLT.sourceCanonicalModulatedEnergy_raw μ (Wk g) (Wk_symm hg) N hN X]
99  have hoff : DGRHS.CanonicalVariance.offDiagonalSum (Function.uncurry (Wk g)) X =
100      -∑ i, ∑ j ∈ Finset.univ.erase i, g (X i - X j) := by
101    rw [offDiagonalSum_eq]
102    simp only [Wk, Finset.sum_neg_distrib]
103  rw [hoff]
104  rw [interactionEnergy_eq hI]
105  simp only [potential_eq, Finset.sum_neg_distrib]
106  unfold modulatedEnergyKernel modulatedEnergy
107  change _ = _
108  simp only [convMeasure, energyPairing] at *

111/-- Row factorial moments of the `DGRHS` centred kernel, from the exponential row estimate
112`periodic_log_row` (Lemma 4.24) with exponent `a = 1`. -/
113theorem rowFactorialMoments (hd : 2 ≤ d) (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ)
114    (hΛ : ∀ x, ρ x ≤ Λ) :
115    ∃ C : ℝ, 1 ≤ C ∧ ∀ p : ℕ, 1 ≤ p → ∀ y,
116      (∫⁻ x, ENNReal.ofReal
117        (|DGRHS.centeredKernel (withDensityReal ρ) (Wk (periodicRiesz d 0)) x y| ^ p)
118          ∂(withDensityReal ρ)) ≤ ENNReal.ofReal (C ^ p * (p.factorial : ℝ)) := by
119  have hs0 : (0 : ℝ) ≤ 0 := le_rfl
120  have hsd : (0 : ℝ) < d := by exact_mod_cast (by omega : 0 < d)
121  have hg : Measurable (periodicRiesz d 0) := PoleCavityAux.measurable_periodicRiesz hd hs0 hsd
122  have hgi : Integrable (periodicRiesz d 0) := KernelAux.integrable_periodicRiesz hd hs0 hsd
123  haveI := isProb_withDensityReal hρ
124  have hI := PreparationAux.integrable_pair_withDensity hg hgi ρ Λ hΛ (ν := withDensityReal ρ) rfl
125  have hev : ∀ z, periodicRiesz d 0 (-z) = periodicRiesz d 0 z := fun z => periodicRiesz_neg z
126  obtain ⟨Cexp, hCexp⟩ := periodic_log_row hd ρ hρ Λ hΛ 1 one_pos (by exact_mod_cast (by omega : 1 < d))
127  refine ⟨max 1 Cexp, le_max_left _ _, fun p hp y => ?_⟩
128  have hsymm : ∀ x, centeredKernel (periodicRiesz d 0) (withDensityReal ρ) x y =
129      centeredKernel (periodicRiesz d 0) (withDensityReal ρ) y x := by
130    intro x
131    unfold centeredKernel
132    rw [← hev (x - y), neg_sub]
133    ring
134  have h1 : ∀ x, ENNReal.ofReal
135      (|DGRHS.centeredKernel (withDensityReal ρ) (Wk (periodicRiesz d 0)) x y| ^ p) ≤
136      ENNReal.ofReal ((p.factorial : ℝ) *
137        Real.exp (1 * |centeredKernel (periodicRiesz d 0) (withDensityReal ρ) y x|)) := by
138    intro x
139    apply ENNReal.ofReal_le_ofReal
140    rw [centeredKernel_eq hev hI, abs_neg, hsymm x]
141    have := Real.pow_div_factorial_le_exp
142      |centeredKernel (periodicRiesz d 0) (withDensityReal ρ) y x| (abs_nonneg _) p
143    have hf : (0 : ℝ) < p.factorial := by exact_mod_cast Nat.factorial_pos p
144    rw [div_le_iff₀ hf] at this
145    rw [one_mul]
146    linarith
147  calc _ ≤ ∫⁻ x, ENNReal.ofReal ((p.factorial : ℝ) *
148        Real.exp (1 * |centeredKernel (periodicRiesz d 0) (withDensityReal ρ) y x|))
149        ∂(withDensityReal ρ) := lintegral_mono h1
150    _ = ENNReal.ofReal (p.factorial : ℝ) * ∫⁻ x, ENNReal.ofReal
151        (Real.exp (1 * |centeredKernel (periodicRiesz d 0) (withDensityReal ρ) y x|))
152        ∂(withDensityReal ρ) := by
153        rw [← lintegral_const_mul']
154        · congr 1; funext x; rw [ENNReal.ofReal_mul (Nat.cast_nonneg _)]
155        · exact ENNReal.ofReal_ne_top
156    _ ≤ ENNReal.ofReal (p.factorial : ℝ) * ENNReal.ofReal Cexp := by
157        gcongr
158        exact hCexp y
159    _ = ENNReal.ofReal ((p.factorial : ℝ) * Cexp) := by
160        rw [ENNReal.ofReal_mul (Nat.cast_nonneg _)]
161    _ ≤ ENNReal.ofReal ((max 1 Cexp) ^ p * (p.factorial : ℝ)) := by
162        apply ENNReal.ofReal_le_ofReal
163        have hC1 : max 1 Cexp ≤ (max 1 Cexp) ^ p := by
164          calc max 1 Cexp = (max 1 Cexp) ^ 1 := (pow_one _).symm
165            _ ≤ (max 1 Cexp) ^ p := pow_le_pow_right₀ (le_max_left _ _) hp
166        have := le_max_right 1 Cexp
167        have hf : (0 : ℝ) ≤ p.factorial := Nat.cast_nonneg _
168        nlinarith

175/-- The attractive half of Proposition 4.25 (`O:eq:small-negative`, `γ ≤ 0`): for the periodic
176logarithm and a bounded probability density there are `a > 0` and `C` with
177`K_{N,γ}(μ) ≤ C` for all `N ≥ 2` and `-a ≤ γ ≤ 0`.  Proof: the cited
178Delgadino-Gvalani-Rosenzweig Theorem 2.6 / Remark 2.1 in the form
179`DGRHS.Attractive.source_uniform_partition`, whose factorial row hypothesis follows from
180Lemma 4.24 (`periodic_log_row`); the potential is bounded by `convMeasure_bounded`. -/
181theorem log_prep_attractive (hd : 2 ≤ d) (ρ : Torus d → ℝ) (hρ : IsProbDensity ρ) (Λ : ℝ)
182    (hΛ : ∀ x, ρ x ≤ Λ) :
183    ∃ a : ℝ, 0 < a ∧ ∃ C : ℝ, ∀ N : ℕ, 2 ≤ N → ∀ γ ∈ Set.Icc (-a) 0,
184      PreparationAux.KE (periodicRiesz d 0) N γ (withDensityReal ρ) ≤ ENNReal.ofReal C := by
185  have hs0 : (0 : ℝ) ≤ 0 := le_rfl
186  have hsd : (0 : ℝ) < d := by exact_mod_cast (by omega : 0 < d)
187  set g := periodicRiesz d 0 with hgdef
188  set μ := withDensityReal ρ with hμ
189  have hg : Measurable g := PoleCavityAux.measurable_periodicRiesz hd hs0 hsd
190  have hgi : Integrable g := KernelAux.integrable_periodicRiesz hd hs0 hsd
191  haveI : IsProbabilityMeasure μ := isProb_withDensityReal hρ
192  have hI : Integrable (fun p : Torus d × Torus d => g (p.1 - p.2)) (μ.prod μ) :=
193    PreparationAux.integrable_pair_withDensity hg hgi ρ Λ hΛ (ν := μ) rfl
194  have hev : ∀ z, g (-z) = g z := fun z => periodicRiesz_neg z
195  obtain ⟨C₁, hC₁, hrow⟩ := rowFactorialMoments hd ρ hρ Λ hΛ
196  obtain ⟨Cu₀, hCu₀⟩ := convMeasure_bounded (s := 0) hd hs0 hsd ρ hρ Λ hΛ
197  set Cu := max Cu₀ 0 with hCu
198  have hCu0 : 0 ≤ Cu := le_max_right _ _
199  have hpot : ∀ x, |DGRHS.potential μ (Wk g) x| ≤ Cu := fun x => by
200    rw [potential_eq, abs_neg]; exact (hCu₀ x).trans (le_max_left _ _)
201  set κ := DGRHS.Attractive.attractiveMomentConstant with hκ
202  have hκpos : 0 < κ := DGRHS.Attractive.attractiveMomentConstant_pos
203  have hC₁pos : 0 < C₁ := by linarith
204  set a := 1 / (2 * (κ * C₁)) with ha
205  have hapos : 0 < a := by positivity
206  set IE := DGRHS.interactionEnergy μ (Wk g) with hIE
207  refine ⟨a, hapos, Real.exp (a * |IE|) * 2 * Real.exp (a * Cu), fun N hN γ hγ => ?_⟩
208  set β := -γ with hβdef
209  have hβ0 : 0 ≤ β := by have := hγ.2; linarith
210  have hβa : β ≤ a := by have := hγ.1; linarith
211  have hsmall' : β * (κ * C₁) ≤ 1 / 2 := by
212    calc β * (κ * C₁) ≤ a * (κ * C₁) := by gcongr
213      _ = 1 / 2 := by rw [ha]; field_simp
214  have hsmall : β * (κ * C₁) < 1 := by linarith
215  have hWm : StronglyMeasurable (Function.uncurry (Wk g)) :=
216    (hg.comp (measurable_fst.sub measurable_snd)).neg.stronglyMeasurable
217  have hWi : Integrable (Function.uncurry (Wk g)) (μ.prod μ) := hI.neg
218  have hpotm : Measurable (DGRHS.potential μ (Wk g)) := by
219    have : DGRHS.potential μ (Wk g) = fun x => -convMeasure g μ x := funext potential_eq
220    rw [this]; exact (PoleCavityAux.measurable_convMeasure hg μ).neg
221  have htail : Integrable (fun x => Real.exp (2 * β * DGRHS.potential μ (Wk g) x)) μ := by
222    refine Integrable.of_bound (C := Real.exp (2 * β * Cu))
223      (hpotm.const_mul (2 * β) |>.exp).aestronglyMeasurable (Filter.Eventually.of_forall fun x => ?_)
224    rw [Real.norm_eq_abs, abs_of_pos (Real.exp_pos _)]
225    apply Real.exp_le_exp.2
226    have := (abs_le.1 (hpot x)).2
227    nlinarith
228  have hN0 : 0 < N := by omega
229  obtain ⟨hint, hbd⟩ := DGRHS.Attractive.source_uniform_partition (μ := μ) (Wk g) hWm hWi
230    (Wk_symm hev) C₁ β (by
231      refine DGRHS.Attractive.rowFactorialMoments_of_lintegral μ _
232        (DGRHS.Gaussian.stronglyMeasurable_hoeffdingCentered μ (Function.uncurry (Wk g))
233          hWm).measurable C₁ (by linarith) hrow)
234    hβ0 hsmall htail N hN0
235  have hKE : PreparationAux.KE g N γ μ = ENNReal.ofReal
236      (∫ x : Fin N → Torus d, Real.exp (-β * (N : ℝ) *
237        DGRHS.CLT.sourceCanonicalModulatedEnergy
238          (fun p => DGRHS.centeredKernel μ (Wk g) p.1 p.2)
239          (DGRHS.potential μ (Wk g)) (DGRHS.interactionEnergy μ (Wk g)) N x)
240        ∂Measure.pi (fun _ : Fin N => μ)) := by
241    rw [ofReal_integral_eq_lintegral_ofReal hint
242      (Filter.Eventually.of_forall fun x => (Real.exp_pos _).le)]
243    unfold PreparationAux.KE
244    refine lintegral_congr fun X => ?_
245    congr 2
246    rw [modulatedEnergyKernel_eq hev hI hN0 X]
247    simp only [hβdef]
248    ring
249  rw [hKE]
250  apply ENNReal.ofReal_le_ofReal
251  refine hbd.trans ?_
252  have h1 : Real.exp (β * |IE|) ≤ Real.exp (a * |IE|) :=
253    Real.exp_le_exp.2 (by gcongr)
254  have h2 : Real.sqrt (2 / (1 - (β * (κ * C₁)) ^ 2)) ≤ 2 := by
255    rw [Real.sqrt_le_left (by norm_num)]
256    have hp : 0 ≤ β * (κ * C₁) := by positivity
257    have : (β * (κ * C₁)) ^ 2 ≤ 1 / 4 := by nlinarith
258    rw [div_le_iff₀ (by nlinarith)]
259    nlinarith
260  have h3 : Real.sqrt (∫ x, Real.exp (2 * β * DGRHS.potential μ (Wk g) x) ∂μ) ≤
261      Real.exp (a * Cu) := by
262    rw [Real.sqrt_le_left (Real.exp_pos _).le]
263    have hle : ∫ x, Real.exp (2 * β * DGRHS.potential μ (Wk g) x) ∂μ ≤
264        Real.exp (2 * β * Cu) := by
265      have := norm_integral_le_of_norm_le_const (μ := μ)
266        (f := fun x => Real.exp (2 * β * DGRHS.potential μ (Wk g) x))
267        (C := Real.exp (2 * β * Cu)) (Filter.Eventually.of_forall fun x => by
268          rw [Real.norm_eq_abs, abs_of_pos (Real.exp_pos _)]
269          apply Real.exp_le_exp.2
270          have := (abs_le.1 (hpot x)).2
271          nlinarith)
272      simpa using (le_abs_self _).trans (by simpa [Real.norm_eq_abs] using this)
273    refine hle.trans ?_
274    rw [← Real.exp_nat_mul]
275    apply Real.exp_le_exp.2
276    push_cast
277    nlinarith
278  have hpos1 : 0 ≤ Real.sqrt (2 / (1 - (β * (κ * C₁)) ^ 2)) := Real.sqrt_nonneg _
279  have hpos2 : 0 ≤ Real.sqrt (∫ x, Real.exp (2 * β * DGRHS.potential μ (Wk g) x) ∂μ) :=
280    Real.sqrt_nonneg _
281  calc _ ≤ Real.exp (a * |IE|) * 2 * Real.exp (a * Cu) := by gcongr
