30/-- Measurability, integrability and near-field Hölder bound (exponent `β ≤ 1`) of `ψ_a = ρ (Mᵀ ∇f)_a`. -/
31theorem pr14_psi_facts (M : Matrix (Fin d) (Fin d) ℝ) {f : Euc d → ℝ} (hf : MemCb 2 f) {ρ : Euc d → ℝ}
32    (hρm : Measurable ρ) (hρint : Integrable ρ) {Bρ Cρ β : ℝ} (hBρ : ∀ x, |ρ x| ≤ Bρ) (hCρ : 0 ≤ Cρ)
33    (hβ0 : 0 ≤ β) (hβ1 : β ≤ 1)
34    (hρH : ∀ x z : Euc d, ‖z‖ ≤ 1 / 4 → |ρ (x - z) - ρ x| ≤ Cρ * ‖z‖ ^ β) (a : Fin d) :
35    Measurable (psiRd f M ρ a) ∧ Integrable (psiRd f M ρ a) ∧
36      ∃ Λ : ℝ, 0 ≤ Λ ∧ ∀ x z : Euc d, ‖z‖ ≤ 1 / 4 →
37        |psiRd f M ρ a (x - z) - psiRd f M ρ a x| ≤ Λ * ‖z‖ ^ β := by
38  obtain ⟨Bv, Lv, hBv0, hLv0, hvc, hBv, hLv⟩ := vcomp_facts_rd M hf a
39  have hBρ0 : 0 ≤ Bρ := (abs_nonneg _).trans (hBρ 0)
40  have hmeas : Measurable (psiRd f M ρ a) := hρm.mul hvc.measurable
41  refine ⟨hmeas, ?_, Cρ * Bv + Bρ * Lv, by positivity, fun x z hz => ?_⟩
42  · refine (hρint.norm.const_mul Bv).mono' hmeas.aestronglyMeasurable ?_
43    refine Filter.Eventually.of_forall fun y => ?_
44    simp only [psiRd, norm_mul, Real.norm_eq_abs]
45    nlinarith [hBv y, abs_nonneg (ρ y)]
46  · exact holder_mul_near_rd hβ0 hβ1 hCρ hLv0 hBρ hBv hρH hLv x z hz
